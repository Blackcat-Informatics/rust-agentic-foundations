    RegisteredCheck projects an actual Result<GateReport, Error> into native Graph
    storage and canonical N-Triples1.2. Its fixed protocol IRI is
    urn:raf:census:registered-check:v1. The sole RegisteredCheck root states
    protocolVersion, status and findingCount. Clean has zero findings; findings has a
    positive count; error has zero findings and exactly one errorMessage. Each finding
    is an owned CensusFinding node at #finding/ORDINAL with a contiguous zero-based
    integer ordinal and message. Order and duplicate messages are preserved.
    Unknown fields/classes/versions, incomplete or orphan nodes, inconsistent
    statuses, noncanonical bytes and responses larger than 16 MiB refuse. The size
    bound governs wire bytes, not total native graph allocation.

    //! Thin host input adapter; migration and debt policy live in raf-census.
    
    use crate::cli::{Args, Command};
    use raf_census::{
        Error, Gate, MAX_INDEX_BYTES, MAX_LEDGER_BYTES, MigrationInput, migrate_index, migrate_legacy,
    };
    use raf_source::{DiskTree, RelPath, Tree as _};
    use std::{io::Write, path::Path};
    
    pub(crate) fn run(args: &Args, output: &mut dyn Write) -> Result<u8, crate::error::Error> {
        let gate = Gate::workspace(&args.workspace);
        match &args.command {
            Command::Check { package, example } => {
                crate::registered::check(&gate.root()?, package, example, output)
            }
            Command::Debt { write: _ } => {
                let receipt = gate.write_debt()?;
                writeln!(output, "{receipt}").map_err(stdout)?;
                Ok(0)
            }
            Command::Migrate {
                legacy,
                ledger,
                index,
                baseline,
            } => {
                let root = gate.root()?;
                let tree = DiskTree::new(&root).map_err(|source| Error::Tree {
                    file: root.display().to_string(),
                    source,
                })?;
                let old = read(ledger, MAX_LEDGER_BYTES)?;
                let companion = if *legacy {
                    baseline.as_ref()
                } else {
                    index.as_ref()
                }
                .ok_or_else(|| {
                    Error::Options(
                        "migration requires its explicitly selected baseline or index".into(),
                    )
                })?;
                let previous = read(
                    companion,
                    if *legacy {
                        MAX_LEDGER_BYTES
                    } else {
                        MAX_INDEX_BYTES
                    },
                )?;
                let old_path = ledger.display().to_string();
                let previous_path = companion.display().to_string();
                let old = MigrationInput {
                    path: &old_path,
                    text: &old,
                };
                let previous = MigrationInput {
                    path: &previous_path,
                    text: &previous,
                };
                let result = if *legacy {
                    migrate_legacy(&tree, old, previous)
                } else {
                    migrate_index(&tree, old, previous)
                }?;
                output.write_all(result.ledger.as_bytes()).map_err(stdout)?;
                Ok(u8::from(!result.unmapped.is_empty()))
            }
        }
    }
    
    fn read(path: &Path, limit: usize) -> Result<String, Error> {
        let canonical = std::fs::canonicalize(path).map_err(|source| Error::Io {
            file: path.display().to_string(),
            source,
        })?;
        let parent = canonical
            .parent()
            .ok_or_else(|| Error::Options(format!("{}: expected an input file", path.display())))?;
        let name = canonical
            .file_name()
            .and_then(std::ffi::OsStr::to_str)
            .ok_or_else(|| {
                Error::Options(format!("{}: input filename is not UTF-8", path.display()))
            })?;
        let relative = RelPath::new(name).map_err(|error| Error::Options(error.to_string()))?;
        let tree = DiskTree::new(parent).map_err(|source| Error::Tree {
            file: path.display().to_string(),
            source,
        })?;
        tree.read(
            &relative,
            u64::try_from(limit).map_err(|error| Error::Options(error.to_string()))?,
        )
        .map_err(|source| Error::Tree {
            file: path.display().to_string(),
            source,
        })
    }
    
    fn stdout(source: std::io::Error) -> Error {
        Error::Io {
            file: "<stdout>".into(),
            source,
        }
    }

    //! Preserve library and bounded host execution failures at the CLI boundary.
    
    use std::{error::Error as StdError, fmt};
    
    #[derive(Debug)]
    pub(crate) enum Error {
        Census(raf_census::Error),
        Transport(String),
        Input {
            context: String,
            source: Box<dyn StdError + Send + Sync>,
        },
    }
    
    impl Error {
        pub(crate) const fn exit_code(&self) -> u8 {
            if matches!(
                self,
                Self::Census(raf_census::Error::RegistrationRequired { .. })
            ) {
                2
            } else {
                1
            }
        }
    }
    
    impl From<raf_census::Error> for Error {
        fn from(error: raf_census::Error) -> Self {
            Self::Census(error)
        }
    }
    
    impl fmt::Display for Error {
        fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
            match self {
                Self::Census(error) => error.fmt(f),
                Self::Transport(message) => f.write_str(message),
                Self::Input { context, source } => write!(f, "{context}: {source}"),
            }
        }
    }
    
    impl StdError for Error {
        fn source(&self) -> Option<&(dyn StdError + 'static)> {
            match self {
                Self::Census(error) => Some(error),
                Self::Input { source, .. } => Some(source.as_ref()),
                Self::Transport(_) => None,
            }
        }
    }

    fn main() -> ExitCode {
        let args = match cli::parse(std::env::args_os()) {
            Ok(args) => args,
            Err(error) => {
                let status = u8::try_from(error.exit_code()).unwrap_or(2);
                return if error.print().is_ok() {
                    ExitCode::from(status)
                } else {
                    ExitCode::FAILURE
                };
            }
        };
        match command::run(&args, &mut io::stdout().lock()) {
            Ok(code) => ExitCode::from(code),
            Err(error) => {
                let code = error.exit_code();
                let _ = writeln!(io::stderr().lock(), "cargo-census: {error}");
                ExitCode::from(code)
            }
        }
    }

    use std::{error::Error as StdError, io::Write, path::Path};

    pub(crate) fn check(
        root: &Path,
        package: &Package,
        example: &ExampleName,
        output: &mut dyn Write,
    ) -> Result<u8, Error> {
        let runner = SystemRunner::new(Programs::new(root));
        run(&runner, package, example, output)
    }
    
    fn run(
        runner: &dyn Runner,
        package: &Package,
        example: &ExampleName,
        output: &mut dyn Write,
    ) -> Result<u8, Error> {
        let context = format!("registered check {package}/{example}");
        let captured = runner
            .run(
                &Invocation::example_run(package, example),
                &Limits::build().with_max_output(MAX_CHECK_RESPONSE_BYTES),
            )
            .map_err(|source| input(&context, source))?;
        if !matches!(captured.code, Some(0 | 1)) {
            return Err(Error::Transport(format!(
                "{context}: target failed with exit {:?}; stderr: {}",
                captured.code,
                captured.stderr.escape_ascii()
            )));
        }
        let text = std::str::from_utf8(&captured.stdout)
            .map_err(|source| input(&diagnostics(&context, &captured.stderr), source))?;
        let response = RegisteredCheck::from_ntriples(text)
            .map_err(|source| input(&diagnostics(&context, &captured.stderr), source))?;
        if captured.code != Some(i32::from(response.exit_code())) {
            return Err(Error::Transport(format!(
                "{}: target exit {:?} disagrees with protocol outcome {}",
                diagnostics(&context, &captured.stderr),
                captured.code,
                response.exit_code()
            )));
        }
        if let Some(error) = response.error_message() {
            return Err(Error::Transport(format!(
                "{context}: repository evaluation failed: {}",
                error.escape_debug()
            )));
        }
        if response.exit_code() == 0 {
            writeln!(output, "census: clean").map_err(|source| input("<stdout>", source))?;
        } else {
            for finding in response.findings() {
                writeln!(output, "{finding}").map_err(|source| input("<stdout>", source))?;
            }
        }
        Ok(response.exit_code())
    }
    
    fn diagnostics(context: &str, stderr: &[u8]) -> String {
        if stderr.is_empty() {
            context.into()
        } else {
            format!("{context}; stderr: {}", stderr.escape_ascii())
        }
    }
    
    fn input(context: &str, source: impl StdError + Send + Sync + 'static) -> Error {
        Error::Input {
            context: context.into(),
            source: Box::new(source),
        }
    }

    //! Read-only repository-target responses over the shared native RDF profile.
    
    use raf_rdf::{Builder, Format, Graph, Node, TermValue as Term};
    
    use crate::report::{shape, vocab, write::put};

    /// The fixed, read-only registered-check response protocol.
    pub const CHECK_PROTOCOL_VERSION: &str = "urn:raf:census:registered-check:v1";
    /// Maximum canonical response size, matching the bounded host stream policy.
    pub const MAX_CHECK_RESPONSE_BYTES: usize = 16 * 1024 * 1024;
    
    #[derive(Clone, Copy, Debug, PartialEq, Eq)]
    enum Outcome {
        Clean,
        Findings,
        Error,
    }
    
    impl Outcome {
        const fn text(self) -> &'static str {
            match self {
                Self::Clean => "clean",
                Self::Findings => "findings",
                Self::Error => "error",
            }
        }
    }
    
    /// One complete registered Gate outcome, retained in native RDF storage.
    ///
    /// Finding ordinals retain order and multiplicity, including equal messages.
    /// This response never authorizes or performs a debt write.
    #[derive(Clone, Debug)]
    pub struct RegisteredCheck {
        graph: Graph,
        outcome: Outcome,
        count: usize,
    }
    
    impl RegisteredCheck {
        /// Project an actual Gate evaluation, preserving findings or its error.
        ///
        /// # Errors
        /// Refuses native storage/profile failures and oversized responses.
        pub fn from_check(result: &Result<GateReport, Error>) -> Result<Self, E> {
            match result {
                Ok(report) => Self::build(
                    if report.is_clean() {
                        Outcome::Clean
                    } else {
                        Outcome::Findings
                    },
                    report.findings().iter().map(String::as_str),
                    None,
                ),
                Err(error) => Self::build(Outcome::Error, std::iter::empty(), Some(&error.to_string())),
            }
        }
    
        fn build<'a>(
            outcome: Outcome,
            messages: impl ExactSizeIterator<Item = &'a str>,
            error: Option<&str>,
        ) -> Result<Self, E> {
            let mut graph = Builder::new();
            let count = messages.len();
            put(&mut graph, "", "type", shape::iri("RegisteredCheck"))?;
            put(
                &mut graph,
                "",
                "protocolVersion",
                Term::iri(CHECK_PROTOCOL_VERSION),
            )?;
            put(
                &mut graph,
                "",
                "status",
                Term::simple_literal(outcome.text()),
            )?;
            put(
                &mut graph,
                "",
                "findingCount",
                shape::integer(count as u128),
            )?;
            let mut bytes = 0usize;
            for (ordinal, message) in messages.enumerate() {
                bytes = bytes.checked_add(message.len()).ok_or_else(oversized)?;
                if bytes > MAX_CHECK_RESPONSE_BYTES {
                    return Err(oversized());
                }
                let id = finding_id(ordinal);
                put(&mut graph, "", "finding", Term::iri(&id))?;
                put(&mut graph, &id, "type", shape::iri("CensusFinding"))?;
                put(&mut graph, &id, "ordinal", shape::integer(ordinal as u128))?;
                put(&mut graph, &id, "message", Term::simple_literal(message))?;
            }
            if let Some(error) = error {
                if error.len() > MAX_CHECK_RESPONSE_BYTES {
                    return Err(oversized());
                }
                put(&mut graph, "", "errorMessage", Term::simple_literal(error))?;
            }
            let graph = graph.finish().map_err(E::Ground)?;

        /// Read one exact canonical N-Triples1.2 response and validate its closed shape.
        ///
        /// # Errors
        /// Refuses size, canonical bytes, protocol version, fields, ordinals and
        /// inconsistent outcomes, including unreferenced or incomplete findings.
        pub fn from_ntriples(text: &str) -> Result<Self, E> {
            if text.len() > MAX_CHECK_RESPONSE_BYTES {
                return Err(oversized());
            }
            let graph = Graph::from_text(text, Format::NTriples).map_err(E::Ground)?;
            let root = graph
                .get("")
                .ok_or_else(|| E::Value("registered response requires its ground root".into()))?;
            if shape::class(&root)? != "RegisteredCheck" {
                return Err(E::Kind {
                    expected: "RegisteredCheck".into(),
                    found: shape::class(&root)?.into(),
                });
            }
            if shape::one(&root, "protocolVersion")?.iri() != Some(CHECK_PROTOCOL_VERSION) {
                return Err(E::Value(format!(
                    "registered check requires protocol {CHECK_PROTOCOL_VERSION}"
                )));
            }
            let outcome = match shape::string(&root, "status")?.as_str() {
                "clean" => Outcome::Clean,
                "findings" => Outcome::Findings,
                "error" => Outcome::Error,
                other => return Err(E::Value(format!("registered check status {other}"))),
            };
            let count = shape::count(&root, "findingCount")?;
            validate_outcome(root, outcome, count)?;
            validate_members(&graph, root, count)?;
            Ok(Self {
                graph,
                outcome,
                count,
            })
        }
    
        /// Emit the native canonical N-Triples1.2 carrier.
        ///
        /// # Errors
        /// Refuses native output failure or a response exceeding the stream limit.
        pub fn to_ntriples(&self) -> Result<String, E> {
            let text = self.graph.to_text(Format::NTriples).map_err(E::Ground)?;
            if text.len() > MAX_CHECK_RESPONSE_BYTES {
                return Err(oversized());
            }
            Ok(text)
        }

        pub const fn exit_code(&self) -> u8 {
            if matches!(self.outcome, Outcome::Clean) {

            } else {

        /// Finding messages in their original order, without a parallel assertion map.
        ///
        /// # Panics
        /// Panics if the private graph loses its constructor-validated finding shape.
        pub fn findings(&self) -> impl ExactSizeIterator<Item = &str> {
            (0..self.count).map(|ordinal| {

                    .get(&finding_id(ordinal))
                    .expect("validated finding sequence");

                    .expect("validated message")
                    .literal()
                    .expect("validated string")
                    .0
            })
        }
    
        /// The original evaluation error, distinct from ordinary findings.
        ///
        /// # Panics
        /// Panics if the private graph loses its constructor-validated error shape.
        #[must_use]
        pub fn error_message(&self) -> Option<&str> {
            if self.outcome != Outcome::Error {
                return None;
            }
            let root = self.graph.get("").expect("validated root");
            Some(
                shape::one(&root, "errorMessage")
                    .expect("validated error")
                    .literal()
                    .expect("validated string")
                    .0,
            )
        }

        format!("#finding/{ordinal}")

    fn oversized() -> E {
        E::Value(format!(
            "registered check response exceeds {MAX_CHECK_RESPONSE_BYTES} bytes"
        ))
    }
    
    fn validate_outcome(root: Node<'_>, outcome: Outcome, count: usize) -> Result<(), E> {
        let errors = root.values(&shape::property("errorMessage")).count();
        if (outcome == Outcome::Findings) != (count > 0) || (outcome == Outcome::Error && count != 0) {
            return Err(E::Inconsistent(
                "registered check status and finding count".into(),
            ));
        }
        if outcome == Outcome::Error {
            shape::string(&root, "errorMessage")?;
        } else if errors != 0 {
            return Err(E::Inconsistent(
                "non-error response contains errorMessage".into(),
            ));
        }
        Ok(())
    }
    
    fn validate_members(graph: &Graph, root: Node<'_>, count: usize) -> Result<(), E> {
        let links = root.values(&shape::property("finding"));
        if links.count() != count {
            return Err(E::Inconsistent(
                "registered findingCount disagrees with membership".into(),
            ));
        }
        for value in root.values(&shape::property("finding")) {
            let id = value
                .iri()
                .ok_or_else(|| E::Value("registered finding reference must be an IRI".into()))?;
            let node = graph.get(id).ok_or_else(|| E::Dangling(id.into()))?;
            let ordinal = shape::count(&node, "ordinal")?;
            if ordinal >= count || id != finding_id(ordinal) {
                return Err(E::Name(id.into()));
            }
            string_value(node, "message")?;
        }
        let mut subjects = 0usize;
        for (id, node) in graph.nodes() {
            subjects += 1;
            let class = shape::class(&node)?;
            let fields: &[&str] = match (id.is_empty(), class) {
                (true, "RegisteredCheck") => &[
                    "protocolVersion",
                    "status",
                    "findingCount",
                    "finding",
                    "errorMessage",
                ],
                (false, "CensusFinding") => &["ordinal", "message"],
                _ => return Err(E::Class(format!("{id}: {class}"))),
            };
            for property in node.properties() {
                if property != raf_rdf::vocab::TYPE
                    && !property
                        .strip_prefix(vocab::CENSUS)
                        .is_some_and(|key| fields.contains(&key))
                {
                    return Err(E::UnknownPredicate(property.into()));
                }
            }
        }
        if subjects.checked_sub(1) != Some(count) {
            return Err(E::Orphan(
                "registered response has unreferenced subjects".into(),
            ));
        }
        Ok(())
    }
    
    fn string_value<'a>(node: Node<'a>, key: &str) -> Result<&'a str, E> {
        shape::one(&node, key)?
            .literal()
            .filter(|(_, datatype)| *datatype == raf_rdf::vocab::STRING)
            .map(|(text, _)| text)
            .ok_or_else(|| E::Value(format!("registered {key}: string expected")))
    }
