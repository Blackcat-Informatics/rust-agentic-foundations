        /// Insert one profile assertion through native set insertion.
        ///
        /// # Errors
        /// Refuses invalid roles, the reserved construction namespace and duplicates.
        pub fn insert(&mut self, mut triple: Triple) -> Result<(), Error> {
            triple.validate()?;
            for value in [
                &mut triple.subject,
                &mut triple.predicate,
                &mut triple.object,
            ] {
                term::expand(value, &self.owner, true)?;
            }
            let quad = QuadValues::triple(triple.subject, triple.predicate, triple.object);
            if !self
                .dataset
                .insert(quad)
                .map_err(|e| Error::Value(e.to_string()))?

                return Err(Error::Repeated);
            }
            Ok(())
        }

        pub(crate) fn validate(&self) -> Result<(), Error> {
            if self.dataset.named_graphs().next().is_some() {
                return Err(Error::Value(
                    "named graphs are outside the report profile".into(),
                ));
            }
            if self.get("").is_none() {
                return Err(Error::Value("the report root is absent".into()));
            }
            let subjects: BTreeSet<_> = self.rows().map(|row| row.s).collect();
            // A native dataset can intentionally retain equal assertions in distinct
            // physical tables. This profile refuses that flat overlap. IDs are
            // compared only inside this one dictionary, never across datasets.
            let mut assertions = purrdf_core::FastSet::default();
            for row in self.rows() {
                if !assertions.insert(row) {
                    return Err(Error::Repeated);
                }
                let subject = self
                    .value(row.s)
                    .iri()
                    .ok_or_else(|| Error::Value("subject must be an IRI".into()))?;
                let TermRef::Iri(predicate) = self.dataset.resolve(row.p) else {
                    return Err(Error::Value("predicate must be an IRI".into()));
                };
                let object = self.value(row.o).to_owned();
                term::validate(subject, predicate, &object)?;
                let check = |value: Value<'_>| -> Result<(), Error> {
                    if value.iri().is_some_and(term::local) && !subjects.contains(&value.id) {
                        return Err(Error::Value(format!(
                            "dangling owned node {}",
                            value.iri().expect("IRI")
                        )));
                    }
                    Ok(())
                };
                let value = self.value(row.o);
                if let Some((s, _, o)) = value.quoted() {
                    check(s)?;
                    check(o)?;
                } else {
                    check(value)?;
                }
            }
            Ok(())
        }
    }

        pub fn term(self) -> TermRef<'a> {
            self.graph.dataset.resolve(self.id)

            match self.term() {
                TermRef::Iri(iri) => Some(self.graph.local(iri)),
                _ => None,

        pub fn literal(self) -> Option<(&'a str, &'a str)> {
            if let TermRef::Literal {
                lexical,
                datatype,
                language: None,
                direction: None,
            } = self.term()
                && let TermRef::Iri(datatype) = self.graph.dataset.resolve(datatype)
            {
                return Some((lexical, datatype));
            }
            None
        }

        pub fn quoted(self) -> Option<(Self, Self, Self)> {
            match self.term() {
                TermRef::Triple { s, p, o } => Some((

                    self.graph.value(p),
                    self.graph.value(o),

                _ => None,
            }
        }

        pub fn to_owned(self) -> TermValue {
            let mut value = self.graph.dataset.term_value(self.id);
            term::normalize(&mut value, &self.graph.owner);
            value
        }
    }

    //! Native RDF parsing followed by the closed report's ownership and byte laws.
    
    use crate::{Error, Format, Graph, vocab, write};
    use purrdf_rdf::{ParseOptions, RdfDataset, TermRef, parse_dataset_with};
    
    fn checked_base(text: &str) -> Result<&str, Error> {
        if text
            .strip_prefix(vocab::BASE)
            .is_none_or(|hex| purrdf_hash::hex::decode_32_canonical(hex).is_none())
        {
            return Err(Error::Value(
                "snapshot BASE requires its namespace and 64 lowercase hexadecimal digits".into(),
            ));
        }
        Ok(text)
    }
    
    fn root_base(dataset: &RdfDataset) -> Result<String, Error> {
        let subjects = dataset
            .quads()
            .map(|row| row.s)
            .chain(dataset.reifiers().map(|(subject, _)| subject))
            .chain(dataset.annotations().map(|(subject, _, _)| subject));
        let mut root = None;
        for subject in subjects {
            if let TermRef::Iri(iri) = dataset.resolve(subject)
                && checked_base(iri).is_ok()
            {
                if root.is_some_and(|before| before != iri) {
                    return Err(Error::Value("more than one snapshot root".into()));
                }
                root = Some(iri);
            }
        }
        root.map(str::to_owned)
            .ok_or_else(|| Error::Value("absolute snapshot root is absent".into()))
    }
    
    pub(crate) fn parse(text: &str, format: Format) -> Result<Graph, Error> {
        let media = match format {
            Format::Turtle => "text/turtle",
            Format::NTriples => "application/n-triples",
        };
        let parsed = parse_dataset_with(text.as_bytes(), media, None, &ParseOptions::default())
            .map_err(|error| {
                let (line, column) = error.location.as_deref().map_or((1, 1), |location| {
                    (
                        location
                            .line
                            .and_then(|line| usize::try_from(line).ok())
                            .unwrap_or(1),
                        location.column.map_or(1, |column| column as usize),
                    )
                });
                Error::Syntax {
                    line,
                    column,
                    reason: error.to_string(),
                }
            })?;
        let base = match format {
            Format::Turtle => {
                let base = parsed
                    .document_base
                    .as_ref()
                    .filter(|base| matches!(base.origin(), purrdf_iri::BaseOrigin::Directive { .. }))
                    .ok_or_else(|| Error::Value("snapshot BASE directive is absent".into()))?;
                checked_base(base.iri().as_str())?.to_owned()
            }
            Format::NTriples => root_base(&parsed.dataset)?,
        };
        if parsed.dataset.named_graphs().next().is_some() {
            return Err(Error::Value(
                "named graphs are outside the report profile".into(),
            ));
        }
        let graph = Graph {
            dataset: parsed.dataset,
            owner: base.clone(),
        };
        graph.validate()?;
        let (computed, canonical) = write::render(&graph, format)?;
        if computed != base {
            return Err(Error::Digest {
                declared: base,
                computed,
            });
        }
        if canonical != text {
            let offset = purrdf_deflate::common_prefix_len(canonical.as_bytes(), text.as_bytes());
            let position = purrdf_iri::LineIndex::new(text)
                .try_locate(text, offset)
                .map_err(|e| Error::Value(e.to_string()))?;
            let line = usize::try_from(position.line)
                .map_err(|_| Error::Value("line exceeds usize".into()))?;
            return Err(Error::NotCanonical { line });
        }
        Ok(graph)
    }

    //! Report roles and ordering over `PurRDF`'s native term representation.
    
    use crate::{Error, TermValue, vocab};
    
    /// A report assertion with native RDF terms. Insertion validates its closed roles.
    #[derive(Clone, Debug, PartialEq, Eq)]
    pub struct Triple {
        /// Snapshot-owned root or fragment IRI.
        pub subject: TermValue,
        /// Absolute vocabulary IRI.
        pub predicate: TermValue,
        /// Atomic value or depth-one triple term.
        pub object: TermValue,
    }
    
    pub(crate) fn local(text: &str) -> bool {
        text.is_empty() || text.starts_with('#')
    }
    
    pub(crate) fn normalize(value: &mut TermValue, base: &str) {
        let atom = |value: &mut TermValue| {
            if let TermValue::Iri(iri) = value {
                if iri == base {
                    iri.clear();
                } else if let Some(fragment) =
                    iri.strip_prefix(base).filter(|tail| tail.starts_with('#'))
                {
                    *iri = fragment.to_owned();
                }
            }
        };
        if let TermValue::Triple { s, o, .. } = value {
            atom(s);
            atom(o);
        } else {
            atom(value);
        }
    }
    
    pub(crate) fn expand(value: &mut TermValue, owner: &str, constructing: bool) -> Result<(), Error> {
        let atom = |value: &mut TermValue| -> Result<(), Error> {
            if let TermValue::Iri(iri) = value {
                if local(iri) {
                    *iri = format!("{owner}{iri}");
                } else if constructing
                    && (iri == owner
                        || iri
                            .strip_prefix(owner)
                            .is_some_and(|tail| tail.starts_with('#')))
                {
                    return Err(Error::Value(
                        "external IRI aliases the reserved construction owner".into(),
                    ));
                }
            }
            if let TermValue::Literal { datatype, .. } = value
                && constructing
                && (datatype == owner
                    || datatype
                        .strip_prefix(owner)
                        .is_some_and(|tail| tail.starts_with('#')))
            {
                return Err(Error::Value(
                    "datatype aliases the reserved construction owner".into(),
                ));
            }
            Ok(())
        };
        if let TermValue::Triple { s, p, o } = value {
            atom(s)?;
            atom(p)?;
            atom(o)?;
        } else {
            atom(value)?;
        }
        Ok(())
    }
    
    pub(crate) fn iri(text: &str, relative: bool) -> Result<(), Error> {
        if relative && local(text) {
            if text.is_empty()
                || (text.len() > 1
                    && text.as_bytes()[1..]
                        .iter()
                        .all(|b| b.is_ascii_alphanumeric() || b"-._~!$&'()*+,;=:@/?%".contains(b)))
            {
                // The native parser admits absolute IRIs. Validate the owned
                // reference in its actual absolute snapshot namespace.
                purrdf_iri::parse(&format!("{}{text}", vocab::BASE))
                    .map_err(|_| Error::Value(format!("owned IRI {text}")))?;
                return Ok(());
            }
        } else if purrdf_iri::parse(text).is_ok_and(|iri| iri.scheme().is_some()) {
            return Ok(());
        }
        Err(Error::Value(format!(
            "{} IRI {text}",
            if relative {
                "absolute or owned"
            } else {
                "absolute"
            }
        )))
    }
    
    fn atom(term: &TermValue) -> Result<(), Error> {
        match term {
            TermValue::Iri(text) => iri(text, true),
            TermValue::Literal {
                datatype,
                language,
                direction,
                ..
            } => {
                iri(datatype, false)?;
                if language.is_some()
                    || direction.is_some()
                    || [
                        purrdf_iri::vocab::rdf::LANG_STRING,
                        purrdf_iri::vocab::rdf::DIR_LANG_STRING,
                    ]
                    .contains(&datatype.as_str())
                {
                    return Err(Error::Value(
                        "language metadata is outside the report profile".into(),
                    ));
                }
                Ok(())
            }
            TermValue::Blank { .. } => Err(Error::Value(
                "blank nodes are outside the report profile".into(),
            )),
            TermValue::Triple { .. } => Err(Error::Value(
                "nested triple terms are outside the report profile".into(),
            )),
        }
    }
    
    impl Triple {
        pub(crate) fn validate(&self) -> Result<(), Error> {
            let subject = self
                .subject
                .as_iri()
                .ok_or_else(|| Error::Value("report subject must be an IRI".into()))?;
            let predicate = self
                .predicate
                .as_iri()
                .ok_or_else(|| Error::Value("predicate must be an IRI".into()))?;
            validate(subject, predicate, &self.object)
        }
    }
    
    pub(crate) fn validate(subject: &str, predicate: &str, object: &TermValue) -> Result<(), Error> {
        iri(subject, true)?;
        if !local(subject) {
            return Err(Error::Value("report subject must be snapshot-owned".into()));
        }
        iri(predicate, false)?;
        if let TermValue::Triple { s, p, o } = object {
            iri(
                s.as_iri()
                    .ok_or_else(|| Error::Value("quoted subject must be an IRI".into()))?,
                true,
            )?;
            iri(
                p.as_iri()
                    .ok_or_else(|| Error::Value("quoted predicate must be an IRI".into()))?,
                false,
            )?;
            atom(o)
        } else {
            atom(object)
        }
    }

    use crate::{Error, Format, Graph, TermValue, term, vocab};
    use purrdf_lex::allocation::{Memory, Resident};
    
    struct Selection<'a> {
        graph: &'a Graph,
        rows: Vec<purrdf_core::QuadIds>,
        keys: std::collections::BTreeMap<purrdf_core::TermId, TermValue>,
    }
    impl<'a> Selection<'a> {
        fn new(graph: &'a Graph) -> Self {
            let rows: Vec<_> = graph.rows().collect();
            let mut keys = std::collections::BTreeMap::new();
            for row in &rows {
                keys.entry(row.o)
                    .or_insert_with(|| graph.value(row.o).to_owned());
            }
            let mut selection = Self { graph, rows, keys };
            let iri = |id| match graph.dataset.resolve(id) {
                purrdf_core::TermRef::Iri(iri) => iri,
                _ => unreachable!("validated IRI role"),
            };
            selection.rows.sort_unstable_by(|a, b| {
                (graph.local(iri(a.s)), iri(a.p), &selection.keys[&a.o]).cmp(&(
                    graph.local(iri(b.s)),
                    iri(b.p),
                    &selection.keys[&b.o],
                ))
            });
            selection
        }
        fn iri(&self, id: purrdf_core::TermId) -> &str {
            match self.graph.dataset.resolve(id) {
                purrdf_core::TermRef::Iri(iri) => iri,
                _ => unreachable!("validated IRI role"),
            }
        }
    }
    
    fn body(selection: &Selection<'_>, base: Option<&str>, out: &mut String) -> Result<(), Error> {
        let mut resident = Resident;
        let mut memory = Memory::new(&mut resident);
        for triple in &selection.rows {
            for (iri, owned) in [
                (selection.graph.local(selection.iri(triple.s)), true),
                (selection.iri(triple.p), false),
            ] {
                if let Some(base) = base
                    && owned
                    && term::local(iri)
                {
                    out.push('<');
                    purrdf_lex::iri_escape::push_escaped(base, out);
                    purrdf_lex::iri_escape::push_escaped(iri, out);
                    out.push('>');
                } else {
                    purrdf_lex::term_syntax::write_iri(iri, out);
                }
                out.push(' ');
            }
            let value = &selection.keys[&triple.o];
            if let Some(base) = base {
                value
                    .try_write_nested_with_memory(
                        out,
                        ["<<( ", " ", " )>>"],
                        |out, leaf, memory| {
                            if let TermValue::Iri(iri) = leaf
                                && term::local(iri)
                            {
                                out.push('<');
                                purrdf_lex::iri_escape::push_escaped(base, out);
                                purrdf_lex::iri_escape::push_escaped(iri, out);
                                out.push('>');
                                Ok(())
                            } else {
                                purrdf_core::write_term_value_with_memory(leaf, out, memory)
                            }
                        },
                        |out, text, _| {
                            out.push_str(text);
                            Ok(())
                        },
                        &mut memory,
                    )
                    .map_err(|error| Error::Value(format!("native term output: {error}")))?;
            } else {
                purrdf_core::write_term_value_with_memory(value, out, &mut memory)
                    .map_err(|error| Error::Value(format!("native term output: {error}")))?;
            }
            out.push_str(" .\n");
        }
        Ok(())
    }
    
    fn normalized(selection: &Selection<'_>) -> Result<String, Error> {
        let mut out = String::from("VERSION \"1.2\"\n");
        body(selection, None, &mut out)?;
        Ok(out)
    }
    
    fn identity(text: &str) -> String {
        format!(
            "{}{}",
            vocab::BASE,
            purrdf_hash::hex::Lower(purrdf_hash::blake3::hash(text.as_bytes()).as_bytes())
        )
    }
    
    pub(crate) fn base(graph: &Graph) -> Result<String, Error> {
        Ok(identity(&normalized(&Selection::new(graph))?))
    }
    
    pub(crate) fn render(graph: &Graph, format: Format) -> Result<(String, String), Error> {
        let selection = Selection::new(graph);
        let mut normalized = normalized(&selection)?;
        let base = identity(&normalized);
        match format {
            Format::Turtle => {
                normalized.insert_str("VERSION \"1.2\"\n".len(), &format!("BASE <{base}>\n"));
                Ok((base, normalized))
            }
            Format::NTriples => {
                drop(normalized);
                let mut out = String::new();
                body(&selection, Some(&base), &mut out)?;
                Ok((base, out))
            }
        }
    }
