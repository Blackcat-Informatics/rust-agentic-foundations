    fn iri(name: &str) -> TermValue {
        TermValue::iri(format!("{}{name}", vocab::REPORT))
    }
    fn local(name: &str) -> TermValue {
        TermValue::iri(name)
    }
    fn value(graph: &mut Builder, id: &str, key: &str, atom: TermValue) -> Answer<()> {
        graph.insert(Triple {
            subject: local(id),
            predicate: iri(key),
            object: atom,
        })
    }
    fn text(graph: &mut Builder, id: &str, key: &str, value_: &str) -> Answer<()> {
        value(graph, id, key, TermValue::simple_literal(value_))
    }
    fn number(graph: &mut Builder, id: &str, key: &str, value_: usize) -> Answer<()> {
        value(

            key,
            TermValue::integer(i128::try_from(value_).expect("supported pointer width fits i128")),
        )
    }
    fn boolean(graph: &mut Builder, id: &str, key: &str, value_: bool) -> Answer<()> {

    fn class(graph: &mut Builder, id: &str, name: &str) -> Answer<()> {
        graph.insert(Triple {
            subject: local(id),
            predicate: TermValue::iri(vocab::TYPE),
            object: iri(name),
        })
    }
    fn child(
        graph: &mut Builder,
        parent: &str,
        key: &str,
        name: &str,
        ordinal: usize,
    ) -> Answer<String> {
        let id = format!("{parent}/{key}/{ordinal}");
        value(graph, parent, key, local(&id))?;
        class(graph, &id, name)?;
        number(graph, &id, "ordinal", ordinal)?;
        Ok(id)
    }
    fn strings(
        graph: &mut Builder,
        id: &str,
        key: &str,
        name: &str,
        field: &str,
        values: &[String],
    ) -> Answer<()> {
        for (ordinal, entry) in values.iter().enumerate() {
            let node = child(graph, id, key, name, ordinal)?;
            text(graph, &node, field, entry)?;
        }
        Ok(())
    }

        Ok(())

    pub(crate) fn graph(command: &str, results: &[&CommandResult]) -> Answer<Graph> {
        if let [CommandResult::Coverage { report, direction }] = results {
            return raf_workplan::report::coverage_graph(report, *direction);
        }
        let mut graph = Builder::new();
        class(&mut graph, "", "Report")?;
        number(&mut graph, "", "reportForm", 1)?;
        text(&mut graph, "", "command", command)?;
        for (ordinal, result) in results.iter().enumerate() {
            let id = format!("#result/{ordinal}");
            value(&mut graph, "", "result", local(&id))?;
            number(&mut graph, &id, "ordinal", ordinal)?;
            project(&mut graph, &id, result)?;
        }
        graph.finish()
    }
    
    fn project(g: &mut Builder, id: &str, result: &CommandResult) -> Answer<()> {
        match result {
            CommandResult::Finding(finding) => raf_workplan::report::finding_into(g, id, finding)?,
            CommandResult::Rewritten { path } | CommandResult::Completed { path } => {
                class(
                    g,
                    id,
                    if matches!(result, CommandResult::Rewritten { .. }) {
                        "Rewritten"
                    } else {
                        "Completed"
                    },
                )?;
                text(g, id, "path", path)?;
            }
            CommandResult::Preview {
                path,
                capability,
                text: body,
            } => {
                class(g, id, "Preview")?;
                text(g, id, "path", path)?;
                text(g, id, "capability", capability)?;
                text(g, id, "text", body)?;
            }
            CommandResult::State {
                state,
                capabilities,
            } => state_fields(g, id, state, capabilities)?,
            CommandResult::OpenChanges(changes) => open_changes_fields(g, id, changes)?,
            CommandResult::NewChange(change) => new_change_fields(g, id, change)?,
            CommandResult::Impact(impact) => impact_fields(g, id, impact)?,
            CommandResult::Obligation(obligation) => {
                class(g, id, "Obligation")?;
                obligation_fields(g, id, obligation)?;
            }
            CommandResult::Refused { failures, message } => {
                class(g, id, "Refused")?;
                number(g, id, "failures", *failures)?;
                text(g, id, "message", message)?;
            }
            CommandResult::Refusal { failure, repeated } => refusal_fields(g, id, failure, *repeated)?,
            CommandResult::Binding { evidence, message } => binding_fields(g, id, evidence, message)?,
            CommandResult::EvidenceRecorded(recorded) => recorded_fields(g, id, recorded)?,
            CommandResult::EvidenceClaim { evidence, claim } => claim_fields(g, id, evidence, claim)?,
            CommandResult::EvidenceSupport {
                capability,
                support,
            } => support_fields(g, id, capability, support)?,
            CommandResult::AssessmentContext(context) => context_fields(g, id, context)?,
            CommandResult::AssessmentRecorded(assessed) => assessed_fields(g, id, assessed)?,
            CommandResult::Assessment(state) => assessment_fields(g, id, state)?,
            CommandResult::History(history) => history_fields(g, id, history)?,
            CommandResult::Claims(report) => claims_fields(g, id, report)?,
            CommandResult::Lines(report) => lines_fields(g, id, report)?,
            CommandResult::DebtWritten(written) => {
                class(g, id, "DebtWritten")?;
                text(g, id, "path", written.path())?;
                text(g, id, "change", written.change().name())?;
                number(g, id, "findings", written.findings())?;
                number(g, id, "lines", written.lines())?;
            }
            CommandResult::Coverage { .. } => {
                return Err(Error::Value(
                    "item coverage must be the sole command result".into(),
                ));
            }
        }
        Ok(())
    }
    fn obligation_fields(
        g: &mut Builder,
        id: &str,
        o: &raf_workplan::obligation::Obligation,

        for (key, v) in [

            ("subject", o.subject.to_string()),
            ("section", o.section.to_string()),
            ("path", o.location.path.clone()),
            ("evidence", o.evidence.to_string()),
        ] {
            text(g, id, key, &v)?;

            number(g, id, "line", line)?;
        }
        Ok(())

    fn recorded_fields(g: &mut Builder, id: &str, recorded: &evidence::Recorded) -> Answer<()> {
        class(g, id, "EvidenceRecorded")?;
        text(g, id, "path", recorded.path.as_str())?;
        let r = &recorded.record;
        for (key, v) in [
            ("id", r.header.id.as_str()),
            ("verdict", r.verdict.name()),
            ("subject", &r.subject.text()),
            ("by", r.by.name()),
        ] {
            text(g, id, key, v)?;
        }
        optional(g, id, "command", r.command.as_deref())?;
        optional(g, id, "revision", r.revision.as_deref())?;
        strings(g, id, "for", "CapabilityReference", "id", &r.covers)?;
        measures(g, id, "value", &r.values)?;
        measures(g, id, "limit", &r.limits)?;
        inputs(g, id, &r.inputs)?;
        for (i, failure) in recorded.failed.iter().enumerate() {
            let node = child(g, id, "failed", "EvidenceFailure", i)?;
            let name = match failure {
                evidence::Failed::Exited(code) => {
                    if let Some(code) = code {
                        value(g, &node, "exitCode", TermValue::integer(*code))?;
                    }
                    "exited"
                }
                evidence::Failed::TimedOut => "timed-out",
                evidence::Failed::OverLimit(names) => {
                    strings(g, &node, "measure", "MeasureReference", "name", names)?;
                    "over-limit"
                }
                evidence::Failed::Binding(outcome) => {
                    text(g, &node, "outcome", outcome.name())?;
                    optional(g, &node, "message", outcome.reason().as_deref())?;
                    "binding"
                }
                _ => return Err(Error::Value("unknown evidence failure".into())),
            };
            text(g, &node, "reason", name)?;
        }
        text(g, id, "message", &recorded.to_string())?;
        Ok(())
    }
    fn context_fields(g: &mut Builder, id: &str, context: &assessment::AssessContext) -> Answer<()> {
        class(g, id, "AssessmentContext")?;
        for (key, v) in [
            ("subject", context.subject.as_str()),
            ("record", context.record.as_str()),
            ("recordPath", context.record_path.as_str()),
            ("document", context.document.as_str()),
            ("section", context.section.as_str()),
        ] {
            text(g, id, key, v)?;
        }
        for (i, symbol) in context.symbols.iter().enumerate() {
            let node = child(g, id, "symbol", "ContextSymbol", i)?;
            text(g, &node, "path", &symbol.path)?;
            boolean(g, &node, "named", symbol.named)?;
            optional(g, &node, "unresolved", symbol.unresolved.as_deref())?;
            for (j, definition) in symbol.definitions.iter().enumerate() {
                let def = child(g, &node, "definition", "Definition", j)?;
                text(g, &def, "file", &definition.file)?;
                text(g, &def, "text", &definition.text)?;
            }
        }
        for (i, evidence) in context.evidence.iter().enumerate() {
            let node = child(g, id, "evidence", "ContextEvidence", i)?;
            text(g, &node, "id", &evidence.id)?;
            text(g, &node, "path", &evidence.path)?;
            optional(
                g,
                &node,
                "verdict",
                evidence.verdict.map(raf_okf::EvidenceVerdict::name),
            )?;
            measures(g, &node, "value", &evidence.values)?;
            measures(g, &node, "limit", &evidence.limits)?;
        }
        for (i, input) in context.inputs.iter().enumerate() {
            let node = child(g, id, "input", "CurrentInput", i)?;
            text(g, &node, "kind", input.kind.name())?;
            text(g, &node, "name", &input.name)?;
            match &input.now {
                assessment::InputNow::Digest(digest) => text(g, &node, "digest", &digest.to_string())?,
                assessment::InputNow::Unresolved(reason) => text(g, &node, "unresolved", reason)?,
                _ => return Err(Error::Value("unknown current input state".into())),
            }
        }
        Ok(())
    }
    fn history_fields(g: &mut Builder, id: &str, history: &history::History) -> Answer<()> {
        class(g, id, "History")?;
        text(g, id, "id", history.id())?;
        text(g, id, "path", history.path())?;
        for (i, entry) in history.entries().iter().enumerate() {
            let node = child(g, id, "entry", "HistoryEntry", i)?;
            text(g, &node, "commit", entry.commit().as_str())?;
            text(g, &node, "date", &entry.date().to_string())?;
            text(g, &node, "subject", entry.subject())?;
            strings(g, &node, "path", "FileReference", "path", entry.paths())?;
            for (j, source) in entry.sources().iter().enumerate() {
                let source_node = child(g, &node, "source", "HistorySource", j)?;
                match source {
                    history::Source::Document => text(g, &source_node, "kind", "document")?,
                    history::Source::Symbol(binding) => {
                        text(g, &source_node, "kind", "symbol")?;
                        text(g, &source_node, "binding", binding)?;
                    }
                    _ => return Err(Error::Value("unknown history source".into())),
                }
            }
        }
        for (i, removed) in history.removed().iter().enumerate() {
            let node = child(g, id, "removed", "RemovedBinding", i)?;
            text(g, &node, "binding", removed.binding())?;
            text(g, &node, "identifier", removed.identifier())?;
            strings(
                g,
                &node,
                "directory",
                "DirectoryReference",
                "path",
                removed.dirs(),
            )?;
            match removed.removal() {
                history::Removal::Found {
                    commit,
                    date,
                    subject,
                    last_resolved,
                } => {
                    text(g, &node, "removal", "found")?;
                    text(g, &node, "commit", commit.as_str())?;
                    text(g, &node, "date", &date.to_string())?;
                    text(g, &node, "subject", subject)?;
                    optional(
                        g,
                        &node,
                        "lastResolved",
                        last_resolved
                            .as_ref()
                            .map(raf_workplan::exec::Commit::as_str),
                    )?;
                }
                history::Removal::NoCommit => text(g, &node, "removal", "no-commit")?,
                history::Removal::NoCrate => text(g, &node, "removal", "no-crate")?,
                _ => return Err(Error::Value("unknown binding removal".into())),
            }
        }
        Ok(())
    }
    fn lines_fields(
        g: &mut Builder,
        id: &str,
        report: &raf_workplan::coverage::lines::LinesReport,
    ) -> Answer<()> {
        class(g, id, "LinesReport")?;
        for (i, row) in report.crates().iter().enumerate() {
            let n = child(g, id, "package", "CrateLines", i)?;
            text(g, &n, "package", row.package())?;
            for (key, v) in [
                ("counted", row.counted()),
                ("covered", row.covered()),
                ("excepted", row.excepted()),
                ("orphan", row.orphan()),
                ("shippingCounted", row.shipping_counted()),
                ("testCounted", row.test_counted()),
                ("shippingOrphan", row.shipping_orphan()),
                ("testOrphan", row.test_orphan()),
                ("debt", row.debt()),
            ] {
                number(g, &n, key, v)?;
            }
            text(g, &n, "ceiling", &row.ceiling_text())?;
            for (j, ceiling) in row.ceilings().iter().enumerate() {
                let c = child(g, &n, "ceilingRow", "Ceiling", j)?;
                text(g, &c, "of", ceiling.of.name())?;
                match &ceiling.limit {
                    raf_okf::CeilingLimit::Lines(v) => text(g, &c, "lines", v.text())?,
                    raf_okf::CeilingLimit::Percent(v) => text(g, &c, "percent", v.text())?,
                }
            }
        }
        for (i, row) in report.modules().iter().enumerate() {
            let n = child(g, id, "module", "ModuleLines", i)?;
            text(g, &n, "package", row.package())?;
            text(g, &n, "module", row.module())?;
            number(g, &n, "counted", row.counted())?;
            number(g, &n, "orphan", row.orphan())?;
        }
        for (i, row) in report.orphans().iter().enumerate() {
            let n = child(g, id, "orphan", "UnitLines", i)?;
            unit_fields(g, &n, row)?;
        }
        for (i, row) in report.excepted().iter().enumerate() {
            let n = child(g, id, "excepted", "ExceptedLines", i)?;
            unit_fields(g, &n, row.lines())?;
            text(g, &n, "exception", row.exception())?;
            text(g, &n, "reason", row.reason())?;
        }
        for (i, error) in report.errors().iter().enumerate() {
            let n = child(g, id, "error", "LinesError", i)?;
            text(g, &n, "message", error)?;
        }
        Ok(())
    }

                &node,
                "programKind",

    }

    //! Closed-profile views over the authoritative native RDF dataset.
    
    use crate::{Error, Format, TermValue, Triple, read, term, write};
    use purrdf_core::{
        DatasetMut, GraphMatch, MutableDataset, QuadIds, QuadValues, RdfDataset, RdfDatasetBuilder,
        TermId, TermRef,

    use std::{collections::BTreeSet, sync::Arc};
    
    // Native storage requires absolute IRIs. Only this exact private owner and its
    // fragments are mapped from local API input; external aliases are refused.
    pub(crate) const CONSTRUCTION_OWNER: &str = "urn:raf:report:construction";
    
    /// An immutable report backed by the original native RDF dataset.
    #[derive(Clone, Debug)]
    pub struct Graph {
        pub(crate) dataset: Arc<RdfDataset>,
        pub(crate) owner: String,
    }
    
    /// Native mutable construction with profile validation and duplicate refusal.
    pub struct Builder {
        dataset: MutableDataset,
        owner: String,
    }
    
    /// A borrowed native term and its report-local IRI presentation.
    #[derive(Clone, Copy)]
    pub struct Value<'a> {
        graph: &'a Graph,
        id: TermId,
    }
    
    /// A subject query against native storage; it owns no assertion collection.
    #[derive(Clone, Copy)]
    pub struct Node<'a> {
        graph: &'a Graph,
        id: TermId,
    }
    
    impl Default for Builder {
        fn default() -> Self {
            Self::new()
        }
    }
    
    impl Builder {
        /// Start an empty native dataset.
        ///
        /// # Panics
        /// If native validation of an empty dataset violates its invariant.
        pub fn new() -> Self {
            let base = RdfDatasetBuilder::new()
                .freeze()
                .expect("empty native RDF dataset");
            Self {
                dataset: MutableDataset::new(base),
                owner: CONSTRUCTION_OWNER.into(),
            }
        }
    
        /// Edit a native base without copying its assertions into another graph.
        pub fn from_graph(graph: &Graph) -> Self {
            Self {
                dataset: MutableDataset::new(Arc::clone(&graph.dataset)),
                owner: graph.owner.clone(),
            }
        }
    
        /// Insert one profile assertion through native set insertion.
        ///
        /// # Errors
        /// Refuses invalid roles, the reserved construction namespace and duplicates.
        pub fn insert(&mut self, mut triple: Triple) -> Result<(), Error> {
            triple.validate()?;
            for value in [
                &mut triple.subject,
                &mut triple.predicate,
                &mut triple.object,
            ] {
                term::expand(value, &self.owner, true)?;
            }
            let quad = QuadValues::triple(triple.subject, triple.predicate, triple.object);
            if !self
                .dataset
                .insert(quad)
                .map_err(|e| Error::Value(e.to_string()))?
            {
                return Err(Error::Repeated);
            }
            Ok(())
        }
    
        /// Query exact assertion membership through the native mutable index.
        ///
        /// # Errors
        /// Refuses invalid roles or an external alias of the construction owner.
        pub fn contains(&self, mut triple: Triple) -> Result<bool, Error> {
            triple.validate()?;
            term::expand(&mut triple.subject, &self.owner, true)?;
            term::expand(&mut triple.predicate, &self.owner, true)?;
            term::expand(&mut triple.object, &self.owner, true)?;
            Ok(self.dataset.contains(&QuadValues::triple(
                triple.subject,
                triple.predicate,
                triple.object,
            )))
        }
    
        /// Remove one exact assertion through native membership semantics.
        ///
        /// # Errors
        /// Refuses invalid profile roles or external owner aliases.
        pub fn remove(&mut self, mut triple: Triple) -> Result<bool, Error> {
            triple.validate()?;
            term::expand(&mut triple.subject, &self.owner, true)?;
            term::expand(&mut triple.predicate, &self.owner, true)?;
            term::expand(&mut triple.object, &self.owner, true)?;
            Ok(self.dataset.remove(&QuadValues::triple(

                triple.predicate,
                triple.object,

            Ok(graph)
        }

    impl Graph {

            let subjects: BTreeSet<_> = self.rows().map(|row| row.s).collect();
            subjects.into_iter().map(|id| {
                let TermRef::Iri(iri) = self.dataset.resolve(id) else {
                    unreachable!("validated subject")

            })
        }

            Value { graph: self, id }
        }
    
        pub(crate) fn validate(&self) -> Result<(), Error> {
            if self.dataset.named_graphs().next().is_some() {
                return Err(Error::Value(
                    "named graphs are outside the report profile".into(),
                ));
            }
            if self.get("").is_none() {
                return Err(Error::Value("the report root is absent".into()));
            }
            let subjects: BTreeSet<_> = self.rows().map(|row| row.s).collect();
            // A native dataset can intentionally retain equal assertions in distinct
            // physical tables. This profile refuses that flat overlap. IDs are
            // compared only inside this one dictionary, never across datasets.
            let mut assertions = purrdf_core::FastSet::default();
            for row in self.rows() {
                if !assertions.insert(row) {
                    return Err(Error::Repeated);
                }
                let subject = self
                    .value(row.s)
                    .iri()
                    .ok_or_else(|| Error::Value("subject must be an IRI".into()))?;
                let TermRef::Iri(predicate) = self.dataset.resolve(row.p) else {
                    return Err(Error::Value("predicate must be an IRI".into()));
                };
                let object = self.value(row.o).to_owned();
                term::validate(subject, predicate, &object)?;
                let check = |value: Value<'_>| -> Result<(), Error> {
                    if value.iri().is_some_and(term::local) && !subjects.contains(&value.id) {
                        return Err(Error::Value(format!(
                            "dangling owned node {}",
                            value.iri().expect("IRI")
                        )));
                    }
                    Ok(())
                };
                let value = self.value(row.o);
                if let Some((s, _, o)) = value.quoted() {
                    check(s)?;
                    check(o)?;
                } else {
                    check(value)?;
                }
            }
            Ok(())
        }
    }

    fn checked_base(text: &str) -> Result<&str, Error> {
        if text
            .strip_prefix(vocab::BASE)
            .is_none_or(|hex| purrdf_hash::hex::decode_32_canonical(hex).is_none())
        {
            return Err(Error::Value(
                "snapshot BASE requires its namespace and 64 lowercase hexadecimal digits".into(),
            ));
        }
        Ok(text)
    }
    
    fn root_base(dataset: &RdfDataset) -> Result<String, Error> {
        let subjects = dataset
            .quads()
            .map(|row| row.s)
            .chain(dataset.reifiers().map(|(subject, _)| subject))
            .chain(dataset.annotations().map(|(subject, _, _)| subject));
        let mut root = None;
        for subject in subjects {
            if let TermRef::Iri(iri) = dataset.resolve(subject)
                && checked_base(iri).is_ok()
            {
                if root.is_some_and(|before| before != iri) {
                    return Err(Error::Value("more than one snapshot root".into()));
                }
                root = Some(iri);
            }
        }
        root.map(str::to_owned)
            .ok_or_else(|| Error::Value("absolute snapshot root is absent".into()))
    }
    
    pub(crate) fn parse(text: &str, format: Format) -> Result<Graph, Error> {
        let media = match format {
            Format::Turtle => "text/turtle",
            Format::NTriples => "application/n-triples",
        };
        let parsed = parse_dataset_with(text.as_bytes(), media, None, &ParseOptions::default())
            .map_err(|error| {
                let (line, column) = error.location.as_deref().map_or((1, 1), |location| {
                    (
                        location
                            .line
                            .and_then(|line| usize::try_from(line).ok())
                            .unwrap_or(1),
                        location.column.map_or(1, |column| column as usize),
                    )
                });
                Error::Syntax {
                    line,
                    column,
                    reason: error.to_string(),
                }
            })?;
        let base = match format {
            Format::Turtle => {
                let base = parsed
                    .document_base
                    .as_ref()
                    .filter(|base| matches!(base.origin(), purrdf_iri::BaseOrigin::Directive { .. }))
                    .ok_or_else(|| Error::Value("snapshot BASE directive is absent".into()))?;
                checked_base(base.iri().as_str())?.to_owned()
            }
            Format::NTriples => root_base(&parsed.dataset)?,
        };
        if parsed.dataset.named_graphs().next().is_some() {
            return Err(Error::Value(
                "named graphs are outside the report profile".into(),
            ));
        }
        let graph = Graph {
            dataset: parsed.dataset,
            owner: base.clone(),
        };
        graph.validate()?;
        let (computed, canonical) = write::render(&graph, format)?;
        if computed != base {
            return Err(Error::Digest {
                declared: base,
                computed,
            });
        }
        if canonical != text {
            let offset = purrdf_deflate::common_prefix_len(canonical.as_bytes(), text.as_bytes());
            let position = purrdf_iri::LineIndex::new(text)
                .try_locate(text, offset)
                .map_err(|e| Error::Value(e.to_string()))?;
            let line = usize::try_from(position.line)
                .map_err(|_| Error::Value("line exceeds usize".into()))?;
            return Err(Error::NotCanonical { line });
        }
        Ok(graph)
    }

    /// A report assertion with native RDF terms. Insertion validates its closed roles.
    #[derive(Clone, Debug, PartialEq, Eq)]
    pub struct Triple {
        /// Snapshot-owned root or fragment IRI.
        pub subject: TermValue,
        /// Absolute vocabulary IRI.
        pub predicate: TermValue,
        /// Atomic value or depth-one triple term.
        pub object: TermValue,
    }
    
    pub(crate) fn local(text: &str) -> bool {
        text.is_empty() || text.starts_with('#')
    }
    
    pub(crate) fn normalize(value: &mut TermValue, base: &str) {
        let atom = |value: &mut TermValue| {
            if let TermValue::Iri(iri) = value {
                if iri == base {
                    iri.clear();
                } else if let Some(fragment) =
                    iri.strip_prefix(base).filter(|tail| tail.starts_with('#'))
                {
                    *iri = fragment.to_owned();
                }
            }
        };
        if let TermValue::Triple { s, o, .. } = value {
            atom(s);
            atom(o);
        } else {
            atom(value);
        }
    }
    
    pub(crate) fn expand(value: &mut TermValue, owner: &str, constructing: bool) -> Result<(), Error> {
        let atom = |value: &mut TermValue| -> Result<(), Error> {
            if let TermValue::Iri(iri) = value {
                if local(iri) {
                    *iri = format!("{owner}{iri}");
                } else if constructing
                    && (iri == owner
                        || iri
                            .strip_prefix(owner)
                            .is_some_and(|tail| tail.starts_with('#')))
                {
                    return Err(Error::Value(
                        "external IRI aliases the reserved construction owner".into(),
                    ));
                }
            }
            if let TermValue::Literal { datatype, .. } = value
                && constructing
                && (datatype == owner
                    || datatype
                        .strip_prefix(owner)
                        .is_some_and(|tail| tail.starts_with('#')))
            {
                return Err(Error::Value(
                    "datatype aliases the reserved construction owner".into(),
                ));
            }
            Ok(())
        };
        if let TermValue::Triple { s, p, o } = value {
            atom(s)?;
            atom(p)?;
            atom(o)?;
        } else {
            atom(value)?;
        }
        Ok(())
    }
    
    pub(crate) fn iri(text: &str, relative: bool) -> Result<(), Error> {
        if relative && local(text) {
            if text.is_empty()
                || (text.len() > 1
                    && text.as_bytes()[1..]
                        .iter()
                        .all(|b| b.is_ascii_alphanumeric() || b"-._~!$&'()*+,;=:@/?%".contains(b)))
            {
                // The native parser admits absolute IRIs. Validate the owned
                // reference in its actual absolute snapshot namespace.
                purrdf_iri::parse(&format!("{}{text}", vocab::BASE))
                    .map_err(|_| Error::Value(format!("owned IRI {text}")))?;
                return Ok(());
            }
        } else if purrdf_iri::parse(text).is_ok_and(|iri| iri.scheme().is_some()) {
            return Ok(());
        }
        Err(Error::Value(format!(
            "{} IRI {text}",
            if relative {
                "absolute or owned"
            } else {
                "absolute"
            }
        )))
    }
    
    fn atom(term: &TermValue) -> Result<(), Error> {
        match term {
            TermValue::Iri(text) => iri(text, true),
            TermValue::Literal {
                datatype,
                language,
                direction,
                ..
            } => {
                iri(datatype, false)?;
                if language.is_some()
                    || direction.is_some()
                    || [
                        purrdf_iri::vocab::rdf::LANG_STRING,
                        purrdf_iri::vocab::rdf::DIR_LANG_STRING,
                    ]
                    .contains(&datatype.as_str())
                {
                    return Err(Error::Value(
                        "language metadata is outside the report profile".into(),
                    ));
                }
                Ok(())
            }
            TermValue::Blank { .. } => Err(Error::Value(
                "blank nodes are outside the report profile".into(),
            )),
            TermValue::Triple { .. } => Err(Error::Value(
                "nested triple terms are outside the report profile".into(),
            )),
        }
    }
    
    impl Triple {
        pub(crate) fn validate(&self) -> Result<(), Error> {
            let subject = self
                .subject
                .as_iri()
                .ok_or_else(|| Error::Value("report subject must be an IRI".into()))?;
            let predicate = self
                .predicate
                .as_iri()
                .ok_or_else(|| Error::Value("predicate must be an IRI".into()))?;
            validate(subject, predicate, &self.object)
        }
    }
    
    pub(crate) fn validate(subject: &str, predicate: &str, object: &TermValue) -> Result<(), Error> {
        iri(subject, true)?;
        if !local(subject) {
            return Err(Error::Value("report subject must be snapshot-owned".into()));
        }
        iri(predicate, false)?;
        if let TermValue::Triple { s, p, o } = object {

                s.as_iri()

            )?;
            iri(
                p.as_iri()
                    .ok_or_else(|| Error::Value("quoted predicate must be an IRI".into()))?,
                false,
            )?;
            atom(o)
        } else {
            atom(object)
        }
    }

    struct Selection<'a> {
        graph: &'a Graph,
        rows: Vec<purrdf_core::QuadIds>,
        keys: std::collections::BTreeMap<purrdf_core::TermId, TermValue>,
    }
    impl<'a> Selection<'a> {
        fn new(graph: &'a Graph) -> Self {
            let rows: Vec<_> = graph.rows().collect();
            let mut keys = std::collections::BTreeMap::new();
            for row in &rows {
                keys.entry(row.o)
                    .or_insert_with(|| graph.value(row.o).to_owned());
            }
            let mut selection = Self { graph, rows, keys };
            let iri = |id| match graph.dataset.resolve(id) {
                purrdf_core::TermRef::Iri(iri) => iri,
                _ => unreachable!("validated IRI role"),
            };
            selection.rows.sort_unstable_by(|a, b| {
                (graph.local(iri(a.s)), iri(a.p), &selection.keys[&a.o]).cmp(&(
                    graph.local(iri(b.s)),
                    iri(b.p),
                    &selection.keys[&b.o],
                ))
            });
            selection
        }
        fn iri(&self, id: purrdf_core::TermId) -> &str {
            match self.graph.dataset.resolve(id) {
                purrdf_core::TermRef::Iri(iri) => iri,
                _ => unreachable!("validated IRI role"),
            }
        }
    }
    
    fn body(selection: &Selection<'_>, base: Option<&str>, out: &mut String) -> Result<(), Error> {
        let mut resident = Resident;
        let mut memory = Memory::new(&mut resident);
        for triple in &selection.rows {
            for (iri, owned) in [
                (selection.graph.local(selection.iri(triple.s)), true),
                (selection.iri(triple.p), false),
            ] {
                if let Some(base) = base
                    && owned
                    && term::local(iri)
                {
                    out.push('<');
                    purrdf_lex::iri_escape::push_escaped(base, out);
                    purrdf_lex::iri_escape::push_escaped(iri, out);
                    out.push('>');
                } else {
                    purrdf_lex::term_syntax::write_iri(iri, out);
                }
                out.push(' ');
            }
            let value = &selection.keys[&triple.o];
            if let Some(base) = base {
                value
                    .try_write_nested_with_memory(
                        out,
                        ["<<( ", " ", " )>>"],
                        |out, leaf, memory| {
                            if let TermValue::Iri(iri) = leaf
                                && term::local(iri)
                            {
                                out.push('<');
                                purrdf_lex::iri_escape::push_escaped(base, out);
                                purrdf_lex::iri_escape::push_escaped(iri, out);
                                out.push('>');
                                Ok(())
                            } else {
                                purrdf_core::write_term_value_with_memory(leaf, out, memory)
                            }
                        },
                        |out, text, _| {
                            out.push_str(text);
                            Ok(())
                        },
                        &mut memory,
                    )
                    .map_err(|error| Error::Value(format!("native term output: {error}")))?;
            } else {
                purrdf_core::write_term_value_with_memory(value, out, &mut memory)
                    .map_err(|error| Error::Value(format!("native term output: {error}")))?;
            }
            out.push_str(" .\n");
        }
        Ok(())
    }
    
    fn normalized(selection: &Selection<'_>) -> Result<String, Error> {
        let mut out = String::from("VERSION \"1.2\"\n");
        body(selection, None, &mut out)?;
        Ok(out)
    }
    
    fn identity(text: &str) -> String {
        format!(
            "{}{}",
            vocab::BASE,
            purrdf_hash::hex::Lower(purrdf_hash::blake3::hash(text.as_bytes()).as_bytes())
        )
    }
    
    pub(crate) fn base(graph: &Graph) -> Result<String, Error> {
        Ok(identity(&normalized(&Selection::new(graph))?))
    }
    
    pub(crate) fn render(graph: &Graph, format: Format) -> Result<(String, String), Error> {
        let selection = Selection::new(graph);
        let mut normalized = normalized(&selection)?;
        let base = identity(&normalized);
        match format {
            Format::Turtle => {
                normalized.insert_str("VERSION \"1.2\"\n".len(), &format!("BASE <{base}>\n"));
                Ok((base, normalized))
            }
            Format::NTriples => {
                drop(normalized);
                let mut out = String::new();
                body(&selection, Some(&base), &mut out)?;
                Ok((base, out))
            }
        }
    }

    /// Append a typed finding at an owned subject shared by command projections.
    /// The caller supplies membership and ordinal links; the finding class and all
    /// location fields are written here. Discard the graph when an error occurs.
    ///
    /// # Errors
    /// Refuses unknown finding kinds, zero lines, unsupported locations and invalid
    /// or duplicate graph terms.
    pub fn finding_into(graph: &mut Builder, id: &str, finding: &Finding) -> Result<(), Error> {
        class(graph, id, "Finding")?;
        if !Finding::KINDS.contains(&finding.kind) {
            return Err(Error::Value(format!(
                "unknown finding kind {}",
                finding.kind
            )));
        }
        for (key, text) in [
            ("kind", finding.kind),
            ("file", finding.file.as_str()),
            ("message", finding.message.as_str()),
            ("fix", finding.fix.as_str()),
        ] {
            add(graph, id, key, TermValue::simple_literal(text))?;
        }
        match &finding.location {
            Location::Line(0) => return Err(Error::Value("finding line must be positive".into())),
            Location::Line(value) => add(graph, id, "line", count(*value))?,
            Location::Section(value) => add(graph, id, "section", TermValue::simple_literal(value))?,
            Location::Key(value) => add(graph, id, "key", TermValue::simple_literal(value))?,
            Location::Document => {}
            _ => return Err(Error::Value("unsupported finding location".into())),
        }
        Ok(())
    }

    fn gap(graph: &mut Builder, id: &str, value: &Gap, in_scope: bool) -> Result<(), Error> {
        for (key, text) in [
            ("kind", value.kind().name()),
            ("package", value.package()),
            ("path", value.path()),
            ("item", value.item()),
            ("file", value.file()),
        ] {
            add(graph, id, key, TermValue::simple_literal(text))?;
        }
        add(graph, id, "line", count(value.line()))?;
        add(graph, id, "inScope", TermValue::boolean(in_scope))?;
        let proof = format!("{id}/source");
        class(graph, &proof, "SourceLocation")?;
        add(graph, id, "source", local(&proof))?;
        graph.insert(Triple {
            subject: local(&proof),
            predicate: TermValue::iri(vocab::REIFIES),
            object: TermValue::Triple {
                s: TermBox::new(local(id)),
                p: TermBox::new(named("item")),
                o: TermBox::new(TermValue::simple_literal(value.item())),
            },
        })?;
        add(
            graph,
            &proof,
            "file",
            TermValue::simple_literal(value.file()),
        )?;
        add(graph, &proof, "line", count(value.line()))?;
        Ok(())
    }
    
    /// Project item/test coverage, including package counts, exclusions and failures.
    ///
    /// # Errors
    /// Refuses an inconsistent package reference or invalid term/graph.
    pub fn coverage_graph(report: &CoverageReport, direction: GapKind) -> Result<Graph, Error> {
        let mut graph = root("coverage")?;
        add(
            &mut graph,
            "",
            "direction",
            TermValue::simple_literal(if direction.is_code() { "code" } else { "tests" }),
        )?;
        let mut ordinal = 0;
        for package in report.packages() {
            let id = member(&mut graph, ordinal, "CoveragePackage")?;
            ordinal += 1;
            add(
                &mut graph,
                &id,
                "package",
                TermValue::simple_literal(package.name()),
            )?;
            add(
                &mut graph,
                &id,
                "inScope",
                TermValue::boolean(package.in_scope()),
            )?;
            for (key, value) in [
                ("items", package.items()),
                ("coveredItems", package.covered_items()),
                ("tests", package.tests()),
                ("coveredTests", package.covered_tests()),
            ] {
                add(&mut graph, &id, key, count(value))?;
            }
        }
        let packages: BTreeMap<_, _> = report
            .packages()
            .iter()
            .map(|p| (p.name(), p.in_scope()))
            .collect();
        let in_scope = |gap: &Gap| -> Result<bool, Error> {
            packages
                .get(gap.package())
                .copied()
                .ok_or_else(|| Error::Value(format!("coverage package absent: {}", gap.package())))
        };
        for value in report.gaps(direction) {
            let id = member(&mut graph, ordinal, "CoverageGap")?;
            ordinal += 1;
            gap(&mut graph, &id, value, in_scope(value)?)?;
        }
        for excluded in report
            .excluded()
            .iter()
            .filter(|e| e.gap().kind().is_code() == direction.is_code())
        {
            let id = member(&mut graph, ordinal, "CoverageExcluded")?;
            ordinal += 1;
            gap(&mut graph, &id, excluded.gap(), in_scope(excluded.gap())?)?;
            add(
                &mut graph,
                &id,
                "exclusion",
                TermValue::simple_literal(excluded.exclusion()),
            )?;
            add(
                &mut graph,
                &id,
                "reason",
                TermValue::simple_literal(excluded.reason()),
            )?;
        }
        if direction.is_code() {
            for error in report.errors() {
                let id = member(&mut graph, ordinal, "CoverageError")?;
                ordinal += 1;
                let scope = packages.get(error.package()).copied().ok_or_else(|| {
                    Error::Value(format!("coverage package absent: {}", error.package()))
                })?;
                surface_error(&mut graph, &id, error, scope)?;
            }
        }
        graph.finish()
    }
    
    fn surface_error(
        graph: &mut Builder,
        id: &str,
        error: &SurfaceError,
        scope: bool,
    ) -> Result<(), Error> {
        add(graph, id, "inScope", TermValue::boolean(scope))?;
        for (key, text) in [
            ("package", error.package()),
            ("target", error.target()),
            ("root", error.root()),
            ("failure", error.failure().name()),
            ("message", error.message()),
        ] {
            add(graph, id, key, TermValue::simple_literal(text))?;
        }
        if let SurfaceFailure::CarryBudget { path } = error.failure() {
            add(graph, id, "failurePath", TermValue::simple_literal(path))?;
        }
        Ok(())
    }
