        ///     &MemoryTree::from([
        ///         ("Cargo.toml", "[package]\nname = \"demo\"\nversion = \"0.1.0\"\n"),
        ///         ("src/lib.rs", "pub fn parse() {}\npub fn parse_partial() {}\n"),
        ///     ]),
        ///     &["."],
        /// )?;
        /// let plan = Workplan::load(&MemoryTree::from([("README.md", "# Demo\n")]), &source);
        /// let report = plan.coverage_report();
        /// let gaps: Vec<&str> = report.code().iter().map(|gap| gap.path()).collect();
        /// assert_eq!(gaps, ["demo::parse", "demo::parse_partial"]);
        /// assert_eq!(report.code()[0].kind(), GapKind::PublicItem);
        /// assert!(!report.packages()[0].in_scope());
        /// # Ok::<(), Box<dyn std::error::Error>>(())
        /// ```

    //! Two-way coverage: the public items that no capability binds (code to
    //! documentation) and the tests that no capability binds (tests to claims).
    //!
    //! [`Workplan::coverage_report`] reads every package of the source workspace
    //! and returns a [`CoverageReport`]; `check` enforces the same computation for
    //! the crates the coverage record `work/coverage/default.md` lists in `scope`.
    //!
    //! # The public surface
    //!
    //! The public items of a package are the items of the public surface of each
    //! of its library and binary targets, as
    //! [`Workspace::exported_items`](raf_source::Workspace::exported_items) lists
    //! them: what another crate can name through the crate root, a `pub` item of a
    //! private module only through a `pub` re-export, the members of an exported
    //! type or trait under its path, and nothing that `#[doc(hidden)]` hides. An
    //! item exported under several paths is one item, the defining one, and is
    //! reported under the first of its exported paths. Binary targets are read by
    //! the same source visibility rules: explicitly public items count, while
    //! private and crate-visible items do not. An ordinary private `main` therefore
    //! creates no public-item gap, but a binary's public helper can create one.
    //!
    //! A public re-export that names no item of the target is a
    //! [`Reexport`]. One that names an item of another crate
    //! of the workspace is counted in that crate, and one of a crate outside the
    //! workspace (its written path starts with a name that is no crate of the
    //! workspace and that the declaring module does not resolve) is not counted.
    //! Any other, a type a macro generates or an enum variant the index cannot
    //! follow, is an entry of the code direction of its own kind,
    //! [`GapKind::Reexport`]: it is never skipped. It is covered when a `code`
    //! binding covers a module that its exported path, or the path it names from
    //! the declaring module, is below, since the re-export itself resolves to no
    //! item a binding could name. The declaring module is the innermost inline
    //! module whose braces hold the declaration, or else the module of its file,
    //! and the path is read from it as `rustc` reads a `use` path: `crate::` from
    //! the crate root, `self::` from the module, bare paths from the crate root in
    //! Rust 2015 or the module in later editions, and each
    //! `super::` one module up.
    //!
    //! A surface that cannot be listed is a [`SurfaceError`] of the report, never
    //! an empty surface: the item-bytes budget of the load passed by the listing
    //! ([`SurfaceFailure::Budget`]) or by deciding whether the public globs carry
    //! a re-export ([`SurfaceFailure::CarryBudget`], the refusal for several
    //! binders of one name), or a public name that is ambiguous, goes through a
    //! cycle of imports or reads a glob of nothing ([`SurfaceFailure::Unresolved`]).
    //! What the listing does not model is a stated limit of the surface read here
    //! (see the Constraints of `raf-source`'s exported items): a name that a local
    //! `pub(crate)` item and a carried re-export of another crate share through
    //! public globs is listed as the re-export, so that item is not counted.
    //!
    //! # Covered, excluded, gap
    //!
    //! A public item is covered when a `code` binding of a capability resolves to
    //! it or to an item it is below: binding a type covers its methods and
    //! associated items, binding a trait its members, and binding a module every
    //! item below it, as a module binding also stands for the module's whole text
    //! in [`Digest::symbol`](crate::digest::Digest::symbol). A binding is followed to the definition, so an item bound
    //! through a re-export is covered under every path it is exported by. A crate
    //! root covers nothing below it: a capability binds the modules and items it
    //! explains, and a whole crate is taken out of enforcement by leaving it out
    //! of `scope`. Only `code` bindings cover items; a documented item bound in
    //! `examples` is a documentation test, not an explanation.
    //!
    //! A test is a function marked `#[test]` that the index knows: in an
    //! integration test target, and in a `#[cfg(test)]` module of a library or a
    //! binary (a unit test), which counts like any other. A documentation test is
    //! not a test function of the index; it is bound in `examples`. A test is
    //! covered when a `tests` binding resolves to it.
    //!
    //! An item or test that is not covered is excluded when an exclusion of the
    //! coverage record matches it: an item path (`raf_okf::Body`) matches the item
    //! exported or defined at that path, and a prefix (`raf_okf::body::*`) every
    //! item or test whose exported or defining path is below the module. The first
    //! exclusion that matches, in the order of the record, gives the reason the
    //! report shows. Anything else is a [`Gap`].
    //!
    //! # What `check` enforces
    //!
    //! For each crate of `scope`, each gap is a finding: `coverage-item-unbound`
    //! and `coverage-test-unbound` at the file and line of the item,
    //! `coverage-reexport-unbound` at the declaration of an unresolved re-export,
    //! and
    //! `coverage-surface-unreadable` at the `scope` of the record for a surface
    //! that cannot be listed. An exclusion that matches no public item and no test
    //! of the workspace is `coverage-exclusion-unmatched`; an exclusion without a
    //! reason, or of a whole crate, is refused by the profile when the record
    //! loads. A repository with capability documents and no coverage record has
    //! one `coverage-record-missing` finding: the record states what is enforced,
    //! and `scope: []` enforces nothing, visibly. Crates outside `scope` are not
    //! enforced and their gaps are in the report. While a capability document
    //! exists and does not load, `check` reports no gap: what that document binds
    //! is not in the graph, so each item and test it binds would read as unbound,
    //! and its own findings already fail the check; the surfaces that cannot be
    //! listed and the exclusions that match nothing are still reported.
    //!
    //! `check` lists the surface of each library and binary target of a crate of
    //! `scope`, and of a crate that an exclusion names by its first path segment
    //! (to judge whether the exclusion matches), once per load; a record whose
    //! `scope` is empty and that has no exclusions lists none. The report lists
    //! every library and binary target of the workspace once.
    //!
    //! Every list is sorted by package and path, so a report is the same on every
    //! run over the same tree.

    /// What a gap or an exclusion is about.
    #[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord, Hash)]
    #[non_exhaustive]
    pub enum GapKind {
        /// An item of the public surface of a library or binary target.
        PublicItem,
        /// A public re-export of a library or binary target that names nothing
        /// the source index can follow, such as a type a macro generates or an
        /// enum variant, and that is not of a crate outside the workspace. It is
        /// in the code direction.
        Reexport,
        /// A function marked `#[test]`.
        Test,
    }
    
    impl GapKind {
        /// The name of the kind as the report writes it: `public item`,
        /// `unresolved re-export` or `test`.
        #[must_use]
        pub fn name(self) -> &'static str {
            match self {
                Self::PublicItem => "public item",
                Self::Reexport => "unresolved re-export",
                Self::Test => "test",
            }
        }
    
        /// Whether the kind is in the code direction (a public item or an
        /// unresolved re-export) rather than the tests direction.
        #[must_use]
        pub fn is_code(self) -> bool {
            !matches!(self, Self::Test)
        }
    }
    
    impl fmt::Display for GapKind {
        fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
            f.write_str(self.name())
        }
    }

    /// Why the public surface of a target could not be listed.
    #[derive(Debug, Clone, PartialEq, Eq)]
    #[non_exhaustive]
    pub enum SurfaceFailure {
        /// What the surface holds passes the item-bytes budget left after the
        /// load.
        Budget,
        /// Deciding whether the public globs carry the re-export at `path` passes
        /// the item-bytes budget, as for several binders of one carried name.
        CarryBudget {
            /// The exported path of the re-export being decided.
            path: String,
        },
        /// A public name is ambiguous, goes through a cycle of imports, or reads
        /// a glob whose path names nothing.
        Unresolved,
        /// Another failure of the listing; the message says which.
        Other,
    }
    
    impl SurfaceFailure {
        /// The stable name of the failure.
        #[must_use]
        pub fn name(&self) -> &'static str {
            match self {
                Self::Budget => "budget",
                Self::CarryBudget { .. } => "carry-budget",
                Self::Unresolved => "unresolved",
                Self::Other => "other",
            }
        }
    
        fn of(error: &ExportError) -> Self {
            match error {
                ExportError::Budget(_) => Self::Budget,
                ExportError::CarryBudget { path, .. } => Self::CarryBudget { path: path.clone() },
                ExportError::Resolve(_) => Self::Unresolved,
                _ => Self::Other,
            }
        }
    }

    /// The coverage of the workspace, or of one package of it: the gaps in each
    /// direction, what the exclusions take out, and the surfaces that could not
    /// be listed, each sorted by package and path.
    #[derive(Debug, Clone, Default, PartialEq, Eq)]
    #[non_exhaustive]
    pub struct CoverageReport {
        packages: Vec<PackageCoverage>,
        code: Vec<Gap>,
        tests: Vec<Gap>,
        excluded: Vec<Excluded>,
        errors: Vec<SurfaceError>,
    }
    
    impl CoverageReport {
        /// The packages, sorted by name.
        #[must_use]
        pub fn packages(&self) -> &[PackageCoverage] {
            &self.packages
        }
    
        /// The public items without a binding or an exclusion.
        #[must_use]
        pub fn code(&self) -> &[Gap] {
            &self.code
        }
    
        /// The tests without a binding or an exclusion.
        #[must_use]
        pub fn tests(&self) -> &[Gap] {
            &self.tests
        }
    
        /// The public items and tests an exclusion takes out.
        #[must_use]
        pub fn excluded(&self) -> &[Excluded] {
            &self.excluded
        }
    
        /// The surfaces that could not be listed.
        #[must_use]
        pub fn errors(&self) -> &[SurfaceError] {
            &self.errors
        }
    
        /// The gaps of one direction: [`CoverageReport::code`] or
        /// [`CoverageReport::tests`].
        #[must_use]
        pub fn gaps(&self, kind: GapKind) -> &[Gap] {
            if kind.is_code() {
                &self.code
            } else {
                &self.tests
            }
        }
    
        /// True when there is no gap in either direction and every surface was
        /// listed.
        #[must_use]
        pub fn is_complete(&self) -> bool {
            self.code.is_empty() && self.tests.is_empty() && self.errors.is_empty()
        }
    
        /// The report of one package, by its name; `None` for a name that is no
        /// package of the report.
        #[must_use]
        pub fn package(&self, name: &str) -> Option<CoverageReport> {
            let package = self.packages.iter().find(|p| p.name == name)?;
            Some(Self {
                packages: vec![package.clone()],
                code: self
                    .code
                    .iter()
                    .filter(|g| g.package == name)
                    .cloned()
                    .collect(),
                tests: self
                    .tests
                    .iter()
                    .filter(|g| g.package == name)
                    .cloned()
                    .collect(),
                excluded: self
                    .excluded
                    .iter()
                    .filter(|e| e.gap.package == name)
                    .cloned()
                    .collect(),
                errors: self
                    .errors
                    .iter()
                    .filter(|e| e.package == name)
                    .cloned()
                    .collect(),
            })
        }
    }

            f.write_str(&render::table(self, GapKind::PublicItem))?;

    impl Workplan<'_> {
        /// The coverage of every package of the source workspace, in both
        /// directions, judged against the bindings of the loaded capabilities and
        /// the exclusions of the coverage record (none without a record). Each
        /// package says whether the record's `scope` lists it. Nothing is
        /// written and no process is started.
        ///
        /// ```rust
        /// use raf_source::{MemoryTree, Workspace};
        /// use raf_workplan::Workplan;
        /// use raf_workplan::coverage::GapKind;
        ///
        /// let source = Workspace::load(
        ///     &MemoryTree::from([
        ///         ("Cargo.toml", "[package]\nname = \"demo\"\nversion = \"0.1.0\"\n"),
        ///         ("src/lib.rs", "pub fn parse() {}\npub fn parse_partial() {}\n"),
        ///     ]),
        ///     &["."],
        /// )?;
        /// let plan = Workplan::load(&MemoryTree::from([("README.md", "# Demo\n")]), &source);
        /// let report = plan.coverage_report();
        /// let gaps: Vec<&str> = report.code().iter().map(|gap| gap.path()).collect();
        /// assert_eq!(gaps, ["demo::parse", "demo::parse_partial"]);
        /// assert_eq!(report.code()[0].kind(), GapKind::PublicItem);
        /// assert!(!report.packages()[0].in_scope());
        /// # Ok::<(), Box<dyn std::error::Error>>(())
        /// ```
        #[must_use]
        pub fn coverage_report(&self) -> CoverageReport {
            let all: BTreeSet<&str> = self
                .source()
                .packages()
                .iter()
                .map(|package| package.name.as_str())
                .collect();
            compute(
                self.source(),
                self.graph(),
                self.coverage().map(Loaded::value),
                &all,
            )
            .report
        }
    }
    
    /// The findings `check` adds for coverage: see the module documentation.
    pub(crate) fn check(plan: &Workplan<'_>) -> Vec<Finding> {
        let Some(loaded) = plan.coverage() else {
            if plan.is_present("coverage:default") || !plan.has_capability_files() {
                return Vec::new();
            }
            return vec![Finding::new(
                kind::COVERAGE_RECORD_MISSING,
                RECORD_PATH,
                Location::Document,
                format!(
                    "the repository has capability documents and no coverage record {RECORD_PATH}, so it does not say which crates must be covered"
                ),
                "add the coverage record with the crates whose coverage is enforced in `scope` and an exclusion with a reason for each item or test that supports no documented guarantee; `scope: []` enforces none",
            )];
        };
        let record = loaded.value();
        let mut out = exceptions_unmatched(plan, loaded);
        if record.scope.is_empty() && record.exclusions.is_empty() {
            return out;
        }
        let scope: BTreeSet<&str> = plan
            .graph()
            .scope()
            .iter()
            .map(ScopeEdge::package)
            .collect();
        let mut needed = scope.clone();
        for exclusion in &record.exclusions {
            let first = first_segment(exclusion_path(exclusion));
            for target in plan.source().targets() {
                if first_segment(&target.crate_path) == first {
                    needed.insert(target.package.as_str());
                }
            }
        }
        let computed = compute(plan.source(), plan.graph(), Some(record), &needed);
        // A capability document that did not load binds nothing in the graph, so
        // every item and test it binds would read as a gap. Its own findings say
        // what is wrong with it, and the gaps are judged once it loads.
        let capability_unloaded = plan
            .unloaded()
            .iter()
            .any(|id| id.starts_with(DocType::Capability.id_prefix()));
        out.extend(findings(loaded, &scope, &computed, !capability_unloaded));
        out
    }
    
    /// A `coverage-exception-unmatched` finding for each exception of the record
    /// that matches nothing of the source index, judged under the complete policy
    /// and without counting a line; a record without exceptions has none.
    fn exceptions_unmatched(plan: &Workplan<'_>, loaded: &Loaded<CoverageRecord>) -> Vec<Finding> {
        let exceptions = &loaded.value().exceptions;
        if exceptions.is_empty() {
            return Vec::new();
        }
        lines::unmatched_exceptions(plan.source(), exceptions)
            .into_iter()
            .map(|position| {
                let target = &exceptions[position].target;
                let names = match target {
                    ExceptionTarget::Item(_) => "no item",
                    ExceptionTarget::Prefix(_) => "no module or item at or below it",
                    ExceptionTarget::File(_) => "no file",
                    ExceptionTarget::Directory(_) => "no file below it",
                };
                Finding::new(
                    kind::COVERAGE_EXCEPTION_UNMATCHED,
                    loaded.path(),
                    Location::Key("exceptions".to_string()),
                    format!(
                        "the exception `{}` names {names} of the source index, so it takes no line out",
                        target.written()
                    ),
                    "correct the path to the item, module, file or directory it means, or remove the exception; an exception of something that no longer exists excepts nothing",
                )
            })
            .collect()
    }
    
    /// The findings of the computed coverage: the gaps of the crates of `scope`
    /// when `gaps` holds, and always the surfaces that cannot be listed and the
    /// exclusions that match nothing.
    fn findings(
        loaded: &Loaded<CoverageRecord>,
        scope: &BTreeSet<&str>,
        computed: &Computed,
        gaps: bool,
    ) -> Vec<Finding> {
        let report = &computed.report;
        let mut out = Vec::new();
        for error in report
            .errors
            .iter()
            .filter(|e| scope.contains(e.package.as_str()))
        {
            out.push(Finding::new(
                kind::COVERAGE_SURFACE_UNREADABLE,
                loaded.path(),
                Location::Key("scope".to_string()),
                format!("`scope` names `{}`, and {error}", error.package),
                "make every public name of the crate resolve to one item (the error names the file and line), raise the item-bytes budget of the load, or remove the crate from `scope`; a crate whose surface cannot be read is not judged covered",
            ));
        }
        let enforced = format!("the coverage record {} enforces", loaded.path());
        for gap in report
            .code
            .iter()
            .chain(&report.tests)
            .filter(|g| gaps && scope.contains(g.package.as_str()))
        {
            out.push(gap_finding(gap, &enforced));
        }
        for (exclusion, matched) in loaded.value().exclusions.iter().zip(&computed.matched) {
            if !matched {
                out.push(Finding::new(
                    kind::COVERAGE_EXCLUSION_UNMATCHED,
                    loaded.path(),
                    Location::Key("exclusions".to_string()),
                    format!(
                        "the exclusion `{}` matches no public item and no test of the workspace",
                        written(exclusion)
                    ),
                    "correct the path to the item, test or module it means, or remove the exclusion; an exclusion of an item that no longer exists, or that is not public, takes nothing out",
                ));
            }
        }
        out
    }
    
    /// The finding of a gap of a package that `enforced` says why it is
    /// enforced, as the clause after `which`: `coverage-item-unbound`,
    /// `coverage-reexport-unbound` or `coverage-test-unbound` at the file and line
    /// of the gap.
    pub(crate) fn gap_finding(gap: &Gap, enforced: &str) -> Finding {
        let (kind, fix) = match gap.kind {
            GapKind::PublicItem => (
                kind::COVERAGE_ITEM_UNBOUND,
                "bind the item, or an item it is below, in the `bindings.code` of the capability that explains it, or exclude it in the coverage record with the reason it needs no explanation",
            ),
            GapKind::Reexport => (
                kind::COVERAGE_REEXPORT_UNBOUND,
                "bind a module the re-export is below in the `bindings.code` of the capability that explains it (the index cannot follow the re-export to an item, so the path itself cannot be bound), or exclude it in the coverage record with the reason it needs no explanation",
            ),
            GapKind::Test => (
                kind::COVERAGE_TEST_UNBOUND,
                "bind the test in the `bindings.tests` of the capability whose guarantee it supports, or exclude it in the coverage record with the reason it supports none",
            ),
        };
        let defined = match gap.kind {
            _ if gap.item == gap.path => String::new(),
            GapKind::Reexport => format!(
                " (naming `{}`, which the source index cannot follow to an item)",
                gap.item
            ),
            _ => format!(" (defined as `{}`)", gap.item),
        };
        Finding::new(
            kind,
            &gap.file,
            Location::Line(gap.line),
            format!(
                "the {} `{}`{defined} of `{}`, which {enforced}, is bound by no capability and excluded by no exclusion",
                gap.kind, gap.path, gap.package,
            ),
            fix,
        )
    }
    
    /// A computed report and, for each exclusion of the record in its order,
    /// whether it matched a public item or a test of the packages computed.
    pub(crate) struct Computed {
        pub(crate) report: CoverageReport,
        matched: Vec<bool>,
    }
    
    /// What covers: the items `code` bindings resolve to, their paths (below
    /// which every item is covered), and the tests `tests` bindings resolve to.
    pub(crate) struct Bound {
        code: BTreeSet<ItemId>,
        below: BTreeSet<String>,
        tests: BTreeSet<ItemId>,
    }
    
    impl Bound {
        pub(crate) fn of(graph: &Graph, source: &Workspace) -> Self {
            let mut bound = Self {
                code: BTreeSet::new(),
                below: BTreeSet::new(),
                tests: BTreeSet::new(),
            };
            for binding in graph.bindings() {
                for id in binding.items() {
                    match binding.kind() {
                        BindingKind::Code => {
                            bound.code.insert(*id);
                            if let Some(item) = source.item(*id) {
                                let root = source
                                    .target(item.target)
                                    .is_some_and(|target| *item.path == *target.crate_path);
                                if !root {
                                    bound.below.insert(item.path.to_string());
                                }
                            }
                        }
                        BindingKind::Test => {
                            bound.tests.insert(*id);
                        }
                        BindingKind::Example => {}
                    }
                }
            }
            bound
        }
    
        /// Whether a `code` binding names `id`, an item of the same source, or an
        /// item that one of them is below.
        pub(crate) fn covers(&self, source: &Workspace, id: ItemId) -> bool {
            let Some(item) = source.item(id) else {
                return false;
            };
            let same = source.same_source(item);
            let ids = if same.is_empty() {
                std::slice::from_ref(&item.id)
            } else {
                same
            };
            ids.iter().any(|id| {
                self.code.contains(id)
                    || source
                        .item(*id)
                        .is_some_and(|item| ancestors(&item.path).any(|p| self.below.contains(p)))
            })
        }
    }
    
    /// The first `code` binding of `graph`, in the order of the graph, that covers
    /// the item `id` as [`Bound::covers`] reads it: it names `id` or an item of the
    /// same source, or one of them is below the item it names. `None` exactly when
    /// `Bound::covers` is false.
    pub(crate) fn covering<'g>(
        graph: &'g Graph,
        source: &Workspace,
        id: ItemId,
    ) -> Option<&'g Binding> {
        let item = source.item(id)?;
        let same = source.same_source(item);
        let ids = if same.is_empty() {
            std::slice::from_ref(&item.id)
        } else {
            same
        };
        let paths: Vec<&str> = ids
            .iter()
            .filter_map(|id| source.item(*id).map(|item| &*item.path))
            .collect();
        graph.bindings().iter().find(|binding| {
            binding.kind() == BindingKind::Code
                && binding.items().iter().any(|bound| {
                    ids.contains(bound)
                        || below_bound(source, *bound)
                            .is_some_and(|at| paths.iter().any(|path| ancestors(path).any(|p| p == at)))
                })
        })
    }
    
    /// The first `code` binding of `graph` that covers the item path `path`, an
    /// item the source index need not hold, such as one at the merge base: it
    /// names an item with that path, or one the path is below.
    pub(crate) fn covering_path<'g>(
        graph: &'g Graph,
        source: &Workspace,
        path: &str,
    ) -> Option<&'g Binding> {
        graph.bindings().iter().find(|binding| {
            binding.kind() == BindingKind::Code
                && binding.items().iter().any(|bound| {
                    source.item(*bound).is_some_and(|item| *item.path == *path)
                        || below_bound(source, *bound)
                            .is_some_and(|at| ancestors(path).any(|p| p == at))
                })
        })
    }
    
    /// The path below which the bound item `id` covers every item, as
    /// [`Bound::of`] keeps it: its own path, unless it is a crate root, which
    /// covers nothing below it.
    fn below_bound(source: &Workspace, id: ItemId) -> Option<&str> {
        let item = source.item(id)?;
        let root = source
            .target(item.target)
            .is_some_and(|target| *item.path == *target.crate_path);
        (!root).then_some(&*item.path)
    }
    
    /// `path` and each path it is below, longest first: `a::b::c`, `a::b`, `a`.
    fn ancestors(path: &str) -> impl Iterator<Item = &str> {
        std::iter::successors(Some(path), |p| p.rsplit_once("::").map(|(head, _)| head))
    }
    
    fn exclusion_path(exclusion: &Exclusion) -> &str {
        match &exclusion.item {
            ExclusionItem::Item(path) | ExclusionItem::Prefix(path) => path,
        }
    }
    
    fn written(exclusion: &Exclusion) -> String {
        match &exclusion.item {
            ExclusionItem::Item(path) => path.clone(),
            ExclusionItem::Prefix(path) => format!("{path}::*"),
        }
    }
    
    fn first_segment(path: &str) -> &str {
        path.split_once("::").map_or(path, |(first, _)| first)
    }
    
    /// The exclusions of the record by the path each names, and whether each has
    /// matched an entry yet.
    ///
    /// An item exclusion is held under its path and matches a path equal to it; a
    /// prefix exclusion is held under its module path and matches a path whose
    /// proper ancestors, as [`ancestors`] gives them, hold that module path. So an
    /// entry costs one lookup for each of its paths and each module above it, and
    /// nothing for each exclusion. The positions held are those of the exclusions
    /// in the record, which are also those of `matched`.
    struct ExclusionIndex<'r> {
        exclusions: &'r [Exclusion],
        items: HashMap<&'r str, Vec<usize>>,
        prefixes: HashMap<&'r str, Vec<usize>>,
        matched: Vec<bool>,
    }
    
    impl<'r> ExclusionIndex<'r> {
        /// The index of `exclusions`, none of them matched.
        fn new(exclusions: &'r [Exclusion]) -> Self {
            let mut items: HashMap<&str, Vec<usize>> = HashMap::new();
            let mut prefixes: HashMap<&str, Vec<usize>> = HashMap::new();
            for (position, exclusion) in exclusions.iter().enumerate() {
                add(Work::ExclusionsIndexed, 1);
                let (by_path, path) = match &exclusion.item {
                    ExclusionItem::Item(path) => (&mut items, path),
                    ExclusionItem::Prefix(path) => (&mut prefixes, path),
                };
                by_path.entry(path.as_str()).or_default().push(position);
            }
            Self {
                exclusions,
                items,
                prefixes,
                matched: vec![false; exclusions.len()],
            }
        }
    
        /// The first exclusion in the order of the record that matches an entry
        /// with these paths, marking every exclusion that does. With no exclusion
        /// nothing is looked up.
        fn exclusion_for(&mut self, paths: &[&str]) -> Option<&'r Exclusion> {
            if self.exclusions.is_empty() {
                return None;
            }
            let mut first = None;
            for path in paths {
                add(Work::IndexLookups, 1);
                mark(self.items.get(path), &mut self.matched, &mut first);
                for module in ancestors(path).skip(1) {
                    add(Work::IndexLookups, 1);
                    mark(self.prefixes.get(module), &mut self.matched, &mut first);
                }
            }
            first.and_then(|position| self.exclusions.get(position))
        }
    }
    
    /// Mark the exclusions at `found` as matched, and keep in `first` the
    /// smallest position seen.
    fn mark(found: Option<&Vec<usize>>, matched: &mut [bool], first: &mut Option<usize>) {
        for &position in found.into_iter().flatten() {
            add(Work::ExclusionMarks, 1);
            matched[position] = true;
            *first = Some(first.map_or(position, |seen| seen.min(position)));
        }
    }
    
    /// The text of a listing error, with the symbols an ambiguous name names, so
    /// the message says where to look.
    fn surface_message(source: &Workspace, error: &ExportError) -> String {
        let mut text = error.to_string();
        if let ExportError::Resolve(ResolveError::Ambiguous(ids)) = error {
            let mut sites: Vec<String> = ids
                .iter()
                .filter_map(|id| source.item(*id))
                .map(|item| format!("`{}` ({}:{})", item.path, item.file, item.name_line))
                .collect();
            sites.sort();
            text.push_str(": ");
            text.push_str(&sites.join(", "));
        }
        text
    }
    
    /// The order of a report: package, path, file and line.
    fn order(gap: &Gap) -> (&str, &str, &str, usize) {
        (&gap.package, &gap.path, &gap.file, gap.line)
    }
    
    /// What judging one package set accumulates: the report, which exclusions
    /// matched, and the counts of each package.
    struct Judge<'a> {
        source: &'a Workspace,
        bound: Bound,
        index: ExclusionIndex<'a>,
        counts: BTreeMap<&'a str, PackageCoverage>,
        report: CoverageReport,
    }
    
    impl Judge<'_> {
        /// Judge one entry of its package: `gap` is what it is reported as when it
        /// is neither covered nor excluded, and `matching` the paths an exclusion
        /// is matched against.
        fn judge(&mut self, gap: Gap, matching: &[&str], covered: bool) {
            add(Work::EntriesJudged, 1);
            let exclusion = self.index.exclusion_for(matching);
            if let Some(count) = self.counts.get_mut(gap.package.as_str()) {
                let (seen, hit) = if gap.kind.is_code() {
                    (&mut count.items, &mut count.covered_items)
                } else {
                    (&mut count.tests, &mut count.covered_tests)
                };
                *seen += 1;
                *hit += usize::from(covered);
            }
            if covered {
                return;
            }
            match exclusion {
                Some(exclusion) => self.report.excluded.push(Excluded {
                    gap,
                    exclusion: written(exclusion),
                    reason: exclusion.reason.clone(),
                }),
                None if gap.kind.is_code() => self.report.code.push(gap),
                None => self.report.tests.push(gap),
            }
        }
    
        /// Judge the item `item` of `package` exported under `paths` (the test
        /// path for a test).
        fn judge_item(
            &mut self,
            kind: GapKind,
            package: &str,
            item: ItemId,
            paths: &[&str],
            covered: bool,
        ) {
            let Some(defining) = self.source.item(item) else {
                return;
            };
            let gap = Gap {
                kind,
                package: package.to_string(),
                path: paths.first().copied().unwrap_or(&defining.path).to_string(),
                item: defining.path.to_string(),
                file: defining.file.to_string(),
                line: defining.name_line,
            };
            let mut matching: Vec<&str> = paths.to_vec();
            matching.push(&defining.path);
            self.judge(gap, &matching, covered);
        }
    
        /// Judge the public surface of one library or binary target, or report
        /// why it cannot be listed.
        fn surface(&mut self, id: TargetId, target: &Target) {
            #[cfg(test)]
            tests::listing(id);
            match self.surface_paths(id) {
                Ok(surface) => {
                    let mut units: BTreeMap<ItemId, Vec<&str>> = BTreeMap::new();
                    for export in &surface.items {
                        units.entry(export.item).or_default().push(&export.path);
                    }
                    for (item, paths) in units {
                        let covered = self.bound.covers(self.source, item);
                        self.judge_item(GapKind::PublicItem, &target.package, item, &paths, covered);
                    }
                    for (reexport, named) in surface.reexports {
                        let mut matching = vec![reexport.path.as_str()];
                        matching.extend(named.iter().map(String::as_str));
                        let covered = matching
                            .iter()
                            .any(|path| ancestors(path).any(|p| self.bound.below.contains(p)));
                        let gap = Gap {
                            kind: GapKind::Reexport,
                            package: target.package.clone(),
                            path: reexport.path.clone(),
                            item: named
                                .first()
                                .cloned()
                                .unwrap_or_else(|| reexport.written.clone()),
                            file: reexport.file.to_string(),
                            line: reexport.line,
                        };
                        self.judge(gap, &matching, covered);
                    }
                }
                Err(error) => self.report.errors.push(SurfaceError {
                    package: target.package.clone(),
                    target: target.crate_path.clone(),
                    root: target.root.to_string(),
                    failure: SurfaceFailure::of(&error),
                    message: surface_message(self.source, &error),
                }),
            }
        }
    }
    
    struct ListedSurface {
        items: Vec<raf_source::ExportedItem>,
        reexports: Vec<(Reexport, Vec<String>)>,
    }
    
    impl Judge<'_> {
        /// Resolve origins before judging anything of a target, so a failed native
        /// lookup cannot leave a partially counted surface.
        fn surface_paths(&self, id: TargetId) -> Result<ListedSurface, ExportError> {
            let mut exports = self.source.exported_items(id)?;
            let mut reexports = Vec::new();
            for reexport in std::mem::take(&mut exports.reexports) {
                if reexport.item.is_some() {
                    continue;
                }
                let named = self
                    .named_paths(id, &reexport)
                    .map_err(ExportError::Resolve)?;
                if !named.is_empty() {
                    reexports.push((reexport, named));
                }
            }
            Ok(ListedSurface {
                items: exports.items,
                reexports,
            })
        }
    
        /// Resolve the known prefix in the declaring module's actual edition and
        /// import context. Only the unknown tail is appended: a macro-generated
        /// item need not exist in the source index.
        fn named_paths(&self, id: TargetId, reexport: &Reexport) -> Result<Vec<String>, ResolveError> {
            let written = reexport.written.as_str();
            let relative = written.strip_prefix("::").unwrap_or(written);
            let first = first_segment(relative);
            let mut rest = relative;
            while let Some(below) = rest.strip_prefix("super::") {
                rest = below;
            }
            let head = if rest.len() == relative.len() {
                &written[..written.len() - relative.len() + first.len()]
            } else {
                &written[..written.len() - rest.len() - 2]
            };
            let mut paths = Vec::new();
            for module in self.declaring_modules(id, reexport) {
                let (prefix, tail) = written.rsplit_once("::").unwrap_or((written, ""));
                match self.source.resolve_from(&module, prefix) {
                    Ok(resolved) => paths.push(if tail.is_empty() {
                        resolved.path
                    } else {
                        format!("{}::{tail}", resolved.path)
                    }),
                    Err(ResolveError::NotFound(_)) if prefix != head => {
                        match self.source.resolve_from(&module, head) {
                            Ok(resolved) => {
                                paths.push(format!("{}{}", resolved.path, &written[head.len()..]));
                            }
                            Err(ResolveError::NotFound(_)) => {}
                            Err(error) => return Err(error),
                        }
                    }
                    Err(ResolveError::NotFound(_)) => {}
                    Err(error) => return Err(error),
                }
            }
            paths.sort();
            paths.dedup();
            Ok(paths)
        }
    
        /// The paths of the module that holds the declaration of a re-export, in
        /// the target, one for each path its file is reached under: the innermost
        /// inline module of the file whose braces hold the declaration's line, or
        /// the module of the file itself.
        fn declaring_modules(&self, id: TargetId, reexport: &Reexport) -> Vec<String> {
            let Some(file) = self.source.file(&reexport.file) else {
                return Vec::new();
            };
            file.reached
                .iter()
                .filter(|reach| reach.target == id)
                .map(|reach| {
                    let below = format!("{}::", reach.module_path);
                    self.source
                        .items_in(&reexport.file)
                        .filter(|item| {
                            item.kind == SymbolKind::Mod
                                && item.target == id
                                && item.path.starts_with(&below)
                                && item
                                    .body
                                    .as_ref()
                                    .is_some_and(|body| body.lines.contains(&reexport.line))
                        })
                        .min_by_key(|item| {
                            item.body
                                .as_ref()
                                .map_or(usize::MAX, |body| body.end_byte - body.start_byte)
                        })
                        .map_or_else(
                            || reach.module_path.to_string(),
                            |item| item.path.to_string(),
                        )
                })
                .collect()
        }
    }
    
    /// The report over the packages named in `packages`.
    pub(crate) fn compute(
        source: &Workspace,
        graph: &Graph,
        record: Option<&CoverageRecord>,
        packages: &BTreeSet<&str>,
    ) -> Computed {
        let exclusions: &[Exclusion] = record.map_or(&[], |record| &record.exclusions);
        let scope: BTreeSet<&str> = record.map_or_else(BTreeSet::new, |record| {
            record.scope.iter().map(String::as_str).collect()
        });
        let counts = packages
            .iter()
            .map(|name| {
                (
                    *name,
                    PackageCoverage {
                        name: (*name).to_string(),
                        in_scope: scope.contains(name),
                        items: 0,
                        covered_items: 0,
                        tests: 0,
                        covered_tests: 0,
                    },
                )
            })
            .collect();
        let mut judge = Judge {
            source,
            bound: Bound::of(graph, source),
            index: ExclusionIndex::new(exclusions),
            counts,
            report: CoverageReport::default(),
        };
        for id in source.target_ids() {
            let Some(target) = source.target(id) else {
                continue;
            };
            if matches!(target.kind, TargetKind::Lib | TargetKind::Bin)
                && packages.contains(target.package.as_str())
            {
                judge.surface(id, target);
            }
        }
        for item in source.items().filter(|item| item.test == TestRole::TestFn) {
            let Some(target) = source.target(item.target) else {
                continue;
            };
            if packages.contains(target.package.as_str()) {
                let covered = judge.bound.tests.contains(&item.id);
                judge.judge_item(
                    GapKind::Test,
                    &target.package,
                    item.id,
                    &[&item.path],
                    covered,
                );
            }
        }
        let Judge {
            mut report,
            index,
            counts,
            ..
        } = judge;

    //! Reverse coverage: the lines of every package of the source workspace that
    //! no capability explains.
    //!
    //! [`Workplan::lines_report`] reads every package the loader reaches, the root
    //! package and the packages that no capability binds included, and returns a
    //! [`LinesReport`].
    //!
    //! # Units and lines
    //!
    //! The units of a file are the units of every target the source index lists:
    //! every item except a module and an `extern crate`, every impl block from its
    //! first attribute to its closing brace, and every item-level macro invocation
    //! and `const _` of a module, found by parsing the file. The counted lines of a
    //! file are the union of the lines of its units; a file is counted once, for
    //! the package of the first target that reaches it, however many targets and
    //! module paths reach it. Every other line (a blank line, a `use` or `mod`
    //! declaration, an inner doc comment, the SPDX header, the braces of an inline
    //! module) is structural and is not counted. Lines are counted by line breaks,
    //! so a file with CRLF endings reads as the same file with LF.
    //!
    //! # Covered
    //!
    //! A line is covered when a covered unit holds it. The covered units are the
    //! closure of the bound items under [`Workspace::reachable`]
    //! (contains, impl-of and references): the roots are the items that a `code`,
    //! `tests` or `examples` binding of a loaded capability resolves to, and for an
    //! `examples` binding of the crate root of an example target every item and
    //! impl block of that target. A unit is covered when any item or block of its
    //! source is reached. A macro invocation or a `const _` is covered when one of
    //! its module paths is a module that a root opens (a bound module, or one
    //! below it), when its target is an example target an `examples` binding
    //! names, or, for an invocation, when a covered unit names a path that resolves
    //! to nothing from the module of that unit and whose last segment is an
    //! identifier inside the invocation; the `macro_rules!` a covered invocation
    //! invokes is then a root too, and the closure is taken again until no
    //! invocation is newly covered.
    //! A direct `#[global_allocator]` static is also a root of its participating
    //! target. Test-only registration requires a participating test context. The
    //! static retains its source role and reaches its real initializer dependencies.
    //! Malformed registrations or registrations without an exact indexed static
    //! refuse the report; macros and `cfg_attr` are not expanded for this purpose.
    //!
    //! # Exceptions
    //!
    //! A line that no covered unit holds is excepted when an exception of the
    //! coverage record matches a unit that holds it, or its file: an `item`
    //! exception the path of the unit or a path the unit is below (a method below
    //! its type, an item below its module, an impl block below its self type and
    //! its module), a `<module>::*` prefix every unit below the module, a `file`
    //! the file, and a directory every file below it. The first matching exception
    //! in the order of the record gives the reason. The exceptions are put into an
    //! index once, and each unit looks up its own paths and the modules above them,
    //! so the work grows with the units plus the exceptions.
    //!
    //! Every other counted line is an orphan line, listed under the innermost unit
    //! that holds it. A line is a test line when that unit is test code (a
    //! `#[cfg(test)]` module or a `#[test]` function) or is in a target that does
    //! not ship (integration tests, benches, examples), and a shipping line
    //! otherwise.
    //!
    //! An exception that matches nothing is judged at the load, without counting a
    //! line: an `item` names an item of the index, a prefix a module or an item of
    //! the index at or below it, a `file` a file of the index and a directory a
    //! file of the index below it.

    impl<'ast> Visit<'ast> for Allocators {
        fn visit_attribute(&mut self, attribute: &'ast syn::Attribute) {
            if attribute.path().is_ident("global_allocator") {
                self.attributes.insert(
                    attribute.span().byte_range().start,
                    attribute.span().start().line,
                );
                if !matches!(attribute.meta, syn::Meta::Path(_)) {
                    self.error = Some(format!(
                        "line {}: global_allocator must be a bare attribute on a static",
                        attribute.span().start().line
                    ));
                }
            }
            syn::visit::visit_attribute(self, attribute);
        }
    
        fn visit_item_static(&mut self, item: &'ast syn::ItemStatic) {
            let attributes: Vec<_> = item
                .attrs
                .iter()
                .filter(|attribute| attribute.path().is_ident("global_allocator"))
                .collect();
            if !attributes.is_empty() {
                if attributes.len() != 1 {
                    self.error = Some(format!(
                        "line {}: a static has repeated global_allocator attributes",
                        item.span().start().line
                    ));
                }
                for attribute in attributes {
                    self.attached.insert(attribute.span().byte_range().start);
                }
                let bytes = item.span().byte_range();
                self.statics.push((bytes.start, bytes.end));
            }
            syn::visit::visit_item_static(self, item);
        }
    }
    
    /// Module units and direct allocator registrations from one bounded parse.
    fn module_units(path: &RelPath, text: &str) -> Result<ParsedUnits, String> {
        parse_source(path, text, |tree| {
            let mut out = ParsedUnits::default();
            found_in(&tree.items, &mut Vec::new(), &mut out.modules);
            let mut allocators = Allocators::default();
            allocators.visit_file(tree);
            if let Some(error) = allocators.error {
                return Err(format!("{path}: {error}"));
            }
            if let Some((_, line)) = allocators
                .attributes
                .iter()
                .find(|(at, _)| !allocators.attached.contains(at))
            {
                return Err(format!(
                    "{path}:{line}: global_allocator is not attached to a static"
                ));
            }
            out.allocators = allocators.statics;
            Ok(out)
        })
        .map_err(|error| error.to_string())?
    }
    
    fn found_in(items: &[SynItem], modules: &mut Vec<String>, out: &mut Vec<Found>) {
        for item in items {
            match item {
                SynItem::Macro(invocation) if invocation.ident.is_none() => {
                    let span = item.span();
                    let mut idents = Vec::new();
                    idents_of(invocation.mac.tokens.clone(), &mut idents);
                    let path: String = quote_path(&invocation.mac.path);
                    out.push(Found {
                        first: span.start().line,
                        last: span.end().line,
                        modules: modules.clone(),
                        invoked: Some(path),
                        idents,
                    });
                }
                SynItem::Const(constant) if constant.ident == "_" => {
                    let span = item.span();
                    out.push(Found {
                        first: span.start().line,
                        last: span.end().line,
                        modules: modules.clone(),
                        invoked: None,
                        idents: Vec::new(),
                    });
                }
                SynItem::Mod(module) => {
                    if let Some((_, content)) = &module.content {
                        let name = module.ident.to_string();
                        modules.push(name.strip_prefix("r#").unwrap_or(&name).to_string());
                        found_in(content, modules, out);
                        modules.pop();
                    }
                }
                _ => {}
            }
        }
    }
    
    /// A macro path as written, without spaces: `kinds`, `crate::kinds`.
    fn quote_path(path: &syn::Path) -> String {
        let mut text = String::new();
        if path.leading_colon.is_some() {
            text.push_str("::");
        }
        let segments: Vec<String> = path
            .segments
            .iter()
            .map(|segment| segment.ident.to_string())
            .collect();
        text.push_str(&segments.join("::"));
        text
    }
    
    /// Every identifier of `stream`, nested groups included, walked on an
    /// explicit stack.
    fn idents_of(stream: TokenStream, out: &mut Vec<String>) {
        let mut stack = vec![stream.into_iter()];
        while let Some(tokens) = stack.last_mut() {
            match tokens.next() {
                Some(TokenTree::Ident(ident)) => {
                    let name = ident.to_string();
                    out.push(name.strip_prefix("r#").unwrap_or(&name).to_string());
                }
                Some(TokenTree::Group(group)) => stack.push(group.stream().into_iter()),
                Some(_) => {}
                None => {
                    stack.pop();
                }
            }
        }
    }

    /// The runs of `text` whose last segment is one of `wanted`, lexed on a
    /// thread of its own so that the span data of the lexer dies with it.
    fn runs_of(path: &RelPath, text: &str, wanted: &BTreeSet<String>) -> Result<Vec<Run>, String> {
        thread::scope(|scope| {
            let worker = thread::Builder::new()
                .name(path.to_string())
                .stack_size(PARSE_STACK_BYTES)
                .spawn_scoped(scope, move || lex_runs(text, wanted))
                .map_err(|error| format!("{path}: the thread that lexes it cannot start: {error}"))?;
            match worker.join() {
                Ok(result) => result.map_err(|error| format!("{path}: {error}")),
                Err(payload) => panic::resume_unwind(payload),
            }
        })
    }
    
    /// [`runs_of`] on the calling thread. A byte order mark and a shebang line
    /// are left out, as `syn` leaves them out, and offsets stay those of `text`.
    fn lex_runs(text: &str, wanted: &BTreeSet<String>) -> Result<Vec<Run>, String> {
        let mut base = 0;
        let mut body = text;
        if let Some(rest) = body.strip_prefix('\u{feff}') {
            base += '\u{feff}'.len_utf8();
            body = rest;
        }
        if let Some(after) = body.strip_prefix("#!")
            && !after.trim_start().starts_with('[')
        {
            let cut = body.find('\n').unwrap_or(body.len());
            base += cut;
            body = &body[cut..];
        }
        let stream = TokenStream::from_str(body).map_err(|error| error.to_string())?;
        let mut runs = Vec::new();
        let mut stack: Vec<(proc_macro2::token_stream::IntoIter, Scan)> =
            vec![(stream.into_iter(), Scan::default())];
        while let Some((tokens, scan)) = stack.last_mut() {
            let Some(tree) = tokens.next() else {
                scan.close(&mut runs, wanted);
                stack.pop();
                if let Some((_, outer)) = stack.last_mut() {
                    outer.push(Tok::Other);
                }
                continue;
            };
            add(Work::LineTokens, 1);
            match tree {
                TokenTree::Group(group) => {
                    scan.close(&mut runs, wanted);
                    stack.push((group.stream().into_iter(), Scan::default()));
                }
                TokenTree::Ident(ident) => {
                    let raw = ident.to_string();
                    let start = base + ident.span().byte_range().start;
                    scan.ident(&raw, start, &mut runs, wanted);
                }
                TokenTree::Punct(punct) => {
                    scan.punct(
                        punct.as_char(),
                        punct.spacing() == Spacing::Joint,
                        &mut runs,
                        wanted,
                    );
                }
                TokenTree::Literal(_) => {
                    scan.close(&mut runs, wanted);
                    scan.push(Tok::Other);
                }
            }
        }
        Ok(runs)
    }
    
    /// A token as the run scanner remembers it.
    #[derive(Debug, Clone, Copy, PartialEq, Eq)]
    enum Tok {
        /// An identifier, and whether the identifier after it is declared by it.
        Ident { declares: bool },
        /// A punctuation character and whether the next one joins it.
        Punct { ch: char, joint: bool },
        /// A literal or a group.
        Other,
    }
    
    /// The run scanner of one level of delimiters: the last three tokens and the
    /// open run.
    #[derive(Default)]
    struct Scan {
        last: [Option<Tok>; 3],
        open: Option<Run>,
    }
    
    impl Scan {
        fn push(&mut self, tok: Tok) {
            self.last = [self.last[1], self.last[2], Some(tok)];
        }
    
        /// True when the last two tokens are `::`.
        fn colons(&self) -> bool {
            matches!(
                (self.last[1], self.last[2]),
                (
                    Some(Tok::Punct {
                        ch: ':',
                        joint: true
                    }),
                    Some(Tok::Punct { ch: ':', .. })
                )
            )
        }
    
        /// True when the last token is a `.` that is not part of `..`.
        fn single_dot(&self) -> bool {
            matches!(
                self.last[2],
                Some(Tok::Punct {
                    ch: '.',
                    joint: false
                })
            ) && !matches!(
                self.last[1],
                Some(Tok::Punct {
                    ch: '.',
                    joint: true
                })
            )
        }
    
        /// End the open run, keeping it when its last segment is wanted.
        fn close(&mut self, runs: &mut Vec<Run>, wanted: &BTreeSet<String>) {
            if let Some(run) = self.open.take()
                && wanted.contains(&run.last)
            {
                runs.push(run);
            }
        }
    
        fn ident(&mut self, raw: &str, start: usize, runs: &mut Vec<Run>, wanted: &BTreeSet<String>) {
            let plain = raw.strip_prefix("r#");
            let word = plain.unwrap_or(raw);
            let path_word = plain.is_none() && matches!(word, "crate" | "self" | "super");
            let keyword = plain.is_none() && KEYWORDS.contains(&word);
            let declares = plain.is_none() && DECLARES.contains(&word);
            if self.colons() && self.open.is_some() && !keyword && !path_word {
                if let Some(run) = self.open.as_mut() {
                    run.text.push_str("::");
                    run.text.push_str(word);
                    run.segments += 1;
                    run.last = word.to_string();
                }
            } else {
                let leading = self.colons() && self.open.is_none();
                self.close(runs, wanted);
                let after_declaring = matches!(self.last[2], Some(Tok::Ident { declares: true }));
                if !keyword && !self.single_dot() && !after_declaring {
                    let text = if leading {
                        format!("::{word}")
                    } else {
                        word.to_string()
                    };
                    self.open = Some(Run {
                        start,
                        text,
                        segments: 1,
                        last: word.to_string(),
                    });
                }
            }
            self.push(Tok::Ident { declares });
        }
    
        fn punct(&mut self, ch: char, joint: bool, runs: &mut Vec<Run>, wanted: &BTreeSet<String>) {
            let second = matches!(
                self.last[2],
                Some(Tok::Punct {
                    ch: ':',
                    joint: true
                })
            );
            if ch == ':' && !joint && !second {
                // A single `:`: a run of one segment before it is a name being
                // bound or a field, not a path.
                if self.open.as_ref().is_some_and(|run| run.segments == 1) {
                    self.open = None;
                } else {
                    self.close(runs, wanted);
                }
            } else if ch != ':' {
                self.close(runs, wanted);
            }
            self.push(Tok::Punct { ch, joint });
        }
    }
    
    /// The exceptions of the record by what each names, so that a unit looks up
    /// its own paths and the modules above them, and a file its own path and the
    /// directories above it, whatever the number of exceptions.
    struct ExceptionIndex<'r> {
        items: HashMap<&'r str, Vec<usize>>,
        prefixes: HashMap<&'r str, Vec<usize>>,
        files: HashMap<&'r str, Vec<usize>>,
        directories: HashMap<&'r str, Vec<usize>>,
    }
    
    /// `path` and each path it is below, longest first: `a::b::c`, `a::b`, `a`.
    fn ancestors(path: &str) -> impl Iterator<Item = &str> {
        std::iter::successors(Some(path), |p| p.rsplit_once("::").map(|(head, _)| head))
    }
    
    /// The directories `file` is below, deepest first: `a/b` and `a` for
    /// `a/b/c.rs`.
    fn directories(file: &str) -> impl Iterator<Item = &str> {
        std::iter::successors(file.rsplit_once('/').map(|(head, _)| head), |p| {
            p.rsplit_once('/').map(|(head, _)| head)
        })
    }
    
    impl<'r> ExceptionIndex<'r> {
        fn new(exceptions: &'r [Exception]) -> Self {
            let mut index = Self {
                items: HashMap::new(),
                prefixes: HashMap::new(),
                files: HashMap::new(),
                directories: HashMap::new(),
            };
            for (position, exception) in exceptions.iter().enumerate() {
                add(Work::ExceptionsIndexed, 1);
                let (map, path) = match &exception.target {
                    ExceptionTarget::Item(path) => (&mut index.items, path),
                    ExceptionTarget::Prefix(path) => (&mut index.prefixes, path),
                    ExceptionTarget::File(path) => (&mut index.files, path),
                    ExceptionTarget::Directory(path) => (&mut index.directories, path),
                };
                map.entry(path.as_str()).or_default().push(position);
            }
            index
        }
    
        fn is_empty(&self) -> bool {
            self.items.is_empty()
                && self.prefixes.is_empty()
                && self.files.is_empty()
                && self.directories.is_empty()
        }
    
        /// The first exception that matches a unit at the item paths `paths` and
        /// in the modules `modules`: an item exception at a path or one it is
        /// below, a prefix exception at a module a path is below, and either at a
        /// module of `modules` or one it is below.
        fn unit(&self, paths: &[&str], modules: &[&str]) -> Option<usize> {
            if self.items.is_empty() && self.prefixes.is_empty() {
                return None;
            }
            let mut first = None;
            let mut take = |found: Option<&Vec<usize>>| {
                add(Work::ExceptionLookups, 1);
                if let Some(&position) = found.and_then(|positions| positions.first()) {
                    first = Some(first.map_or(position, |seen: usize| seen.min(position)));
                }
            };
            for path in paths {
                for (depth, at) in ancestors(path).enumerate() {
                    take(self.items.get(at));
                    if depth > 0 {
                        take(self.prefixes.get(at));
                    }
                }
            }
            for module in modules {
                for at in ancestors(module) {
                    take(self.items.get(at));
                    take(self.prefixes.get(at));
                }
            }
            first
        }
    
        /// The first exception that matches the file `file`, by its path or a
        /// directory it is below.
        fn file(&self, file: &str) -> Option<usize> {
            if self.files.is_empty() && self.directories.is_empty() {
                return None;
            }
            add(Work::ExceptionLookups, 1);
            let mut first = self
                .files
                .get(file)
                .and_then(|found| found.first())
                .copied();
            for directory in directories(file) {
                add(Work::ExceptionLookups, 1);
                if let Some(&position) = self
                    .directories
                    .get(directory)
                    .and_then(|found| found.first())
                {
                    first = Some(first.map_or(position, |seen| seen.min(position)));
                }
            }
            first
        }
    }
    
    /// The positions of the exceptions of `exceptions` that match nothing of the
    /// index of `source`: an item exception that names no item, a prefix that
    /// names no module or item at or below it, a file that is no file of the
    /// index and a directory that holds none. No line is counted to judge it.
    pub(crate) fn unmatched_exceptions(source: &Workspace, exceptions: &[Exception]) -> Vec<usize> {
        let mut matched = vec![false; exceptions.len()];
        let mut files: HashMap<&str, Vec<usize>> = HashMap::new();
        let mut dirs: HashMap<&str, Vec<usize>> = HashMap::new();
        for (position, exception) in exceptions.iter().enumerate() {
            add(Work::ExceptionsIndexed, 1);
            match &exception.target {
                ExceptionTarget::Item(path) => {
                    add(Work::ExceptionLookups, 1);
                    matched[position] = source
                        .items_under(path)
                        .next()
                        .is_some_and(|item| *item.path == **path);
                }
                ExceptionTarget::Prefix(path) => {
                    add(Work::ExceptionLookups, 1);
                    matched[position] = source.items_under(path).next().is_some();
                }
                ExceptionTarget::File(path) => files.entry(path).or_default().push(position),
                ExceptionTarget::Directory(path) => dirs.entry(path).or_default().push(position),
            }
        }
        if !files.is_empty() || !dirs.is_empty() {
            for file in source.files() {
                let path = file.path.as_str();
                add(Work::ExceptionLookups, 1);
                for position in files.get(path).into_iter().flatten() {
                    matched[*position] = true;
                }
                for directory in directories(path) {
                    add(Work::ExceptionLookups, 1);
                    for position in dirs.get(directory).into_iter().flatten() {
                        matched[*position] = true;
                    }
                }
            }
        }
        matched
            .iter()
            .enumerate()
            .filter(|(_, hit)| !**hit)
            .map(|(position, _)| position)
            .collect()
    }
    
    /// The name of the kind of an item, as the report writes it.
    fn kind_name(kind: SymbolKind) -> &'static str {
        match kind {
            SymbolKind::Fn
            | SymbolKind::ForeignFn
            | SymbolKind::InherentMethod
            | SymbolKind::TraitImplMethod
            | SymbolKind::TraitDefaultMethod
            | SymbolKind::TraitRequiredMethod => "fn",
            SymbolKind::Struct => "struct",
            SymbolKind::Enum => "enum",
            SymbolKind::Union => "union",
            SymbolKind::Trait => "trait",
            SymbolKind::TypeAlias | SymbolKind::ForeignType | SymbolKind::AssocType => "type",
            SymbolKind::Const | SymbolKind::AssocConst => "const",
            SymbolKind::Static | SymbolKind::ForeignStatic => "static",
            SymbolKind::MacroRules => "macro_rules",
            _ => "item",
        }
    }
    
    /// An impl block as the report names it: `<module>::<impl Type>` or
    /// `<module>::<impl Trait for Type>`.
    fn impl_name(block: &ImplBlock) -> String {
        match &block.trait_path {
            Some(trait_path) => format!(
                "{}::<impl {trait_path} for {}>",
                block.module, block.self_type
            ),
            None => format!("{}::<impl {}>", block.module, block.self_type),
        }
    }
    
    /// `path` without its last segment.
    fn parent(path: &str) -> &str {
        path.rsplit_once("::").map_or(path, |(head, _)| head)
    }
    
    /// The module of an item: the module of its impl block for a member of one,
    /// the module of its trait for a member of a trait, and the module its path
    /// names otherwise.
    fn item_module(source: &Workspace, item: &Item, block_of: &HashMap<ItemId, ImplId>) -> String {
        if let Some(block) = block_of.get(&item.id).and_then(|id| source.impl_block(*id)) {
            return block.module.to_string();
        }
        match item.kind {
            SymbolKind::TraitRequiredMethod
            | SymbolKind::TraitDefaultMethod
            | SymbolKind::AssocConst
            | SymbolKind::AssocType => parent(parent(&item.path)).to_string(),
            _ => parent(&item.path).to_string(),
        }
    }
    
    /// The unit of the items of one source, the item of the first path first.
    fn item_unit(
        source: &Workspace,
        file: usize,
        mut copies: Vec<&Item>,
        block_of: &HashMap<ItemId, ImplId>,
    ) -> Unit {
        copies.sort_by(|a, b| (&a.path, a.id).cmp(&(&b.path, b.id)));
        let first = copies[0];
        let block = block_of
            .get(&first.id)
            .and_then(|id| source.impl_block(*id));
        let item = match (first.self_key, block) {
            (Some(SelfType::Trait), Some(block)) => format!("{}::{}", impl_name(block), first.name),
            _ => first.path.to_string(),
        };
        Unit {
            file,
            first: first.span.lines.start,
            last: first
                .span
                .lines
                .end
                .saturating_sub(1)
                .max(first.span.lines.start),
            bytes: (first.span.start_byte, first.span.end_byte),
            module: item_module(source, first, block_of),
            item,
            kind: kind_name(first.kind),
            shipping: copies.iter().any(|copy| source.is_shipping(copy)),
            what: What::Item(copies.iter().map(|copy| copy.id).collect()),
            covered: false,
            exception: None,
        }
    }
    
    /// The unit of the impl blocks of one source.
    fn impl_unit(source: &Workspace, file: usize, copies: &[&ImplBlock]) -> Unit {
        let first = copies[0];
        Unit {
            file,
            first: first.span.lines.start,
            last: first
                .span
                .lines
                .end
                .saturating_sub(1)
                .max(first.span.lines.start),
            bytes: (first.span.start_byte, first.span.end_byte),
            module: first.module.to_string(),
            item: impl_name(first),
            kind: "impl",
            shipping: copies.iter().any(|block| {
                block.test == TestRole::None
                    && source
                        .target(block.target)
                        .is_some_and(|target| target.shipping)
            }),
            what: What::Impl(copies.iter().map(|block| block.id).collect()),
            covered: false,
            exception: None,
        }
    }
    
    /// The unit of a macro invocation or a `const _` that a parse of `file`
    /// found, once for each module path the file is reached under.
    fn module_unit(source: &Workspace, index: usize, file: &SourceFile, found: Found) -> Unit {
        let suffix = found.modules.join("::");
        let mut copies: Vec<(String, TargetId, bool)> = file
            .reached
            .iter()
            .map(|reach| {
                let module = if suffix.is_empty() {
                    reach.module_path.to_string()
                } else {
                    format!("{}::{suffix}", reach.module_path)
                };
                // An inline module guarded by `test` makes what it holds test
                // code, as it makes its items.
                let inner_test = !suffix.is_empty()
                    && source
                        .items_under(&module)
                        .take_while(|item| *item.path == *module)
                        .any(|item| {
                            item.kind == SymbolKind::Mod
                                && item.target == reach.target
                                && item.test != TestRole::None
                        });
                (module, reach.target, reach.test || inner_test)
            })
            .collect();
        copies.sort();
        copies.dedup();
        let shipping = copies.iter().any(|(_, target, test)| {
            !test && source.target(*target).is_some_and(|target| target.shipping)
        });
        let module = copies
            .first()
            .map_or_else(String::new, |(module, _, _)| module.clone());
        let (item, kind) = match &found.invoked {
            Some(path) => (format!("{module}::{path}!"), "macro"),
            None => (format!("{module}::_"), "const"),
        };
        Unit {
            file: index,
            first: found.first,
            last: found.last.max(found.first),
            bytes: (0, 0),
            module,
            item,
            kind,
            shipping,
            what: What::Module(ModuleUnit {
                copies,
                invoked: found.invoked,
                idents: found.idents,
            }),
            covered: false,
            exception: None,
        }
    }
    
    /// The units of every file of the source workspace.
    struct Gathered<'s> {
        files: Vec<File<'s>>,
        units: Vec<Unit>,
        /// The units of each file, as a range of `units`.
        of_file: Vec<std::ops::Range<usize>>,
        /// The runs whose last segment a macro invocation holds, each with the
        /// innermost item or impl unit that holds its start.
        runs: Vec<(Run, Option<usize>)>,
        /// Exact indexed definitions bearing a compiler allocator registration.
        allocators: Vec<ItemId>,
    }
    
    /// The units of every file of `source`, file by file in the order of the
    /// files.
    fn collect<'s>(source: &'s Workspace, judged: &BTreeSet<&str>) -> Result<Gathered<'s>, String> {
        let mut block_of: HashMap<ItemId, ImplId> = HashMap::new();
        let mut blocks: BTreeMap<&RelPath, BTreeMap<(usize, usize), Vec<&ImplBlock>>> = BTreeMap::new();
        for block in source.impls() {
            for member in &block.members {
                block_of.insert(*member, block.id);
            }
            blocks
                .entry(&block.file)
                .or_default()
                .entry((block.span.start_byte, block.span.end_byte))
                .or_default()
                .push(block);
        }
        let mut out = Gathered {
            files: Vec::new(),
            units: Vec::new(),
            of_file: Vec::new(),
            runs: Vec::new(),
            allocators: Vec::new(),
        };
        for (index, file) in source
            .files()
            .filter(|file| judged.contains(file.package.as_str()))
            .enumerate()
        {
            let begin = out.units.len();
            out.files.push(File {
                path: &file.path,
                package: &file.package,
                exception: None,
            });
            let mut groups: BTreeMap<(usize, usize), Vec<&Item>> = BTreeMap::new();
            for item in source
                .items_in(&file.path)
                .filter(|item| !matches!(item.kind, SymbolKind::Mod | SymbolKind::ExternCrate))
            {
                groups
                    .entry((item.span.start_byte, item.span.end_byte))
                    .or_default()
                    .push(item);
            }
            let parsed = module_units(&file.path, &file.text)?;
            for bytes in parsed.allocators {
                let copies = groups.get(&bytes).filter(|copies| copies.iter().all(|item| item.kind == SymbolKind::Static)).ok_or_else(|| format!("{}: global_allocator static at bytes {}..{} has no exact indexed static definition", file.path, bytes.0, bytes.1))?;
                out.allocators.extend(copies.iter().map(|item| item.id));
            }
            for copies in groups.into_values() {
                out.units.push(item_unit(source, index, copies, &block_of));
            }
            for copies in blocks
                .get(&file.path)
                .into_iter()
                .flat_map(BTreeMap::values)
            {
                out.units.push(impl_unit(source, index, copies));
            }
            for found in parsed.modules {
                out.units.push(module_unit(source, index, file, found));
            }
            out.of_file.push(begin..out.units.len());
        }
        let wanted: BTreeSet<String> = out
            .units
            .iter()
            .filter_map(|unit| match &unit.what {
                What::Module(module) if module.invoked.is_some() => Some(&module.idents),
                _ => None,
            })
            .flatten()
            .cloned()
            .collect();
        if !wanted.is_empty() {
            for (index, file) in source
                .files()
                .filter(|file| judged.contains(file.package.as_str()))
                .enumerate()
            {
                let runs = runs_of(&file.path, &file.text, &wanted)?;
                let range = out.of_file[index].clone();
                place(&out.units, range, runs, &mut out.runs);
            }
        }
        Ok(out)
    }
    
    /// Give each run of a file to the innermost item or impl unit of `range`
    /// whose bytes hold its start, in one sweep of both in byte order.
    fn place(
        units: &[Unit],
        range: std::ops::Range<usize>,
        mut runs: Vec<Run>,
        out: &mut Vec<(Run, Option<usize>)>,
    ) {
        let mut spans: Vec<(usize, usize, usize)> = range
            .filter(|at| !matches!(units[*at].what, What::Module(_)))
            .map(|at| (units[at].bytes.0, units[at].bytes.1, at))
            .collect();
        // Outer units first where two start at one byte.
        spans.sort_by(|a, b| a.0.cmp(&b.0).then(b.1.cmp(&a.1)));
        runs.sort_by_key(|run| run.start);
        let mut open: Vec<(usize, usize, usize)> = Vec::new();
        let mut next = 0;
        for run in runs {
            while let Some(&span) = spans.get(next).filter(|span| span.0 <= run.start) {
                while open.last().is_some_and(|top| top.1 <= span.0) {
                    open.pop();
                }
                open.push(span);
                next += 1;
            }
            while open.last().is_some_and(|top| top.1 <= run.start) {
                open.pop();
            }
            let holder = open.last().map(|span| span.2);
            out.push((run, holder));
        }
    }
    
    /// The roots of the walk: the bound items, and every item and impl block of
    /// an example target an `examples` binding names; the paths of the bound
    /// modules, which open what they hold; and those example targets.
    struct Roots {
        nodes: BTreeSet<Node>,
        opened: BTreeSet<String>,
        whole: BTreeSet<TargetId>,
    }
    
    fn roots(plan: &Workplan<'_>) -> Roots {
        let source = plan.source();
        let mut roots = Roots {
            nodes: BTreeSet::new(),
            opened: BTreeSet::new(),
            whole: BTreeSet::new(),
        };
        for binding in plan.graph().bindings() {
            for id in binding.items() {
                let Some(item) = source.item(*id) else {
                    continue;
                };
                let target = source.target(item.target);
                let crate_root = target.is_some_and(|target| *item.path == *target.crate_path);
                if binding.kind() == BindingKind::Example
                    && crate_root
                    && target.is_some_and(|target| target.kind == TargetKind::Example)
                {
                    roots.whole.insert(item.target);
                    continue;
                }
                roots.nodes.insert(Node::Item(*id));
                if item.kind == SymbolKind::Mod && !crate_root {
                    roots.opened.insert(item.path.to_string());
                }
            }
        }
        if !roots.whole.is_empty() {
            for item in source
                .items()
                .filter(|item| roots.whole.contains(&item.target))
            {
                roots.nodes.insert(Node::Item(item.id));
            }
            for block in source
                .impls()
                .filter(|block| roots.whole.contains(&block.target))
            {
                roots.nodes.insert(Node::Impl(block.id));
            }
        }
        roots
    }
    
    /// Mark the covered units: the closure of the roots, the macro invocations
    /// and `const _` items of the opened modules and the example targets, and the
    /// invocations a covered unit names a generated item of, whose `macro_rules!`
    /// join the roots until no invocation is newly covered. `extra` roots are
    /// walked beside them.
    fn cover(
        source: &Workspace,
        units: &mut Gathered<'_>,
        roots: &mut Roots,
        extra: &[Node],
    ) -> Result<(), String> {
        let mut participating = BTreeSet::new();
        let mut tests = BTreeSet::new();
        for node in roots.nodes.iter().chain(extra) {
            let (target, test) = match node {
                Node::Item(id) => {
                    let item = source
                        .item(*id)
                        .ok_or_else(|| "a bound coverage root has no indexed item".to_string())?;
                    (item.target, item.test)
                }
                Node::Impl(id) => {
                    let block = source
                        .impl_block(*id)
                        .ok_or_else(|| "a bound coverage root has no indexed impl".to_string())?;
                    (block.target, block.test)
                }
            };
            participating.insert(target);
            if test != TestRole::None {
                tests.insert(target);
            }
        }
        for id in &units.allocators {
            let item = source
                .item(*id)
                .ok_or_else(|| "a registered allocator has no indexed static".to_string())?;
            if participating.contains(&item.target)
                && (item.test == TestRole::None || tests.contains(&item.target))
            {
                roots.nodes.insert(Node::Item(*id));
            }
        }
        let mut unresolved: HashMap<(String, String), bool> = HashMap::new();
        loop {
            add(Work::LineRounds, 1);
            let mut nodes: Vec<Node> = roots.nodes.iter().copied().collect();
            nodes.extend_from_slice(extra);
            let reach = source
                .reachable(&nodes)
                .map_err(|error| format!("the walk of the bound items failed: {error}"))?;
            for unit in &mut units.units {
                unit.covered = match &unit.what {
                    What::Item(ids) => ids.iter().any(|id| reach.contains(Node::Item(*id))),
                    What::Impl(ids) => ids.iter().any(|id| reach.contains(Node::Impl(*id))),
                    What::Module(module) => module.copies.iter().any(|(path, target, _)| {
                        roots.whole.contains(target)
                            || ancestors(path).any(|at| roots.opened.contains(at))
                    }),
                };
            }
            let mut named: BTreeSet<String> = BTreeSet::new();
            for (run, holder) in &units.runs {
                add(Work::LineRuns, 1);
                let Some(unit) = holder
                    .map(|at| &units.units[at])
                    .filter(|unit| unit.covered)
                else {
                    continue;
                };
                let key = (unit.module.clone(), run.text.clone());
                let missing = *unresolved.entry(key).or_insert_with(|| {
                    matches!(
                        source.resolve_from(&unit.module, &run.text),
                        Err(ResolveError::NotFound(_))
                    )
                });
                if missing {
                    named.insert(run.last.clone());
                }
            }
            let mut grew = false;
            for unit in &mut units.units {
                let What::Module(module) = &unit.what else {
                    continue;
                };
                if !unit.covered && module.idents.iter().any(|ident| named.contains(ident)) {
                    unit.covered = true;
                }
                let (true, Some(invoked)) = (unit.covered, &module.invoked) else {
                    continue;
                };
                for (path, _, _) in &module.copies {
                    for site in invoked_sites(source, path, invoked) {
                        grew |= roots.nodes.insert(Node::Item(site));
                    }
                }
            }
            if !grew {
                return Ok(());
            }
        }
    }
    
    /// The macros that the invocation of `invoked` in the module `module` names:
    /// the path read from the module in the macro name space, and for a name of
    /// one segment from each module above it too, as a `macro_rules!` of a parent
    /// module is in the textual scope of the modules declared after it.
    fn invoked_sites(source: &Workspace, module: &str, invoked: &str) -> Vec<ItemId> {
        let modules: Vec<&str> = if invoked.contains("::") {
            vec![module]
        } else {
            ancestors(module).collect()
        };
        // `resolve_from_in` reads an import path: a bare 2015 import starts at
        // the crate root. A textual macro lookup instead starts in each module.
        let local;
        let path = if invoked.contains("::") {
            invoked
        } else {
            local = format!("self::{invoked}");
            &local
        };
        for at in modules {
            if let Ok(found) = source.resolve_from_in(at, path, Namespace::Macro) {
                return found.sites;
            }
        }
        Vec::new()
    }
    
    /// The paths and the modules a unit is matched against the exceptions by:
    /// the paths of an item, the self types and the modules of an impl block, and
    /// the modules of a macro invocation or a `const _`.
    fn match_paths(source: &Workspace, what: &What) -> (Vec<String>, Vec<String>) {
        match what {
            What::Item(ids) => (
                ids.iter()
                    .filter_map(|id| source.item(*id))
                    .map(|item| item.path.to_string())
                    .collect(),
                Vec::new(),
            ),
            What::Impl(ids) => {
                let blocks: Vec<&ImplBlock> =
                    ids.iter().filter_map(|id| source.impl_block(*id)).collect();
                (
                    blocks
                        .iter()
                        .flat_map(|block| &block.self_items)
                        .filter_map(|id| source.item(*id))
                        .map(|item| item.path.to_string())
                        .collect(),
                    blocks
                        .iter()
                        .map(|block| block.module.to_string())
                        .collect(),
                )
            }
            What::Module(module) => (
                Vec::new(),
                module
                    .copies
                    .iter()
                    .map(|(path, _, _)| path.clone())
                    .collect(),
            ),
        }
    }
    
    /// What the lines of the units add up to.
    #[derive(Default)]
    struct Totals {
        crates: BTreeMap<String, CrateLines>,
        modules: BTreeMap<(String, String), (usize, usize)>,
        orphans: BTreeMap<usize, Vec<usize>>,
        excepted: BTreeMap<(usize, usize), Vec<usize>>,
    }
    
    /// Judge every counted line of one file: covered when a covered unit holds
    /// it, excepted when an exception matches a unit that holds it or the file,
    /// and orphan otherwise, each under the innermost unit that holds it.
    fn sweep(units: &[Unit], range: std::ops::Range<usize>, file: &File<'_>, totals: &mut Totals) {
        let Some(last) = range.clone().map(|at| units[at].last).max() else {
            return;
        };
        let mut covered = vec![false; last + 1];
        let mut exception: Vec<Option<usize>> = vec![file.exception; last + 1];
        let mut inner: Vec<Option<usize>> = vec![None; last + 1];
        let mut order: Vec<usize> = range.collect();
        // Longer units first, so the innermost unit of a line is the last to mark
        // it, and of two units of one length the first in the file.
        order.sort_by(|a, b| {
            let (a, b) = (&units[*a], &units[*b]);
            (b.last - b.first, b.first, b.bytes.0).cmp(&(a.last - a.first, a.first, a.bytes.0))
        });
        for at in order {
            add(Work::LineUnits, 1);
            let unit = &units[at];
            for line in unit.first..=unit.last {
                add(Work::LineMarks, 1);
                covered[line] |= unit.covered;
                exception[line] = match (exception[line], unit.exception) {
                    (Some(a), Some(b)) => Some(a.min(b)),
                    (a, b) => a.or(b),
                };
                inner[line] = Some(at);
            }
        }
        for (line, holder) in inner.iter().enumerate() {
            let Some(at) = *holder else {
                continue;
            };
            let unit = &units[at];
            let row = totals
                .crates
                .entry(file.package.to_string())
                .or_insert_with(|| empty_row(file.package));
            let module = totals
                .modules
                .entry((file.package.to_string(), unit.module.clone()))
                .or_default();
            module.0 += 1;
            if unit.shipping {
                row.shipping_counted += 1;
            } else {
                row.test_counted += 1;
            }
            row.counted += 1;
            if covered[line] {
                row.covered += 1;
            } else if let Some(position) = exception[line] {
                row.excepted += 1;
                totals
                    .excepted
                    .entry((at, position))
                    .or_default()
                    .push(line);
            } else {
                module.1 += 1;
                if unit.shipping {
                    row.shipping_orphan += 1;
                } else {
                    row.test_orphan += 1;
                }
                totals.orphans.entry(at).or_default().push(line);
            }
        }
    }
    
    /// Set the debt of each package of `report`: for each item with orphan lines
    /// that a row of `record` names, the fewer of its lines and its row, counted
    /// for the package of the first of its units.
    fn hold_debt(report: &mut LinesReport, record: &raf_okf::DebtRecord) {
        let recorded: BTreeMap<&str, usize> = record
            .lines
            .iter()
            .map(|line| (line.item.as_str(), line.orphans))
            .collect();
        let mut present: BTreeMap<&str, (&str, usize)> = BTreeMap::new();
        for unit in &report.orphans {
            let entry = present
                .entry(unit.item.as_str())
                .or_insert((unit.package.as_str(), 0));
            entry.1 += unit.lines();
        }
        let mut held: BTreeMap<String, usize> = BTreeMap::new();
        for (item, (package, lines)) in present {
            if let Some(row) = recorded.get(item) {
                *held.entry(package.to_string()).or_default() += lines.min(*row);
            }
        }
        for row in &mut report.crates {
            row.debt = held.get(&row.package).copied().unwrap_or(0);
        }
    }
    
    /// The row of a package with no line.
    fn empty_row(package: &str) -> CrateLines {
        CrateLines {
            package: package.to_string(),
            counted: 0,
            covered: 0,
            excepted: 0,
            shipping_counted: 0,
            test_counted: 0,
            shipping_orphan: 0,
            test_orphan: 0,
            ceilings: Vec::new(),
            debt: 0,
        }
    }
    
    /// Sorted lines as merged ranges.
    fn ranges(lines: &[usize]) -> Vec<(usize, usize)> {
        let mut out: Vec<(usize, usize)> = Vec::new();
        for line in lines {
            match out.last_mut() {
                Some((_, last)) if *last + 1 == *line => *last = *line,
                _ => out.push((*line, *line)),
            }
        }
        out
    }
    
    /// The lines of the unit `unit` of the file `file`, as rows of the report.
    fn unit_lines(unit: &Unit, file: &File<'_>, lines: &[usize]) -> UnitLines {
        UnitLines {
            package: file.package.to_string(),
            module: unit.module.clone(),
            item: unit.item.clone(),
            kind: unit.kind,
            file: file.path.to_string(),
            ranges: ranges(lines),
        }
    }
    
    /// The order of the rows of units: package, module path, file, first line.
    fn unit_order(lines: &UnitLines) -> (&str, &str, &str, usize) {
        (
            &lines.package,
            &lines.module,
            &lines.file,
            lines.ranges.first().map_or(0, |range| range.0),
        )
    }
    
    /// The packages the report judges: every package of `source`.
    fn judged_packages(source: &Workspace) -> BTreeSet<&str> {
        source
            .packages()
            .iter()
            .map(|package| package.name.as_str())
            .collect()
    }
    
    /// The report of `plan`, with `extra` roots walked beside the bound items.
    pub(crate) fn compute(plan: &Workplan<'_>, extra: &[Node]) -> LinesReport {
        match judged(plan, extra) {
            Ok(report) => report,
            Err(error) => LinesReport {
                errors: vec![error],
                ..LinesReport::default()
            },
        }
    }
    
    fn judged(plan: &Workplan<'_>, extra: &[Node]) -> Result<LinesReport, String> {
        let source = plan.source();
        let record = plan.coverage().map(Loaded::value);
        let exceptions: &[Exception] = record.map_or(&[], |record| &record.exceptions);
        let judged = judged_packages(source);
        let mut units = collect(source, &judged)?;
        let mut roots = roots(plan);
        cover(source, &mut units, &mut roots, extra)?;
        let index = ExceptionIndex::new(exceptions);
        if !index.is_empty() {
            for unit in &mut units.units {
                let (paths, modules) = match_paths(source, &unit.what);
                let paths: Vec<&str> = paths.iter().map(String::as_str).collect();
                let modules: Vec<&str> = modules.iter().map(String::as_str).collect();
                unit.exception = index.unit(&paths, &modules);
            }
            for file in &mut units.files {
                file.exception = index.file(file.path.as_str());
            }
        }
        let mut totals = Totals::default();
        for (file, range) in units.files.iter().zip(&units.of_file) {
            sweep(&units.units, range.clone(), file, &mut totals);
        }
        let mut report = LinesReport::default();
        for package in source
            .packages()
            .iter()
            .filter(|package| judged.contains(package.name.as_str()))
        {
            let mut row = totals
                .crates
                .remove(&package.name)
                .unwrap_or_else(|| empty_row(&package.name));
            row.ceilings = record.map_or_else(Vec::new, |record| {
                record
                    .ceilings
                    .iter()
                    .filter(|ceiling| ceiling.package == package.name)
                    .cloned()
                    .collect()
            });
            report.crates.push(row);
        }
        report.crates.sort_by(|a, b| a.package.cmp(&b.package));
        report.modules = totals
            .modules
            .into_iter()
            .filter(|(_, (_, orphan))| *orphan > 0)
            .map(|((package, module), (counted, orphan))| ModuleLines {
                package,
                module,
                counted,
                orphan,
            })
            .collect();
        for (at, lines) in &totals.orphans {
            let unit = &units.units[*at];
            report
                .orphans
                .push(unit_lines(unit, &units.files[unit.file], lines));
        }
        report
            .orphans
            .sort_by(|a, b| unit_order(a).cmp(&unit_order(b)));
        if let Some(recorded) = plan.debt() {
            hold_debt(&mut report, recorded.value());
        }
        let mut excepted: Vec<(usize, ExceptedLines)> = Vec::new();
        for ((at, position), lines) in &totals.excepted {
            let unit = &units.units[*at];
            let exception = &exceptions[*position];
            excepted.push((
                *position,
                ExceptedLines {
                    lines: unit_lines(unit, &units.files[unit.file], lines),
                    exception: exception.target.written(),
                    reason: exception.reason.clone(),
                },
            ));
        }
        excepted.sort_by(|a, b| (unit_order(&a.1.lines), a.0).cmp(&(unit_order(&b.1.lines), b.0)));
        report.excepted = excepted.into_iter().map(|(_, row)| row).collect();
        Ok(report)
    }
    
    /// Whether the exact line count or percentage is within its ceiling.
    /// Comparing products avoids rounded division, including when no line is counted.
    fn within(orphan: usize, counted: usize, limit: &CeilingLimit) -> bool {
        use purrdf_xsd::exact::{Decimal, Integer};
        let count = |value: usize| Decimal::from_integer(Integer::from(value as u128));
        match limit {
            CeilingLimit::Lines(lines) => count(orphan).le(lines.exact()),
            CeilingLimit::Percent(percent) => {
                let left = count(orphan)
                    .try_mul(&Decimal::from(100))
                    .expect("integer product preserves decimal scale");
                let right = percent
                    .exact()
                    .try_mul(&count(counted))
                    .expect("integer product preserves decimal scale");
                left <= right
            }
        }
    }
    
    /// A package over a ceiling of the coverage record, with the finding that
    /// says so. A report that could not count the lines is one `Over` that names
    /// no package.
    #[derive(Debug, Clone)]
    pub(crate) struct Over {
        /// The package, or none for a report that could not count.
        pub(crate) package: Option<String>,
        /// `coverage-lines-over-ceiling` at `ceilings`.
        pub(crate) finding: Finding,
    }
    
    /// The report of the lines and the classes of the packages whose orphan lines
    /// are over their ceilings: each
    /// class of a package whose orphan lines are over its ceiling is
    /// `coverage-lines-over-ceiling` at `ceilings`. A package's shipping and test
    /// lines are each judged against the row of their class, held to 0 when no
    /// row of their class and no `all` row bounds them, and their sum against an
    /// `all` row. Without a coverage record that loads, or while a capability
    /// document does not load, nothing is judged: the record's own finding, or the
    /// document's, fails the check, and a document that did not load binds
    /// nothing. A report that could not count the lines is one finding.
    pub(crate) fn over_ceilings(plan: &Workplan<'_>) -> Option<(LinesReport, Vec<Over>)> {
        let loaded = plan.coverage()?;
        let capability_unloaded = plan
            .unloaded()
            .iter()
            .any(|id| id.starts_with(raf_okf::DocType::Capability.id_prefix()));
        if capability_unloaded {
            return None;
        }
        let report = plan.lines_report();
        let overs = ceiling_findings(loaded, &report);
        Some((report, overs))
    }
    
    /// The classes over their ceilings in `report`, at the record `loaded`.
    pub(crate) fn ceiling_findings(
        loaded: &Loaded<raf_okf::CoverageRecord>,
        report: &LinesReport,
    ) -> Vec<Over> {
        let finding = |message: String| {
            Finding::new(
                kind::COVERAGE_LINES_OVER_CEILING,
                loaded.path(),
                Location::Key("ceilings".to_string()),
                message,
                "bind or reach the items that explain the orphan lines, add an exception with its reason, raise the crate's ceiling, or hold the lines as debt with `cargo workplan debt --write`",
            )
        };
        if let Some(error) = report.errors().first() {
            return vec![Over {
                package: None,
                finding: finding(format!(
                    "the orphan lines of the workspace cannot be counted, so no package is judged under its ceiling: {error}"
                )),
            }];
        }
        let mut out = Vec::new();
        for row in report.crates() {
            let class = |of: CeilingClass| row.ceilings.iter().find(|ceiling| ceiling.of == of);
            let all = class(CeilingClass::All);
            let mut judge =
                |of: CeilingClass, orphan: usize, counted: usize, ceiling: Option<&Ceiling>| {
                    let held = ceiling.map_or(orphan == 0, |ceiling| {
                        within(orphan, counted, &ceiling.limit)
                    });
                    if !held {
                        out.push(Over {
                            package: Some(row.package.clone()),
                            finding: finding(over_message(report, row, of, orphan, counted, ceiling)),
                        });
                    }
                };
            for (of, orphan, counted) in [
                (
                    CeilingClass::Shipping,
                    row.shipping_orphan,
                    row.shipping_counted,
                ),
                (CeilingClass::Test, row.test_orphan, row.test_counted),
            ] {
                match class(of) {
                    Some(ceiling) => judge(of, orphan, counted, Some(ceiling)),
                    None if all.is_none() => judge(of, orphan, counted, None),
                    None => {}
                }
            }
            if let Some(ceiling) = all {
                judge(CeilingClass::All, row.orphan(), row.counted, Some(ceiling));
            }
        }
        out
    }
    
    /// The message of a class of a package over its ceiling.
    fn over_message(
        report: &LinesReport,
        row: &CrateLines,
        of: CeilingClass,
        orphan: usize,
        counted: usize,
        ceiling: Option<&Ceiling>,
    ) -> String {
        let lines = match of {
            CeilingClass::Shipping => "shipping orphan lines",
            CeilingClass::Test => "test orphan lines",
            CeilingClass::All => "orphan lines",
        };
        let bound = ceiling.map_or_else(
            || "the default of 0 lines, since no row of `ceilings` bounds them".to_string(),
            |ceiling| format!("its ceiling of {}", ceiling_text(ceiling)),
        );
        let mut largest: Vec<&UnitLines> = report
            .orphans()
            .iter()
            .filter(|unit| unit.package == row.package)
            .collect();
        largest.sort_by(|a, b| b.lines().cmp(&a.lines()).then(a.item.cmp(&b.item)));
        let named: Vec<String> = largest
            .iter()
            .take(3)
            .map(|unit| format!("`{}` ({})", unit.item, unit.lines()))
            .collect();
        format!(
            "`{}` has {} orphan {} ({} shipping, {} test) of {} counted, and its {lines}, {orphan} of {counted} counted, are over {bound}; the largest orphan items are {}",
            row.package,
            row.orphan(),
            if row.orphan() == 1 { "line" } else { "lines" },
            row.shipping_orphan,
            row.test_orphan,
            row.counted,
            named.join(", ")
        )
    }

    //! The work of the line report, counted and never timed, the exact
    //! comparison of a percent ceiling, the paths that may name what a macro
    //! generates, and a walk that fails.

    //! Each unit is swept once over its lines and looks up its own paths and the

    //! the work grows with the units plus the exceptions and not with their
    //! product.

    /// The functions of a module of the crate `big`.
    const FUNCTIONS: usize = 10;
    /// The modules of one unit of units.
    const MODULES_PER_UNIT: usize = 10;
    /// The exceptions of one unit of exceptions.
    const EXCEPTIONS_PER_UNIT: usize = 20;
    /// What one unit of the report, or one exception, may cost at most: a sweep,
    /// one mark for its one line and a lookup for each segment of its path and
    /// each module above it.
    const PER_ENTRY: usize = 8;
    
    /// The work of the report of a crate of `modules` modules of ten private
    /// functions beside a bound `run`, under `exceptions` exceptions of items
    /// that are no units, with every kind of work held to `ceiling`; and the
    /// units the report swept.
    fn work(modules: usize, exceptions: usize, ceiling: usize) -> (usize, usize) {
        let mut lib = String::from("pub fn run() {}\n");
        for m in 0..modules {
            let _ = writeln!(lib, "mod m{m} {{");
            for f in 0..FUNCTIONS {
                let _ = writeln!(lib, "    fn f{f}() {{}}");
            }
            let _ = writeln!(lib, "}}");
        }
        let source = Workspace::load(
            &MemoryTree::from([
                (
                    "Cargo.toml",
                    "[package]\nname = \"big\"\nversion = \"0.1.0\"\nedition = \"2021\"\n",
                ),
                ("src/lib.rs", lib.as_str()),
            ]),
            &["."],
        )
        .expect("a workspace");
        let mut record =
            String::from("---\ntype: Coverage\nid: coverage:default\nscope: []\nexclusions: []\n");
        if exceptions > 0 {
            record.push_str("exceptions:\n");
            for e in 0..exceptions {
                let _ = write!(
                    record,
                    "  - item: big::m0::f0::x{e}\n    reason: \"Nothing of it.\"\n"
                );
            }
        }
        record.push_str("---\n");

    fn a_percent_is_compared_as_whole_numbers_without_division_or_overflow() {
        let percent = |text: &str| CeilingLimit::Percent(Number::parse(text).expect("a number"));
        // 1 of 3 is 33.33...%.
        assert!(within(1, 3, &percent("33.334")));
        assert!(!within(1, 3, &percent("33.333")));
        assert!(within(1, 3, &percent("33.3340000")));
        assert!(within(0, 0, &percent("0")));
        assert!(!within(1, 0, &percent("100")));
        // A fraction too long for any machine word compares exactly.
        let long = format!("33.{}4", "3".repeat(80));
        assert!(within(1, 3, &percent(&long)));
        let short = format!("33.{}", "3".repeat(80));
        assert!(!within(1, 3, &percent(&short)));
        assert!(within(usize::MAX, usize::MAX, &percent("100")));
        assert!(within(1, 8, &percent("12.5")));

        assert!(within(0, 3, &percent("-0.000")));
        assert!(!within(1, 3, &percent("-0.000")));
        let lines = |text: &str| CeilingLimit::Lines(Number::parse(text).expect("a number"));
        assert!(within(3, 9, &lines("3")));
        assert!(!within(4, 9, &lines("3")));
        assert!(within(4, 9, &lines("99999999999999999999999999")));

    fn excluded_of(report: &CoverageReport, kind: GapKind) -> impl Iterator<Item = &Excluded> {
        report
            .excluded()
            .iter()
            .filter(move |excluded| excluded.gap().kind().is_code() == kind.is_code())
    }

        if kind.is_code() { report.errors() } else { &[] }
    }
    
    /// One direction of the report as text: the gaps, then the exclusion hits and
    /// the surfaces that could not be listed when there are any.
    #[must_use]
    pub fn table(report: &CoverageReport, kind: GapKind) -> String {
        let mut out = String::from("| Gap | Crate | Path | Location |\n| --- | --- | --- | --- |\n");
        for gap in report.gaps(kind) {
            out.push_str(&gap_row(gap));
            out.push('\n');
        }
        let mut excluded = excluded_of(report, kind).peekable();
        if excluded.peek().is_some() {
            out.push_str(
                "\n| Excluded | Crate | Path | Location | Exclusion | Reason |\n| --- | --- | --- | --- | --- | --- |\n",
            );
            for hit in excluded {
                out.push_str(&excluded_row(hit));
                out.push('\n');
            }
        }
        let errors = errors_of(report, kind);
        if !errors.is_empty() {
            out.push_str("\n| Crate | Target | Error |\n| --- | --- | --- |\n");
            for error in errors {
                out.push_str(&error_row(error));
                out.push('\n');
            }
        }
        out
    }

    /// The reverse coverage as text: the crate, module and item tables, then the
    /// excepted lines and the errors when there are any.
    ///
    /// ```rust
    /// use raf_workplan::coverage::lines::LinesReport;
    /// use raf_workplan::coverage::render;
    ///
    /// assert_eq!(
    ///     render::lines_table(&LinesReport::default()),
    ///     "| Crate | Counted | Covered | Excepted | Orphan | Shipping orphan | Test orphan | Ceiling | Debt |\n\
    ///      | --- | --- | --- | --- | --- | --- | --- | --- | --- |\n\n\
    ///      | Crate | Module | Counted | Orphan |\n| --- | --- | --- | --- |\n\n\
    ///      | Crate | Item | Kind | File | Orphan lines |\n| --- | --- | --- | --- | --- |\n"
    /// );
    /// ```
    #[must_use]
    pub fn lines_table(report: &LinesReport) -> String {
        let mut out = String::from(
            "| Crate | Counted | Covered | Excepted | Orphan | Shipping orphan | Test orphan | Ceiling | Debt |\n| --- | --- | --- | --- | --- | --- | --- | --- | --- |\n",
        );
        for row in report.crates() {
            out.push_str(&crate_row(row));
            out.push('\n');
        }
        out.push_str("\n| Crate | Module | Counted | Orphan |\n| --- | --- | --- | --- |\n");
        for row in report.modules() {
            out.push_str(&module_row(row));
            out.push('\n');
        }
        out.push_str(
            "\n| Crate | Item | Kind | File | Orphan lines |\n| --- | --- | --- | --- | --- |\n",
        );
        for unit in report.orphans() {
            out.push_str(&unit_cells(unit));
            out.push('\n');
        }
        if !report.excepted().is_empty() {
            out.push_str(
                "\n| Crate | Item | Kind | File | Excepted lines | Exception | Reason |\n| --- | --- | --- | --- | --- | --- | --- |\n",
            );
            for row in report.excepted() {
                out.push_str(&excepted_row(row));
                out.push('\n');
            }
        }
        if !report.errors().is_empty() {
            out.push_str("\n| Error |\n| --- |\n");
            for error in report.errors() {
                out.push_str("| ");
                out.push_str(&table_cell(error));
                out.push_str(" |\n");
            }
        }
        out
    }

    fn collect_terms(graph: &raf_rdf::Graph, terms: &mut std::collections::BTreeSet<String>) {
        fn iri(value: &str, terms: &mut std::collections::BTreeSet<String>) {
            if let Some(name) = value.strip_prefix(raf_rdf::vocab::REPORT) {
                terms.insert(name.into());
            }
        }
        for t in graph.assertions() {
            for atom in [&t.subject, &t.predicate, &t.object] {
                let _ = atom.visit_terms(|v| {
                    match v {
                        TermValue::Iri(v) => iri(v, terms),
                        TermValue::Literal { datatype, .. } => iri(datatype, terms),
                        _ => {}
                    }
                    std::ops::ControlFlow::<()>::Continue(())
                });
            }
        }
    }

    //! The work of judging coverage, counted, and the index of the exclusions held
    //! to the reading of an exclusion it stands for.
    //!
    //! Every public item and every test of a crate is judged once. The exclusions
    //! of the record are put into an index once, and judging an entry looks up each
    //! of its paths and each module above them, so the work grows with the entries
    //! plus the exclusions. The index finds every exclusion that matches an entry,
    //! not only the first: each is marked, so an exclusion that matches nothing is
    //! reported, and the first in the order of the record gives the reason. The
    //! reading the index stands for is the one stated here as `oracle`: each
    //! exclusion of the record tested against each path of the entry.

    thread_local! {
        /// Exact attempts at the real surface operation, bounded by the phase's
        /// expected target count. Other tests do not retain a receipt.
        static LISTINGS: RefCell<Option<(usize, Vec<TargetId>)>> = const { RefCell::new(None) };
    }
    
    /// Observe the operation immediately before its real source-index call.
    pub(super) fn listing(id: TargetId) {
        LISTINGS.with(|slot| {
            if let Some((limit, calls)) = slot.borrow_mut().as_mut() {
                assert!(calls.len() < *limit, "surface phase exceeded {limit} calls");
                calls.push(id);
            }
        });
    }
    
    fn listed<T>(source: &Workspace, packages: &[&str], run: impl FnOnce() -> T) -> T {
        struct Clear;
        impl Drop for Clear {
            fn drop(&mut self) {
                LISTINGS.with(|slot| *slot.borrow_mut() = None);
            }
        }
        let mut expected: Vec<_> = packages
            .iter()
            .flat_map(|package| {
                [
                    format!("crates/{package}/src/lib.rs"),
                    format!("crates/{package}/src/main.rs"),
                ]
            })
            .collect();
        expected.sort();
        LISTINGS.with(|slot| {
            assert!(slot.borrow().is_none(), "surface receipts do not nest");
            *slot.borrow_mut() = Some((expected.len(), Vec::new()));
        });
        let _clear = Clear;
        let value = run();
        let calls = LISTINGS.with(|slot| slot.borrow_mut().take().expect("active receipt").1);
        let mut actual: Vec<_> = calls
            .into_iter()
            .map(|id| source.target(id).expect("real target").root.to_string())
            .collect();
        actual.sort();
        assert_eq!(actual, expected);
        value
    }

    /// The functions of a module of the crate `big`.
    const FUNCTIONS: usize = 10;
    /// The tests of the crate `big`, `TESTS_PER_MODULE` for each of `TEST_MODULES`.
    const TESTS_PER_MODULE: usize = 10;
    const TEST_MODULES: usize = 12;
    /// The public items of the module count one each, and the module itself one.
    const ENTRIES_PER_MODULE: usize = FUNCTIONS + 1;
    /// The modules of one unit of entries.
    const MODULES_PER_UNIT: usize = 10;
    /// The exclusions of one unit of exclusions.
    const EXCLUSIONS_PER_UNIT: usize = 20;
    /// The paths an entry of `big` is matched by: its exported path and its
    /// defining path, which are the same there.
    const PATHS_PER_ENTRY: usize = 2;
    /// The most segments of a path of an entry of `big`: `big::test_crate::it::t0_0`.
    const DEEPEST: usize = 4;
    /// The most lookups judging one entry of `big` makes: one for each segment of
    /// each of its paths.
    const LOOKUPS_PER_ENTRY: usize = PATHS_PER_ENTRY * DEEPEST;
    
    /// What judging counted, and what the report said.
    struct Judged {
        entries: usize,
        indexed: usize,
        lookups: usize,
        marks: usize,
        total: usize,
        reported: usize,
        excluded: usize,
    }
    
    /// The names of the tests of `big`, the first `count` of them in the order of
    /// the exclusions.
    fn test_names(count: usize) -> Vec<String> {
        (0..TEST_MODULES)
            .flat_map(|m| (0..TESTS_PER_MODULE).map(move |i| format!("big::test_crate::it::t{m}_{i}")))
            .take(count)
            .collect()
    }
    
    /// The paths of the entries of `big` with `modules` modules: each module, each
    /// of its functions, and each test.
    fn entry_paths(modules: usize) -> Vec<String> {
        let mut paths = Vec::new();
        for m in 0..modules {
            paths.push(format!("big::m{m}"));
            paths.extend((0..FUNCTIONS).map(|i| format!("big::m{m}::f{i}")));
        }
        paths.extend(test_names(TEST_MODULES * TESTS_PER_MODULE));
        paths
    }
    
    /// The entries of `big` with `modules` modules.
    const fn entries_of(modules: usize) -> usize {
        modules * ENTRIES_PER_MODULE + TEST_MODULES * TESTS_PER_MODULE
    }
    
    /// A ceiling linear in the entries plus the exclusions: what an index may
    /// count of any kind of work, and less than what testing each exclusion
    /// against each entry counts once both are more than a few.
    const fn linear(entries: usize, exclusions: usize) -> usize {
        LOOKUPS_PER_ENTRY * (entries + exclusions)
    }

    /// `package` with an exclusion of each of `exclusions`.
    fn documents(package: &str, code: &[String], exclusions: &[String]) -> MemoryTree {

    /// Judge a crate of `modules` modules of ten public functions and
    /// `TEST_MODULES * TESTS_PER_MODULE` tests, half the modules bound by a
    /// capability and `exclusions` of the tests excluded by name, one exclusion
    /// each, with every kind of work held to `ceiling`; and the work counted.
    fn judged(modules: usize, exclusions: usize, ceiling: usize) -> Judged {
        let mut lib = String::new();
        for m in 0..modules {
            let _ = writeln!(lib, "pub mod m{m} {{");
            for i in 0..FUNCTIONS {
                let _ = writeln!(lib, "    pub fn f{i}() {{}}");
            }
            let _ = writeln!(lib, "}}");
        }
        let mut tests = String::new();
        for m in 0..TEST_MODULES {
            for i in 0..TESTS_PER_MODULE {
                let _ = writeln!(tests, "#[test]\nfn t{m}_{i}() {{}}");
            }
        }
        let source = Workspace::load(
            &MemoryTree::from([
                (
                    "crates/big/Cargo.toml",
                    "[package]\nname = \"big\"\nversion = \"0.1.0\"\n",
                ),
                ("crates/big/src/lib.rs", lib.as_str()),
                ("crates/big/tests/it.rs", tests.as_str()),
            ]),
            &["crates"],
        )
        .expect("a workspace");
        let bound: Vec<String> = (0..modules / 2).map(|m| format!("big::m{m}")).collect();
        let names = test_names(exclusions);
        // Every exclusion asked for is in the record.
        assert_eq!(names.len(), exclusions);
        let docs = documents("big", &bound, &names);
        let plan = Workplan::load(&docs, &source);
    
        let (report, tally) = measure(ceiling, || plan.coverage_report());
        let big = report.package("big").expect("big");
        let counts = &big.packages()[0];
        // The count is of what the report says as well as of what is ticked.
        assert_eq!(counts.items(), modules * ENTRIES_PER_MODULE);
        assert_eq!(counts.tests(), TEST_MODULES * TESTS_PER_MODULE);
        assert_eq!(counts.covered_items(), modules / 2 * ENTRIES_PER_MODULE);
        Judged {
            entries: tally.get(Work::EntriesJudged),
            indexed: tally.get(Work::ExclusionsIndexed),
            lookups: tally.get(Work::IndexLookups),
            marks: tally.get(Work::ExclusionMarks),
            total: tally.total(),
            reported: big.code().len() + big.tests().len(),
            excluded: big.excluded().len(),
        }
    }
    
    /// Each exclusion is put into the index once, each entry looks up each of its
    /// paths and each module above them, and that is the same for ten times the
    /// exclusions: the lookups do not depend on how many exclusions there are.
    #[test]
    fn the_exclusions_are_indexed_once_and_each_entry_looks_up_its_paths_and_their_modules() {
        let modules = 4;
        let entries = entries_of(modules);
        let lookups: usize = entry_paths(modules)
            .iter()
            .map(|path| PATHS_PER_ENTRY * path.split("::").count())
            .sum();
        let mut seen = Vec::new();
        for exclusions in [12, 120] {
            let work = judged(modules, exclusions, linear(entries, exclusions));
            assert_eq!(work.entries, entries, "{exclusions}");
            assert_eq!(work.indexed, exclusions, "{exclusions}");
            assert_eq!(work.lookups, lookups, "{exclusions}");
            // Each excluded test is found once by each of its two paths.
            assert_eq!(work.marks, PATHS_PER_ENTRY * exclusions, "{exclusions}");
            // Judging reports what it judged: the unbound half of the items and
            // the tests that no exclusion takes, and the excluded tests.
            assert_eq!(work.excluded, exclusions, "{exclusions}");
            assert_eq!(
                work.reported,
                modules / 2 * ENTRIES_PER_MODULE + TEST_MODULES * TESTS_PER_MODULE - exclusions,
                "{exclusions}"
            );
            seen.push(work.lookups);
        }
        assert_eq!(seen[0], seen[1]);
    }

    /// Whether `exclusion` matches an entry with these paths, read off the
    /// exclusion: an item path equal to one of them, or a prefix one of them is
    /// below.
    fn oracle_matches(exclusion: &Exclusion, paths: &[&str]) -> bool {
        match &exclusion.item {
            ExclusionItem::Item(item) => paths.iter().any(|path| path == item),
            ExclusionItem::Prefix(prefix) => paths.iter().any(|path| {
                path.strip_prefix(prefix.as_str())
                    .is_some_and(|rest| rest.starts_with("::"))
            }),
        }
    }
    
    /// The position of the first exclusion that matches the paths, each
    /// exclusion tested against them in the order of the record, marking every
    /// one that does.
    fn oracle(exclusions: &[Exclusion], matched: &mut [bool], paths: &[&str]) -> Option<usize> {
        let mut first = None;
        for (position, (exclusion, hit)) in exclusions.iter().zip(matched.iter_mut()).enumerate() {
            if oracle_matches(exclusion, paths) {
                *hit = true;
                first.get_or_insert(position);
            }
        }
        first
    }
    
    /// The position in `exclusions` of the exclusion the index gives.
    fn position_of(exclusions: &[Exclusion], found: Option<&Exclusion>) -> Option<usize> {
        found.map(|found| {
            exclusions
                .iter()
                .position(|exclusion| std::ptr::eq(exclusion, found))
                .expect("the index gives an exclusion of the record")
        })
    }
    
    /// Each path of one to four segments of `a`, `b` and `bc`, so that `a::b` and
    /// `a::bc` share their text, and every path has a sibling, a parent and a
    /// child.
    fn alphabet_paths() -> Vec<String> {
        let mut paths: Vec<String> = Vec::new();
        let mut last = vec![String::new()];
        for _ in 0..4 {
            last = last
                .iter()
                .flat_map(|head| {

    /// Every exclusion of the alphabet: an item and a prefix of each path, so a
    /// prefix equal to a whole path, a prefix longer than a path, prefixes shared
    /// by several paths and an exclusion that is a prefix of another are all here.
    fn alphabet_exclusions() -> Vec<Exclusion> {
        alphabet_paths()
            .into_iter()
            .flat_map(|path| {
                [
                    Exclusion::new(ExclusionItem::Item(path.clone()), "Item."),
                    Exclusion::new(ExclusionItem::Prefix(path), "Prefix."),
                ]
            })
            .collect()
    }
    
    /// For each entry of `entries`, alone and one after another, the index of
    /// `exclusions` gives the first match of the oracle and marks what it marks.
    fn agrees(exclusions: &[Exclusion], entries: &[Vec<&str>]) {
        let mut alone = ExclusionIndex::new(exclusions);
        let mut together = ExclusionIndex::new(exclusions);
        let mut marked = vec![false; exclusions.len()];
        for paths in entries {
            alone.matched.fill(false);
            let mut expected = vec![false; exclusions.len()];
            let first = oracle(exclusions, &mut expected, paths);
            assert_eq!(
                position_of(exclusions, alone.exclusion_for(paths)),
                first,
                "{paths:?}"
            );
            assert_eq!(alone.matched, expected, "{paths:?}");
            assert_eq!(
                position_of(exclusions, together.exclusion_for(paths)),
                oracle(exclusions, &mut marked, paths),
                "{paths:?}"
            );
        }
        assert_eq!(together.matched, marked);
    }
    
    /// The index finds what testing each exclusion against each path finds, on
    /// every exclusion of the alphabet against every entry of one or two of its
    /// paths: the same first match and the same exclusions marked, in the order of
    /// the record and in the reverse order, alone and in one list, and the same on
    /// an index built again.
    #[test]
    fn the_index_finds_the_first_match_and_marks_what_testing_every_exclusion_finds() {
        let paths = alphabet_paths();
        let mut entries: Vec<Vec<&str>> = paths.iter().map(|path| vec![path.as_str()]).collect();
        for one in &paths {
            for two in &paths {
                entries.push(vec![one.as_str(), two.as_str()]);
            }
        }
        let exclusions = alphabet_exclusions();
        for exclusion in &exclusions {
            agrees(std::slice::from_ref(exclusion), &entries[..paths.len()]);
        }
        let mut reversed = exclusions.clone();
        reversed.reverse();
        for list in [&exclusions, &reversed] {
            // Twice: each index is built with its own hash state.
            agrees(list, &entries);
            agrees(list, &entries);
        }
    }
    
    /// Two equal exclusions, which a record that loads never holds, are both
    /// kept and both marked, and the first of them is the match.
    #[test]
    fn equal_exclusions_on_a_hand_built_list_are_both_marked_and_the_first_is_the_match() {
        let exclusions = [

        assert_eq!(position_of(&exclusions, found), Some(0));
        assert_eq!(index.matched, [true; 4]);
        agrees(
            &exclusions,
            &[vec!["a::b"], vec!["a"], vec!["a::bc", "a::b"]],
        );
    }
