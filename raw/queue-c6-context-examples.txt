    Examples
    
    These two functions are one algorithm in the value form, and the census puts them
    in one `body` group:
    
    ```rust
    fn total(items: &[u64]) -> u64 {
        let mut sum = 0;
        for item in items.iter() {
            sum += *item;
            if sum > 1_000_000 { sum = 1_000_000; }
        }
        sum
    }
    
    fn add_up(values: &[u64]) -> u64 {
        let mut acc: u64 = 0;
        for v in values.iter() {
            acc += *v;
            if acc > 1_000_000 { acc = 1_000_000; }
        }
        acc
    }
    ```
    
    The `baseline` example of the crate prints both reports of a built-in workspace
    (`cargo run -p raf-census --example baseline`); given a directory it prints the
    reports of that directory, with no jobs. The workspace has one body group (`alpha`
    and `beta`), one shim group (`one` and `two`, forwarders to `helper`), a
    hexadecimal digit table, the constant 65536 and one job `sum` whose home is
    `alpha` and which forbids the body group's fingerprint and the constant 65536. The
    baseline report:

symbol raf_census::Census::index (bound by the capability; the section names no bound symbol by a path)
  in crates/raf-census/src/report/mod.rs
    /// The index report of the loaded workspace, with the jobs the caller
        /// matched through [`Census::match_jobs`].
        ///
        /// # Errors
        /// Refuses duplicate job ids.
        pub fn index(&self, jobs: &[JobReport]) -> Result<Index, Error> {
            let mut units: Vec<IndexUnit> = self
                .units()
                .iter()
                .map(|unit| IndexUnit {
                    symbol: unit.symbol.clone(),
                    package: unit.package.clone(),
                    file: unit.file.clone(),
                    line: unit.line,
                    start_byte: unit.start_byte,
                    shipping: unit.shipping,
                    tokens: unit.print.as_ref().map_or(0, |print| print.tokens),
                    body: unit
                        .print
                        .as_ref()
                        .and_then(|print| print.body.as_ref())
                        .map(|hashed| hashed.fingerprint.clone()),
                    shim: unit.shim_key.clone(),
                })
                .collect();
            units.sort_by(|a, b| (&a.symbol, &a.file, a.line).cmp(&(&b.symbol, &b.file, b.line)));
            let mut report = Index {
                normal_form: NORMAL_FORM,
                units,
                groups: self
                    .groups()
                    .iter()
                    .map(|group| IndexGroup {
                        tokens: group.tokens,
                        kind: group.kind,
                        fingerprint: group.fingerprint.clone(),
                        members: rows(&group.members),
                    })
                    .collect(),
                jobs: jobs
                    .iter()
                    .map(|job| JobRow {
                        id: job.id.clone(),
                        home: job.home.clone(),
                        matches: job
                            .matches
                            .iter()
                            .map(|found| MatchRow {
                                symbol: found.symbol.clone(),
                                file: found.file.clone(),
                                line: found.line,
                                start_byte: found.start_byte,
                                class: found.class,
                            })
                            .collect(),
                        copies: job.copies,
                    })
                    .collect(),
            };
            report.jobs.sort_by(|a, b| a.id.cmp(&b.id));
            if let Some(pair) = report.jobs.windows(2).find(|pair| pair[0].id == pair[1].id) {
                return Err(ReportError::DuplicateJob(pair[0].id.clone()).into());
            }
            Ok(report)
        }
symbol raf_census::Census::load (bound by the capability; the section names no bound symbol by a path)
  in crates/raf-census/src/census.rs
    /// Loads the workspace under `roots` of `tree` and collects its units.
        ///
        /// `options.max_file_bytes` is the cap on one source file: a larger file
        /// is a load error that names it.
        ///
        /// # Errors
        ///
        /// [`Error::Load`] when the workspace does not load, and [`Error::Source`]
        /// when a body does not lex or a signature or an initialiser does not
        /// parse or the source index is inconsistent, and [`Error::Collision`] when two different forms have one
        /// fingerprint. Nothing falls back to a partial result.
        pub fn load(tree: &impl Tree, roots: &[&str], options: &CensusOptions) -> Result<Self, Error> {
            let mut load = LoadOptions::default();
            load.max_file_bytes = options.max_file_bytes;
            let workspace = Workspace::load_with(tree, roots, &load)?;
            let units = units::collect(&workspace, options)?;
            let grouping = group::group(&units)?;
            let files = workspace
                .files()
                .map(|file| {
                    let target_kinds = file
                        .reached
                        .iter()
                        .map(|reach| {
                            workspace
                                .target(reach.target)
                                .map(|target| target.kind)
                                .ok_or_else(|| Error::Source {
                                    file: file.path.to_string(),
                                    line: 1,
                                    reason: "a target that reaches the file does not resolve"
                                        .to_owned(),
                                })
                        })
                        .collect::<Result<_, _>>()?;
                    Ok(FileCtx {
                        package: file.package.clone(),
                        path: file.path.to_string(),
                        shipping: file.shipping,
                        target_kinds,
                    })
                })
                .collect::<Result<_, Error>>()?;
            Ok(Self {
                options: options.clone(),
                units,
                files,
                grouping,
            })
        }
symbol raf_census::Census::match_jobs (bound by the capability; the section names no bound symbol by a path)
  in crates/raf-census/src/job.rs
    /// Matches every job against the units: a unit matches when its `body`
        /// fingerprint, its shim key, or the `table:` id of a table it holds, is
        /// among the job's forbidden fingerprints, or one of its integer constants
        /// equals a forbidden constant by value. A shim id matches the units that
        /// group under it: a method of a trait implementation and a forwarder
        /// that calls through `Self::` have no shim key and never match one. One report per job, in the order given.
        #[must_use]
        pub fn match_jobs(&self, jobs: &[Job]) -> Vec<JobReport> {
            jobs.iter()
                .map(|job| {
                    let mut matches: Vec<JobMatch> = self
                        .units()
                        .iter()
                        .filter(|unit| job.matches(unit))
                        .map(|unit| JobMatch {
                            symbol: unit.symbol.clone(),
                            file: unit.file.clone(),
                            line: unit.line,
                            start_byte: unit.start_byte,
                            class: job.classify(&unit.symbol),
                        })
                        .collect();
                    matches.sort_by(|a, b| {
                        (&a.symbol, &a.file, a.line).cmp(&(&b.symbol, &b.file, b.line))
                    });
                    let copies = matches
                        .iter()
                        .filter(|found| found.class == MatchClass::Copy)
                        .count();
                    JobReport {
                        id: job.id.clone(),
                        home: job.home.clone(),
                        matches,
                        copies,
                    }
                })
                .collect()
        }

symbol raf_census::Normalizer::explain (bound by the capability; the section names no bound symbol by a path)
  in crates/raf-census/src/normalize/mod.rs
    /// Compares two slices level by level and names where they part.
        ///
        /// # Errors
        ///
        /// [`Error::Source`] as [`Tower::new`], for either slice.
        pub fn explain(&self, a: &Slice<'_>, b: &Slice<'_>) -> Result<Divergence, Error> {
            let (a, b) = (Tower::new(a)?, Tower::new(b)?);
            let same = |level: Level| a.form(level).texts() == b.form(level).texts();
            let differs_through = Level::ALL.into_iter().rev().find(|level| !same(*level));
            let equal_from = Level::ALL
                .into_iter()
                .rev()
                .take_while(|level| same(*level))
                .last();
            let first = differs_through.and_then(|level| {
                let (left, right) = (a.form(level).tokens(), b.form(level).tokens());
                (0..left.len().max(right.len()))
                    .find(|index| {
                        left.get(*index).map(|t| &t.text) != right.get(*index).map(|t| &t.text)
                    })
                    .map(|index| TokenDiff {
                        index,
                        a: left.get(index).cloned(),
                        b: right.get(index).cloned(),
                    })
            });
            Ok(Divergence {
                differs_through,
                equal_from,
                first,
            })
        }

symbol raf_census::Tower::new (bound by the capability; the section names no bound symbol by a path)
  in crates/raf-census/src/normalize/mod.rs
    /// Lexes `slice` and applies the passes in order.
        ///
        /// # Errors
        ///
        /// [`Error::Source`] when the slice does not lex or nests delimiters
        /// deeper than [`MAX_NESTING`].
        pub fn new(slice: &Slice<'_>) -> Result<Self, Error> {
            if u32::try_from(slice.text.len()).map_or(true, |len| len > u32::MAX / 2) {
                return Err(Error::Source {
                    file: slice.file.to_owned(),
                    line: slice.first_line,
                    reason: "the source slice is larger than 2 GiB".to_owned(),
                });
            }
            let verbatim = Form::lex(slice.file, slice.first_line, slice.text)?;
            let shim_candidate = shim::candidate(&verbatim);
            let shim = passes::rename_params(&verbatim, &slice.params);
            let literal_tier = passes::rename_locals(&shim);
            let value_tokens = passes::held_len(&literal_tier);
            let structural = passes::kind_literals(&literal_tier, value_tokens < LITERAL_TIER);
            let value = passes::erase_held(&structural);
            Ok(Self {
                forms: [verbatim, shim, literal_tier, structural, value],
                value_tokens,
                shim_candidate,
            })
        }

    /// The fingerprints this slice has under `min_tokens`.
        #[must_use]
        pub fn print(&self, min_tokens: usize) -> Print {
            let tokens = self.value_tokens();
            let named = |level: Level, make: fn(&[u8; 32]) -> Fingerprint| {
                let digest = fingerprint::digest(self.form(level));
                Hashed {
                    fingerprint: make(&digest),
                    digest,
                }
            };
            let shim = (self.shim_candidate && (MIN_SHIM_TOKENS..SHIM_MAX_TOKENS).contains(&tokens))
                .then(|| named(Level::Shim, Fingerprint::shim));
            let body = (tokens >= min_tokens).then(|| named(Level::Value, Fingerprint::body));
            Print { tokens, body, shim }
        }
