    /// Write the debt record as [`Gate::write_debt`] does, with the SPDX
        /// lines of `options` when the record is created or replaced and `options`
        /// gives some, and the lines the record has otherwise.
        ///
        /// # Errors
        ///
        /// The errors of [`Gate::write_debt`] other than opening the repository.
        pub fn write_debt(&self, options: &RecordOptions) -> Result<DebtWrite, DebtError> {
            let plan = self.workplan();
            let judged = judge(&plan);
            let mut blocking = plan.base_findings();
            blocking.extend(judged.unheld.iter().cloned());
            if !blocking.is_empty() {
                return Err(DebtError::Findings(blocking));
            }
            if plan.debt().is_some() && !judged.growth.is_empty() {
                return Err(DebtError::Growth(judged.growth));
            }
            let debt = DebtRecord::new(
                judged
                    .findings
                    .iter()
                    .map(|key| DebtFinding::new(key.kind, key.subject.as_str()))
                    .collect(),
                judged
                    .lines
                    .iter()
                    .map(|(item, count)| DebtLine::new(item.as_str(), *count))
                    .collect(),
            );
            let (findings, lines) = (debt.findings.len(), debt.lines.len());
            let record = Record::Debt(debt);
            let options = kept(&plan, options);
            let text = record.text(&options)?;
            let path = record.path()?;
            let change = match plan.debt() {
                Some(existing) if existing.text() == text => DebtChange::Unchanged,
                Some(_) => {
                    record::replace(self.dir(), &record, &options)?;
                    DebtChange::Replaced
                }
                None => {
                    record::create(self.dir(), &record, &options)?;
                    DebtChange::Created
                }
            };
            Ok(DebtWrite {
                path: path.to_string(),
                change,
                findings,
                lines,
            })
        }

        /// and the paths its tested-by markers name, under the complete policy. It
        /// is the report of `cargo workplan coverage claims`, and it writes
        /// nothing.
        ///
        /// ```rust
        /// use raf_source::{MemoryTree, Workspace};
        /// use raf_workplan::Workplan;
        ///
        /// let source = Workspace::load(
        ///     &MemoryTree::from([
        ///         ("Cargo.toml", "[package]\nname = \"demo\"\nversion = \"0.1.0\"\n"),
        ///         ("src/lib.rs", "pub fn run() {}\n"),
        ///         ("tests/it.rs", "#[test]\nfn case() {}\n"),
        ///     ]),
        ///     &["."],
        /// )?;
        /// let section = |name: &str, text: &str| format!("## {name}\n\n{text}\n\n");
        /// let document = format!(
        ///     "---\ntype: Capability\nid: cap:demo\ntitle: Demo\ndescription: Runs.\n\
        ///      state: delivered\nbindings:\n  code: [demo::run]\n  tests: [demo::test_crate::it::case]\n  \
        ///      examples: []\ndepends: []\n---\n\n# Demo\n\n{}{}{}{}{}{}{}",
        ///     section("Purpose", "Runs."),
        ///     section(
        ///         "Contract",
        ///         "**Running.** It runs.\n<!-- raf:tested-by demo::test_crate::it::case -->\n\n\
        ///          **Stopping.** It stops.",
        ///     ),
        ///     section("Usage", "Call it."),
        ///     section("Examples", "None."),
        ///     section("Constraints", "- One at a time."),
        ///     section("Explanation", "Simple."),
        ///     section("Verification", "Tested."),
        /// );
        /// let tree = MemoryTree::from([("docs/capabilities/demo.md", document.as_str())]);
        /// let report = Workplan::load(&tree, &source).claims_report();
        /// let claims: Vec<(String, &[String])> = report
        ///     .claims()
        ///     .iter()
        ///     .map(|claim| (claim.key(), claim.tests()))
        ///     .collect();
        /// assert_eq!(claims[1].0, "cap:demo#contract/running");
        /// assert_eq!(claims[1].1, ["demo::test_crate::it::case"]);
        /// assert_eq!(claims[2].0, "cap:demo#contract/stopping");
        /// assert!(claims[2].1.is_empty());
        /// assert!(!report.is_complete());
        /// # Ok::<(), Box<dyn std::error::Error>>(())
        /// ```
        #[must_use]
        pub fn claims_report(&self) -> ClaimsReport {
            let mut claims: Vec<ClaimCoverage> = self
                .capabilities()
                .iter()
                .filter(|loaded| loaded.value().state == State::Delivered)
                .flat_map(|loaded| {
                    let id = &loaded.value().header.id;
                    units(loaded.body())
                        .units
                        .into_iter()
                        .map(|unit| ClaimCoverage {
                            capability: id.clone(),
                            section: unit.section,
                            claim: unit.key,
                            line: unit.line,
                            tests: unit.tests,
                        })
                })
                .collect();
            claims.sort_by(|a, b| {
                (&a.capability, a.section.slug(), a.line).cmp(&(
                    &b.capability,
                    b.section.slug(),
                    b.line,
                ))
            });
            ClaimsReport { claims }
        }

    //! Reverse coverage: the lines of every package of the source workspace that
    //! no capability explains.
    //!
    //! [`Workplan::lines_report`] reads every package the loader reaches, the root
    //! package and the packages that no capability binds included, and returns a
    //! [`LinesReport`].
    //!
    //! # Units and lines
    //!
    //! The units of a file are the units of every target the source index lists:
    //! every item except a module and an `extern crate`, every impl block from its
    //! first attribute to its closing brace, and every item-level macro invocation
    //! and `const _` of a module, found by parsing the file. The counted lines of a
    //! file are the union of the lines of its units; a file is counted once, for
    //! the package of the first target that reaches it, however many targets and
    //! module paths reach it. Every other line (a blank line, a `use` or `mod`
    //! declaration, an inner doc comment, the SPDX header, the braces of an inline
    //! module) is structural and is not counted. Lines are counted by line breaks,
    //! so a file with CRLF endings reads as the same file with LF.
    //!
    //! # Covered
    //!
    //! A line is covered when a covered unit holds it. The covered units are the
    //! closure of the bound items under [`Workspace::reachable`]
    //! (contains, impl-of and references): the roots are the items that a `code`,
    //! `tests` or `examples` binding of a loaded capability resolves to, and for an
    //! `examples` binding of the crate root of an example target every item and
    //! impl block of that target. A unit is covered when any item or block of its
    //! source is reached. A macro invocation or a `const _` is covered when one of
    //! its module paths is a module that a root opens (a bound module, or one
    //! below it), when its target is an example target an `examples` binding
    //! names, or, for an invocation, when a covered unit names a path that resolves
    //! to nothing from the module of that unit and whose last segment is an
    //! identifier inside the invocation; the `macro_rules!` a covered invocation
    //! invokes is then a root too, and the closure is taken again until no
    //! invocation is newly covered.
    //! A direct `#[global_allocator]` static is also a root of its participating
    //! target. Test-only registration requires a participating test context. The
    //! static retains its source role and reaches its real initializer dependencies.
    //! Malformed registrations or registrations without an exact indexed static
    //! refuse the report; macros and `cfg_attr` are not expanded for this purpose.
    //!
    //! # Exceptions
    //!
    //! A line that no covered unit holds is excepted when an exception of the
    //! coverage record matches a unit that holds it, or its file: an `item`
    //! exception the path of the unit or a path the unit is below (a method below
    //! its type, an item below its module, an impl block below its self type and
    //! its module), a `<module>::*` prefix every unit below the module, a `file`
    //! the file, and a directory every file below it. The first matching exception
    //! in the order of the record gives the reason. The exceptions are put into an
    //! index once, and each unit looks up its own paths and the modules above them,
    //! so the work grows with the units plus the exceptions.
    //!
    //! Every other counted line is an orphan line, listed under the innermost unit
    //! that holds it. A line is a test line when that unit is test code (a
    //! `#[cfg(test)]` module or a `#[test]` function) or is in a target that does
    //! not ship (integration tests, benches, examples), and a shipping line
    //! otherwise.
    //!
    //! An exception that matches nothing is judged at the load, without counting a
    //! line: an `item` names an item of the index, a prefix a module or an item of
    //! the index at or below it, a `file` a file of the index and a directory a
    //! file of the index below it.

        pub fn ceiling_text(&self) -> String {
            if self.ceilings.is_empty() {
                return "0 lines".to_string();
            }
            let rows: Vec<String> = self.ceilings.iter().map(ceiling_text).collect();
            rows.join(", ")
        }
    
        /// The orphan lines of the package that the debt record holds as debt:
        /// for each item of the package that holds orphan lines and has a row in
        /// the `lines` of the debt record, the fewer of the lines it holds and the
        /// lines its row records. None without a debt record, and none for an item
        /// with no row.
        #[must_use]
        pub fn debt(&self) -> usize {
            self.debt
        }
    }
    
    /// One ceiling row as the report writes it.
    fn ceiling_text(ceiling: &Ceiling) -> String {
        let class = match ceiling.of {
            CeilingClass::All => String::new(),
            class => format!("{} ", class.name()),
        };
        match &ceiling.limit {
            CeilingLimit::Lines(lines) => format!("{class}{} lines", lines.text()),
            CeilingLimit::Percent(percent) => format!("{class}{}%", percent.text()),
        }
    }

        /// block as `<module>::<impl Type>` or `<module>::<impl Trait for Type>`
        /// and a member of an impl of a trait on a type of another crate below
        /// it, a macro invocation as `<module>::<path>!` and a `const _` as
        /// `<module>::_`.

    /// The reverse coverage of the workspace, or of one package of it: a row for
    /// each package, the modules that hold an orphan line, the orphan lines of
    /// each unit, the lines the exceptions take out, and why the lines could not
    /// be counted, if they could not. The rows are sorted by package, module
    /// path, file and first line.
    #[derive(Debug, Clone, Default, PartialEq, Eq)]
    #[non_exhaustive]
    pub struct LinesReport {
        crates: Vec<CrateLines>,
        modules: Vec<ModuleLines>,
        orphans: Vec<UnitLines>,
        excepted: Vec<ExceptedLines>,
        errors: Vec<String>,
    }
    
    impl LinesReport {
        /// Every package, sorted by name.
        #[must_use]
        pub fn crates(&self) -> &[CrateLines] {
            &self.crates
        }
    
        /// The modules that hold an orphan line.
        #[must_use]
        pub fn modules(&self) -> &[ModuleLines] {
            &self.modules
        }
    
        /// The orphan lines of each unit that holds one, each line under the
        /// innermost unit that holds it.
        #[must_use]
        pub fn orphans(&self) -> &[UnitLines] {
            &self.orphans
        }
    
        /// The lines each exception takes out of each unit.
        #[must_use]
        pub fn excepted(&self) -> &[ExceptedLines] {
            &self.excepted
        }
    
        /// Why the lines could not be counted: a file that does not parse or
        /// lex, a malformed or unindexed allocator registration, or a walk of the
        /// index that failed. A report with an error holds no row, since no line
        /// could be judged.
        #[must_use]
        pub fn errors(&self) -> &[String] {
            &self.errors
        }
    
        /// True when the report holds no orphan line and no error.
        #[must_use]
        pub fn is_complete(&self) -> bool {
            self.orphans.is_empty() && self.errors.is_empty()
        }
    
        /// The report of one package, by its name; `None` for a name that is no
        /// package of the report.
        #[must_use]
        pub fn package(&self, name: &str) -> Option<LinesReport> {
            let row = self.crates.iter().find(|row| row.package == name)?;
            Some(Self {
                crates: vec![row.clone()],
                modules: self
                    .modules
                    .iter()
                    .filter(|row| row.package == name)
                    .cloned()
                    .collect(),
                orphans: self
                    .orphans
                    .iter()
                    .filter(|row| row.package == name)
                    .cloned()
                    .collect(),
                excepted: self
                    .excepted
                    .iter()
                    .filter(|row| row.lines.package == name)
                    .cloned()
                    .collect(),
                errors: self.errors.clone(),
            })
        }
    }

    /// The item-level macro invocations and `const _` items one parse of a file
    /// found.
    struct Found {

        /// The inline modules around it, outermost first.
        modules: Vec<String>,
        invoked: Option<String>,
        idents: Vec<String>,
    }
    
    /// Compiler registrations observed during the same bounded source parse.
    #[derive(Default)]
    struct ParsedUnits {
        modules: Vec<Found>,
        allocators: Vec<(usize, usize)>,
    }
    
    #[derive(Default)]
    struct Allocators {
        attributes: BTreeMap<usize, usize>,
        attached: BTreeSet<usize>,
        statics: Vec<(usize, usize)>,
        error: Option<String>,
    }
    
    impl<'ast> Visit<'ast> for Allocators {
        fn visit_attribute(&mut self, attribute: &'ast syn::Attribute) {
            if attribute.path().is_ident("global_allocator") {
                self.attributes.insert(
                    attribute.span().byte_range().start,
                    attribute.span().start().line,
                );
                if !matches!(attribute.meta, syn::Meta::Path(_)) {
                    self.error = Some(format!(
                        "line {}: global_allocator must be a bare attribute on a static",
                        attribute.span().start().line
                    ));
                }
            }
            syn::visit::visit_attribute(self, attribute);
        }
    
        fn visit_item_static(&mut self, item: &'ast syn::ItemStatic) {
            let attributes: Vec<_> = item
                .attrs
                .iter()
                .filter(|attribute| attribute.path().is_ident("global_allocator"))
                .collect();
            if !attributes.is_empty() {
                if attributes.len() != 1 {
                    self.error = Some(format!(
                        "line {}: a static has repeated global_allocator attributes",
                        item.span().start().line
                    ));
                }
                for attribute in attributes {
                    self.attached.insert(attribute.span().byte_range().start);
                }
                let bytes = item.span().byte_range();
                self.statics.push((bytes.start, bytes.end));
            }
            syn::visit::visit_item_static(self, item);
        }
    }
    
    /// Module units and direct allocator registrations from one bounded parse.
    fn module_units(path: &RelPath, text: &str) -> Result<ParsedUnits, String> {
        parse_source(path, text, |tree| {
            let mut out = ParsedUnits::default();
            found_in(&tree.items, &mut Vec::new(), &mut out.modules);
            let mut allocators = Allocators::default();
            allocators.visit_file(tree);
            if let Some(error) = allocators.error {
                return Err(format!("{path}: {error}"));
            }
            if let Some((_, line)) = allocators
                .attributes
                .iter()
                .find(|(at, _)| !allocators.attached.contains(at))
            {
                return Err(format!(
                    "{path}:{line}: global_allocator is not attached to a static"
                ));
            }
            out.allocators = allocators.statics;
            Ok(out)
        })
        .map_err(|error| error.to_string())?
    }
    
    fn found_in(items: &[SynItem], modules: &mut Vec<String>, out: &mut Vec<Found>) {
        for item in items {
            match item {
                SynItem::Macro(invocation) if invocation.ident.is_none() => {
                    let span = item.span();
                    let mut idents = Vec::new();
                    idents_of(invocation.mac.tokens.clone(), &mut idents);
                    let path: String = quote_path(&invocation.mac.path);
                    out.push(Found {
                        first: span.start().line,
                        last: span.end().line,
                        modules: modules.clone(),
                        invoked: Some(path),
                        idents,
                    });
                }
                SynItem::Const(constant) if constant.ident == "_" => {
                    let span = item.span();
                    out.push(Found {
                        first: span.start().line,
                        last: span.end().line,
                        modules: modules.clone(),
                        invoked: None,
                        idents: Vec::new(),
                    });
                }
                SynItem::Mod(module) => {
                    if let Some((_, content)) = &module.content {
                        let name = module.ident.to_string();
                        modules.push(name.strip_prefix("r#").unwrap_or(&name).to_string());
                        found_in(content, modules, out);
                        modules.pop();
                    }
                }
                _ => {}
            }
        }
    }
    
    /// A macro path as written, without spaces: `kinds`, `crate::kinds`.
    fn quote_path(path: &syn::Path) -> String {
        let mut text = String::new();
        if path.leading_colon.is_some() {
            text.push_str("::");
        }
        let segments: Vec<String> = path
            .segments
            .iter()
            .map(|segment| segment.ident.to_string())
            .collect();
        text.push_str(&segments.join("::"));
        text
    }
    
    /// Every identifier of `stream`, nested groups included, walked on an
    /// explicit stack.
    fn idents_of(stream: TokenStream, out: &mut Vec<String>) {
        let mut stack = vec![stream.into_iter()];
        while let Some(tokens) = stack.last_mut() {
            match tokens.next() {
                Some(TokenTree::Ident(ident)) => {
                    let name = ident.to_string();
                    out.push(name.strip_prefix("r#").unwrap_or(&name).to_string());
                }
                Some(TokenTree::Group(group)) => stack.push(group.stream().into_iter()),
                Some(_) => {}
                None => {
                    stack.pop();
                }
            }
        }
    }
    
    /// A path run of a file whose last segment is an identifier that a macro
    /// invocation holds: the byte where it starts, its text and that segment.
    #[derive(Debug, Clone, PartialEq, Eq)]
    struct Run {
        start: usize,
        text: String,
        segments: usize,
        last: String,
    }
    
    /// The words that are no segment of a path, beside `crate`, `self` and
    /// `super`, which start one.
    const KEYWORDS: [&str; 49] = [
        "Self", "abstract", "as", "async", "await", "become", "box", "break", "const", "continue",
        "do", "dyn", "else", "enum", "extern", "false", "final", "fn", "for", "gen", "if", "impl",
        "in", "let", "loop", "macro", "match", "mod", "move", "mut", "override", "priv", "pub", "ref",
        "return", "static", "struct", "trait", "true", "try", "type", "typeof", "unsafe", "unsized",
        "use", "virtual", "where", "while", "yield",
    ];
    
    /// The words after which an identifier is a name being declared or bound,
    /// not a path.
    const DECLARES: [&str; 10] = [
        "fn", "struct", "enum", "union", "trait", "type", "mod", "const", "static", "let",
    ];
    
    /// The runs of `text` whose last segment is one of `wanted`, lexed on a
    /// thread of its own so that the span data of the lexer dies with it.
    fn runs_of(path: &RelPath, text: &str, wanted: &BTreeSet<String>) -> Result<Vec<Run>, String> {
        thread::scope(|scope| {
            let worker = thread::Builder::new()
                .name(path.to_string())
                .stack_size(PARSE_STACK_BYTES)
                .spawn_scoped(scope, move || lex_runs(text, wanted))
                .map_err(|error| format!("{path}: the thread that lexes it cannot start: {error}"))?;
            match worker.join() {
                Ok(result) => result.map_err(|error| format!("{path}: {error}")),
                Err(payload) => panic::resume_unwind(payload),
            }
        })
    }
    
    /// [`runs_of`] on the calling thread. A byte order mark and a shebang line
    /// are left out, as `syn` leaves them out, and offsets stay those of `text`.
    fn lex_runs(text: &str, wanted: &BTreeSet<String>) -> Result<Vec<Run>, String> {
        let mut base = 0;
        let mut body = text;
        if let Some(rest) = body.strip_prefix('\u{feff}') {
            base += '\u{feff}'.len_utf8();
            body = rest;
        }
        if let Some(after) = body.strip_prefix("#!")
            && !after.trim_start().starts_with('[')
        {
            let cut = body.find('\n').unwrap_or(body.len());
            base += cut;
            body = &body[cut..];
        }
        let stream = TokenStream::from_str(body).map_err(|error| error.to_string())?;
        let mut runs = Vec::new();
        let mut stack: Vec<(proc_macro2::token_stream::IntoIter, Scan)> =
            vec![(stream.into_iter(), Scan::default())];
        while let Some((tokens, scan)) = stack.last_mut() {
            let Some(tree) = tokens.next() else {
                scan.close(&mut runs, wanted);
                stack.pop();
                if let Some((_, outer)) = stack.last_mut() {
                    outer.push(Tok::Other);
                }
                continue;
            };
            add(Work::LineTokens, 1);
            match tree {
                TokenTree::Group(group) => {
                    scan.close(&mut runs, wanted);
                    stack.push((group.stream().into_iter(), Scan::default()));
                }
                TokenTree::Ident(ident) => {
                    let raw = ident.to_string();
                    let start = base + ident.span().byte_range().start;
                    scan.ident(&raw, start, &mut runs, wanted);
                }
                TokenTree::Punct(punct) => {
                    scan.punct(
                        punct.as_char(),
                        punct.spacing() == Spacing::Joint,
                        &mut runs,
                        wanted,
                    );
                }
                TokenTree::Literal(_) => {
                    scan.close(&mut runs, wanted);
                    scan.push(Tok::Other);
                }
            }
        }
        Ok(runs)
    }
    
    /// A token as the run scanner remembers it.
    #[derive(Debug, Clone, Copy, PartialEq, Eq)]
    enum Tok {
        /// An identifier, and whether the identifier after it is declared by it.
        Ident { declares: bool },
        /// A punctuation character and whether the next one joins it.
        Punct { ch: char, joint: bool },
        /// A literal or a group.
        Other,
    }
    
    /// The run scanner of one level of delimiters: the last three tokens and the
    /// open run.
    #[derive(Default)]
    struct Scan {
        last: [Option<Tok>; 3],
        open: Option<Run>,
    }
    
    impl Scan {
        fn push(&mut self, tok: Tok) {
            self.last = [self.last[1], self.last[2], Some(tok)];
        }
    
        /// True when the last two tokens are `::`.
        fn colons(&self) -> bool {
            matches!(
                (self.last[1], self.last[2]),
                (
                    Some(Tok::Punct {
                        ch: ':',
                        joint: true
                    }),
                    Some(Tok::Punct { ch: ':', .. })
                )
            )
        }
    
        /// True when the last token is a `.` that is not part of `..`.
        fn single_dot(&self) -> bool {
            matches!(
                self.last[2],
                Some(Tok::Punct {
                    ch: '.',
                    joint: false
                })
            ) && !matches!(
                self.last[1],
                Some(Tok::Punct {
                    ch: '.',
                    joint: true
                })
            )
        }
    
        /// End the open run, keeping it when its last segment is wanted.
        fn close(&mut self, runs: &mut Vec<Run>, wanted: &BTreeSet<String>) {
            if let Some(run) = self.open.take()
                && wanted.contains(&run.last)
            {
                runs.push(run);
            }
        }
    
        fn ident(&mut self, raw: &str, start: usize, runs: &mut Vec<Run>, wanted: &BTreeSet<String>) {
            let plain = raw.strip_prefix("r#");
            let word = plain.unwrap_or(raw);
            let path_word = plain.is_none() && matches!(word, "crate" | "self" | "super");
            let keyword = plain.is_none() && KEYWORDS.contains(&word);
            let declares = plain.is_none() && DECLARES.contains(&word);
            if self.colons() && self.open.is_some() && !keyword && !path_word {
                if let Some(run) = self.open.as_mut() {
                    run.text.push_str("::");
                    run.text.push_str(word);
                    run.segments += 1;
                    run.last = word.to_string();
                }
            } else {
                let leading = self.colons() && self.open.is_none();
                self.close(runs, wanted);
                let after_declaring = matches!(self.last[2], Some(Tok::Ident { declares: true }));
                if !keyword && !self.single_dot() && !after_declaring {
                    let text = if leading {
                        format!("::{word}")
                    } else {
                        word.to_string()
                    };
                    self.open = Some(Run {
                        start,
                        text,
                        segments: 1,
                        last: word.to_string(),
                    });
                }
            }
            self.push(Tok::Ident { declares });
        }
    
        fn punct(&mut self, ch: char, joint: bool, runs: &mut Vec<Run>, wanted: &BTreeSet<String>) {
            let second = matches!(
                self.last[2],
                Some(Tok::Punct {
                    ch: ':',
                    joint: true
                })
            );
            if ch == ':' && !joint && !second {
                // A single `:`: a run of one segment before it is a name being
                // bound or a field, not a path.
                if self.open.as_ref().is_some_and(|run| run.segments == 1) {
                    self.open = None;
                } else {
                    self.close(runs, wanted);
                }
            } else if ch != ':' {
                self.close(runs, wanted);
            }
            self.push(Tok::Punct { ch, joint });
        }
    }
    
    /// The exceptions of the record by what each names, so that a unit looks up
    /// its own paths and the modules above them, and a file its own path and the
    /// directories above it, whatever the number of exceptions.
    struct ExceptionIndex<'r> {
        items: HashMap<&'r str, Vec<usize>>,
        prefixes: HashMap<&'r str, Vec<usize>>,
        files: HashMap<&'r str, Vec<usize>>,
        directories: HashMap<&'r str, Vec<usize>>,
    }
    
    /// `path` and each path it is below, longest first: `a::b::c`, `a::b`, `a`.
    fn ancestors(path: &str) -> impl Iterator<Item = &str> {
        std::iter::successors(Some(path), |p| p.rsplit_once("::").map(|(head, _)| head))
    }
    
    /// The directories `file` is below, deepest first: `a/b` and `a` for
    /// `a/b/c.rs`.
    fn directories(file: &str) -> impl Iterator<Item = &str> {
        std::iter::successors(file.rsplit_once('/').map(|(head, _)| head), |p| {
            p.rsplit_once('/').map(|(head, _)| head)
        })
    }
    
    impl<'r> ExceptionIndex<'r> {
        fn new(exceptions: &'r [Exception]) -> Self {
            let mut index = Self {
                items: HashMap::new(),
                prefixes: HashMap::new(),
                files: HashMap::new(),
                directories: HashMap::new(),
            };
            for (position, exception) in exceptions.iter().enumerate() {
                add(Work::ExceptionsIndexed, 1);
                let (map, path) = match &exception.target {
                    ExceptionTarget::Item(path) => (&mut index.items, path),
                    ExceptionTarget::Prefix(path) => (&mut index.prefixes, path),
                    ExceptionTarget::File(path) => (&mut index.files, path),
                    ExceptionTarget::Directory(path) => (&mut index.directories, path),
                };
                map.entry(path.as_str()).or_default().push(position);
            }
            index
        }
    
        fn is_empty(&self) -> bool {
            self.items.is_empty()
                && self.prefixes.is_empty()
                && self.files.is_empty()
                && self.directories.is_empty()
        }
    
        /// The first exception that matches a unit at the item paths `paths` and
        /// in the modules `modules`: an item exception at a path or one it is
        /// below, a prefix exception at a module a path is below, and either at a
        /// module of `modules` or one it is below.
        fn unit(&self, paths: &[&str], modules: &[&str]) -> Option<usize> {
            if self.items.is_empty() && self.prefixes.is_empty() {
                return None;
            }
            let mut first = None;
            let mut take = |found: Option<&Vec<usize>>| {
                add(Work::ExceptionLookups, 1);
                if let Some(&position) = found.and_then(|positions| positions.first()) {
                    first = Some(first.map_or(position, |seen: usize| seen.min(position)));
                }
            };
            for path in paths {
                for (depth, at) in ancestors(path).enumerate() {
                    take(self.items.get(at));
                    if depth > 0 {
                        take(self.prefixes.get(at));
                    }
                }
            }
            for module in modules {
                for at in ancestors(module) {
                    take(self.items.get(at));
                    take(self.prefixes.get(at));
                }
            }
            first
        }
    
        /// The first exception that matches the file `file`, by its path or a
        /// directory it is below.
        fn file(&self, file: &str) -> Option<usize> {
            if self.files.is_empty() && self.directories.is_empty() {
                return None;
            }
            add(Work::ExceptionLookups, 1);
            let mut first = self
                .files
                .get(file)
                .and_then(|found| found.first())
                .copied();
            for directory in directories(file) {
                add(Work::ExceptionLookups, 1);
                if let Some(&position) = self
                    .directories
                    .get(directory)
                    .and_then(|found| found.first())
                {
                    first = Some(first.map_or(position, |seen| seen.min(position)));
                }
            }
            first
        }
    }
    
    /// The positions of the exceptions of `exceptions` that match nothing of the
    /// index of `source`: an item exception that names no item, a prefix that
    /// names no module or item at or below it, a file that is no file of the
    /// index and a directory that holds none. No line is counted to judge it.
    pub(crate) fn unmatched_exceptions(source: &Workspace, exceptions: &[Exception]) -> Vec<usize> {
        let mut matched = vec![false; exceptions.len()];
        let mut files: HashMap<&str, Vec<usize>> = HashMap::new();
        let mut dirs: HashMap<&str, Vec<usize>> = HashMap::new();
        for (position, exception) in exceptions.iter().enumerate() {
            add(Work::ExceptionsIndexed, 1);
            match &exception.target {
                ExceptionTarget::Item(path) => {
                    add(Work::ExceptionLookups, 1);
                    matched[position] = source
                        .items_under(path)
                        .next()
                        .is_some_and(|item| *item.path == **path);
                }
                ExceptionTarget::Prefix(path) => {
                    add(Work::ExceptionLookups, 1);
                    matched[position] = source.items_under(path).next().is_some();
                }
                ExceptionTarget::File(path) => files.entry(path).or_default().push(position),
                ExceptionTarget::Directory(path) => dirs.entry(path).or_default().push(position),
            }
        }
        if !files.is_empty() || !dirs.is_empty() {
            for file in source.files() {
                let path = file.path.as_str();
                add(Work::ExceptionLookups, 1);
                for position in files.get(path).into_iter().flatten() {
                    matched[*position] = true;
                }
                for directory in directories(path) {
                    add(Work::ExceptionLookups, 1);
                    for position in dirs.get(directory).into_iter().flatten() {
                        matched[*position] = true;
                    }
                }
            }
        }
        matched
            .iter()
            .enumerate()
            .filter(|(_, hit)| !**hit)
            .map(|(position, _)| position)
            .collect()
    }
    
    /// The name of the kind of an item, as the report writes it.
    fn kind_name(kind: SymbolKind) -> &'static str {
        match kind {
            SymbolKind::Fn
            | SymbolKind::ForeignFn
            | SymbolKind::InherentMethod
            | SymbolKind::TraitImplMethod
            | SymbolKind::TraitDefaultMethod
            | SymbolKind::TraitRequiredMethod => "fn",
            SymbolKind::Struct => "struct",
            SymbolKind::Enum => "enum",
            SymbolKind::Union => "union",
            SymbolKind::Trait => "trait",
            SymbolKind::TypeAlias | SymbolKind::ForeignType | SymbolKind::AssocType => "type",
            SymbolKind::Const | SymbolKind::AssocConst => "const",
            SymbolKind::Static | SymbolKind::ForeignStatic => "static",
            SymbolKind::MacroRules => "macro_rules",
            _ => "item",
        }
    }
    
    /// An impl block as the report names it: `<module>::<impl Type>` or
    /// `<module>::<impl Trait for Type>`.
    fn impl_name(block: &ImplBlock) -> String {
        match &block.trait_path {
            Some(trait_path) => format!(
                "{}::<impl {trait_path} for {}>",
                block.module, block.self_type
            ),
            None => format!("{}::<impl {}>", block.module, block.self_type),
        }
    }
    
    /// `path` without its last segment.
    fn parent(path: &str) -> &str {
        path.rsplit_once("::").map_or(path, |(head, _)| head)
    }
    
    /// The module of an item: the module of its impl block for a member of one,
    /// the module of its trait for a member of a trait, and the module its path
    /// names otherwise.
    fn item_module(source: &Workspace, item: &Item, block_of: &HashMap<ItemId, ImplId>) -> String {
        if let Some(block) = block_of.get(&item.id).and_then(|id| source.impl_block(*id)) {
            return block.module.to_string();
        }
        match item.kind {
            SymbolKind::TraitRequiredMethod
            | SymbolKind::TraitDefaultMethod
            | SymbolKind::AssocConst
            | SymbolKind::AssocType => parent(parent(&item.path)).to_string(),
            _ => parent(&item.path).to_string(),
        }
    }
    
    /// The unit of the items of one source, the item of the first path first.
    fn item_unit(
        source: &Workspace,
        file: usize,
        mut copies: Vec<&Item>,
        block_of: &HashMap<ItemId, ImplId>,
    ) -> Unit {
        copies.sort_by(|a, b| (&a.path, a.id).cmp(&(&b.path, b.id)));
        let first = copies[0];
        let block = block_of
            .get(&first.id)
            .and_then(|id| source.impl_block(*id));
        let item = match (first.self_key, block) {
            (Some(SelfType::Trait), Some(block)) => format!("{}::{}", impl_name(block), first.name),
            _ => first.path.to_string(),
        };
        Unit {
            file,
            first: first.span.lines.start,
            last: first
                .span
                .lines
                .end
                .saturating_sub(1)
                .max(first.span.lines.start),
            bytes: (first.span.start_byte, first.span.end_byte),
            module: item_module(source, first, block_of),
            item,
            kind: kind_name(first.kind),
            shipping: copies.iter().any(|copy| source.is_shipping(copy)),
            what: What::Item(copies.iter().map(|copy| copy.id).collect()),
            covered: false,
            exception: None,
        }
    }
    
    /// The unit of the impl blocks of one source.
    fn impl_unit(source: &Workspace, file: usize, copies: &[&ImplBlock]) -> Unit {
        let first = copies[0];
        Unit {
            file,
            first: first.span.lines.start,
            last: first
                .span
                .lines
                .end
                .saturating_sub(1)
                .max(first.span.lines.start),
            bytes: (first.span.start_byte, first.span.end_byte),
            module: first.module.to_string(),
            item: impl_name(first),
            kind: "impl",
            shipping: copies.iter().any(|block| {
                block.test == TestRole::None
                    && source
                        .target(block.target)
                        .is_some_and(|target| target.shipping)
            }),
            what: What::Impl(copies.iter().map(|block| block.id).collect()),
            covered: false,
            exception: None,
        }
    }
    
    /// The unit of a macro invocation or a `const _` that a parse of `file`
    /// found, once for each module path the file is reached under.
    fn module_unit(source: &Workspace, index: usize, file: &SourceFile, found: Found) -> Unit {
        let suffix = found.modules.join("::");
        let mut copies: Vec<(String, TargetId, bool)> = file
            .reached
            .iter()
            .map(|reach| {
                let module = if suffix.is_empty() {
                    reach.module_path.to_string()
                } else {
                    format!("{}::{suffix}", reach.module_path)
                };
                // An inline module guarded by `test` makes what it holds test
                // code, as it makes its items.
                let inner_test = !suffix.is_empty()
                    && source
                        .items_under(&module)
                        .take_while(|item| *item.path == *module)
                        .any(|item| {
                            item.kind == SymbolKind::Mod
                                && item.target == reach.target
                                && item.test != TestRole::None
                        });
                (module, reach.target, reach.test || inner_test)
            })
            .collect();
        copies.sort();
        copies.dedup();
        let shipping = copies.iter().any(|(_, target, test)| {
            !test && source.target(*target).is_some_and(|target| target.shipping)
        });
        let module = copies
            .first()
            .map_or_else(String::new, |(module, _, _)| module.clone());
        let (item, kind) = match &found.invoked {
            Some(path) => (format!("{module}::{path}!"), "macro"),
            None => (format!("{module}::_"), "const"),
        };
        Unit {
            file: index,
            first: found.first,
            last: found.last.max(found.first),
            bytes: (0, 0),
            module,
            item,
            kind,
            shipping,
            what: What::Module(ModuleUnit {
                copies,
                invoked: found.invoked,
                idents: found.idents,
            }),
            covered: false,
            exception: None,
        }
    }
    
    /// The units of every file of the source workspace.
    struct Gathered<'s> {
        files: Vec<File<'s>>,
        units: Vec<Unit>,
        /// The units of each file, as a range of `units`.
        of_file: Vec<std::ops::Range<usize>>,
        /// The runs whose last segment a macro invocation holds, each with the
        /// innermost item or impl unit that holds its start.
        runs: Vec<(Run, Option<usize>)>,
        /// Exact indexed definitions bearing a compiler allocator registration.
        allocators: Vec<ItemId>,
    }
    
    /// The units of every file of `source`, file by file in the order of the
    /// files.
    fn collect<'s>(source: &'s Workspace, judged: &BTreeSet<&str>) -> Result<Gathered<'s>, String> {
        let mut block_of: HashMap<ItemId, ImplId> = HashMap::new();
        let mut blocks: BTreeMap<&RelPath, BTreeMap<(usize, usize), Vec<&ImplBlock>>> = BTreeMap::new();
        for block in source.impls() {
            for member in &block.members {
                block_of.insert(*member, block.id);
            }
            blocks
                .entry(&block.file)
                .or_default()
                .entry((block.span.start_byte, block.span.end_byte))
                .or_default()
                .push(block);
        }
        let mut out = Gathered {
            files: Vec::new(),
            units: Vec::new(),
            of_file: Vec::new(),
            runs: Vec::new(),
            allocators: Vec::new(),
        };
        for (index, file) in source
            .files()
            .filter(|file| judged.contains(file.package.as_str()))
            .enumerate()
        {
            let begin = out.units.len();
            out.files.push(File {
                path: &file.path,
                package: &file.package,
                exception: None,
            });
            let mut groups: BTreeMap<(usize, usize), Vec<&Item>> = BTreeMap::new();
            for item in source
                .items_in(&file.path)
                .filter(|item| !matches!(item.kind, SymbolKind::Mod | SymbolKind::ExternCrate))
            {
                groups
                    .entry((item.span.start_byte, item.span.end_byte))
                    .or_default()
                    .push(item);
            }
            let parsed = module_units(&file.path, &file.text)?;
            for bytes in parsed.allocators {
                let copies = groups.get(&bytes).filter(|copies| copies.iter().all(|item| item.kind == SymbolKind::Static)).ok_or_else(|| format!("{}: global_allocator static at bytes {}..{} has no exact indexed static definition", file.path, bytes.0, bytes.1))?;
                out.allocators.extend(copies.iter().map(|item| item.id));
            }
            for copies in groups.into_values() {
                out.units.push(item_unit(source, index, copies, &block_of));
            }
            for copies in blocks
                .get(&file.path)
                .into_iter()
                .flat_map(BTreeMap::values)
            {
                out.units.push(impl_unit(source, index, copies));
            }
            for found in parsed.modules {
                out.units.push(module_unit(source, index, file, found));
            }
            out.of_file.push(begin..out.units.len());
        }
        let wanted: BTreeSet<String> = out
            .units
            .iter()
            .filter_map(|unit| match &unit.what {
                What::Module(module) if module.invoked.is_some() => Some(&module.idents),
                _ => None,
            })
            .flatten()
            .cloned()
            .collect();
        if !wanted.is_empty() {
            for (index, file) in source
                .files()
                .filter(|file| judged.contains(file.package.as_str()))
                .enumerate()
            {
                let runs = runs_of(&file.path, &file.text, &wanted)?;
                let range = out.of_file[index].clone();
                place(&out.units, range, runs, &mut out.runs);
            }
        }
        Ok(out)
    }
    
    /// Give each run of a file to the innermost item or impl unit of `range`
    /// whose bytes hold its start, in one sweep of both in byte order.
    fn place(
        units: &[Unit],
        range: std::ops::Range<usize>,
        mut runs: Vec<Run>,
        out: &mut Vec<(Run, Option<usize>)>,
    ) {
        let mut spans: Vec<(usize, usize, usize)> = range
            .filter(|at| !matches!(units[*at].what, What::Module(_)))
            .map(|at| (units[at].bytes.0, units[at].bytes.1, at))
            .collect();
        // Outer units first where two start at one byte.
        spans.sort_by(|a, b| a.0.cmp(&b.0).then(b.1.cmp(&a.1)));
        runs.sort_by_key(|run| run.start);
        let mut open: Vec<(usize, usize, usize)> = Vec::new();
        let mut next = 0;
        for run in runs {
            while let Some(&span) = spans.get(next).filter(|span| span.0 <= run.start) {
                while open.last().is_some_and(|top| top.1 <= span.0) {
                    open.pop();
                }
                open.push(span);
                next += 1;
            }
            while open.last().is_some_and(|top| top.1 <= run.start) {
                open.pop();
            }
            let holder = open.last().map(|span| span.2);
            out.push((run, holder));
        }
    }
    
    /// The roots of the walk: the bound items, and every item and impl block of
    /// an example target an `examples` binding names; the paths of the bound
    /// modules, which open what they hold; and those example targets.
    struct Roots {
        nodes: BTreeSet<Node>,
        opened: BTreeSet<String>,
        whole: BTreeSet<TargetId>,
    }
    
    fn roots(plan: &Workplan<'_>) -> Roots {
        let source = plan.source();
        let mut roots = Roots {
            nodes: BTreeSet::new(),
            opened: BTreeSet::new(),
            whole: BTreeSet::new(),
        };
        for binding in plan.graph().bindings() {
            for id in binding.items() {
                let Some(item) = source.item(*id) else {
                    continue;
                };
                let target = source.target(item.target);
                let crate_root = target.is_some_and(|target| *item.path == *target.crate_path);
                if binding.kind() == BindingKind::Example
                    && crate_root
                    && target.is_some_and(|target| target.kind == TargetKind::Example)
                {
                    roots.whole.insert(item.target);
                    continue;
                }
                roots.nodes.insert(Node::Item(*id));
                if item.kind == SymbolKind::Mod && !crate_root {
                    roots.opened.insert(item.path.to_string());
                }
            }
        }
        if !roots.whole.is_empty() {
            for item in source
                .items()
                .filter(|item| roots.whole.contains(&item.target))
            {
                roots.nodes.insert(Node::Item(item.id));
            }
            for block in source
                .impls()
                .filter(|block| roots.whole.contains(&block.target))
            {
                roots.nodes.insert(Node::Impl(block.id));
            }
        }
        roots
    }
    
    /// Mark the covered units: the closure of the roots, the macro invocations
    /// and `const _` items of the opened modules and the example targets, and the
    /// invocations a covered unit names a generated item of, whose `macro_rules!`
    /// join the roots until no invocation is newly covered. `extra` roots are
    /// walked beside them.
    fn cover(
        source: &Workspace,
        units: &mut Gathered<'_>,
        roots: &mut Roots,
        extra: &[Node],
    ) -> Result<(), String> {
        let mut participating = BTreeSet::new();
        let mut tests = BTreeSet::new();
        for node in roots.nodes.iter().chain(extra) {
            // Registration context is auxiliary; every root remains an input of
            // the real walk below, which owns invalid item and impl refusals.
            let context = match node {
                Node::Item(id) => source.item(*id).map(|item| (item.target, item.test)),
                Node::Impl(id) => source
                    .impl_block(*id)
                    .map(|block| (block.target, block.test)),
            };
            if let Some((target, test)) = context {
                participating.insert(target);
                if test != TestRole::None {
                    tests.insert(target);
                }
            }
        }
        for id in &units.allocators {
            let item = source
                .item(*id)
                .ok_or_else(|| "a registered allocator has no indexed static".to_string())?;
            if participating.contains(&item.target)
                && (item.test == TestRole::None || tests.contains(&item.target))
            {
                roots.nodes.insert(Node::Item(*id));
            }
        }
        let mut unresolved: HashMap<(String, String), bool> = HashMap::new();
        loop {
            add(Work::LineRounds, 1);
            let mut nodes: Vec<Node> = roots.nodes.iter().copied().collect();
            nodes.extend_from_slice(extra);
            let reach = source
                .reachable(&nodes)
                .map_err(|error| format!("the walk of the bound items failed: {error}"))?;
            for unit in &mut units.units {
                unit.covered = match &unit.what {
                    What::Item(ids) => ids.iter().any(|id| reach.contains(Node::Item(*id))),
                    What::Impl(ids) => ids.iter().any(|id| reach.contains(Node::Impl(*id))),
                    What::Module(module) => module.copies.iter().any(|(path, target, _)| {
                        roots.whole.contains(target)
                            || ancestors(path).any(|at| roots.opened.contains(at))
                    }),
                };
            }
            let mut named: BTreeSet<String> = BTreeSet::new();
            for (run, holder) in &units.runs {
                add(Work::LineRuns, 1);
                let Some(unit) = holder
                    .map(|at| &units.units[at])
                    .filter(|unit| unit.covered)
                else {
                    continue;
                };
                let key = (unit.module.clone(), run.text.clone());
                let missing = *unresolved.entry(key).or_insert_with(|| {
                    matches!(
                        source.resolve_from(&unit.module, &run.text),
                        Err(ResolveError::NotFound(_))
                    )
                });
                if missing {
                    named.insert(run.last.clone());
                }
            }
            let mut grew = false;
            for unit in &mut units.units {
                let What::Module(module) = &unit.what else {
                    continue;
                };
                if !unit.covered && module.idents.iter().any(|ident| named.contains(ident)) {
                    unit.covered = true;
                }
                let (true, Some(invoked)) = (unit.covered, &module.invoked) else {
                    continue;
                };
                for (path, _, _) in &module.copies {
                    for site in invoked_sites(source, path, invoked) {
                        grew |= roots.nodes.insert(Node::Item(site));
                    }
                }
            }
            if !grew {
                return Ok(());
            }
        }
    }
    
    /// The macros that the invocation of `invoked` in the module `module` names:
    /// the path read from the module in the macro name space, and for a name of
    /// one segment from each module above it too, as a `macro_rules!` of a parent
    /// module is in the textual scope of the modules declared after it.
    fn invoked_sites(source: &Workspace, module: &str, invoked: &str) -> Vec<ItemId> {
        let modules: Vec<&str> = if invoked.contains("::") {
            vec![module]
        } else {
            ancestors(module).collect()
        };
        // `resolve_from_in` reads an import path: a bare 2015 import starts at
        // the crate root. A textual macro lookup instead starts in each module.
        let local;
        let path = if invoked.contains("::") {
            invoked
        } else {
            local = format!("self::{invoked}");
            &local
        };
        for at in modules {
            if let Ok(found) = source.resolve_from_in(at, path, Namespace::Macro) {
                return found.sites;
            }
        }
        Vec::new()
    }
    
    /// The paths and the modules a unit is matched against the exceptions by:
    /// the paths of an item, the self types and the modules of an impl block, and
    /// the modules of a macro invocation or a `const _`.
    fn match_paths(source: &Workspace, what: &What) -> (Vec<String>, Vec<String>) {
        match what {
            What::Item(ids) => (
                ids.iter()
                    .filter_map(|id| source.item(*id))
                    .map(|item| item.path.to_string())
                    .collect(),
                Vec::new(),
            ),
            What::Impl(ids) => {
                let blocks: Vec<&ImplBlock> =
                    ids.iter().filter_map(|id| source.impl_block(*id)).collect();
                (
                    blocks
                        .iter()
                        .flat_map(|block| &block.self_items)
                        .filter_map(|id| source.item(*id))
                        .map(|item| item.path.to_string())
                        .collect(),
                    blocks
                        .iter()
                        .map(|block| block.module.to_string())
                        .collect(),
                )
            }
            What::Module(module) => (
                Vec::new(),
                module
                    .copies
                    .iter()
                    .map(|(path, _, _)| path.clone())
                    .collect(),
            ),
        }
    }
    
    /// What the lines of the units add up to.
    #[derive(Default)]
    struct Totals {
        crates: BTreeMap<String, CrateLines>,
        modules: BTreeMap<(String, String), (usize, usize)>,
        orphans: BTreeMap<usize, Vec<usize>>,
        excepted: BTreeMap<(usize, usize), Vec<usize>>,
    }
    
    /// Judge every counted line of one file: covered when a covered unit holds
    /// it, excepted when an exception matches a unit that holds it or the file,
    /// and orphan otherwise, each under the innermost unit that holds it.
    fn sweep(units: &[Unit], range: std::ops::Range<usize>, file: &File<'_>, totals: &mut Totals) {
        let Some(last) = range.clone().map(|at| units[at].last).max() else {
            return;
        };
        let mut covered = vec![false; last + 1];
        let mut exception: Vec<Option<usize>> = vec![file.exception; last + 1];
        let mut inner: Vec<Option<usize>> = vec![None; last + 1];
        let mut order: Vec<usize> = range.collect();
        // Longer units first, so the innermost unit of a line is the last to mark
        // it, and of two units of one length the first in the file.
        order.sort_by(|a, b| {
            let (a, b) = (&units[*a], &units[*b]);
            (b.last - b.first, b.first, b.bytes.0).cmp(&(a.last - a.first, a.first, a.bytes.0))
        });
        for at in order {
            add(Work::LineUnits, 1);
            let unit = &units[at];
            for line in unit.first..=unit.last {
                add(Work::LineMarks, 1);
                covered[line] |= unit.covered;
                exception[line] = match (exception[line], unit.exception) {
                    (Some(a), Some(b)) => Some(a.min(b)),
                    (a, b) => a.or(b),
                };
                inner[line] = Some(at);
            }
        }
        for (line, holder) in inner.iter().enumerate() {
            let Some(at) = *holder else {
                continue;
            };
            let unit = &units[at];
            let row = totals
                .crates
                .entry(file.package.to_string())
                .or_insert_with(|| empty_row(file.package));
            let module = totals
                .modules
                .entry((file.package.to_string(), unit.module.clone()))
                .or_default();
            module.0 += 1;
            if unit.shipping {
                row.shipping_counted += 1;
            } else {
                row.test_counted += 1;
            }
            row.counted += 1;
            if covered[line] {
                row.covered += 1;
            } else if let Some(position) = exception[line] {
                row.excepted += 1;
                totals
                    .excepted
                    .entry((at, position))
                    .or_default()
                    .push(line);
            } else {
                module.1 += 1;
                if unit.shipping {
                    row.shipping_orphan += 1;
                } else {
                    row.test_orphan += 1;
                }
                totals.orphans.entry(at).or_default().push(line);
            }
        }
    }
    
    /// Set the debt of each package of `report`: for each item with orphan lines
    /// that a row of `record` names, the fewer of its lines and its row, counted
    /// for the package of the first of its units.
    fn hold_debt(report: &mut LinesReport, record: &raf_okf::DebtRecord) {
        let recorded: BTreeMap<&str, usize> = record
            .lines
            .iter()
            .map(|line| (line.item.as_str(), line.orphans))
            .collect();
        let mut present: BTreeMap<&str, (&str, usize)> = BTreeMap::new();
        for unit in &report.orphans {
            let entry = present
                .entry(unit.item.as_str())
                .or_insert((unit.package.as_str(), 0));
            entry.1 += unit.lines();
        }
        let mut held: BTreeMap<String, usize> = BTreeMap::new();
        for (item, (package, lines)) in present {
            if let Some(row) = recorded.get(item) {
                *held.entry(package.to_string()).or_default() += lines.min(*row);
            }
        }
        for row in &mut report.crates {
            row.debt = held.get(&row.package).copied().unwrap_or(0);
        }
    }
    
    /// The row of a package with no line.
    fn empty_row(package: &str) -> CrateLines {
        CrateLines {
            package: package.to_string(),
            counted: 0,
            covered: 0,
            excepted: 0,
            shipping_counted: 0,
            test_counted: 0,
            shipping_orphan: 0,
            test_orphan: 0,
            ceilings: Vec::new(),
            debt: 0,
        }
    }
    
    /// Sorted lines as merged ranges.
    fn ranges(lines: &[usize]) -> Vec<(usize, usize)> {
        let mut out: Vec<(usize, usize)> = Vec::new();
        for line in lines {
            match out.last_mut() {
                Some((_, last)) if *last + 1 == *line => *last = *line,
                _ => out.push((*line, *line)),
            }
        }
        out
    }
    
    /// The lines of the unit `unit` of the file `file`, as rows of the report.
    fn unit_lines(unit: &Unit, file: &File<'_>, lines: &[usize]) -> UnitLines {
        UnitLines {
            package: file.package.to_string(),
            module: unit.module.clone(),
            item: unit.item.clone(),
            kind: unit.kind,
            file: file.path.to_string(),
            ranges: ranges(lines),
        }
    }
    
    /// The order of the rows of units: package, module path, file, first line.
    fn unit_order(lines: &UnitLines) -> (&str, &str, &str, usize) {
        (
            &lines.package,
            &lines.module,
            &lines.file,
            lines.ranges.first().map_or(0, |range| range.0),
        )
    }
    
    /// The packages the report judges: every package of `source`.
    fn judged_packages(source: &Workspace) -> BTreeSet<&str> {
        source
            .packages()
            .iter()
            .map(|package| package.name.as_str())
            .collect()
    }
    
    /// The report of `plan`, with `extra` roots walked beside the bound items.
    pub(crate) fn compute(plan: &Workplan<'_>, extra: &[Node]) -> LinesReport {
        match judged(plan, extra) {
            Ok(report) => report,
            Err(error) => LinesReport {
                errors: vec![error],
                ..LinesReport::default()
            },
        }
    }
    
    fn judged(plan: &Workplan<'_>, extra: &[Node]) -> Result<LinesReport, String> {
        let source = plan.source();
        let record = plan.coverage().map(Loaded::value);
        let exceptions: &[Exception] = record.map_or(&[], |record| &record.exceptions);
        let judged = judged_packages(source);
        let mut units = collect(source, &judged)?;
        let mut roots = roots(plan);
        cover(source, &mut units, &mut roots, extra)?;
        let index = ExceptionIndex::new(exceptions);
        if !index.is_empty() {
            for unit in &mut units.units {
                let (paths, modules) = match_paths(source, &unit.what);
                let paths: Vec<&str> = paths.iter().map(String::as_str).collect();
                let modules: Vec<&str> = modules.iter().map(String::as_str).collect();
                unit.exception = index.unit(&paths, &modules);
            }
            for file in &mut units.files {
                file.exception = index.file(file.path.as_str());
            }
        }
        let mut totals = Totals::default();
        for (file, range) in units.files.iter().zip(&units.of_file) {
            sweep(&units.units, range.clone(), file, &mut totals);
        }
        let mut report = LinesReport::default();
        for package in source
            .packages()
            .iter()
            .filter(|package| judged.contains(package.name.as_str()))
        {
            let mut row = totals
                .crates
                .remove(&package.name)
                .unwrap_or_else(|| empty_row(&package.name));
            row.ceilings = record.map_or_else(Vec::new, |record| {
                record
                    .ceilings
                    .iter()
                    .filter(|ceiling| ceiling.package == package.name)
                    .cloned()
                    .collect()
            });
            report.crates.push(row);
        }
        report.crates.sort_by(|a, b| a.package.cmp(&b.package));
        report.modules = totals
            .modules
            .into_iter()
            .filter(|(_, (_, orphan))| *orphan > 0)
            .map(|((package, module), (counted, orphan))| ModuleLines {
                package,
                module,
                counted,
                orphan,
            })
            .collect();
        for (at, lines) in &totals.orphans {
            let unit = &units.units[*at];
            report
                .orphans
                .push(unit_lines(unit, &units.files[unit.file], lines));
        }
        report
            .orphans
            .sort_by(|a, b| unit_order(a).cmp(&unit_order(b)));
        if let Some(recorded) = plan.debt() {
            hold_debt(&mut report, recorded.value());
        }
        let mut excepted: Vec<(usize, ExceptedLines)> = Vec::new();
        for ((at, position), lines) in &totals.excepted {
            let unit = &units.units[*at];
            let exception = &exceptions[*position];
            excepted.push((
                *position,
                ExceptedLines {
                    lines: unit_lines(unit, &units.files[unit.file], lines),
                    exception: exception.target.written(),
                    reason: exception.reason.clone(),
                },
            ));
        }
        excepted.sort_by(|a, b| (unit_order(&a.1.lines), a.0).cmp(&(unit_order(&b.1.lines), b.0)));
        report.excepted = excepted.into_iter().map(|(_, row)| row).collect();
        Ok(report)
    }
    
    /// Whether the exact line count or percentage is within its ceiling.
    /// Comparing products avoids rounded division, including when no line is counted.
    fn within(orphan: usize, counted: usize, limit: &CeilingLimit) -> bool {
        use purrdf_xsd::exact::{Decimal, Integer};
        let count = |value: usize| Decimal::from_integer(Integer::from(value as u128));
        match limit {
            CeilingLimit::Lines(lines) => count(orphan).le(lines.exact()),
            CeilingLimit::Percent(percent) => {
                let left = count(orphan)
                    .try_mul(&Decimal::from(100))
                    .expect("integer product preserves decimal scale");
                let right = percent
                    .exact()
                    .try_mul(&count(counted))
                    .expect("integer product preserves decimal scale");
                left <= right
            }
        }
    }
    
    /// A package over a ceiling of the coverage record, with the finding that
    /// says so. A report that could not count the lines is one `Over` that names
    /// no package.
    #[derive(Debug, Clone)]
    pub(crate) struct Over {
        /// The package, or none for a report that could not count.
        pub(crate) package: Option<String>,
        /// `coverage-lines-over-ceiling` at `ceilings`.
        pub(crate) finding: Finding,
    }
    
    /// The report of the lines and the classes of the packages whose orphan lines
    /// are over their ceilings: each
    /// class of a package whose orphan lines are over its ceiling is
    /// `coverage-lines-over-ceiling` at `ceilings`. A package's shipping and test
    /// lines are each judged against the row of their class, held to 0 when no
    /// row of their class and no `all` row bounds them, and their sum against an
    /// `all` row. Without a coverage record that loads, or while a capability
    /// document does not load, nothing is judged: the record's own finding, or the
    /// document's, fails the check, and a document that did not load binds
    /// nothing. A report that could not count the lines is one finding.
    pub(crate) fn over_ceilings(plan: &Workplan<'_>) -> Option<(LinesReport, Vec<Over>)> {
        let loaded = plan.coverage()?;
        let capability_unloaded = plan
            .unloaded()
            .iter()
            .any(|id| id.starts_with(raf_okf::DocType::Capability.id_prefix()));
        if capability_unloaded {
            return None;
        }
        let report = plan.lines_report();
        let overs = ceiling_findings(loaded, &report);
        Some((report, overs))
    }
    
    /// The classes over their ceilings in `report`, at the record `loaded`.
    pub(crate) fn ceiling_findings(
        loaded: &Loaded<raf_okf::CoverageRecord>,
        report: &LinesReport,
    ) -> Vec<Over> {
        let finding = |message: String| {
            Finding::new(
                kind::COVERAGE_LINES_OVER_CEILING,
                loaded.path(),
                Location::Key("ceilings".to_string()),
                message,
                "bind or reach the items that explain the orphan lines, add an exception with its reason, raise the crate's ceiling, or hold the lines as debt with `cargo workplan debt --write`",
            )
        };
        if let Some(error) = report.errors().first() {
            return vec![Over {
                package: None,
                finding: finding(format!(
                    "the orphan lines of the workspace cannot be counted, so no package is judged under its ceiling: {error}"
                )),
            }];
        }
        let mut out = Vec::new();
        for row in report.crates() {
            let class = |of: CeilingClass| row.ceilings.iter().find(|ceiling| ceiling.of == of);
            let all = class(CeilingClass::All);
            let mut judge =
                |of: CeilingClass, orphan: usize, counted: usize, ceiling: Option<&Ceiling>| {
                    let held = ceiling.map_or(orphan == 0, |ceiling| {
                        within(orphan, counted, &ceiling.limit)
                    });
                    if !held {
                        out.push(Over {
                            package: Some(row.package.clone()),
                            finding: finding(over_message(report, row, of, orphan, counted, ceiling)),
                        });
                    }
                };
            for (of, orphan, counted) in [
                (
                    CeilingClass::Shipping,
                    row.shipping_orphan,
                    row.shipping_counted,
                ),
                (CeilingClass::Test, row.test_orphan, row.test_counted),
            ] {
                match class(of) {
                    Some(ceiling) => judge(of, orphan, counted, Some(ceiling)),
                    None if all.is_none() => judge(of, orphan, counted, None),
                    None => {}
                }
            }
            if let Some(ceiling) = all {
                judge(CeilingClass::All, row.orphan(), row.counted, Some(ceiling));
            }
        }
        out
    }
    
    /// The message of a class of a package over its ceiling.
    fn over_message(
        report: &LinesReport,
        row: &CrateLines,
        of: CeilingClass,
        orphan: usize,
        counted: usize,
        ceiling: Option<&Ceiling>,
    ) -> String {
        let lines = match of {
            CeilingClass::Shipping => "shipping orphan lines",
            CeilingClass::Test => "test orphan lines",
            CeilingClass::All => "orphan lines",
        };
        let bound = ceiling.map_or_else(
            || "the default of 0 lines, since no row of `ceilings` bounds them".to_string(),
            |ceiling| format!("its ceiling of {}", ceiling_text(ceiling)),
        );
        let mut largest: Vec<&UnitLines> = report
            .orphans()
            .iter()
            .filter(|unit| unit.package == row.package)
            .collect();
        largest.sort_by(|a, b| b.lines().cmp(&a.lines()).then(a.item.cmp(&b.item)));
        let named: Vec<String> = largest
            .iter()
            .take(3)
            .map(|unit| format!("`{}` ({})", unit.item, unit.lines()))
            .collect();
        format!(
            "`{}` has {} orphan {} ({} shipping, {} test) of {} counted, and its {lines}, {orphan} of {counted} counted, are over {bound}; the largest orphan items are {}",
            row.package,
            row.orphan(),
            if row.orphan() == 1 { "line" } else { "lines" },
            row.shipping_orphan,
            row.test_orphan,
            row.counted,
            named.join(", ")
        )
    }

    //! The work of the line report, counted and never timed, the exact
    //! comparison of a percent ceiling, the paths that may name what a macro
    //! generates, and a walk that fails.
    //!
    //! Each unit is swept once over its lines and looks up its own paths and the
    //! modules above them in the index of the exceptions, which is built once, so
    //! the work grows with the units plus the exceptions and not with their
    //! product.
    
    use std::collections::BTreeSet;
    use std::fmt::Write as _;
    
    use raf_okf::{CeilingLimit, Number};
    use raf_source::{MemoryTree, Namespace, Node, Workspace};

    /// The functions of a module of the crate `big`.
    const FUNCTIONS: usize = 10;
    /// The modules of one unit of units.
    const MODULES_PER_UNIT: usize = 10;
    /// The exceptions of one unit of exceptions.
    const EXCEPTIONS_PER_UNIT: usize = 20;
    /// What one unit of the report, or one exception, may cost at most: a sweep,
    /// one mark for its one line and a lookup for each segment of its path and
    /// each module above it.
    const PER_ENTRY: usize = 8;
    
    /// The work of the report of a crate of `modules` modules of ten private
    /// functions beside a bound `run`, under `exceptions` exceptions of items
    /// that are no units, with every kind of work held to `ceiling`; and the
    /// units the report swept.
    fn work(modules: usize, exceptions: usize, ceiling: usize) -> (usize, usize) {
        let mut lib = String::from("pub fn run() {}\n");
        for m in 0..modules {
            let _ = writeln!(lib, "mod m{m} {{");
            for f in 0..FUNCTIONS {
                let _ = writeln!(lib, "    fn f{f}() {{}}");
            }
            let _ = writeln!(lib, "}}");
        }
        let source = Workspace::load(
            &MemoryTree::from([
                (
                    "Cargo.toml",
                    "[package]\nname = \"big\"\nversion = \"0.1.0\"\nedition = \"2021\"\n",
                ),
                ("src/lib.rs", lib.as_str()),
            ]),
            &["."],
        )
        .expect("a workspace");
        let mut record =
            String::from("---\ntype: Coverage\nid: coverage:default\nscope: []\nexclusions: []\n");
        if exceptions > 0 {
            record.push_str("exceptions:\n");
            for e in 0..exceptions {
                let _ = write!(
                    record,
                    "  - item: big::m0::f0::x{e}\n    reason: \"Nothing of it.\"\n"
                );
            }
        }
        record.push_str("---\n");
        let capability = "---\ntype: Capability\nid: cap:big\ntitle: Big\ndescription: Big.\n\
             state: implementing\nbindings:\n  code: [big::run]\n  tests: []\n  examples: []\n\
             depends: []\n---\n\n# Big\n\n## Purpose\n\nText.\n\n## Contract\n\nText.\n\n## Usage\n\n\
             Text.\n\n## Examples\n\nText.\n\n## Constraints\n\nText.\n\n## Explanation\n\nText.\n\n\
             ## Verification\n\nText.\n";
        let docs = MemoryTree::from([
            ("docs/capabilities/big.md", capability),
            ("work/coverage/default.md", record.as_str()),
        ]);
        let plan = Workplan::load(&docs, &source);
        let (report, tally) = measure(ceiling, || plan.lines_report());
        assert!(report.errors().is_empty(), "{:?}", report.errors());
        assert_eq!(report.orphans().len(), modules * FUNCTIONS);
        (tally.total(), tally.get(Work::LineUnits))
    }
    
    /// The units of `big` with `modules` modules: its functions and `run`.
    const fn units_of(modules: usize) -> usize {
        modules * FUNCTIONS + 1
    }
    
    #[test]
    fn the_work_grows_with_the_units_and_no_faster_at_fixed_exceptions() {
        let exceptions = 3 * EXCEPTIONS_PER_UNIT;
        let mut total = [0; 3];
        for units in 1..=3 {
            let (work, swept) = work(units * MODULES_PER_UNIT, exceptions, usize::MAX);
            assert_eq!(swept, units_of(units * MODULES_PER_UNIT));
            total[units - 1] = work;
        }
        let [one, two, three] = total;
        assert!(one > 0);
        assert!(three + one <= 2 * two, "{one}, {two}, {three}");
    }
    
    #[test]
    fn the_work_grows_with_the_exceptions_and_no_faster_at_fixed_units() {
        let modules = 2 * MODULES_PER_UNIT;
        let mut total = [0; 3];
        for units in 1..=3 {
            total[units - 1] = work(modules, units * EXCEPTIONS_PER_UNIT, usize::MAX).0;
        }
        let [one, two, three] = total;
        assert!(one > 0);
        assert!(three + one <= 2 * two, "{one}, {two}, {three}");
    }
    
    /// Every kind of work is held to a ceiling linear in the units plus the
    /// exceptions, below their product, so a report that tests each exception
    /// against each unit passes the ceiling and stops at once.
    #[test]
    fn every_kind_of_work_is_held_to_a_ceiling_linear_in_the_units_and_the_exceptions() {
        let modules = 3 * MODULES_PER_UNIT;
        let exceptions = 3 * EXCEPTIONS_PER_UNIT;
        let units = units_of(modules);
        let ceiling = PER_ENTRY * (units + exceptions);
        assert!(ceiling < units * exceptions);
        let (_, swept) = work(modules, exceptions, ceiling);
        assert_eq!(swept, units);
    }

    #[test]
    fn a_percent_is_compared_as_whole_numbers_without_division_or_overflow() {
        let percent = |text: &str| CeilingLimit::Percent(Number::parse(text).expect("a number"));
        // 1 of 3 is 33.33...%.
        assert!(within(1, 3, &percent("33.334")));
        assert!(!within(1, 3, &percent("33.333")));
        assert!(within(1, 3, &percent("33.3340000")));
        assert!(within(0, 0, &percent("0")));
        assert!(!within(1, 0, &percent("100")));
        // A fraction too long for any machine word compares exactly.
        let long = format!("33.{}4", "3".repeat(80));
        assert!(within(1, 3, &percent(&long)));
        let short = format!("33.{}", "3".repeat(80));
        assert!(!within(1, 3, &percent(&short)));
        assert!(within(usize::MAX, usize::MAX, &percent("100")));
        assert!(within(1, 8, &percent("12.5")));
        assert!(!within(2, 8, &percent("12.5")));
        assert!(within(0, 3, &percent("-0.000")));
        assert!(!within(1, 3, &percent("-0.000")));
        let lines = |text: &str| CeilingLimit::Lines(Number::parse(text).expect("a number"));
        assert!(within(3, 9, &lines("3")));
        assert!(!within(4, 9, &lines("3")));
        assert!(within(4, 9, &lines("99999999999999999999999999")));
    }
    
    #[test]
    fn a_path_that_may_name_what_a_macro_generates_is_a_run_and_a_method_a_field_or_a_declared_name_is_not()
     {
        let wanted: BTreeSet<String> = ["made".to_string()].into();
        let texts = |text: &str| -> Vec<String> {
            lex_runs(text, &wanted)
                .expect("lexes")
                .into_iter()
                .map(|run| run.text)
                .collect()
        };
        assert_eq!(
            texts("fn f() { made(); a::made(); ::b::made; crate::made(); }"),
            ["made", "a::made", "::b::made", "crate::made"]
        );
        assert_eq!(
            texts("fn f() { x.made(); let made = 1; fn made() {} }"),
            [""; 0]
        );

        assert_eq!(texts("fn f() { let made = || {}; made(); }"), ["made"]);

        let text = "\u{feff}fn f() { made() }";
        let runs = lex_runs(text, &wanted).expect("lexes");
        assert_eq!(runs[0].start, text.find("made").expect("made"));

    //! The debt record of the complete policy and the ratchet over it.
    //!
    //! A repository that adopts [the complete policy](crate::strict) with findings it has
    //! not yet paid records them in `work/debt/default.md`, a record of type `Debt`
    //! that a tool writes ([`Gate::write_debt`], `cargo workplan debt --write`) and
    //! a person reviews. The rule is the one `arch:debt-ratchet` states for every
    //! ratchet of the repository, applied to the findings of the complete policy:
    //!
    //! * the record is owned by this tool and holds only its debt;
    //! * an entry is keyed, and a finding that is present and not recorded is
    //!   growth: it is reported as it is;
    //! * a recorded entry that is not present, or a count that is below its
    //!   record, fails as [`debt-resolved`](crate::Finding::KINDS) until the write
    //!   lowers the record, so the record always equals the present debt;
    //! * the write refuses growth against a record that exists, creates the record
    //!   when none exists (adoption), refuses while a finding that is not debt
    //!   exists, and takes the present state;
    //! * the control is the review of the diff of the record.
    //!
    //! # The entries
    //!
    //! The `findings` rows hold a finding of one of [`DEBT_KINDS`], keyed by the
    //! kind and a subject: the claim key `cap:<slug>#<section>/<key>` for
    //! `strict-claim-uncovered`, the capability `cap:<slug>` for
    //! `strict-example-missing`, the section `cap:<slug>#<section>` for the
    //! other content rules and `strict-claim-unit-collapsed` (the section the rule
    //! judges: Verification, Usage, Constraints and Contract for the four with
    //! one, the section of the finding for `strict-section-thin`), and the
    //! reported path of the item or test for the three coverage kinds. A finding
    //! of another kind is never debt: a finding of the default rules, the
    //! obligations of a change (they belong to an open change), an unreadable
    //! surface, and `coverage-lines-over-ceiling` as a row of `findings`.
    //!
    //! The `lines` rows hold orphan lines (see [`crate::coverage::lines`]) as a
    //! count for each item that holds some, in a package that is over a ceiling of
    //! the coverage record: the crate-qualified item path and the number of
    //! orphan lines it holds. Ranges are reported by `coverage lines` and never
    //! recorded. A package over a ceiling with a row for some of its items is
    //! judged by its rows: an item above its row, or with no row, is
    //! `coverage-lines-over-ceiling` naming the row, and an item below its row is
    //! `debt-resolved`. A package over a ceiling with no row is judged by its
    //! ceilings. A package that is within its ceilings holds no debt, and its rows
    //! are `debt-resolved`.
    //!
    //! # The write
    //!
    //! [`Gate::write_debt`] and [`Opened::write_debt`] write the present debt:
    //!
    //! * [`DebtError::Findings`] while a finding exists that no row can hold: a
    //!   finding of the default rules, a generated block that is stale, a surface
    //!   that cannot be listed, a report that cannot count the lines. Nothing is
    //!   written;
    //! * [`DebtError::Growth`] when the record exists and a finding is present that
    //!   it does not hold, or an item holds more lines than its row, or has no
    //!   row. Nothing is written;
    //! * otherwise the record is created when none exists, and replaced when the
    //!   present debt differs from it; two writes of the same state write the same
    //!   bytes, and the second leaves the file untouched.
    //!
    //! Only `work/debt/default.md` is written. A record with SPDX comment lines
    //! keeps them when the write is given none.
    //!
    //! ```rust
    //! use std::fs;
    //!
    //! use raf_workplan::debt::DebtChange;
    //! use raf_workplan::{Gate, SyncOptions, Workplan};
    //!
    //! let scratch = purrdf_testkit::TempDir::new_in(std::env::temp_dir())?;
    //! let dir = scratch.path();

    //!     .iter()
    //!     .map(|name| format!("## {name}\n\nThe {name} of the demo.\n\n"))
    //!     .collect();

    //!      depends: []\n---\n\n# Demo\n\n{sections}"
    //! );

        pub(crate) unheld: Vec<Finding>,
        /// The findings that are growth against a record that exists.
        pub(crate) growth: Vec<Finding>,

    /// Judge the findings of the complete policy against the debt record.
    pub(crate) fn judge(plan: &Workplan<'_>) -> Judged {
        let record = plan.debt();
        let (invalid_rows, recorded) = record.map_or_else(Default::default, invalid);
        let mut out = Judged::default();
        for owed in strict::owed(plan) {
            let Some(key) = owed.key else {
                out.unheld.push(owed.finding.clone());
                out.reported.push(owed.finding);
                continue;
            };
            let held = recorded
                .findings
                .contains(&(key.kind, key.subject.as_str()));
            out.findings.insert(key);
            if !held {
                if record.is_some() {
                    out.growth.push(owed.finding.clone());
                }
                out.reported.push(owed.finding);
            }
        }
        let mut resolved = Vec::new();
        if let Some(judged) = lines::over_ceilings(plan) {
            lines_judged(plan, judged, &recorded, &mut out, &mut resolved);
        }
        out.reported.extend(invalid_rows);
        if let Some(record) = record {
            for row in &record.value().findings {
                let pair = (row.kind.as_str(), row.subject.as_str());
                if !recorded.findings.contains(&pair) {
                    continue;
                }
                let present = out
                    .findings
                    .iter()
                    .any(|key| (key.kind, key.subject.as_str()) == pair);
                if !present {
                    out.reported.push(Finding::new(
                        kind::DEBT_RESOLVED,
                        record.path(),
                        Location::Key("findings".to_string()),
                        format!(
                            "the recorded `{}` for `{}` is no longer raised, so the record holds debt that is paid",
                            row.kind, row.subject
                        ),
                        RESOLVED_FIX,
                    ));
                }
            }
        }
        out.reported.extend(resolved);
        out
    }

    /// The `debt-entry-invalid` findings of the rows of the record, in record
    /// order, and the rows that are valid.
    fn invalid(record: &Loaded<DebtRecord>) -> (Vec<Finding>, Recorded<'_>) {
        let mut out = Vec::new();
        let mut recorded = Recorded::default();
        for row in &record.value().findings {
            let problem = if !DEBT_KINDS.contains(&row.kind.as_str()) {
                Some(format!(
                    "`{}` is not a finding a debt record can hold; the kinds are {}",
                    row.kind,
                    DEBT_KINDS.join(", ")
                ))
            } else if !subject_fits(&row.kind, &row.subject) {
                Some(format!(
                    "`{}` is not the subject of a `{}`: {}",
                    row.subject,
                    row.kind,
                    subject_form(&row.kind)
                ))
            } else {
                None
            };
            match problem {
                Some(problem) => out.push(entry_invalid(record, "findings", &problem)),
                None => {
                    recorded
                        .findings
                        .insert((row.kind.as_str(), row.subject.as_str()));
                }
            }
        }
        for row in &record.value().lines {
            if !item_fits(&row.item) {
                out.push(entry_invalid(
                    record,
                    "lines",
                    &format!(
                        "`{}` is not an item path: it is the crate-qualified path of an item, with a `::` and no space or control character at either end",
                        row.item
                    ),
                ));
            } else if row.orphans == 0 {
                out.push(entry_invalid(
                    record,
                    "lines",
                    &format!(
                        "`{}` records 0 orphan lines, which is no debt; a row holds at least one line",
                        row.item
                    ),
                ));
            } else {
                recorded.lines.insert(row.item.as_str(), row.orphans);
            }
        }
        (out, recorded)
    }

    /// Whether `text` is a capability id.
    fn capability(text: &str) -> bool {
        text.strip_prefix("cap:").is_some_and(is_kebab)
    }
    
    /// Whether `text` is `cap:<slug>#<section>`, of `only` when given.
    fn section(text: &str, only: Option<CapabilitySection>) -> bool {
        text.split_once('#').is_some_and(|(id, slug)| {
            capability(id)
                && CapabilitySection::from_slug(slug)
                    .is_some_and(|found| only.is_none_or(|w| w == found))
        })
    }
    
    /// Whether `text` is a claim key `cap:<slug>#<section>/<key>` of the Contract
    /// or the Constraints.
    fn claim(text: &str) -> bool {
        text.split_once('#').is_some_and(|(id, rest)| {
            rest.split_once('/').is_some_and(|(slug, key)| {
                capability(id)
                    && matches!(
                        CapabilitySection::from_slug(slug),
                        Some(CapabilitySection::Contract | CapabilitySection::Constraints)
                    )
                    && !key.is_empty()
                    && key.chars().all(|c| c.is_alphanumeric() || c == '-')
            })
        })
    }
    
    /// Whether `text` is an item path: it has a `::` and neither a control
    /// character nor a space at either end.
    fn item_fits(text: &str) -> bool {
        text.contains("::")
            && text == text.trim()
            && !text.chars().any(char::is_control)
            && !text.starts_with("::")
            && !text.ends_with("::")
    }
    
    /// Whether `subject` has the form of the subject of `kind`.
    fn subject_fits(kind_name: &str, subject: &str) -> bool {
        match kind_name {
            kind::STRICT_EXAMPLE_MISSING => capability(subject),
            kind::STRICT_MATRIX_MISSING => section(subject, Some(CapabilitySection::Verification)),
            kind::STRICT_USAGE_UNBOUND => section(subject, Some(CapabilitySection::Usage)),
            kind::STRICT_LIMITATIONS_MISSING => section(subject, Some(CapabilitySection::Constraints)),
            kind::STRICT_CLAIM_UNIT_COLLAPSED => section(subject, Some(CapabilitySection::Contract)),
            kind::STRICT_SECTION_THIN => section(subject, None),
            kind::STRICT_CLAIM_UNCOVERED => claim(subject),
            kind::COVERAGE_ITEM_UNBOUND
            | kind::COVERAGE_REEXPORT_UNBOUND
            | kind::COVERAGE_TEST_UNBOUND => item_fits(subject) && !subject.contains(' '),
            _ => false,
        }
    }
    
    /// The form of the subject of `kind`, for a message.
    fn subject_form(kind_name: &str) -> &'static str {
        match kind_name {
            kind::STRICT_EXAMPLE_MISSING => "it is a capability, `cap:<slug>`",
            kind::STRICT_MATRIX_MISSING => "it is `cap:<slug>#verification`",
            kind::STRICT_USAGE_UNBOUND => "it is `cap:<slug>#usage`",
            kind::STRICT_LIMITATIONS_MISSING => "it is `cap:<slug>#constraints`",
            kind::STRICT_CLAIM_UNIT_COLLAPSED => "it is `cap:<slug>#contract`",
            kind::STRICT_SECTION_THIN => "it is a section, `cap:<slug>#<section>`",
            kind::STRICT_CLAIM_UNCOVERED => {
                "it is a claim key, `cap:<slug>#contract/<key>` or `cap:<slug>#constraints/<key>`"
            }
            _ => "it is the reported path of an item or a test, such as `crate::item`",
        }
    }
    
    /// The orphan lines of each item that holds some: the package of its first
    /// unit and its lines.
    type Present<'r> = BTreeMap<&'r str, (&'r str, usize)>;
    
    /// Judge the orphan lines against the ceilings and the rows of `lines`: the
    /// findings that stand go to `out.reported`, those of rows now paid to
    /// `resolved`.
    fn lines_judged(
        plan: &Workplan<'_>,
        (report, overs): (LinesReport, Vec<lines::Over>),
        recorded: &Recorded<'_>,
        out: &mut Judged,
        resolved: &mut Vec<Finding>,
    ) {
        if !report.errors().is_empty() {
            for over in overs {
                out.unheld.push(over.finding.clone());
                out.reported.push(over.finding);
            }
            return;
        }
        let record = plan.debt();
        let over_packages: BTreeSet<&str> = overs
            .iter()
            .filter_map(|over| over.package.as_deref())
            .collect();
        let mut present: Present<'_> = BTreeMap::new();
        for unit in report.orphans() {
            let entry = present.entry(unit.item()).or_insert((unit.package(), 0));
            entry.1 += unit.lines();
        }
        // The present debt: the items of the packages over a ceiling.
        let held: Present<'_> = present
            .iter()
            .filter(|(_, (package, _))| over_packages.contains(package))
            .map(|(item, found)| (*item, *found))
            .collect();
        out.lines = held
            .iter()
            .map(|(item, (_, count))| ((*item).to_string(), *count))
            .collect();
        // A package with a row for one of its items is judged by its rows.
        let by_rows: BTreeSet<&str> = held
            .iter()
            .filter(|(item, _)| recorded.lines.contains_key(*item))
            .map(|(_, (package, _))| *package)
            .collect();
        for over in overs {
            let by_row = over
                .package
                .as_deref()
                .is_some_and(|package| by_rows.contains(package));
            if !by_row {
                if record.is_some() {
                    out.growth.push(over.finding.clone());
                }
                out.reported.push(over.finding);
            }
        }
        let debt_path = record.map_or("work/debt/default.md", |record| record.path());
        for (item, (package, count)) in &held {
            if !by_rows.contains(package) {
                continue;
            }
            let message = match recorded.lines.get(item) {
                None => format!(
                    "`{item}` of `{package}` holds {} and the debt record has no row for it, so its lines are over the ceilings of `{package}`",
                    lines_text(*count)
                ),
                Some(row) if count > row => format!(
                    "`{item}` of `{package}` holds {}, over the {} its row of `lines` records",
                    lines_text(*count),
                    lines_text(*row)
                ),
                Some(_) => continue,
            };
            let finding = Finding::new(
                kind::COVERAGE_LINES_OVER_CEILING,
                debt_path,
                Location::Key("lines".to_string()),
                message,
                "bind or reach the items that explain the orphan lines, or add an exception with its reason; the debt record only ever shrinks, and is lowered with `cargo workplan debt --write`",
            );
            out.growth.push(finding.clone());
            out.reported.push(finding);
        }
        if let Some(record) = record {
            for (item, row) in &recorded.lines {
                let reason = match (held.get(item), present.get(item)) {
                    (Some((_, count)), _) if count < row => {
                        format!("now holds {}", lines_text(*count))
                    }
                    (Some(_), _) => continue,
                    (None, Some((package, count))) => format!(
                        "holds {} and `{package}` is within its ceilings",
                        lines_text(*count)
                    ),
                    (None, None) => "holds no orphan line".to_string(),
                };
                resolved.push(Finding::new(
                    kind::DEBT_RESOLVED,
                    record.path(),
                    Location::Key("lines".to_string()),
                    format!(
                        "the record holds {} of `{item}` as debt, and the item {reason}",
                        lines_text(*row)
                    ),
                    RESOLVED_FIX,
                ));
            }
        }
    }
    
    /// `1 orphan line`, `5 orphan lines`.
    fn lines_text(count: usize) -> String {
        if count == 1 {
            "1 orphan line".to_string()
        } else {
            format!("{count} orphan lines")
        }
    }

    /// Why the debt record was not written. Nothing is written in any case.
    #[derive(Debug)]
    #[non_exhaustive]
    pub enum DebtError {
        /// The repository cannot be opened.
        Open(GateError),
        /// A finding exists that no row can hold: a finding of the default rules
        /// or one the complete policy cannot record. Pay it first.
        Findings(Vec<Finding>),
        /// The debt would grow: the findings are the present debt that the record
        /// does not hold. A debt only shrinks; pay it, or remove the record to
        /// adopt anew.
        Growth(Vec<Finding>),
        /// The record could not be written.
        Record(RecordError),
    }
    
    impl fmt::Display for DebtError {
        fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
            match self {
                Self::Open(error) => write!(f, "{error}"),
                Self::Findings(findings) => {
                    write!(
                        f,
                        "{} finding(s) that no debt record can hold stand, so nothing is written; pay them first",
                        findings.len()
                    )?;
                    crate::lookup::write_findings(f, findings)
                }
                Self::Growth(findings) => {
                    write!(
                        f,
                        "the debt would grow by {} finding(s), so nothing is written; a debt record only ever shrinks, so pay them",
                        findings.len()
                    )?;
                    crate::lookup::write_findings(f, findings)
                }
                Self::Record(error) => write!(f, "{error}"),
            }
        }
    }
    
    impl Error for DebtError {
        fn source(&self) -> Option<&(dyn Error + 'static)> {
            match self {
                Self::Open(error) => Some(error),
                Self::Record(error) => Some(error),
                Self::Findings(_) | Self::Growth(_) => None,
            }
        }
    }
    
    impl From<GateError> for DebtError {
        fn from(error: GateError) -> Self {
            Self::Open(error)
        }
    }
    
    impl From<RecordError> for DebtError {
        fn from(error: RecordError) -> Self {
            Self::Record(error)
        }
    }
    
    impl Gate {
        /// Write the debt record `work/debt/default.md`: the present debt of
        /// the complete policy, as [the module documentation](crate::debt) states, with
        /// the SPDX lines of [`Gate::record_options`] on a record it creates and
        /// the lines a record already has otherwise.
        ///
        /// # Errors
        ///
        /// A [`DebtError`]: the repository cannot be opened, is not judged strict,
        /// holds a finding no row can hold or a debt that grew, or the record
        /// cannot be written. Nothing is written in any of them.
        pub fn write_debt(&self) -> Result<DebtWrite, DebtError> {
            self.open()?.write_debt(self.record_spdx())
        }
    }
    
    impl Opened {
        /// Write the debt record as [`Gate::write_debt`] does, with the SPDX
        /// lines of `options` when the record is created or replaced and `options`
        /// gives some, and the lines the record has otherwise.
        ///
        /// # Errors
        ///
        /// The errors of [`Gate::write_debt`] other than opening the repository.
        pub fn write_debt(&self, options: &RecordOptions) -> Result<DebtWrite, DebtError> {
            let plan = self.workplan();
            let judged = judge(&plan);
            let mut blocking = plan.base_findings();
            blocking.extend(judged.unheld.iter().cloned());
            if !blocking.is_empty() {
                return Err(DebtError::Findings(blocking));
            }
            if plan.debt().is_some() && !judged.growth.is_empty() {
                return Err(DebtError::Growth(judged.growth));
            }
            let debt = DebtRecord::new(
                judged
                    .findings
                    .iter()
                    .map(|key| DebtFinding::new(key.kind, key.subject.as_str()))
                    .collect(),
                judged
                    .lines
                    .iter()
                    .map(|(item, count)| DebtLine::new(item.as_str(), *count))
                    .collect(),
            );
            let (findings, lines) = (debt.findings.len(), debt.lines.len());
            let record = Record::Debt(debt);
            let options = kept(&plan, options);
            let text = record.text(&options)?;
            let path = record.path()?;
            let change = match plan.debt() {
                Some(existing) if existing.text() == text => DebtChange::Unchanged,
                Some(_) => {
                    record::replace(self.dir(), &record, &options)?;
                    DebtChange::Replaced
                }
                None => {
                    record::create(self.dir(), &record, &options)?;
                    DebtChange::Created
                }
            };
            Ok(DebtWrite {
                path: path.to_string(),
                change,
                findings,
                lines,
            })
        }
    }
    
    /// `options`, or the SPDX lines of the record that exists when `options`
    /// gives none.
    fn kept(plan: &Workplan<'_>, options: &RecordOptions) -> RecordOptions {
        if options.copyright.is_some() || options.licence.is_some() {
            return options.clone();
        }
        let Some((copyright, licence)) = plan.debt().and_then(|record| spdx::read(record.head()))
        else {
            return options.clone();
        };
        RecordOptions::new().copyright(copyright).licence(licence)
    }

    //! The complete content and binding policy of every workplan check.
    //!
    //! The complete check judges these rules after the document-model checks.
    //! Loading documents retains its own findings without judging these rules;
    //! every complete check applies them, without a policy switch.
    //!
    //! The findings that a repository has not yet paid can be recorded in the debt
    //! record ([`crate::debt`]): [`Workplan::check`] reports, of the findings of this
    //! module and of the ceilings of reverse coverage, only those the record does not
    //! hold, and reports the rows of the record that are paid.
    //!
    //! # Examples, the matrix and the sections
    //!
    //! For each loaded `delivered` capability:
    //!
    //! * `strict-example-missing` at the key `bindings`: `bindings.examples` is
    //!   empty. That a bound example runs is proven where the default rules prove
    //!   it, by `verify` and condition 5 of the completion gate, which refuse a
    //!   documentation test that is `no_run` or `compile_fail`.
    //! * `strict-matrix-missing` at the section Verification: the Verification
    //!   section holds no `verification-matrix` generated block.
    //! * `strict-section-thin` at the section: one of the seven sections holds
    //!   fewer than [`MIN_SECTION_WORDS`] words of prose. The prose of a section
    //!   is its lines below its heading outside fenced code, HTML comments, code
    //!   spans, PROPOSED blocks and generated blocks, cut into words as the tense
    //!   scan of reconcile cuts them: runs of letters, digits and `_`, with the
    //!   destination of a link left out.
    //! * `strict-usage-unbound` at the section Usage: no path of
    //!   `bindings.examples` substantiates Usage, that is, none is named in a code
    //!   span of the Usage section outside an HTML comment, so the verification
    //!   matrix lists Usage for no example that `verify` runs.
    //! * `strict-limitations-missing` at the section Constraints: the Constraints
    //!   section holds no list item with text or a code span after its marker; an
    //!   empty item and a thematic break such as `* * *` are none.
    //!
    //! # Claims
    //!
    //! A claim is a unit of the authored prose of the Contract and the
    //! Constraints of a delivered capability, read outside fenced code, PROPOSED
    //! blocks and generated blocks. In the Contract it is a clause: it starts at a
    //! paragraph whose first line opens with a bold run (`**The setting.**`) or
    //! at a `###` or `####` heading outside fenced code, and runs to the next
    //! clause start or the end of the section; text before the first clause start
    //! is one clause, the opening, and the statement of a heading clause is its
    //! heading line. In the Constraints it is each top-level list item and each
    //! paragraph or table outside a list; a thematic break and a list item with
    //! nothing after its marker are none. A blank line, a comment line, a heading of
    //! any level, a fence and a hidden block each end a paragraph, and a list item
    //! goes on after a blank or a comment line only with the lines indented to its
    //! content.
    //!
    //! Its key, `cap:<slug>#<section>/<key>` ([`ClaimCoverage::key`]), takes the
    //! kebab slug of the bold lead or the heading, `opening` for the opening, and
    //! for a Constraints unit `c-` and the first 16 hexadecimal digits of the
    //! SHA-256 digest of its text with its comments left out and each run of
    //! whitespace read as one space; a key that repeats in its section takes the
    //! first of `-2`, `-3` and on that no other unit of the section has or takes, in
    //! document order, so the keys of a section are distinct.
    //!
    //! A Contract clause is covered by a tested-by marker on any of its lines, a
    //! Constraints unit by a marker whose statement starts in it. Under strict:
    //!
    //! * `strict-claim-uncovered` at the first line of each claim that no marker
    //!   covers, its message naming the key and the first 60 characters of its
    //!   lead or item;
    //! * `strict-claim-unit-collapsed` at the Contract when it has no clause start
    //!   and holds more than [`MIN_SECTION_WORDS`] words, as the section rule counts
    //!   them, whether or not a marker covers it.
    //!
    //! [`Workplan::claims_report`] lists every claim of every delivered capability
    //! on every complete check. That a marker names only paths the capability
    //! binds in `tests` or `examples` is a rule of the load, `claim-test-unbound`.
    //!
    //! # Explained items
    //!
    //! The code and the tests directions of two-way coverage
    //! ([`crate::coverage`]) are enforced for every package that holds an item a
    //! `code`, `tests` or `examples` binding of a loaded capability resolves to and
    //! that the coverage record's `scope` does not name, with the kinds the
    //! default rule gives a gap: `coverage-item-unbound`,
    //! `coverage-reexport-unbound` and `coverage-test-unbound` at the file and line
    //! of the gap, and `coverage-surface-unreadable` at the key `bindings` of the
    //! capability whose binding reaches the package. The message names that
    //! capability and the path it binds, the first binding of the graph that
    //! reaches the package, as the reason the complete policy enforces it. The exclusions
    //! of the coverage record apply as they do to `scope`, and a repository whose
    //! coverage record is missing or does not load has no gap of this rule, as it
    //! has none of `scope`: the record's own finding fails the check. A package that no
    //! binding reaches and `scope` does not name is not enforced, and while a
    //! capability document does not load no gap is reported, as for `scope`. This
    //! coverage is computed here: the default coverage check is the same under
    //! every complete check.

    /// The gaps of two-way coverage of the packages a binding reaches and `scope`
    /// does not name. Without a coverage record that loads there are none: the
    /// record's absence, or its own findings, fail the check first.
    fn explained(plan: &Workplan<'_>) -> Vec<Owed> {
        let Some(record) = plan.coverage() else {
            return Vec::new();
        };
        let reasons = reaching_bindings(plan);
        if reasons.is_empty() {
            return Vec::new();
        }
        add(Work::StrictUnits, reasons.len());
        let packages: BTreeSet<&str> = reasons.keys().copied().collect();
        let Computed { report, .. } =
            coverage::compute(plan.source(), plan.graph(), Some(record.value()), &packages);
        let mut out = Vec::new();
        for error in report.errors() {
            let Some(binding) = reasons.get(error.package()) else {
                continue;
            };
            // An unreadable surface is an error and no debt: no row can hold it.
            out.push(Owed::unheld(Finding::new(
                kind::COVERAGE_SURFACE_UNREADABLE,
                capability_path(plan, binding.capability()),
                Location::Key("bindings".to_string()),
                format!(
                    "`{}` binds `{}` of `{}`, which the complete policy therefore enforces, and {error}",
                    binding.capability(),
                    binding.path(),
                    error.package()
                ),
                "make every public name of the crate resolve to one item (the error names the file and line), or raise the item-bytes budget of the load; under the complete policy a package that a binding reaches is not judged covered while its surface cannot be read",
            )));
        }
        // A capability document that did not load binds nothing in the graph, so
        // every item and test it binds would read as a gap, as for `scope`.
        let capability_unloaded = plan
            .unloaded()
            .iter()
            .any(|id| id.starts_with(DocType::Capability.id_prefix()));
        if capability_unloaded {
            return out;
        }
        for gap in report.code().iter().chain(report.tests()) {
            let Some(binding) = reasons.get(gap.package()) else {
                continue;
            };
            let enforced = format!(
                "the complete policy enforces because `{}` binds `{}`",
                binding.capability(),
                binding.path()
            );
            out.push(Owed::held(
                coverage::gap_finding(gap, &enforced),
                gap.path(),
            ));
        }
        out
    }
    
    /// Each package outside `scope` that a binding of a loaded capability
    /// reaches, with the first binding of the graph that reaches it.
    fn reaching_bindings<'p>(plan: &'p Workplan<'_>) -> BTreeMap<&'p str, &'p Binding> {
        let source = plan.source();
        let scope: BTreeSet<&str> = plan
            .graph()
            .scope()
            .iter()
            .map(ScopeEdge::package)
            .collect();
        let mut reasons = BTreeMap::new();
        for binding in plan.graph().bindings() {
            for id in binding.items() {
                let Some(target) = source.item(*id).and_then(|item| source.target(item.target)) else {
                    continue;
                };
                let package = target.package.as_str();
                if !scope.contains(package) {
                    reasons.entry(package).or_insert(binding);
                }
            }
        }
        reasons
    }

    /// The list item `raw` opens: after an indentation of spaces, `-`, `*` or
    /// `+`, or one to nine digits and `.` or `)`, then a space or the end of the
    /// line. A thematic break such as `* * *` opens one too; [`is_thematic_break`]
    /// tells it apart.
    fn list_item(raw: &str) -> Option<ListItem<'_>> {
        let indent = raw.len() - raw.trim_start_matches(' ').len();
        let text = &raw[indent..];
        let width = if text.starts_with(['-', '*', '+']) {
            1
        } else {
            let digits = text.bytes().take_while(u8::is_ascii_digit).count();
            if digits == 0 || digits > 9 || !text[digits..].starts_with(['.', ')']) {
                return None;
            }
            digits + 1
        };
        let after = &text[width..];
        let content = if after.is_empty() {
            after
        } else {
            after.strip_prefix(' ')?
        };
        Some(ListItem {
            column: indent + width + 1,
            content,
        })
    }
    
    /// Whether a line is a thematic break: three or more of one of `-`, `*` or `_`
    /// and nothing else but spaces and tabs.
    fn is_thematic_break(raw: &str) -> bool {
        let marks: Vec<char> = raw.chars().filter(|c| !matches!(c, ' ' | '\t')).collect();
        marks.len() >= 3
            && ['-', '*', '_']
                .iter()
                .any(|mark| marks.iter().all(|c| c == mark))
    }
    
    /// Whether `raw` opens a list item that states something: an item with text
    /// or a code span after its marker, and not a thematic break.
    fn states_an_item(raw: &str) -> bool {
        list_item(raw).is_some_and(|item| !item.content.trim().is_empty()) && !is_thematic_break(raw)
    }

    fn capability_path<'p>(plan: &'p Workplan<'_>, id: &'p str) -> &'p str {
        plan.capabilities()
            .iter()
            .find(|capability| capability.value().header.id == id)
            .map_or(id, Loaded::path)

    /// The claim units of the Contract and the Constraints of `body`, each with
    /// the paths of the markers that cover it.
    fn units(body: &Body) -> Units {
        let mut found = Units::default();
        for section in body.sections() {
            let rows = rows(body, section);
            match CapabilitySection::from_slug(section.slug()) {
                Some(CapabilitySection::Contract) => {
                    let (units, collapsed) = contract(body, section, &rows);
                    found.units.extend(units);
                    found.collapsed = found.collapsed.or(collapsed);
                }
                Some(CapabilitySection::Constraints) => found.units.extend(constraints(&rows)),
                _ => {}
            }
        }
        number_repeats(&mut found.units);
        cover(body, &mut found.units);
        found
    }
    
    /// The lines of a section below its heading.
    fn rows<'b>(body: &'b Body, section: &Section) -> Vec<Row<'b>> {
        let Some(text) = body.section_text(section) else {
            return Vec::new();
        };
        let first = section.content().lines.start;
        let within = |lines: &Range<usize>, line: usize| lines.contains(&line);
        text.split_inclusive('\n')
            .enumerate()
            .map(|(index, raw)| {
                let line = first + index;
                let raw = raw.trim_end_matches(['\n', '\r']);
                let hidden = body
                    .proposed_blocks()
                    .iter()
                    .any(|block| within(&block.span().lines, line))
                    || body
                        .generated_blocks()
                        .iter()
                        .any(|block| within(&block.outer().lines, line));
                let fenced = body
                    .fenced_blocks()
                    .iter()
                    .any(|block| within(&block.span().lines, line));
                let block = if hidden {
                    Block::Hidden
                } else if fenced {
                    Block::Fenced
                } else {
                    Block::Text
                };
                Row {
                    line,
                    raw,
                    text: without_comments(raw),
                    block,
                }
            })
            .collect()
    }
    
    /// `raw` with each comment that opens and closes on the line removed.
    fn without_comments(raw: &str) -> String {
        let mut out = String::with_capacity(raw.len());
        let mut rest = raw;
        while let Some(open) = rest.find("<!--") {
            let Some(close) = rest[open + 4..].find("-->") else {
                break;
            };
            out.push_str(&rest[..open]);
            rest = &rest[open + 4 + close + 3..];
        }
        out.push_str(rest);
        out
    }
    
    /// The clause starts and the opening of the Contract, and its words when it
    /// has no clause start and is longer than [`MIN_SECTION_WORDS`].
    fn contract(body: &Body, section: &Section, rows: &[Row<'_>]) -> (Vec<Unit>, Option<usize>) {
        let end = section.content().lines.end;
        // (line, text, whether it starts a clause): every heading ends a
        // paragraph, and a `###` or `####` heading starts a clause.
        let headings: Vec<(usize, &str, bool)> = body
            .headings()
            .iter()
            .map(|heading| {
                (
                    heading.line(),
                    heading.text(),
                    matches!(heading.level(), 3 | 4),
                )
            })
            .collect();
        // (index of the row, key base, lead)
        let mut starts: Vec<(usize, String, String)> = Vec::new();
        let mut boundary = true;
        for (index, row) in rows.iter().enumerate() {
            if row.block != Block::Text {
                boundary = true;
                continue;
            }
            if let Some((_, text, clause)) = headings.iter().find(|(line, ..)| *line == row.line) {
                if *clause {
                    starts.push((index, kebab(text), lead(text)));
                }
                boundary = true;
                continue;
            }
            if row.is_empty() {
                boundary = true;
                continue;
            }
            if boundary && let Some(bold) = bold_lead(row.raw) {
                starts.push((index, kebab(bold), lead(&paragraph(rows, index))));
            }
            boundary = false;
        }
        let content = |range: Range<usize>| {
            rows[range]
                .iter()
                .position(|row| row.block == Block::Text && !row.is_empty())
        };
        let mut units = Vec::new();
        let first_start = starts.first().map_or(rows.len(), |(index, _, _)| *index);
        if let Some(at) = content(0..first_start) {
            units.push(Unit {
                section: CapabilitySection::Contract,
                key: "opening".to_string(),
                line: rows[at].line,
                lines: rows[at].line
                    ..starts
                        .first()
                        .map_or(end, |(index, _, _)| rows[*index].line),
                lead: lead(&paragraph(rows, at)),
                tests: Vec::new(),
            });
        }
        for (position, (index, key, text)) in starts.iter().enumerate() {
            let until = starts
                .get(position + 1)
                .map_or(end, |(next, _, _)| rows[*next].line);
            units.push(Unit {
                section: CapabilitySection::Contract,
                key: key.clone(),
                line: rows[*index].line,
                lines: rows[*index].line..until,
                lead: text.clone(),
                tests: Vec::new(),
            });
        }
        let collapsed = if starts.is_empty() {
            let words = contract_words(body, section);
            (words > MIN_SECTION_WORDS).then_some(words)
        } else {
            None
        };
        (units, collapsed)
    }
    
    /// The words of prose of the Contract, counted as the section rule of strict
    /// mode counts them.
    fn contract_words(body: &Body, section: &Section) -> usize {
        body.prose_lines()
            .iter()
            .filter(|line| {
                line.section() == Some(section.slug())
                    && line.line() != section.line()
                    && !line.in_proposed()

    /// What a Constraints unit is while it is read.
    #[derive(Debug, Clone, Copy, PartialEq, Eq)]
    enum Shape {
        /// A top-level list item and the column its content starts at.
        Item(usize),
        Paragraph,
        Table,
    }
    
    /// The units of the Constraints: each top-level list item, and each
    /// paragraph or table outside a list. A blank line or a comment line ends a
    /// paragraph or a table, after which a list item goes on only with the lines
    /// indented to its content; a heading, a fence and a hidden block end every
    /// unit, and a thematic break and an item with nothing after its marker end
    /// the open unit and open none.
    fn constraints(rows: &[Row<'_>]) -> Vec<Unit> {
        let mut units = Vec::new();
        // (shape, first row, last row with text, text)
        let mut open: Option<(Shape, usize, usize, Vec<String>)> = None;
        // A blank line, or a comment line, since the last line of text: a list
        // item continues after one only with lines indented to its content.
        let mut after_blank = false;
        let close = |open: &mut Option<(Shape, usize, usize, Vec<String>)>, units: &mut Vec<Unit>| {
            if let Some((_, first, last, text)) = open.take() {
                units.push(constraint_unit(rows, first, last, &text));
            }
        };
        for (index, row) in rows.iter().enumerate() {
            let in_item = matches!(open, Some((Shape::Item(_), ..)));
            match row.block {
                Block::Hidden => {
                    close(&mut open, &mut units);
                    after_blank = false;
                    continue;
                }
                Block::Fenced => {
                    // A fence indented under a list item belongs to the item;
                    // any other fence ends the unit and is no unit of its own.
                    if in_item && row.indent() > 0 {
                        if let Some((_, _, last, _)) = open.as_mut() {
                            *last = index;
                        }
                    } else {
                        close(&mut open, &mut units);
                    }
                    after_blank = false;
                    continue;
                }
                Block::Text => {}
            }
            if row.raw.trim().is_empty() {
                after_blank = true;
                if !in_item {
                    close(&mut open, &mut units);
                }
                continue;
            }
            if row.is_empty() {
                // A comment line, such as a marker, ends a paragraph or a table;
                // a list item goes on with the lines indented to its content.
                after_blank = true;
                if !in_item {
                    close(&mut open, &mut units);
                }
                continue;
            }
            let trimmed = row.raw.trim_start();
            if trimmed.starts_with('#') && is_heading(trimmed) {
                close(&mut open, &mut units);
                after_blank = false;
                continue;
            }
            // A thematic break and a list item with nothing after its marker
            // state nothing: each ends the open unit and opens none.
            let item = list_item(row.raw);
            if is_thematic_break(row.raw) || item.is_some_and(|item| item.content.trim().is_empty()) {
                close(&mut open, &mut units);
                after_blank = false;
                continue;
            }
            let item = item.map(|item| item.column);
            let starts_item = match (item, open.as_ref()) {
                (Some(_), Some((Shape::Item(column), ..))) => row.indent() < *column,
                (Some(_), _) => true,
                (None, _) => false,
            };
            let continues = match open.as_ref() {
                Some((Shape::Item(column), ..)) => !after_blank || row.indent() >= *column,
                Some((Shape::Paragraph | Shape::Table, ..)) => !after_blank,
                None => false,
            };
            if starts_item {
                close(&mut open, &mut units);
                let column = item.unwrap_or_default();
                open = Some((Shape::Item(column), index, index, vec![row.text.clone()]));
            } else if continues {
                if let Some((_, _, last, text)) = open.as_mut() {
                    *last = index;
                    text.push(row.text.clone());
                }
            } else {
                close(&mut open, &mut units);
                let shape = if trimmed.starts_with('|') {
                    Shape::Table
                } else {
                    Shape::Paragraph
                };
                open = Some((shape, index, index, vec![row.text.clone()]));
            }
            after_blank = false;
        }
        close(&mut open, &mut units);
        units
    }
    
    /// Whether a line, without its indentation, is an ATX heading.
    fn is_heading(trimmed: &str) -> bool {
        let marks = trimmed.bytes().take_while(|b| *b == b'#').count();
        (1..=6).contains(&marks) && trimmed[marks..].starts_with([' ', '\t'])
    }
    
    /// The unit of the Constraints from row `first` to row `last`, its key the
    /// digest of its text with each run of whitespace read as one space.
    fn constraint_unit(rows: &[Row<'_>], first: usize, last: usize, text: &[String]) -> Unit {
        let folded = fold(&text.join(" "));
        let hash = purrdf_core::content_store::ContentDigest::of(folded.as_bytes());
        let mut key = String::from("c-");
        purrdf_hash::hex::encode_into(&hash.as_bytes()[..DIGEST_DIGITS / 2], &mut key);
        let item = folded
            .strip_prefix(['-', '*', '+'])
            .map_or(folded.as_str(), str::trim_start);
        Unit {
            section: CapabilitySection::Constraints,
            key,
            line: rows[first].line,
            lines: rows[first].line..rows[last].line + 1,
            lead: lead(item),
            tests: Vec::new(),
        }
    }
    
    /// `text` with each run of whitespace read as one space, trimmed.
    fn fold(text: &str) -> String {
        text.split_whitespace().collect::<Vec<_>>().join(" ")
    }
    
    /// The first [`LEAD_CHARS`] characters of `text`, folded, without the `**` of
    /// bold runs and without a space they end in.
    fn lead(text: &str) -> String {
        let cut: String = fold(&text.replace("**", ""))
            .chars()
            .take(LEAD_CHARS)
            .collect();
        cut.trim_end().to_string()
    }
    
    /// The kebab slug of a lead or a heading: its letters and digits in lowercase,
    /// each other run of characters one `-`, none at either end; `clause` for a
    /// text with no letter or digit.
    fn kebab(text: &str) -> String {
        let mut out = String::new();
        let mut gap = false;
        for c in text.chars() {
            if c.is_alphanumeric() {
                if gap && !out.is_empty() {
                    out.push('-');
                }
                gap = false;
                out.extend(c.to_lowercase());
            } else {
                gap = true;
            }
        }
        if out.is_empty() {
            "clause".to_string()
        } else {
            out
        }
    }
    
    /// Number the repeats of a key within its section, in document order: a key
    /// that an earlier unit of the section has takes the first of `-2`, `-3` and on
    /// that is no key a unit of the section has or takes, so the keys of a section
    /// are distinct whatever its leads are (`**Foo.**`, `**Foo.**` and `**Foo 2.**`
    /// give `foo`, `foo-3` and `foo-2`).
    fn number_repeats(units: &mut [Unit]) {
        let mut taken: BTreeSet<(&'static str, String)> = units
            .iter()
            .map(|unit| (unit.section.slug(), unit.key.clone()))
            .collect();
        let mut seen: BTreeSet<(&'static str, String)> = BTreeSet::new();
        for unit in units {
            let section = unit.section.slug();
            if seen.insert((section, unit.key.clone())) {
                continue;
            }
            // Fewer keys are taken than the numbers tried, so one is free.
            let key = (2..=taken.len() + 2)
                .map(|count| format!("{}-{count}", unit.key))
                .find(|key| !taken.contains(&(section, key.clone())))
                .unwrap_or_default();
            taken.insert((section, key.clone()));
            seen.insert((section, key.clone()));
            unit.key = key;
        }
    }
    
    /// The paths each unit's markers name: a Contract clause holds the markers on
    /// its lines, and a Constraints unit the markers whose statement starts in it.
    fn cover(body: &Body, units: &mut [Unit]) {
        for marker in body.tested_by() {
            let section = marker.section().and_then(CapabilitySection::from_slug);
            let line = match section {
                Some(CapabilitySection::Contract) => marker.line(),
                Some(CapabilitySection::Constraints) => marker.statement().lines.start,
                _ => continue,
            };
            if let Some(unit) = units
                .iter_mut()
                .find(|unit| Some(unit.section) == section && unit.lines.contains(&line))
            {
                for path in marker.paths() {
                    if !unit.tests.contains(path) {
                        unit.tests.push(path.clone());
                    }
                }
            }
        }
    }

    /// The findings of the content rules, capability by capability in the order
    /// of the load, each with the debt key it is held against.
    pub(super) fn check(plan: &Workplan<'_>) -> Vec<Owed> {
        let mut out = Vec::new();
        for loaded in plan.capabilities() {
            if loaded.value().state == State::Delivered {
                add(Work::StrictUnits, 1);
                judge(plan, loaded, &mut out);

    /// The content rules of one delivered capability.
    fn judge(plan: &Workplan<'_>, loaded: &Loaded<Capability>, out: &mut Vec<Owed>) {
        let path = loaded.path();
        let capability = loaded.value();
        let body = loaded.body();
        let id = capability.header.id.as_str();
        let section_subject = |which: CapabilitySection| format!("{id}#{}", which.slug());
        if capability.bindings.examples.is_empty() {
            out.push(Owed::held(
                Finding::new(
                    kind::STRICT_EXAMPLE_MISSING,
                    path,
                    Location::Key("bindings".to_string()),
                    "the delivered capability binds no entry in bindings.examples",
                    "bind at least one example target or runnable documentation test in `bindings.examples`; the complete policy asks every delivered capability for a usage that runs",
                ),
                id,
            ));
        }
        if !holds_matrix(body) {
            out.push(Owed::held(
                Finding::new(
                    kind::STRICT_MATRIX_MISSING,
                    path,
                    section(CapabilitySection::Verification),
                    "the Verification section holds no verification-matrix generated block",
                    "add `<!-- raf:generated:verification-matrix -->` and `<!-- raf:generated:end -->` to the Verification section and run `cargo workplan docs sync`",
                ),
                section_subject(CapabilitySection::Verification),
            ));
        }
        for (heading, words) in section_words(body) {
            if words < MIN_SECTION_WORDS {
                let subject = CapabilitySection::ALL
                    .into_iter()
                    .find(|which| which.heading() == heading)
                    .map_or_else(|| id.to_string(), section_subject);
                out.push(Owed::held(
                    Finding::new(
                        kind::STRICT_SECTION_THIN,
                        path,
                        Location::Section(heading.to_string()),
                        format!(
                            "the section holds {words} words of prose, fewer than {MIN_SECTION_WORDS}"
                        ),
                        format!(
                            "write at least {MIN_SECTION_WORDS} words of prose saying what the section covers; a section of a delivered capability is read on its own, and a generated block, a code span and fenced code do not count as prose"
                        ),
                    ),
                    subject,
                ));
            }
        }
        let usage_bound = plan
            .graph()
            .bindings_of(&capability.header.id)
            .filter(|binding| binding.kind() == BindingKind::Example)
            .any(|binding| binding.sections().contains(&CapabilitySection::Usage));
        if !usage_bound {
            out.push(Owed::held(
                Finding::new(
                    kind::STRICT_USAGE_UNBOUND,
                    path,
                    section(CapabilitySection::Usage),
                    "no path of bindings.examples is named in the section",
                    "name a path of bindings.examples in a code span in the Usage section, so the verification matrix lists Usage for an example that verify runs",
                ),
                section_subject(CapabilitySection::Usage),
            ));
        }
        if !lists_an_item(body, CapabilitySection::Constraints) {
            out.push(Owed::held(
                Finding::new(
                    kind::STRICT_LIMITATIONS_MISSING,
                    path,
                    section(CapabilitySection::Constraints),
                    "the Constraints section holds no list item",
                    "state each limitation of the capability as a list item of Constraints",
                ),
                section_subject(CapabilitySection::Constraints),
            ));
        }
    }
    
    /// The location of one of the seven sections.
    fn section(section: CapabilitySection) -> Location {
        Location::Section(section.heading().to_string())
    }
    
    /// The second-level section of `body` that is `which`.
    fn find(body: &Body, which: CapabilitySection) -> Option<&Section> {
        body.sections()
            .iter()
            .find(|section| section.slug() == which.slug())
    }
    
    /// Whether the Verification section holds a `verification-matrix` block.
    fn holds_matrix(body: &Body) -> bool {
        let Some(verification) = find(body, CapabilitySection::Verification) else {
            return false;
        };
        let lines = &verification.content().lines;
        body.generated_blocks()
            .iter()
            .any(|block| block.kind() == MATRIX && lines.contains(&block.outer().lines.start))
    }
    
    /// The prose lines of one section: below its heading, outside PROPOSED and
    /// generated blocks.
    fn prose_lines<'b>(body: &'b Body, section: &'b Section) -> impl Iterator<Item = &'b ProseLine> {
        body.prose_lines().iter().filter(move |line| {
            line.section() == Some(section.slug())
                && line.line() != section.line()
                && !line.in_proposed()
                && !line.in_generated()
        })
    }
    
    /// The text of the prose lines of one section.
    fn prose<'b>(body: &'b Body, section: &'b Section) -> impl Iterator<Item = &'b str> {
        prose_lines(body, section).map(ProseLine::text)
    }
    
    /// The heading and the number of words of prose of each of the seven
    /// sections, in document order.
    fn section_words(body: &Body) -> Vec<(&str, usize)> {
        body.sections()
            .iter()
            .filter(|section| CapabilitySection::from_slug(section.slug()).is_some())
            .map(|section| {
                let words = prose(body, section)
                    .map(|text| word_spans(text).len())
                    .sum();
                (section.heading(), words)
            })
            .collect()
    }
    
    /// Whether the section `which` holds a list item: a prose line that starts,
    /// after its indentation, with `-`, `*` or `+` and a space, or with one to nine
    /// digits, `.` or `)` and a space, and holds text after the marker, which may be
    /// a code span. A thematic break such as `* * *` is no list item.
    fn lists_an_item(body: &Body, which: CapabilitySection) -> bool {
        let Some(section) = find(body, which) else {
            return false;
        };
        let first = body.span().lines.start;
        prose_lines(body, section).any(|line| {
            let raw = body.text().lines().nth(line.line() - first).unwrap_or("");
            list_item(line.text().trim_start()).is_some() && states_an_item(raw.trim_start())
        })
    }

        /// Whether the set holds `value`.
        fn holds(&self, value: &Q) -> bool;
    }
    
    impl<T: Eq + Hash + Borrow<Q>, Q: Eq + Hash + ?Sized> Holds<Q> for HashSet<T> {
        fn holds(&self, value: &Q) -> bool {
            self.contains(value)
        }
    }
    
    impl<T: Ord + Borrow<Q>, Q: Ord + ?Sized> Holds<Q> for BTreeSet<T> {
        fn holds(&self, value: &Q) -> bool {
            self.contains(value)
        }
    }
    
    impl<S: Holds<Q>, Q: ?Sized> Holds<Q> for &S {
        fn holds(&self, value: &Q) -> bool {
            (**self).holds(value)
        }
    }
    
    /// A set whose every lookup is counted as one unit of a kind of [`Work`]. The
    /// count is part of the lookup, so a lookup that does not go through the set
    /// is not counted.
    pub(crate) struct Counted<S> {
        set: S,
        work: Work,
    }
    
    impl<S> Counted<S> {
        /// `set`, counting its lookups as `work`.
        pub(crate) const fn new(work: Work, set: S) -> Self {
            Self { set, work }
        }
    
        /// Whether the set holds `value`; counts one lookup.
        pub(crate) fn contains<Q: ?Sized>(&self, value: &Q) -> bool
        where
            S: Holds<Q>,
        {
            add(self.work, 1);
            self.set.holds(value)
        }
    }

    /// What one [`measure`] counted, per kind of [`Work`].
    #[cfg(test)]
    #[derive(Debug, Clone, Copy, PartialEq, Eq, Default)]
    pub(crate) struct Tally([usize; KINDS]);
    
    #[cfg(test)]
    impl Tally {
        /// The count of one kind of work.
        pub(crate) const fn get(&self, work: Work) -> usize {
            self.0[work as usize]
        }
    
        /// The count of every kind of work together.
        pub(crate) fn total(&self) -> usize {
            self.0.iter().sum()
        }
    }
    
    #[cfg(test)]
    mod counting {
        use std::cell::Cell;
    
        use super::{KINDS, Tally, Work};
    
        thread_local! {
            /// The counts of this thread since the last reset.
            static TALLY: Cell<Tally> = const { Cell::new(Tally([0; KINDS])) };
            /// The count a kind may not pass while a measure runs: the one kind
            /// it limits, or every kind when that is `None`.
            static CEILING: Cell<Option<(Option<Work>, usize)>> = const { Cell::new(None) };
        }
    
        /// Count `amount` units of `work`, and panic if that passes the ceiling.
        pub(crate) fn add(work: Work, amount: usize) {
            let mut tally = TALLY.get();
            let slot = &mut tally.0[work as usize];
            *slot = slot.saturating_add(amount);
            let count = *slot;
            TALLY.set(tally);
            if let Some((limited, ceiling)) = CEILING.get()
                && limited.is_none_or(|kind| kind == work)
            {
                assert!(
                    count <= ceiling,
                    "{work:?} work passed the ceiling of {ceiling}: {count}"
                );
            }
        }
    
        /// Clears the ceiling when a measure ends, even by a panic.
        struct Armed;
    
        impl Drop for Armed {
            fn drop(&mut self) {
                CEILING.set(None);
            }
        }
    
        /// Run `run` against fresh tallies and return its result with what it
        /// counted. Panics the moment any kind of work passes `ceiling`.
        pub(crate) fn measure<T>(ceiling: usize, run: impl FnOnce() -> T) -> (T, Tally) {
            measured(None, ceiling, run)
        }
    
        /// Like [`measure`], but only `limited` is held to `ceiling`; the other
        /// kinds are counted without a limit.
        pub(crate) fn measure_kind<T>(
            limited: Work,
            ceiling: usize,
            run: impl FnOnce() -> T,
        ) -> (T, Tally) {
            measured(Some(limited), ceiling, run)
        }
    
        fn measured<T>(limited: Option<Work>, ceiling: usize, run: impl FnOnce() -> T) -> (T, Tally) {
            TALLY.set(Tally::default());
            CEILING.set(Some((limited, ceiling)));
            let _armed = Armed;
            let result = run();
            (result, TALLY.get())
        }
    }

    #[cfg(test)]
    mod tests {
        use super::{ALL, Work, add, measure, measure_kind};
    
        #[test]
        fn a_counter_increments_by_what_is_added() {
            let ((), tally) = measure(100, || {
                add(Work::ClaimChecks, 1);
                add(Work::ClaimChecks, 4);
                add(Work::ContractLines, 7);
            });
            assert_eq!(tally.get(Work::ClaimChecks), 5);
            assert_eq!(tally.get(Work::ContractLines), 7);
            assert_eq!(tally.get(Work::BlankedSteps), 0);
        }
    
        #[test]
        fn every_kind_has_its_own_counter() {
            let ((), tally) = measure(100, || {
                for (index, work) in ALL.into_iter().enumerate() {
                    add(work, index + 1);
                }
            });
            for (index, work) in ALL.into_iter().enumerate() {
                assert_eq!(tally.get(work), index + 1, "{work:?}");
            }
            assert_eq!(tally.total(), ALL.len() * (ALL.len() + 1) / 2);
        }
    
        #[test]
        fn the_result_of_the_closure_is_returned() {
            let (value, tally) = measure(10, || 42);
            assert_eq!(value, 42);
            assert_eq!(tally, super::Tally::default());
        }
    
        #[test]
        #[should_panic(expected = "ProseChars work passed the ceiling of 3: 4")]
        fn passing_the_ceiling_panics_at_once() {
            measure(3, || {
                for _ in 0..4 {
                    add(Work::ProseChars, 1);
                }
            });
        }
    
        #[test]
        #[should_panic(expected = "ProseChars work passed the ceiling of 3: 4")]
        fn the_panic_comes_before_the_work_that_follows_it() {
            measure(3, || {
                add(Work::ProseChars, 4);
                unreachable!("the ceiling should have stopped the closure");
            });
        }
    
        #[test]
        fn reaching_the_ceiling_is_allowed() {
            let ((), tally) = measure(3, || add(Work::ProseChars, 3));
            assert_eq!(tally.get(Work::ProseChars), 3);
        }
    
        #[test]
        #[should_panic(expected = "BlankedSteps work passed the ceiling of 2: 3")]
        fn a_limited_kind_panics_at_its_ceiling() {
            measure_kind(Work::BlankedSteps, 2, || add(Work::BlankedSteps, 3));
        }
    
        #[test]
        fn a_limited_measure_counts_the_other_kinds_without_a_ceiling() {
            let ((), tally) = measure_kind(Work::BlankedSteps, 2, || {
                add(Work::BlankedSteps, 2);
                add(Work::ClaimChecks, 500);
            });
            assert_eq!(tally.get(Work::ClaimChecks), 500);
        }
    
        #[test]
        fn the_tallies_are_reset_between_measures() {
            let ((), first) = measure(10, || add(Work::ProseChars, 5));
            let ((), second) = measure(10, || add(Work::ProseChars, 2));
            assert_eq!(first.get(Work::ProseChars), 5);
            assert_eq!(second.get(Work::ProseChars), 2);
        }
    
        #[test]
        fn a_measure_that_panicked_leaves_neither_counts_nor_ceiling_behind() {
            let caught = std::panic::catch_unwind(|| measure(1, || add(Work::ClaimChecks, 2)));
            assert!(caught.is_err());
            let ((), tally) = measure(10, || {});
            assert_eq!(tally.get(Work::ClaimChecks), 0);
            // With no measure running there is no ceiling.
            add(Work::ClaimChecks, 1000);
        }
