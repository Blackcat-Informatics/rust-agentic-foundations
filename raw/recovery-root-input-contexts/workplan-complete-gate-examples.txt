    //! The execution of the tests, documentation tests and examples that a
    //! capability binds.
    //!
    //! An [`Executor`] takes a [`Binding`](crate::Binding) of the graph, decides
    //! from the facts of the source index what it runs as (a [`Plan`]), runs it
    //! through an [`exec::Runner`](crate::exec::Runner) and returns the
    //! [`Evidence`]: the invocations that ran, the counts of the proof and the
    //! verdict. Nothing is written and nothing is kept.
    //!
    //! # Planning
    //!
    //! Before any process starts, the binding is judged from the index. A path
    //! that does not resolve, a function that is not a `#[test]`, a target with
    //! `harness = false` or `required-features`, a test in a benchmark, example or
    //! build-script target, and a value that fails its pattern are each a
    //! [`Refusal`] that names the binding and the target, and nothing reaches
    //! `cargo`. The libtest name of a test is its path within its crate
    //! (`documents::x` for a test of an integration test target, `bindings::tests::x`
    //! for a `cfg(test)` module of the library); the target is pinned with `--lib`,
    //! `--bin=<bin>` or `--test=<test>` under the Cargo name of the target, which
    //! keeps its hyphens where the index path spells them with underscores.
    //!
    //! # Proof
    //!
    //! * **Test.** The list pass shows exactly one `<name>: test` entry, and the run
    //!   pass reports `1 passed; 0 failed; 0 ignored` in one `test result:` line.
    //!   Zero entries, two entries, `0 passed` and an `#[ignore]`d test (listed as a
    //!   test, reported as ignored) leave the binding [`BoundOutcome::Unresolved`].
    //! * **Documentation test.** The list pass for the package selects the entries
    //!   whose item part equals the bound item path within the crate (impl items as
    //!   rustdoc writes them, `Wrapper<T>::get`); the run uses the bare item path as
    //!   a substring filter without `--exact`; each selected entry must appear as the
    //!   whole line `test <name> ... ok`. A `no_run` test (`- compile ... ok`) and a
    //!   `compile_fail` test (`- compile fail ... ok`) were not built and run, and
    //!   are refused with a request to bind a runnable doc test. A documentation
    //!   test that the substring also matches, such as `foo_more` for `foo`, is not
    //!   counted, but if it fails it fails the exit status and the binding.
    //! * **Example.** `cargo run` with standard input null; the exit status within
    //!   the time limit decides.
    //!
    //! A process that cannot be started is an [`ExecError`](crate::exec::ExecError)
    //! and the run has no verdict; a process killed at its limit is a failed
    //! binding that reports how long it ran.
    //!
    //! The proof guards against a filter that selects nothing, more than one test
    //! or the wrong one; it is not a defence against the code it runs. A bound test
    //! or example is workspace code and trusted build input (see `SECURITY.md`), so
    //! output from it that is shaped like a result, such as a test that prints its
    //! own `test result:` line and exits with status 0, is read as the run reports
    //! it.
    //!
    //! ```rust
    //! use raf_source::{MemoryTree, Workspace};
    //! use raf_workplan::Workplan;
    //! use raf_workplan::bound::{BoundOutcome, Executor};
    //! use raf_workplan::exec::{FakeRunner, Output, Program};
    //!
    //! let source = Workspace::load(
    //!     &MemoryTree::from([
    //!         ("Cargo.toml", "[package]\nname = \"demo\"\nversion = \"0.1.0\"\n"),
    //!         ("src/lib.rs", "#[cfg(test)]\nmod tests {\n    #[test]\n    fn case() {}\n}\n"),
    //!     ]),
    //!     &["."],
    //! )?;
    //! let sections: String = [
    //!     "Purpose", "Contract", "Usage", "Examples", "Constraints", "Explanation", "Verification",
    //! ]
    //! .iter()
    //! .map(|name| format!("## {name}\n\nText.\n\n"))
    //! .collect();
    //! let document = format!(
    //!     "---\ntype: Capability\nid: cap:demo\ntitle: Demo\ndescription: Demo.\n\
    //!      state: implementing\nbindings:\n  code: []\n  tests: [demo::tests::case]\n  \
    //!      examples: []\ndepends: []\n---\n\n# Demo\n\n{sections}"
    //! );
    //! let docs = MemoryTree::from([("docs/capabilities/demo.md", document.as_str())]);
    //! let plan = Workplan::load(&docs, &source);
    //! let binding = &plan.graph().bindings()[0];
    //!
    //! // Nothing is started here: the runner answers from a script.
    //! let ok = |text: &str| Ok(Output::new(Some(0), text.as_bytes().to_vec(), Vec::new()));
    //! let runner = FakeRunner::new()
    //!     .on(
    //!         Program::Cargo,
    //!         &["test", "--frozen", "--package=demo", "--lib", "--", "--list", "--exact", "--", "tests::case"],
    //!         ok("tests::case: test\n\n1 test, 0 benchmarks\n"),
    //!     )
    //!     .on(
    //!         Program::Cargo,
    //!         &["test", "--frozen", "--package=demo", "--lib", "--", "--exact", "--", "tests::case"],
    //!         ok("running 1 test\ntest tests::case ... ok\n\ntest result: ok. 1 passed; 0 failed; 0 ignored; 0 measured; 0 filtered out; finished in 0.00s\n"),
    //!     );
    //! let evidence = Executor::new(&source, &runner)?.run(binding)?;
    //! assert_eq!(evidence.outcome(), &BoundOutcome::Passed);
    //! assert_eq!(evidence.runs().len(), 2);
    //! # Ok::<(), Box<dyn std::error::Error>>(())
    //! ```

    //! What a run of a binding leaves behind: the exact invocations, the proof the
    //! kind of binding requires, the verdict and the time it took. It is a value;
    //! nothing is written anywhere.
    
    use std::fmt;
    use std::time::Duration;
    
    use crate::BindingKind;
    use crate::exec::Invocation;
    
    use super::plan::Refusal;
    
    /// The verdict on a binding.
    #[derive(Debug, Clone, PartialEq, Eq)]
    #[non_exhaustive]
    pub enum BoundOutcome {
        /// The proof its kind requires holds.
        Passed,
        /// The binding ran or tried to and it failed: the build or the run exited
        /// with a failing status, a test or example failed, or the process ran
        /// past its time limit or its output bound and was killed.
        Failed {
            /// What failed.
            reason: String,
        },
        /// The binding is not something that can be run and counted: the index
        /// facts or a value refuse it, or the run proves nothing (zero tests, an
        /// `#[ignore]`d test, a `no_run` documentation test).
        Unresolved {
            /// Why.
            reason: Refusal,
        },
    }
    
    impl BoundOutcome {
        /// The stable name of the verdict: `passed`, `failed` or `unresolved`.
        #[must_use]
        pub const fn name(&self) -> &'static str {
            match self {
                Self::Passed => "passed",
                Self::Failed { .. } => "failed",
                Self::Unresolved { .. } => "unresolved",
            }
        }
    
        /// Why the binding did not pass, in full; `None` for a pass.
        #[must_use]
        pub fn reason(&self) -> Option<String> {
            match self {
                Self::Passed => None,
                Self::Failed { reason } => Some(reason.clone()),
                Self::Unresolved { reason } => Some(reason.to_string()),
            }
        }
    
        /// Whether the binding passed.
        #[must_use]
        pub fn is_passed(&self) -> bool {
            matches!(self, Self::Passed)
        }
    }
    
    /// How a process of a run ended.
    #[derive(Debug, Clone, Copy, PartialEq, Eq)]
    #[non_exhaustive]
    pub enum Ended {
        /// It exited with this code, or `None` when a signal ended it.
        Exited(Option<i32>),
        /// It was killed at its wall-clock limit.
        TimedOut,
        /// It was killed for writing more than the output bound.
        Overflowed,
    }
    
    /// One process of a run.
    #[derive(Debug, Clone, PartialEq, Eq)]
    pub struct Run {
        invocation: Invocation,
        ended: Ended,
        elapsed: Duration,
    }
    
    impl Run {
        pub(super) fn new(invocation: Invocation, ended: Ended, elapsed: Duration) -> Self {
            Self {
                invocation,
                ended,
                elapsed,
            }
        }
    
        /// What was run.
        #[must_use]
        pub fn invocation(&self) -> &Invocation {
            &self.invocation
        }
    
        /// The argv vector the process received, without the program.
        #[must_use]
        pub fn argv(&self) -> Vec<String> {
            self.invocation.argv()
        }
    
        /// How it ended.
        #[must_use]
        pub fn ended(&self) -> Ended {
            self.ended
        }
    
        /// How long it ran.
        #[must_use]
        pub fn elapsed(&self) -> Duration {
            self.elapsed
        }
    }
    
    /// The counts a proof is made of. A field is `None` when the binding's kind
    /// has no such count or the run did not get that far.
    #[derive(Debug, Clone, Copy, Default, PartialEq, Eq)]
    #[non_exhaustive]
    pub struct Proof {
        /// A test binding: the `<name>: test` entries of the list pass. A
        /// doc-test binding: the entries of the list pass for the package.
        pub listed: Option<usize>,
        /// A doc-test binding: the entries whose item part is the bound item.
        pub selected: Option<usize>,
        /// A test binding: tests passed, from `test result:`.
        pub passed: Option<usize>,
        /// A test binding: tests failed.
        pub failed: Option<usize>,
        /// A test binding: tests ignored.
        pub ignored: Option<usize>,
        /// A doc-test binding: the selected entries with a `test <name> ... ok`
        /// line in the run output.
        pub ok_lines: Option<usize>,
    }
    
    impl fmt::Display for Proof {
        fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
            let mut parts = Vec::new();
            for (what, count) in [
                ("listed", self.listed),
                ("selected", self.selected),
                ("passed", self.passed),
                ("failed", self.failed),
                ("ignored", self.ignored),
                ("ok lines", self.ok_lines),
            ] {
                if let Some(count) = count {
                    parts.push(format!("{count} {what}"));
                }
            }
            if parts.is_empty() {
                f.write_str("no counts")
            } else {
                f.write_str(&parts.join(", "))
            }
        }
    }
    
    /// The evidence of one binding: what it was, what was run, the proof, the
    /// verdict and the wall time. [`Evidence`] is returned by the executor and is
    /// not persisted.
    #[derive(Debug, Clone, PartialEq, Eq)]
    pub struct Evidence {
        capability: String,
        kind: BindingKind,
        path: String,
        runs: Vec<Run>,
        proof: Proof,
        outcome: BoundOutcome,
        wall: Duration,
    }
    
    impl Evidence {
        pub(super) fn new(
            capability: &str,
            kind: BindingKind,
            path: &str,
            runs: Vec<Run>,
            proof: Proof,
            outcome: BoundOutcome,
        ) -> Self {
            let wall = runs.iter().map(Run::elapsed).sum();
            Self {
                capability: capability.to_string(),
                kind,
                path: path.to_string(),
                runs,
                proof,
                outcome,
                wall,
            }
        }
    
        /// The id of the capability that lists the binding.
        #[must_use]
        pub fn capability(&self) -> &str {
            &self.capability
        }
    
        /// The list the binding is in.
        #[must_use]
        pub fn kind(&self) -> BindingKind {
            self.kind
        }
    
        /// The bound path, as the document writes it.
        #[must_use]
        pub fn path(&self) -> &str {
            &self.path
        }
    
        /// Every process of the run, in order. Empty when the binding was refused
        /// before any process started.
        #[must_use]
        pub fn runs(&self) -> &[Run] {
            &self.runs
        }
    
        /// The counts of the proof.
        #[must_use]
        pub fn proof(&self) -> &Proof {
            &self.proof
        }
    
        /// The verdict.
        #[must_use]
        pub fn outcome(&self) -> &BoundOutcome {
            &self.outcome
        }
    
        /// Whether the binding passed.
        #[must_use]
        pub fn passed(&self) -> bool {
            self.outcome.is_passed()
        }
    
        /// The wall time of every process of the run, added up.
        #[must_use]
        pub fn wall(&self) -> Duration {
            self.wall
        }
    }
    
    fn list_name(kind: BindingKind) -> &'static str {
        match kind {
            BindingKind::Code => "code",
            BindingKind::Test => "tests",
            BindingKind::Example => "examples",
        }
    }
    
    impl fmt::Display for Evidence {
        /// A line for the verdict, then a line for each process:
        ///
        /// ```text
        /// pass cap:demo tests `demo::case` (1 listed, 1 passed, 0 failed, 0 ignored; 1.2 s)
        ///   ran cargo test --frozen --package=demo --lib -- --list --exact -- case
        /// ```
        fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
            let verdict = match &self.outcome {
                BoundOutcome::Passed => "pass",
                BoundOutcome::Failed { .. } => "fail",
                BoundOutcome::Unresolved { .. } => "unresolved",
            };
            write!(
                f,
                "{verdict} {} {} `{}` ({}; {:.1} s)",
                self.capability,
                list_name(self.kind),
                self.path,
                self.proof,
                self.wall.as_secs_f64()
            )?;
            match &self.outcome {
                BoundOutcome::Passed => {}
                BoundOutcome::Failed { reason } => write!(f, ": {reason}")?,
                BoundOutcome::Unresolved { reason } => write!(f, ": {reason}")?,
            }
            for run in &self.runs {
                write!(f, "\n  ran {}", run.invocation)?;
            }
            Ok(())
        }
    }

    //! Pure parsers for what libtest and rustdoc print.
    //!
    //! None of them starts a process or fails: a line that is not of the form a
    //! parser reads is noise and is skipped, and the caller decides what a missing
    //! entry means. Every match is on a whole line, so a name that is a prefix or
    //! a suffix of another never stands for it.
    
    use crate::exec::Pattern;
    
    /// What a line of `--list` output says an entry is.
    #[derive(Debug, Clone, Copy, PartialEq, Eq)]
    pub(crate) enum EntryKind {
        /// `<name>: test`. An `#[ignore]`d test is listed as a test too.
        Test,
        /// `<name>: benchmark`.
        Benchmark,
    }
    
    /// One entry of libtest's `--list` output.
    #[derive(Debug, Clone, PartialEq, Eq)]
    pub(crate) struct ListEntry {
        pub(crate) name: String,
        pub(crate) kind: EntryKind,
    }
    
    /// The entries of a libtest `--list` run: every line `<name>: test` or
    /// `<name>: benchmark`. The count line (`1 test, 0 benchmarks`), blank lines
    /// and anything else are skipped.
    pub(crate) fn test_list(stdout: &str) -> Vec<ListEntry> {
        stdout
            .lines()
            .filter_map(|line| {
                let (name, kind) = if let Some(name) = line.strip_suffix(": test") {
                    (name, EntryKind::Test)
                } else {
                    (line.strip_suffix(": benchmark")?, EntryKind::Benchmark)
                };
                // A name is a path of identifiers (the pattern of a test name), so
                // free text before the colon of a diagnostic is noise.
                Pattern::Name.matches(name).then(|| ListEntry {
                    name: name.to_string(),
                    kind,
                })
            })
            .collect()
    }
    
    /// The counts of one `test result:` line.
    #[derive(Debug, Clone, Copy, PartialEq, Eq)]
    pub(crate) struct Counts {
        pub(crate) passed: usize,
        pub(crate) failed: usize,
        pub(crate) ignored: usize,
    }
    
    /// A `test result:` line that could not be read.
    #[derive(Debug, Clone, PartialEq, Eq)]
    pub(crate) struct BadResult(pub(crate) String);
    
    /// Every `test result: ok. 1 passed; 0 failed; 0 ignored; ...` line, in order.
    /// A line that starts so and does not carry the three counts is an error: it
    /// is a result that cannot be counted.
    pub(crate) fn results(stdout: &str) -> Result<Vec<Counts>, BadResult> {
        stdout
            .lines()
            .filter_map(|line| line.strip_prefix("test result: ").map(|rest| (line, rest)))
            .map(|(line, rest)| count_line(rest).ok_or_else(|| BadResult(line.to_string())))
            .collect()
    }
    
    /// `ok. 1 passed; 0 failed; 0 ignored; 0 measured; 0 filtered out; finished in 0.00s`
    /// without its `test result: ` prefix.
    fn count_line(rest: &str) -> Option<Counts> {
        let (status, counts) = rest.split_once(". ")?;
        if !matches!(status, "ok" | "FAILED") {
            return None;
        }
        let mut passed = None;
        let mut failed = None;
        let mut ignored = None;
        for part in counts.split("; ") {
            let (number, what) = part.split_once(' ')?;
            let slot = match what {
                "passed" => &mut passed,
                "failed" => &mut failed,
                "ignored" => &mut ignored,
                _ => continue,
            };
            if slot.is_some() {
                return None;
            }
            *slot = Some(crate::unsigned::count(number)?);
        }
        Some(Counts {
            passed: passed?,
            failed: failed?,
            ignored: ignored?,
        })
    }
    
    /// One entry of rustdoc's `--list` output.
    #[derive(Debug, Clone, PartialEq, Eq)]
    pub(crate) struct DocEntry {
        /// The whole name, as `test <name> ... ok` repeats it:
        /// `src/lib.rs - inner::foo (line 22)`.
        pub(crate) name: String,
        /// The item part: the text between ` - ` and ` (line `. `None` for the
        /// documentation of a crate or module root, which has no item part
        /// (`src/lib.rs - (line 3)`).
        pub(crate) item: Option<String>,
    }
    
    /// The documentation tests of a rustdoc `--list` run: every line
    /// `<relative-source-path> - <item-path> (line <n>): test`. Rustdoc prints one
    /// block for the tests it runs together and another for the `compile_fail`
    /// ones; both are read. Other lines are skipped.
    pub(crate) fn doc_list(stdout: &str) -> Vec<DocEntry> {
        stdout.lines().filter_map(doc_entry).collect()
    }
    
    fn doc_entry(line: &str) -> Option<DocEntry> {
        let name = line.strip_suffix(": test")?;
        let (file, rest) = name.split_once(" - ")?;
        // rustdoc prints the file part with the platform's separator; the pattern is
        // checked on the `/` form, and the printed name is kept for the `ok`-line match.
        if !Pattern::SourcePath.matches(&file.replace('\\', "/")) {
            return None;
        }
        let item = if let Some(at) = rest.rfind(" (line ") {
            line_number(&rest[at + " (line ".len()..])?;
            Some(rest[..at].to_string())
        } else {
            line_number(rest.strip_prefix("(line ")?)?;
            None
        };
        if item.as_deref() == Some("") {
            return None;
        }
        Some(DocEntry {
            name: name.to_string(),
            item,
        })
    }
    
    /// `<digits>)`: the end of a name.
    fn line_number(tail: &str) -> Option<()> {
        let digits = tail.strip_suffix(')')?;
        (!digits.is_empty() && digits.bytes().all(|b| b.is_ascii_digit())).then_some(())
    }
    
    /// What the run output says about one documentation test.
    #[derive(Debug, Clone, Copy, PartialEq, Eq)]
    pub(crate) enum DocStatus {
        /// `test <name> ... ok`: built and run, and passed.
        Ok,
        /// `test <name> - compile ... ok`: a `no_run` test, built and not run.
        CompileOnly,
        /// `test <name> - compile fail ... ok`: a `compile_fail` test, which
        /// passes when the build fails.
        CompileFail,
        /// The line ends in `FAILED`.
        Failed,
        /// The test was ignored.
        Ignored,
        /// No line names the test.
        Missing,
    }
    
    /// The status of the documentation test `name` in a rustdoc run, read from
    /// the whole line `test <name> ... ok` (and its `- compile` forms) with
    /// nothing between the name and `...` that this function does not name.
    ///
    /// The first line that names the test decides. The name ends in `)`, so the
    /// name of a test is never a prefix of another's.
    pub(crate) fn doc_status(stdout: &str, name: &str) -> DocStatus {
        let prefix = format!("test {name} ");
        for line in stdout.lines() {
            let Some(rest) = line.strip_prefix(&prefix) else {
                continue;
            };
            return match rest {
                "... ok" => DocStatus::Ok,
                "- compile ... ok" => DocStatus::CompileOnly,
                "- compile fail ... ok" => DocStatus::CompileFail,
                _ if rest.ends_with("FAILED") => DocStatus::Failed,
                _ if rest.starts_with("... ignored") || rest.contains(" ... ignored") => {
                    DocStatus::Ignored
                }
                // A line that starts with the name and says something else is not
                // proof of a run.
                _ => DocStatus::Missing,
            };
        }
        DocStatus::Missing
    }

    //! What a binding runs as, decided from the facts of the source index before
    //! any process is started.

    /// Why a binding was not run, or why a run proves nothing. Each case names the
    /// binding and what to change; none is a skip.
    #[derive(Debug, Clone, PartialEq, Eq)]
    #[non_exhaustive]
    pub enum Refusal {
        /// A `code` binding: nothing runs it.
        NotRunnable {
            /// The bound path.
            path: String,
        },
        /// The path does not resolve in the source index.
        Unresolved {
            /// The bound path.
            path: String,
            /// What the index said.
            error: String,
        },
        /// The path resolves to something that is not a function marked
        /// `#[test]`.
        NotATest {
            /// The bound path.
            path: String,
            /// What it is, with its place.
            found: String,
        },
        /// An `examples` binding that is neither an example target nor a
        /// documented item of a library whose documentation tests run.
        NotAnExample {
            /// The bound path.
            path: String,
        },
        /// The test is in a target that `cargo test` runs in a way the argv table
        /// cannot select a test in: a benchmark, an example or the build script.
        TargetKind {
            /// The bound path.
            path: String,
            /// The Cargo name of the target.
            target: String,
            /// What the target is.
            kind: TargetKind,
        },
        /// The target has `harness = false`: its output is not libtest's, so no
        /// test can be listed or counted in it.
        NoHarness {
            /// The bound path.
            path: String,
            /// The Cargo name of the target.
            target: String,
        },
        /// The target has `required-features`, which the argv table does not
        /// enable.
        RequiredFeatures {
            /// The bound path.
            path: String,
            /// The Cargo name of the target.
            target: String,
            /// The features it requires.
            features: Vec<String>,
        },
        /// The path of a documented item cannot be written as the name rustdoc
        /// prints, for instance the root of a crate, which has no item part.
        NoDocItem {
            /// The bound path.
            path: String,
        },
        /// A value taken from the index does not match its pattern.
        Pattern {
            /// The bound path.
            path: String,
            /// Which value it is.
            what: &'static str,
            /// The refusal of the pattern.
            error: PatternError,
        },
        /// The definitions the path resolves to would run as different things.
        Divergent {
            /// The bound path.
            path: String,
            /// What each definition would run as.
            plans: Vec<String>,
        },
        /// The list pass shows no `<name>: test` entry.
        NoTestListed {
            /// The resolved test name.
            name: String,
            /// The target the list pass was pinned to.
            target: String,
        },
        /// The list pass shows more than one entry, or one under another name.
        TestListedWrong {
            /// The resolved test name.
            name: String,
            /// The target the list pass was pinned to.
            target: String,
            /// The entries listed.
            listed: Vec<String>,
        },
        /// The run pass did not report exactly one test passed and none failed or
        /// ignored.
        TestCounts {
            /// The resolved test name.
            name: String,
            /// Tests passed.
            passed: usize,
            /// Tests failed.
            failed: usize,
            /// Tests ignored.
            ignored: usize,
        },
        /// The test is `#[ignore]`d: it is listed and it did not run.
        TestIgnored {
            /// The resolved test name.
            name: String,
        },
        /// The output of a pass does not have the form the proof reads.
        Unparsable {
            /// What was read.
            what: &'static str,
            /// The offending text.
            text: String,
        },
        /// The list pass selects no documentation test of the item.
        NoDocTest {
            /// The item path within the crate.
            item: String,
            /// The package.
            package: String,
        },
        /// A selected documentation test was built and not run (`no_run`), or
        /// passes when it does not build (`compile_fail`).
        DocNotRunnable {
            /// The item path within the crate.
            item: String,
            /// The names of the tests, as rustdoc prints them.
            names: Vec<String>,
        },
        /// A selected documentation test has no `... ok` line in the run output
        /// and did not fail: it was ignored or did not appear.
        DocNotRun {
            /// The item path within the crate.
            item: String,
            /// The names of the tests, as rustdoc prints them.
            names: Vec<String>,
        },
    }
    
    impl fmt::Display for Refusal {
        fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
            match self {
                Self::NotRunnable { path } => write!(
                    f,
                    "`{path}` is a code binding: nothing runs it; bind a test or an example to have it run"
                ),
                Self::Unresolved { path, error } => write!(
                    f,
                    "`{path}` does not resolve in the source index ({error}); nothing is passed to cargo"
                ),
                Self::NotATest { path, found } => write!(
                    f,
                    "`{path}` resolves to {found}, not to a function marked `#[test]`, so it would run zero tests; bind a `#[test]` function"
                ),
                Self::NotAnExample { path } => write!(
                    f,
                    "`{path}` is neither the crate root of an example target nor a documented item of a library whose documentation tests run; bind an example or a documented item"
                ),
                Self::TargetKind { path, target, kind } => write!(
                    f,
                    "`{path}` is in the {} target `{target}`, and `cargo test` cannot select a test there with `--lib`, `--bin` or `--test`; move the test to the library, a binary or an integration test",
                    kind_name(*kind)
                ),
                Self::NoHarness { path, target } => write!(
                    f,
                    "`{path}` is in the target `{target}`, which has `harness = false` in its manifest: its output is not libtest's, so no test can be listed or counted; bind a test of a target with the libtest harness"
                ),
                Self::RequiredFeatures {
                    path,
                    target,
                    features,
                } => write!(
                    f,
                    "`{path}` is in the target `{target}`, which has `required-features = [{}]`; the argv table enables no feature, so the target would not build; bind a test of a target with no required features",
                    features.join(", ")
                ),
                Self::NoDocItem { path } => write!(
                    f,
                    "`{path}` has no item part in a documentation test name (the documentation of a crate root is named `src/lib.rs - (line n)`); bind a documented item"
                ),
                Self::Pattern { path, what, error } => write!(
                    f,
                    "`{path}`: the {what} is refused before any process starts: {error}"
                ),
                Self::Divergent { path, plans } => write!(
                    f,
                    "`{path}` has several definitions that would run as different things ({}); bind the canonical path of one definition",
                    plans.join("; ")
                ),
                Self::NoTestListed { name, target } => write!(
                    f,
                    "the list pass of `{target}` shows no test named `{name}`, so a run would prove nothing (a test behind a disabled `cfg(feature)`, or a name that differs from the libtest name, lists nothing); bind a test that the target lists"
                ),
                Self::TestListedWrong {
                    name,
                    target,
                    listed,
                } => write!(
                    f,
                    "the list pass of `{target}` for `{name}` shows {} entries ({}), not exactly one test named `{name}`; bind a test that the target lists once",
                    listed.len(),
                    listed.join(", ")
                ),
                Self::TestCounts {
                    name,
                    passed,
                    failed,
                    ignored,
                } => write!(
                    f,
                    "the run of `{name}` reported {passed} passed; {failed} failed; {ignored} ignored, and the proof needs exactly 1 passed; 0 failed; 0 ignored"
                ),
                Self::TestIgnored { name } => write!(
                    f,
                    "`{name}` is marked `#[ignore]`: it is listed as a test and the run did not run it; remove `#[ignore]` or bind a test that runs"
                ),
                Self::Unparsable { what, text } => write!(
                    f,
                    "the {what} cannot be read: `{}`; the run has no proof",
                    text.escape_debug()
                ),
                Self::NoDocTest { item, package } => write!(
                    f,
                    "no documentation test of the package `{package}` is listed for the item `{item}`; document the item with a runnable example, or bind another item"
                ),
                Self::DocNotRunnable { item, names } => write!(
                    f,
                    "the documentation test of `{item}` ({}) was built and not run (`no_run`) or passes when it does not build (`compile_fail`); bind a runnable doc test",
                    names.join(", ")
                ),
                Self::DocNotRun { item, names } => write!(
                    f,
                    "the documentation test of `{item}` ({}) has no `... ok` line in the run output; it was ignored or did not run",
                    names.join(", ")
                ),
            }
        }
    }

    /// The plan of a binding, decided from the source index.
    ///
    /// `tests` is the index's `#[test]` functions by item.
    pub(crate) fn plan(
        source: &Workspace,
        tests: &std::collections::HashMap<ItemId, TestFn>,
        binding: &Binding,
    ) -> Result<Plan, Refusal> {
        let path = binding.path();
        let sites = binding.resolution().map_err(|error| Refusal::Unresolved {
            path: path.to_string(),
            error: error.to_string(),
        })?;
        let mut plans = Vec::new();
        for id in sites {
            let plan = match binding.kind() {
                BindingKind::Code => {
                    return Err(Refusal::NotRunnable {
                        path: path.to_string(),
                    });
                }
                BindingKind::Test => test_site(source, tests, path, *id)?,
                BindingKind::Example => example_site(source, binding, path, *id)?,
            };
            if !plans.contains(&plan) {
                plans.push(plan);
            }
        }
        let mut plans = plans.into_iter();
        match (plans.next(), plans.next()) {
            (Some(only), None) => Ok(only),
            (Some(first), Some(second)) => Err(Refusal::Divergent {
                path: path.to_string(),
                plans: [first, second]
                    .into_iter()
                    .chain(plans)
                    .map(|plan| plan.to_string())
                    .collect(),
            }),
            // A resolution is never empty.
            (None, _) => Err(Refusal::Unresolved {
                path: path.to_string(),
                error: "the path resolves to no definition".to_string(),
            }),
        }
    }
    
    fn test_site(
        source: &Workspace,
        tests: &std::collections::HashMap<ItemId, TestFn>,
        path: &str,
        id: ItemId,
    ) -> Result<Plan, Refusal> {
        let (Some(item), Some(test)) = (source.item(id), tests.get(&id)) else {
            return Err(Refusal::NotATest {
                path: path.to_string(),
                found: source
                    .item(id)
                    .map_or_else(|| "nothing the index holds".to_string(), describe),
            });
        };
        let Some(target) = source.target(item.target) else {
            return Err(Refusal::NotATest {
                path: path.to_string(),
                found: "an item whose target the index does not hold".to_string(),
            });
        };
        test_plan(
            path,
            &TestFacts {
                package: &test.package,
                kind: test.kind,
                target: &test.target,
                name: &test.name,
                harness: target.harness,
                required_features: &target.required_features,
            },
        )
        .map(Plan::Test)
    }
    
    /// What the index says of an item that is not a test.
    fn describe(item: &Item) -> String {
        format!(
            "{} defined at {}:{}",
            format!("{:?}", item.kind).to_lowercase(),
            item.file,
            item.name_line
        )
    }
    
    /// The index facts of a test function, as text. The tests of this module
    /// build them by hand, with values no manifest would hold, to show that each
    /// one is judged before it reaches an invocation.
    pub(crate) struct TestFacts<'a> {
        pub(crate) package: &'a str,
        pub(crate) kind: TargetKind,
        pub(crate) target: &'a str,
        pub(crate) name: &'a str,
        pub(crate) harness: bool,
        pub(crate) required_features: &'a [String],
    }
    
    pub(crate) fn test_plan(path: &str, facts: &TestFacts<'_>) -> Result<TestPlan, Refusal> {
        let pinned =
            |name: &str| TargetName::new(name).map_err(|error| pattern(path, "target name", error));
        let target = match facts.kind {
            TargetKind::Lib => Target::Lib,
            TargetKind::Bin => Target::Bin(pinned(facts.target)?),
            TargetKind::Test => Target::Test(pinned(facts.target)?),
            kind => {
                return Err(Refusal::TargetKind {
                    path: path.to_string(),
                    target: facts.target.to_string(),
                    kind,
                });
            }
        };
        if !facts.harness {
            return Err(Refusal::NoHarness {
                path: path.to_string(),
                target: facts.target.to_string(),
            });
        }
        if !facts.required_features.is_empty() {
            return Err(Refusal::RequiredFeatures {
                path: path.to_string(),
                target: facts.target.to_string(),
                features: facts.required_features.to_vec(),
            });
        }
        Ok(TestPlan {
            package: Package::new(facts.package)
                .map_err(|error| pattern(path, "package name", error))?,
            target,
            name: Filter::new(facts.name).map_err(|error| pattern(path, "test name", error))?,
        })
    }
    
    fn example_site(
        source: &Workspace,
        binding: &Binding,
        path: &str,
        id: ItemId,
    ) -> Result<Plan, Refusal> {
        let not_an_example = || Refusal::NotAnExample {
            path: path.to_string(),
        };
        let item = source.item(id).ok_or_else(not_an_example)?;
        let target = source.target(item.target).ok_or_else(not_an_example)?;
        match binding.example_kind(id).ok_or_else(not_an_example)? {
            ExampleKind::Target => example_plan(
                path,
                &ExampleFacts {
                    package: &target.package,
                    example: &target.name,
                    required_features: &target.required_features,
                },
            )
            .map(Plan::Example),
            ExampleKind::Doctest => {
                let file_module = file_module(source, item, target);
                let within = doc_item_path(item, target, file_module.as_deref()).ok_or_else(|| {
                    Refusal::NoDocItem {
                        path: path.to_string(),
                    }
                })?;
                doc_plan(path, &target.package, &within).map(Plan::DocTest)
            }
        }
    }

        pub(crate) example: &'a str,

    /// The module path, within its crate, of the file that holds `item` in the
    /// target of the item: `""` for the crate root file, `sync` for `src/sync.rs`.
    /// `None` when the index does not say.
    fn file_module(source: &Workspace, item: &Item, target: &SourceTarget) -> Option<String> {
        let file = source.file(&item.file)?;
        let reach = file
            .reached
            .iter()
            .find(|reach| reach.target == item.target)?;
        let module: &str = &reach.module_path;
        if module == target.crate_path {
            return Some(String::new());
        }
        module
            .strip_prefix(&target.crate_path)?
            .strip_prefix("::")
            .map(str::to_string)
    }
    
    /// The item part of the name rustdoc gives the documentation tests of `item`:
    /// its path within the crate, with an inherent impl's method written under the
    /// self type as the impl writes it (`Wrapper<T>::get`). `None` for the crate
    /// root, which has no item part.
    ///
    /// rustdoc names a method by the module of its impl block. The impl is taken to
    /// sit in the module of its self type when the file that holds it is that
    /// module or one that contains it (an inline module of the file), and in the
    /// module of that file otherwise (`sync::Workplan<'_>::sync` for an impl of
    /// `model::Workplan` written in `src/sync.rs`). An impl in an inline module of
    /// another file, or under another spelling of its self type, is named
    /// differently by rustdoc and the list pass then selects nothing, which is a
    /// refusal.
    fn doc_item_path(item: &Item, target: &SourceTarget, file_module: Option<&str>) -> Option<String> {
        let within = item
            .path
            .strip_prefix(&target.crate_path)?
            .strip_prefix("::")?;
        if let (Some(self_type), Some(SelfType::Local)) = (&item.self_type, &item.self_key) {
            let (owner, method) = within.rsplit_once("::")?;
            let (modules, type_name) = owner
                .rsplit_once("::")
                .map_or(("", owner), |(modules, name)| (modules, name));
            let modules = match file_module {
                Some(file) if !holds(file, modules) => file,
                _ => modules,
            };
            if self_type.split('<').next() == Some(type_name) {
                return Some(if modules.is_empty() {
                    format!("{self_type}::{method}")
                } else {
                    format!("{modules}::{self_type}::{method}")
                });
            }
        }
        Some(within.to_string())
    }
    
    /// Whether the module `outer` is the module `inner` or contains it.
    fn holds(outer: &str, inner: &str) -> bool {
        outer.is_empty()
            || inner == outer
            || inner
                .strip_prefix(outer)
                .is_some_and(|rest| rest.starts_with("::"))
    }

    /// Runs the tests, documentation tests and examples that capabilities bind.
    ///
    /// Every process goes through the [`Runner`]; the executor starts nothing
    /// itself, and it builds an [`Invocation`] only from a [`Plan`], whose values
    /// are validated (see [`Refusal`]).
    ///
    /// The limits are those of the completion gate: [`Limits::build`] (1800
    /// seconds, `LC_ALL=C`) for each pass of a test or documentation test, which
    /// may compile first, and [`Limits::example`] (300 seconds) for the one
    /// `cargo run` of an example. That 300 seconds covers the build `cargo run`
    /// does first, so an example that needs a longer build is killed and fails.
    /// [`Executor::with_limits`] replaces them.
    pub struct Executor<'a, R> {
        source: &'a Workspace,
        runner: R,
        tests: HashMap<ItemId, TestFn>,
        build: Limits,
        example: Limits,
    }
    
    impl<'a, R: Runner> Executor<'a, R> {
        /// An executor over the source index and a runner.
        ///
        /// # Errors
        ///
        /// The [`LoadError`] of the index when it cannot say which functions are
        /// tests, which its own invariants rule out.
        pub fn new(source: &'a Workspace, runner: R) -> Result<Self, LoadError> {
            let tests = source
                .test_fns()?
                .into_iter()
                .map(|test| (test.id, test))
                .collect();
            Ok(Self {
                source,
                runner,
                tests,
                build: Limits::build(),
                example: Limits::example(),
            })

        /// The same executor with other limits: `build` for each pass of a test

        /// What the binding runs as, decided from the facts of the source index
        /// alone. No process is started.
        ///
        /// # Errors
        ///
        /// A [`Refusal`] when the path does not resolve, resolves to something
        /// `cargo` cannot run as the argv table runs it (an item that is not a
        /// `#[test]`, a target with `harness = false` or `required-features`, a
        /// benchmark), or yields a value that fails its pattern.
        pub fn plan(&self, binding: &Binding) -> Result<Plan, Refusal> {
            plan::plan(self.source, &self.tests, binding)
        }
    
        /// Run the binding and judge it.
        ///
        /// A binding that cannot be planned is [`BoundOutcome::Unresolved`] with
        /// no process started. A process killed at its limit, or for its output,
        /// is [`BoundOutcome::Failed`], with the elapsed time in the reason.
        ///
        /// # Errors
        ///
        /// An [`ExecError`] when a process cannot be started or read to its end:
        /// the run has no verdict, which is not the same as a failing binding.
        pub fn run(&self, binding: &Binding) -> Result<Evidence, ExecError> {
            let mut run = Judged::new(binding);
            match self.plan(binding) {
                Err(refusal) => run.unresolved(refusal),
                Ok(Plan::Test(plan)) => self.test(&plan, &mut run)?,
                Ok(Plan::DocTest(plan)) => self.doc(&plan, &mut run)?,
                Ok(Plan::Example(plan)) => self.example(&plan, &mut run)?,
            }
            Ok(run.finish())
        }
    
        /// Start one process. `None` when it was killed at a limit, which the run
        /// has recorded as its failure.
        fn start(
            &self,
            run: &mut Judged<'_>,
            invocation: Invocation,
            limits: &Limits,
        ) -> Result<Option<Output>, ExecError> {
            match self.runner.run(&invocation, limits) {
                Ok(output) => {
                    run.runs.push(Run::new(
                        invocation,
                        Ended::Exited(output.code),
                        output.elapsed,
                    ));
                    Ok(Some(output))
                }
                Err(error @ ExecError::Timeout { elapsed, .. }) => {
                    run.runs
                        .push(Run::new(invocation, Ended::TimedOut, elapsed));
                    run.fail(error.to_string());
                    Ok(None)
                }
                Err(error @ ExecError::OutputTooLarge { elapsed, .. }) => {
                    run.runs
                        .push(Run::new(invocation, Ended::Overflowed, elapsed));
                    run.fail(error.to_string());
                    Ok(None)
                }
                Err(error) => Err(error),
            }
        }
    
        /// The list pass and the run pass of a test.
        fn test(&self, plan: &TestPlan, run: &mut Judged<'_>) -> Result<(), ExecError> {
            let name = plan.name().as_str();
            let target = plan.target().to_string();
            let list = plan.list();
            let Some(output) = self.start(run, list.clone(), &self.build)? else {
                return Ok(());
            };
            if !output.success() {
                run.fail(not_success(&list, &output));
                return Ok(());
            }
            let text = String::from_utf8_lossy(&output.stdout);
            let listed: Vec<_> = parse::test_list(&text)
                .into_iter()
                .filter(|entry| entry.kind == EntryKind::Test)
                .collect();
            run.proof.listed = Some(listed.len());
            match listed.as_slice() {
                [] => {
                    run.unresolved(Refusal::NoTestListed {
                        name: name.to_string(),
                        target,
                    });
                    return Ok(());
                }
                [only] if only.name == name => {}
                _ => {
                    run.unresolved(Refusal::TestListedWrong {
                        name: name.to_string(),
                        target,
                        listed: listed.iter().map(|entry| entry.name.clone()).collect(),
                    });
                    return Ok(());
                }
            }
    
            let invocation = plan.run();
            let Some(output) = self.start(run, invocation.clone(), &self.build)? else {
                return Ok(());
            };
            let text = String::from_utf8_lossy(&output.stdout);
            let counts = match parse::results(&text) {
                Ok(counts) => counts,
                Err(parse::BadResult(line)) => {
                    run.unresolved(Refusal::Unparsable {
                        what: "result line of the run pass",
                        text: line,
                    });
                    return Ok(());
                }
            };
            let [counts] = counts.as_slice() else {
                if output.success() {
                    run.unresolved(Refusal::Unparsable {
                        what: "result of the run pass (it must hold exactly one `test result:` line)",
                        text: format!("{} result lines", counts.len()),
                    });
                } else {
                    run.fail(not_success(&invocation, &output));
                }
                return Ok(());
            };
            run.proof.passed = Some(counts.passed);
            run.proof.failed = Some(counts.failed);
            run.proof.ignored = Some(counts.ignored);
            if !output.success() || counts.failed > 0 {
                run.fail(format!(
                    "{}; {} passed, {} failed, {} ignored",
                    not_success(&invocation, &output),
                    counts.passed,
                    counts.failed,
                    counts.ignored
                ));
            } else if counts.ignored > 0 {
                run.unresolved(Refusal::TestIgnored {
                    name: name.to_string(),
                });
            } else if counts.passed == 1 {
                run.outcome = Some(BoundOutcome::Passed);
            } else {
                run.unresolved(Refusal::TestCounts {
                    name: name.to_string(),
                    passed: counts.passed,
                    failed: counts.failed,
                    ignored: counts.ignored,
                });
            }
            Ok(())
        }
    
        /// The list pass over the documentation tests of the package and the run
        /// pass with the item path as the filter.
        fn doc(&self, plan: &DocPlan, run: &mut Judged<'_>) -> Result<(), ExecError> {
            let item = plan.item().as_str();
            let list = plan.list();
            let Some(output) = self.start(run, list.clone(), &self.build)? else {
                return Ok(());
            };
            if !output.success() {
                run.fail(not_success(&list, &output));
                return Ok(());
            }
            let entries = parse::doc_list(&String::from_utf8_lossy(&output.stdout));
            let selected: Vec<_> = entries
                .iter()
                .filter(|entry| entry.item.as_deref() == Some(item))
                .collect();
            run.proof.listed = Some(entries.len());
            run.proof.selected = Some(selected.len());
            if selected.is_empty() {
                run.unresolved(Refusal::NoDocTest {
                    item: item.to_string(),
                    package: plan.package().to_string(),
                });
                return Ok(());
            }
    
            let invocation = plan.run();
            let Some(output) = self.start(run, invocation.clone(), &self.build)? else {
                return Ok(());
            };
            let text = String::from_utf8_lossy(&output.stdout);
            let by_status = |wanted: &[DocStatus]| -> Vec<String> {
                selected
                    .iter()
                    .filter(|entry| wanted.contains(&parse::doc_status(&text, &entry.name)))
                    .map(|entry| entry.name.clone())
                    .collect()
            };
            let ok = by_status(&[DocStatus::Ok]).len();
            run.proof.ok_lines = Some(ok);
            let not_runnable = by_status(&[DocStatus::CompileOnly, DocStatus::CompileFail]);
            let failed = by_status(&[DocStatus::Failed]);
            let not_run = by_status(&[DocStatus::Ignored, DocStatus::Missing]);
            if !not_runnable.is_empty() {
                run.unresolved(Refusal::DocNotRunnable {
                    item: item.to_string(),
                    names: not_runnable,
                });
            } else if !output.success() || !failed.is_empty() {
                // The exit status covers the documentation tests the substring
                // also matched (`foo_more` for `foo`), which the gate does not
                // count and which still refuse the binding.
                let named = if failed.is_empty() {
                    String::new()
                } else {
                    format!("; failed: {}", failed.join(", "))
                };
                run.fail(format!("{}{named}", not_success(&invocation, &output)));
            } else if !not_run.is_empty() {
                run.unresolved(Refusal::DocNotRun {
                    item: item.to_string(),
                    names: not_run,
                });
            } else if ok == selected.len() {
                run.outcome = Some(BoundOutcome::Passed);
            }
            Ok(())
        }
    
        /// The run of an example: its exit status decides.
        fn example(&self, plan: &ExamplePlan, run: &mut Judged<'_>) -> Result<(), ExecError> {
            let invocation = plan.run();
            let Some(output) = self.start(run, invocation.clone(), &self.example)? else {
                return Ok(());
            };
            if output.success() {
                run.outcome = Some(BoundOutcome::Passed);
            } else {
                run.fail(not_success(&invocation, &output));
            }
            Ok(())
        }
    }
    
    /// The reason for a process that did not exit with code 0: the code and the
    /// start of what it wrote to standard error.
    fn not_success(invocation: &Invocation, output: &Output) -> String {
        let how = output.code.map_or_else(
            || "was ended by a signal".to_string(),
            |code| format!("exited with code {code}"),
        );
        let stderr = output.stderr_text();
        if stderr.is_empty() {
            format!("`{invocation}` {how}")
        } else {
            format!("`{invocation}` {how}: {stderr}")
        }
    }
    
    /// A run in progress.
    struct Judged<'b> {
        binding: &'b Binding,
        runs: Vec<Run>,
        proof: Proof,
        outcome: Option<BoundOutcome>,
    }
    
    impl<'b> Judged<'b> {
        fn new(binding: &'b Binding) -> Self {
            Self {
                binding,
                runs: Vec::new(),
                proof: Proof::default(),
                outcome: None,
            }
        }
    
        fn fail(&mut self, reason: String) {
            self.outcome = Some(BoundOutcome::Failed { reason });
        }
    
        fn unresolved(&mut self, reason: Refusal) {
            self.outcome = Some(BoundOutcome::Unresolved { reason });
        }
    
        /// The evidence. A run that reached no verdict is a failure: a pass is
        /// only ever set by the proof of its kind.
        fn finish(self) -> Evidence {
            let outcome = self.outcome.unwrap_or_else(|| BoundOutcome::Failed {
                reason: "the run ended without the proof its kind requires".to_string(),
            });
            Evidence::new(
                self.binding.capability(),
                self.binding.kind(),
                self.binding.path(),
                self.runs,
                self.proof,
                outcome,
            )
        }
    }

    /// The seven sections of a capability, each with a line of text.
    fn sections() -> String {
        let mut text = String::new();
        for name in [
            "Purpose",
            "Contract",
            "Usage",
            "Examples",
            "Constraints",
            "Explanation",
            "Verification",
        ] {
            write!(text, "## {name}\n\nText for {name}.\n\n").unwrap();
        }
        text
    }
    
    /// The capability document that lists `entry` in `list`.
    fn capability(list: &str, entry: &str) -> String {
        let sections = sections();
        let entry = format!("\"{entry}\"");
        let (code, tests, examples) = match list {
            "code" => (entry.as_str(), "", ""),
            "tests" => ("", entry.as_str(), ""),
            _ => ("", "", entry.as_str()),
        };
        format!(
            "---\ntype: Capability\nid: cap:demo\ntitle: Demo\ndescription: Demo.\n\
             state: implementing\nbindings:\n  code: [{code}]\n  tests: [{tests}]\n  \
             examples: [{examples}]\ndepends: []\n---\n\n# Demo\n\n{sections}"
        )
    }
    
    fn workspace() -> Workspace {
        let tree: MemoryTree = files().into_iter().collect();
        Workspace::load(&tree, &["."]).expect("the demo workspace loads")
    }
    
    /// Run the binding `entry` of `list` over the demo workspace.
    fn run(
        list: &str,
        entry: &str,
        runner: &FakeRunner,
        limits: Option<(Limits, Limits)>,
    ) -> Result<Evidence, ExecError> {
        let source = workspace();
        let doc = capability(list, entry);
        let docs = MemoryTree::from([("docs/capabilities/demo.md", doc.as_str())]);
        let plan = Workplan::load(&docs, &source);
        let binding = plan
            .graph()
            .bindings()
            .first()
            .expect("the capability lists a binding");
        let mut executor = Executor::new(&source, runner).expect("the index lists its tests");
        if let Some((build, example)) = limits {
            executor = executor.with_limits(build, example);
        }
        executor.run(binding)
    }
    
    #[allow(
        clippy::unnecessary_wraps,
        reason = "a script answers with a Result, like the runner"
    )]
    fn output(code: i32, stdout: &str) -> Result<Output, ExecError> {
        Ok(Output::new(
            Some(code),
            stdout.as_bytes().to_vec(),
            Vec::new(),
        ))
    }
    
    fn ok(stdout: &str) -> Result<Output, ExecError> {
        output(0, stdout)
    }

    /// A runner for a test binding: the list pass and the run pass.
    fn test_runner(target: &str, name: &str, list: &str, run: Result<Output, ExecError>) -> FakeRunner {
        FakeRunner::new()
            .on(
                Program::Cargo,
                &[
                    "test",
                    "--frozen",
                    "--package=demo-pkg",
                    target,
                    "--",
                    "--list",
                    "--exact",
                    "--",
                    name,
                ],
                ok(list),
            )
            .on(
                Program::Cargo,
                &[
                    "test",
                    "--frozen",
                    "--package=demo-pkg",
                    target,
                    "--",
                    "--exact",
                    "--",
                    name,
                ],
                run,
            )
    }
    
    fn commands(runner: &FakeRunner) -> Vec<String> {
        runner
            .calls()
            .iter()
            .map(crate::exec::Call::command)
            .collect()
    }

    /// Run a binding that must be refused from the index alone and return the
    /// refusal; the runner is asked nothing.
    fn refused(list: &str, entry: &str) -> Refusal {
        let runner = FakeRunner::new();
        let evidence = run(list, entry, &runner, None).unwrap();
        assert!(runner.calls().is_empty(), "{:?}", runner.calls());
        assert_eq!(evidence.runs().len(), 0);
        let BoundOutcome::Unresolved { reason } = evidence.outcome().clone() else {
            panic!("{evidence}");
        };
        reason
    }

    /// A workspace whose manifest holds a name that no pattern allows. The index
    /// reads the manifest as it is, so the pattern is the last defence between a
    /// manifest and a command line.
    fn hostile_workspace(package: &str, extra: &str, files: &[(&str, &str)]) -> Workspace {
        let manifest = format!("[package]\nname = \"{package}\"\nversion = \"0.1.0\"\n\n{extra}");
        let tree: MemoryTree = files
            .iter()
            .copied()
            .chain([("Cargo.toml", manifest.as_str())])
            .collect();
        Workspace::load(&tree, &["."]).expect("the index reads the manifest as it is")
    }
    
    #[test]
    fn a_hostile_name_in_the_manifest_never_reaches_the_runner() {
        const CASE: &str = "#[test]\nfn case() {}\n";
        /// A package, the manifest text and files that hold the hostile name, the
        /// binding that reaches it and the value the pattern refuses.
        type Case<'a> = (&'a str, &'a str, &'a [(&'a str, &'a str)], &'a str, &'a str);
        let cases: [Case<'_>; 4] = [
            (
                "demo",
                "[[test]]\nname = \"-Zunstable-options\"\npath = \"tests/hostile.rs\"\n",
                &[("src/lib.rs", ""), ("tests/hostile.rs", CASE)],
                "demo::test_crate::_Zunstable_options::case",
                "target name",
            ),
            (
                "demo",
                "[[test]]\nname = \"--exact\"\npath = \"tests/hostile.rs\"\n",
                &[("src/lib.rs", ""), ("tests/hostile.rs", CASE)],
                "demo::test_crate::__exact::case",
                "target name",
            ),
            (
                "demo",
                "[[bin]]\nname = \"--include-ignored\"\npath = \"src/hostile.rs\"\n",
                &[
                    ("src/lib.rs", ""),
                    ("src/hostile.rs", "fn main() {}\n#[test]\nfn case() {}\n"),
                ],
                "demo::bin_crate::__include_ignored::case",
                "target name",
            ),
            (
                "--frozen",
                "",
                &[("src/lib.rs", CASE)],
                "__frozen::case",
                "package name",
            ),
        ];
        for (package, extra, files, path, what) in cases {
            let source = hostile_workspace(package, extra, files);
            let doc = capability("tests", path);
            let docs = MemoryTree::from([("docs/capabilities/demo.md", doc.as_str())]);
            let plan = Workplan::load(&docs, &source);
            let runner = FakeRunner::new();
            let executor = Executor::new(&source, &runner).unwrap();
            let binding = &plan.graph().bindings()[0];
            assert!(binding.resolution().is_ok(), "{path} must resolve");
            let evidence = executor.run(binding).unwrap();
            assert!(
                matches!(
                    unresolved(&evidence),
                    Refusal::Pattern { what: found, .. } if *found == what
                ),
                "{path}: {evidence}"
            );
            assert!(runner.calls().is_empty(), "{path} reached the runner");
        }
    }

    //! The one place a process is started.
    //!
    //! Every `cargo` and `git` process of the workplan is an [`Invocation`] given to
    //! a [`Runner`], and so is the one further process, the caller's own command of
    //! `evidence record` ([`Invocation::trusted`] of a [`TrustedArgv`], which only
    //! the command line of the running process can give). The invocation is a [`Program`] and a list of [`Arg`]s; an
    //! argument is fixed text or a validated value, and a value can be built only
    //! through one of the patterns of the completion gate (see [`Pattern`]), so text
    //! from a document cannot reach a command line by any other route.
    //!
    //! [`SystemRunner`] starts the process directly with no shell and the argv
    //! vector as given, with its working directory set to [`Programs::cwd`] and its
    //! standard input null. The environment is the caller's, plus `LC_ALL=C` for a
    //! `cargo` or `git` process whose output is parsed ([`Limits::c_locale`]); the
    //! caller's own command always runs in the caller's environment unchanged. Standard output and
    //! standard error are captured, each up to a bound; a process that writes more,
    //! or runs longer than its wall-clock limit, is killed and the run is an
    //! [`ExecError`]. On Unix the process leads a process group of its own and a
    //! kill (SIGKILL to the group, sent directly and not through a `kill` program)
    //! reaches the group, so a test binary that `cargo` started does not outlive its
    //! timeout. A descendant that leaves the group, as one that calls `setsid` or
    //! `setpgid` does, is not reached and may outlive the kill; elsewhere than Unix
    //! only the process itself is killed. A kill that cannot be sent is an
    //! [`ExecError::Io`] that says the process may still be running.
    //!
    //! [`FakeRunner`] answers invocations from a script, in process and on every
    //! platform, and records what it was asked.
    //!
    //! ```rust
    //! use raf_workplan::exec::{FakeRunner, Invocation, Limits, Output, Program, Runner};
    //!
    //! let runner = FakeRunner::new().on(
    //!     Program::Git,
    //!     &["--version"],

    //! );

    //! assert!(output.success());

        pub code: Option<i32>,
        /// What it wrote to standard output.
        pub stdout: Vec<u8>,

        pub stderr: Vec<u8>,

        #[must_use]
        pub fn new(code: Option<i32>, stdout: Vec<u8>, stderr: Vec<u8>) -> Self {

                stderr,

        #[must_use]

            self.elapsed = elapsed;

        #[must_use]
        pub fn success(&self) -> bool {
            self.code == Some(0)

        /// for a message: invalid UTF-8 is replaced and a long text is cut.
        pub(crate) fn stderr_text(&self) -> String {
            let text = String::from_utf8_lossy(&self.stderr);
            let text = text.trim();
            let mut cut: String = text.chars().take(MESSAGE_TEXT).collect();
            if text.chars().nth(MESSAGE_TEXT).is_some() {
                cut.push_str("...");
            }
            cut
        }

    const MESSAGE_TEXT: usize = 2048;

    pub enum Stream {
        /// Standard output.
        Stdout,
        /// Standard error.
        Stderr,
    }
    
    impl fmt::Display for Stream {
        fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {

    /// Why a process did not run to a result. Each case is an error and none is a
    /// skip: a process that could not be started, ran too long or wrote too much
    /// has no verdict.
    #[derive(Debug, Clone, PartialEq, Eq)]
    #[non_exhaustive]
    pub enum ExecError {
        /// The process could not be started.
        Spawn {
            /// The invocation, as text for the message.
            command: String,
            /// The program that was looked for.
            program: String,
            /// The kind of the operating-system error.
            kind: io::ErrorKind,
            /// The operating-system error.
            message: String,
        },
        /// The process ran past its wall-clock limit and was killed.
        Timeout {
            /// The invocation, as text for the message.
            command: String,
            /// The limit.
            timeout: Duration,
            /// How long it had run when it was killed.
            elapsed: Duration,
        },
        /// The process wrote more than the bound to one stream and was killed.
        OutputTooLarge {
            /// The invocation, as text for the message.
            command: String,
            /// The stream that overflowed.
            stream: Stream,
            /// The bound, in bytes.
            limit: usize,
            /// How long it had run when it was killed.
            elapsed: Duration,
        },
        /// Reading the output or waiting for the process failed, or a process that
        /// had to be killed could not be.
        Io {
            /// The invocation, as text for the message.
            command: String,
            /// The operating-system error.
            message: String,
        },
        /// A [`FakeRunner`] was asked for an invocation it has no answer for.
        Unscripted {
            /// The invocation, as text for the message.
            command: String,
        },
    }
    
    impl fmt::Display for ExecError {
        fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
            match self {
                Self::Spawn {
                    command,
                    program,
                    message,
                    ..
                } => write!(
                    f,
                    "cannot start `{command}`: `{program}` {message}; install it, put it on PATH, or for cargo set $CARGO"
                ),
                Self::Timeout {
                    command,
                    timeout,
                    elapsed,
                } => write!(
                    f,
                    "`{command}` ran {:.1} s of its {:.0} s limit and was killed; the run has no verdict",
                    elapsed.as_secs_f64(),
                    timeout.as_secs_f64()
                ),
                Self::OutputTooLarge {
                    command,
                    stream,
                    limit,
                    elapsed,
                } => write!(
                    f,
                    "`{command}` wrote more than {limit} bytes to {stream} in {:.1} s and was killed; the run has no verdict",
                    elapsed.as_secs_f64()
                ),
                Self::Io { command, message } => {
                    write!(f, "`{command}` could not be read to its end: {message}")
                }
                Self::Unscripted { command } => write!(
                    f,
                    "no scripted answer for `{command}`; add one with `FakeRunner::on`"
                ),
            }
        }
    }
    
    impl Error for ExecError {}
    
    /// Starts the processes of the workplan.
    pub trait Runner {
        /// Run the invocation to its end under the limits.
        ///
        /// # Errors
        ///
        /// An [`ExecError`] when the process cannot be started, runs past the
        /// wall-clock limit, writes more than the output bound to a stream, or
        /// cannot be read to its end. A process that exits with a failing code is
        /// an `Ok` [`Output`]; judging the code is the caller's.
        fn run(&self, invocation: &Invocation, limits: &Limits) -> Result<Output, ExecError>;
    }
    
    impl<R: Runner + ?Sized> Runner for &R {
        fn run(&self, invocation: &Invocation, limits: &Limits) -> Result<Output, ExecError> {
            (**self).run(invocation, limits)
        }
    }

    //! The one place in this crate that reads a clock or waits on one.
    //!
    //! The runner measures how long a process has run and pauses between two looks
    //! at it. Both go through [`Clock`], so the runner's decisions can be driven by
    //! a clock that a test controls, and no other file has to name a clock at all.
    //! [`WallClock`] is the real reading; it is the only implementation outside
    //! tests.

        clippy::disallowed_types,
        clippy::disallowed_methods,

    use std::time::{Duration, Instant};
    
    /// A reading of the time since a run started, and a way to wait.
    pub(super) trait Clock {
        /// The time since the clock started.
        fn elapsed(&self) -> Duration;
    
        /// Wait for this long before the next look.
        fn pause(&mut self, pause: Duration);
    
        /// The process of the run has started with this id and is not collected
        /// yet. The clock of the machine has no use for it; a clock of a test may
        /// watch the process.
        fn spawned(&mut self, _process: u32) {}
    }
    
    /// The clock of the machine: time since [`WallClock::start`], and a sleep.
    pub(super) struct WallClock(Instant);
    
    impl WallClock {
        /// Start counting now.
        pub(super) fn start() -> Self {
            Self(Instant::now())
        }
    }
    
    impl Clock for WallClock {
        fn elapsed(&self) -> Duration {
            self.0.elapsed()
        }
    
        fn pause(&mut self, pause: Duration) {
            thread::sleep(pause);
        }
    }

    //! The in-process runner for tests: scripted answers, recorded calls.
    
    use std::sync::{Mutex, MutexGuard, PoisonError};
    
    use super::{ExecError, Invocation, Limits, Output, Program, Runner, Stream};
    
    /// One invocation a [`FakeRunner`] was asked to run.
    #[derive(Debug, Clone, PartialEq, Eq)]
    pub struct Call {
        program: Program,
        args: Vec<String>,
        limits: Limits,
    }
    
    impl Call {
        /// The program.
        #[must_use]
        pub fn program(&self) -> Program {
            self.program
        }
    
        /// The argv vector, without the program.
        #[must_use]
        pub fn args(&self) -> &[String] {
            &self.args
        }
    
        /// The limits the caller asked for.
        #[must_use]
        pub fn limits(&self) -> &Limits {
            &self.limits
        }
    
        /// `program arg arg ...`, as the invocation reads in a message; for the
        /// caller's own command, its words.
        #[must_use]
        pub fn command(&self) -> String {
            if self.program == Program::Trusted {
                return self.args.join(" ");
            }
            let mut text = self.program.to_string();
            for arg in &self.args {
                text.push(' ');
                text.push_str(arg);
            }
            text
        }
    }
    
    /// A scripted answer for one invocation.
    #[derive(Debug)]
    struct Rule {
        program: Program,
        args: Vec<String>,
        answer: Result<Output, ExecError>,
    }
    
    /// A runner that starts nothing. It answers an invocation from the script it
    /// was built with and records every call, on every platform.
    ///
    /// Answers repeat: an invocation gets the same answer each time it is asked.
    /// An invocation with no answer is [`ExecError::Unscripted`], so a test that
    /// runs a command it did not expect fails. The bounds of the [`Limits`] are
    /// applied to a scripted [`Output`] as the real runner applies them to a
    /// process: more bytes than the output bound on a stream is
    /// [`ExecError::OutputTooLarge`], and an elapsed time over the time limit is
    /// [`ExecError::Timeout`].
    #[derive(Debug, Default)]
    pub struct FakeRunner {
        rules: Vec<Rule>,
        calls: Mutex<Vec<Call>>,
    }
    
    impl FakeRunner {
        /// A runner with no answers.
        #[must_use]
        pub fn new() -> Self {
            Self::default()
        }
    
        /// Answer the invocation with exactly this program and argv vector. A
        /// later answer for the same invocation replaces the earlier one.
        #[must_use]
        pub fn on(
            mut self,
            program: Program,
            args: &[&str],
            answer: Result<Output, ExecError>,
        ) -> Self {
            let args: Vec<String> = args.iter().map(|arg| (*arg).to_string()).collect();
            self.rules
                .retain(|rule| !(rule.program == program && rule.args == args));
            self.rules.push(Rule {
                program,
                args,
                answer,
            });
            self
        }
    
        /// Every call so far, in order.
        #[must_use]
        pub fn calls(&self) -> Vec<Call> {
            self.recorded().clone()
        }
    
        fn recorded(&self) -> MutexGuard<'_, Vec<Call>> {
            self.calls.lock().unwrap_or_else(PoisonError::into_inner)
        }
    }
    
    impl Runner for FakeRunner {
        fn run(&self, invocation: &Invocation, limits: &Limits) -> Result<Output, ExecError> {
            let call = Call {
                program: invocation.program(),
                args: invocation.argv(),
                limits: *limits,
            };
            let command = call.command();
            let answer = self
                .rules
                .iter()
                .find(|rule| rule.program == call.program && rule.args == call.args)
                .map(|rule| rule.answer.clone());
            self.recorded().push(call);
            let output = answer.ok_or(ExecError::Unscripted {
                command: command.clone(),
            })??;
            if output.elapsed > limits.timeout() {
                return Err(ExecError::Timeout {
                    command,
                    timeout: limits.timeout(),
                    elapsed: output.elapsed,
                });
            }
            for (stream, bytes) in [
                (Stream::Stdout, &output.stdout),
                (Stream::Stderr, &output.stderr),
            ] {
                if bytes.len() > limits.max_output() {
                    return Err(ExecError::OutputTooLarge {
                        command,
                        stream,
                        limit: limits.max_output(),
                        elapsed: output.elapsed,
                    });
                }
            }
            Ok(output)
        }
    }

    //! The patterns that decide which text may reach a command line, and the
    //! validated newtypes that can only be built through them.
    //!
    //! The four patterns of the completion gate are written by hand, byte by byte,
    //! over ASCII only:
    //!
    //! * `<name>`: `[A-Za-z_][A-Za-z0-9_]*(::[A-Za-z_][A-Za-z0-9_]*)*`;
    //! * `<pkg>`, `<bin>`, `<test>` and `<example>`: `[A-Za-z0-9_][A-Za-z0-9_-]*`;
    //! * `<item-path>`: `<name>` with an optional generic argument list
    //!   `<[A-Za-z0-9_,':&]*>` after each segment;
    //! * `<relative-source-path>`: `[A-Za-z0-9_][A-Za-z0-9_./-]*`.
    //!
    //! Five further validators serve the Git commands: a full reference name, a
    //! commit id, a tree path (a file or directory of the workspace as the
    //! history command names it), an identifier (the text of a `-S` pickaxe) and a
    //! date (the value of `--since`, which the history compares with the author
    //! date of each commit and gives to no process). Each newtype keeps its text private, so a
    //! value of the type is proof that the text matched.
    
    use std::error::Error;
    use std::fmt;
    
    /// The longest value shown in an error message, in characters.
    const SHOWN: usize = 64;
    
    /// The longest reference name accepted, in bytes.
    const REF_NAME_LIMIT: usize = 1024;
    
    /// The longest tree path accepted, in bytes.
    const TREE_PATH_LIMIT: usize = 4096;
    
    /// A pattern a value must match before it reaches a command line.
    #[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
    #[non_exhaustive]
    pub enum Pattern {
        /// `<name>`: a path of identifiers within a crate, such as `a::b::c`.
        Name,
        /// `<pkg>`, `<bin>`, `<test>` and `<example>`: a package or target name,
        /// which may hold hyphens and never begins with one.
        TargetName,
        /// `<item-path>`: a `<name>` whose segments may carry a generic argument
        /// list, such as `Wrapper<T>::get`.
        ItemPath,
        /// `<relative-source-path>`: a relative path of the list output, with no
        /// empty, `.` or `..` segment.
        SourcePath,
        /// A full Git reference name, such as `refs/remotes/origin/main`.
        RefName,
        /// A Git commit id: 40 or 64 lowercase hexadecimal digits.
        Commit,
        /// `<tree-path>`: a path relative to the workspace root as the history
        /// command gives it to `git`, only after `--` or after the colon of one
        /// `-L` argument, so it may begin with `-`. Its segments are separated by
        /// `/` and none is empty, `.` or `..`; it holds no control character and
        /// no `\`, `*`, `?` or `[`, does not begin with `/` or `:` (so it is never
        /// a pathspec pattern or magic), and is at most 4096 bytes.
        TreePath,
        /// `<identifier>`: a Rust identifier, the text of a `-S` argument.
        Identifier,
        /// `<date>`: `YYYY-MM-DD`, a day of the Gregorian calendar from
        /// 1970-01-01 to 2099-12-31, the value of `--since`. It reaches no
        /// process: the history compares the start of that day in UTC with the
        /// author date of each commit, in process.
        Date,
    }
    
    impl Pattern {
        /// Every pattern.
        pub const ALL: [Self; 9] = [
            Self::Name,
            Self::TargetName,
            Self::ItemPath,
            Self::SourcePath,
            Self::RefName,
            Self::Commit,
            Self::TreePath,
            Self::Identifier,
            Self::Date,
        ];
    
        /// The pattern as the capability documents write it.
        #[must_use]
        pub const fn expression(self) -> &'static str {
            match self {
                Self::Name => "[A-Za-z_][A-Za-z0-9_]*(::[A-Za-z_][A-Za-z0-9_]*)*",
                Self::TargetName => "[A-Za-z0-9_][A-Za-z0-9_-]*",
                Self::ItemPath => {
                    "[A-Za-z_][A-Za-z0-9_]*(<[A-Za-z0-9_,':&]*>)?(::[A-Za-z_][A-Za-z0-9_]*(<[A-Za-z0-9_,':&]*>)?)*"
                }
                Self::SourcePath => "[A-Za-z0-9_][A-Za-z0-9_./-]*",
                Self::RefName => "refs/<segment>(/<segment>)*",
                Self::Commit => "[0-9a-f]{40}|[0-9a-f]{64}",
                Self::TreePath => "[^/:][^/]*(/[^/]+)*",
                Self::Identifier => "[A-Za-z_][A-Za-z0-9_]*",
                Self::Date => "(19[7-9][0-9]|20[0-9][0-9])-[0-9]{2}-[0-9]{2}",
            }
        }
    
        /// What a value of the pattern names.
        const fn what(self) -> &'static str {
            match self {
                Self::Name => "item path within a crate",
                Self::TargetName => "package or target name",
                Self::ItemPath => "documented item path",
                Self::SourcePath => "relative source path",
                Self::RefName => "full reference name",
                Self::Commit => "commit id",
                Self::TreePath => "tree path",
                Self::Identifier => "identifier",
                Self::Date => "date of the form YYYY-MM-DD from 1970-01-01 to 2099-12-31",
            }
        }
    
        /// Whether the whole text matches the pattern.
        ///
        /// The relative source path and the tree path are narrower than their
        /// expressions: a segment that is empty, `.` or `..` does not match, so a
        /// path stays inside the tree it is read from, and a tree path holds no
        /// control character, no `\`, `*`, `?` or `[` and at most 4096 bytes. A
        /// date names a real day: `2026-02-29` does not match and `2028-02-29`
        /// does.
        #[must_use]
        pub fn matches(self, text: &str) -> bool {
            match self {
                Self::Name => is_path(text, false),
                Self::TargetName => is_target_name(text),
                Self::ItemPath => is_path(text, true),
                Self::SourcePath => is_source_path(text),
                Self::RefName => is_ref_name(text),
                Self::Commit => is_commit(text),
                Self::TreePath => is_tree_path(text),
                Self::Identifier => is_identifier(text),
                Self::Date => is_date(text),
            }
        }
    
        /// `text` unchanged when it matches.
        fn check(self, text: &str) -> Result<String, PatternError> {
            if self.matches(text) {
                Ok(text.to_string())
            } else {
                Err(PatternError {
                    pattern: self,
                    value: shown(text),
                })
            }
        }
    }
    
    /// A value that does not match its pattern. The value is kept for the message
    /// only; it is never placed on a command line.
    #[derive(Debug, Clone, PartialEq, Eq)]
    pub struct PatternError {
        pattern: Pattern,
        value: String,
    }
    
    impl PatternError {
        /// The pattern the value had to match.
        #[must_use]
        pub fn pattern(&self) -> Pattern {
            self.pattern
        }
    
        /// The refused value, cut to 64 characters.
        #[must_use]
        pub fn value(&self) -> &str {
            &self.value
        }
    }
    
    impl fmt::Display for PatternError {
        fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
            write!(
                f,
                "`{}` is not a {}: it must match `{}`; the value is not passed to a process",

                self.pattern.what(),

    }
    
    impl Error for PatternError {}
    
    /// The first `SHOWN` characters of a value, marked when cut.
    fn shown(text: &str) -> String {
        let mut cut: String = text.chars().take(SHOWN).collect();
        if text.chars().nth(SHOWN).is_some() {
            cut.push_str("...");
        }
        cut
    }
    
    /// `<name>` when `generics` is false, `<item-path>` when it is true.
    fn is_path(text: &str, generics: bool) -> bool {
        let bytes = text.as_bytes();
        let mut at = 0;
        loop {
            if !bytes.get(at).is_some_and(|b| is_ident_start(*b)) {
                return false;
            }
            at += 1;
            while bytes.get(at).is_some_and(|b| is_ident_continue(*b)) {
                at += 1;
            }
            if generics && bytes.get(at) == Some(&b'<') {
                at += 1;
                while bytes.get(at).is_some_and(|b| is_generic(*b)) {
                    at += 1;
                }
                if bytes.get(at) != Some(&b'>') {
                    return false;
                }
                at += 1;
            }
            if at == bytes.len() {
                return true;
            }
            if bytes[at..].starts_with(b"::") {
                at += 2;
            } else {
                return false;
            }
        }
    }
    
    fn is_ident_start(byte: u8) -> bool {
        byte.is_ascii_alphabetic() || byte == b'_'
    }
    
    fn is_ident_continue(byte: u8) -> bool {
        byte.is_ascii_alphanumeric() || byte == b'_'
    }
    
    fn is_generic(byte: u8) -> bool {
        byte.is_ascii_alphanumeric() || matches!(byte, b'_' | b',' | b'\'' | b':' | b'&')
    }
    
    /// `[A-Za-z0-9_][A-Za-z0-9_-]*`.
    fn is_target_name(text: &str) -> bool {
        let bytes = text.as_bytes();
        bytes
            .first()
            .is_some_and(|b| b.is_ascii_alphanumeric() || *b == b'_')
            && bytes
                .iter()
                .all(|b| b.is_ascii_alphanumeric() || matches!(*b, b'_' | b'-'))
    }
    
    /// `[A-Za-z0-9_][A-Za-z0-9_./-]*`, with no empty, `.` or `..` segment.
    fn is_source_path(text: &str) -> bool {
        let bytes = text.as_bytes();
        bytes
            .first()
            .is_some_and(|b| b.is_ascii_alphanumeric() || *b == b'_')
            && bytes
                .iter()
                .all(|b| b.is_ascii_alphanumeric() || matches!(*b, b'_' | b'.' | b'/' | b'-'))
            && text
                .split('/')
                .all(|segment| !matches!(segment, "" | "." | ".."))
    }
    
    /// A full reference name: `refs/` and then segments of letters, digits and
    /// `. _ - = + @ , #` (or any non-ASCII letter or digit), none empty, none
    /// beginning with `.` or ending with `.` or `.lock`, and no `..`.
    fn is_ref_name(text: &str) -> bool {
        let Some(rest) = text.strip_prefix("refs/") else {
            return false;
        };
        if text.len() > REF_NAME_LIMIT || text.contains("..") {
            return false;
        }
        rest.split('/').all(|segment| {
            !segment.is_empty()
                && !segment.starts_with('.')
                && !segment.ends_with('.')
                && segment.strip_suffix(".lock").is_none()
                && segment.chars().all(|c| {
                    if c.is_ascii() {
                        c.is_ascii_alphanumeric()
                            || matches!(c, '.' | '_' | '-' | '=' | '+' | '@' | ',' | '#')
                    } else {
                        c.is_alphanumeric()
                    }
                })
        })
    }
    
    /// 40 (SHA-1) or 64 (SHA-256) lowercase hexadecimal digits.
    fn is_commit(text: &str) -> bool {
        matches!(text.len(), 40 | 64)
            && text
                .bytes()
                .all(|byte| purrdf_hash::hex::nibble_canonical(byte).is_some())
    }
    
    /// A tree path: see [`Pattern::TreePath`].
    fn is_tree_path(text: &str) -> bool {
        !text.is_empty()
            && text.len() <= TREE_PATH_LIMIT
            && !text.starts_with([':', '/'])
            && !text
                .chars()
                .any(|c| c.is_control() || matches!(c, '\\' | '*' | '?' | '['))
            && text
                .split('/')
                .all(|segment| !matches!(segment, "" | "." | ".."))
    }
    
    /// `[A-Za-z_][A-Za-z0-9_]*`.
    fn is_identifier(text: &str) -> bool {
        let bytes = text.as_bytes();
        bytes.first().is_some_and(|b| is_ident_start(*b)) && bytes.iter().all(|b| is_ident_continue(*b))
    }
    
    /// `YYYY-MM-DD`, a day of the Gregorian calendar of the years 1970 to 2099.
    fn is_date(text: &str) -> bool {
        purrdf_xsd::rfc3339::parse_date(text).is_ok_and(|(year, _, _)| (1970..=2099).contains(&year))
    }
    
    macro_rules! validated {
        ($(#[$meta:meta])* $name:ident, $pattern:expr) => {
            $(#[$meta])*
            #[derive(Debug, Clone, PartialEq, Eq, Hash)]

            impl $name {

                ///
                /// # Errors

                }
            }
    
            impl fmt::Display for $name {
                fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
                    f.write_str(&self.0)
                }
            }
        };
    }

    /// The longest pause between two looks at a running process.
    const POLL_CEILING: Duration = Duration::from_millis(20);
    
    /// Starts real processes: no shell, the argv vector as given, standard input
    /// null, standard output and standard error captured within the bound of
    /// [`Limits`], and a kill when the wall-clock limit passes.
    ///
    /// A bare program name is looked up by the operating system on the `PATH` of
    /// this process, or on the directories given to [`SystemRunner::search_path`].
    #[derive(Debug, Clone)]
    pub struct SystemRunner {
        programs: Programs,
        search_path: Option<OsString>,
    }
    
    impl SystemRunner {
        /// A runner for these programs.
        #[must_use]
        pub fn new(programs: Programs) -> Self {
            Self {
                programs,
                search_path: None,
            }
        }
    
        /// Look a bare program name up in these directories (separated as `PATH`
        /// is) instead of on the `PATH` of this process. The environment of the
        /// started process is not changed.
        #[must_use]
        pub fn search_path(mut self, directories: impl Into<OsString>) -> Self {
            self.search_path = Some(directories.into());
            self
        }
    
        /// The programs this runner starts.
        #[must_use]
        pub fn programs(&self) -> &Programs {
            &self.programs
        }
    
        /// The path to start for the invocation, or the error when a search path
        /// is set and the name is not in it.
        fn program_path(&self, invocation: &Invocation) -> Result<PathBuf, ExecError> {
            // The caller's own command names its program as its first word.
            let first = PathBuf::from(invocation.argv().first().cloned().unwrap_or_default());
            let named = self
                .programs
                .path_of(invocation.program())
                .unwrap_or(first.as_path());
            let Some(directories) = &self.search_path else {
                return Ok(named.to_path_buf());
            };
            if named.components().count() != 1 {
                return Ok(named.to_path_buf());
            }
            std::env::split_paths(directories)
                .map(|directory| directory.join(named))
                .find(|candidate| candidate.is_file())
                .ok_or_else(|| ExecError::Spawn {
                    command: invocation.to_string(),
                    program: named.display().to_string(),
                    kind: io::ErrorKind::NotFound,
                    message: "is not in the search path".to_string(),
                })
        }
    }
    
    impl Runner for SystemRunner {
        fn run(&self, invocation: &Invocation, limits: &Limits) -> Result<Output, ExecError> {
            self.run_with(invocation, limits, WallClock::start)
        }
    }
    
    impl SystemRunner {
        /// Run the invocation, starting `start` just before the process is.
        fn run_with<C: Clock>(
            &self,
            invocation: &Invocation,
            limits: &Limits,
            start: impl FnOnce() -> C,
        ) -> Result<Output, ExecError> {
            let command_text = invocation.to_string();
            let program = self.program_path(invocation)?;
            let trusted = invocation.program() == Program::Trusted;
            let mut argv = invocation.argv();
            if trusted {
                // The first word of the caller's own command is its program.
                argv.remove(0);
            }
            let mut command = Command::new(&program);
            command
                .args(argv)
                .current_dir(&self.programs.cwd)
                .stdin(Stdio::null())
                .stdout(Stdio::piped())
                .stderr(Stdio::piped());
            // The caller's own command keeps the caller's environment.
            if limits.c_locale() && !trusted {
                command.env("LC_ALL", "C");
            }
            #[cfg(unix)]
            {
                use std::os::unix::process::CommandExt as _;
                command.process_group(0);
            }
            let mut clock = start();
            let mut child = command.spawn().map_err(|error| ExecError::Spawn {
                command: command_text.clone(),
                program: program.display().to_string(),
                kind: error.kind(),
                message: error.to_string(),
            })?;
            clock.spawned(child.id());
            let (Some(stdout), Some(stderr)) = (child.stdout.take(), child.stderr.take()) else {
                let failure = ExecError::Io {
                    command: command_text.clone(),
                    message: "a standard stream was not captured".to_string(),
                };
                return Err(abandon(&mut child, failure, command_text));
            };
            let bound = limits.max_output();
            let out = Capture::start(stdout, bound);
            let err = Capture::start(stderr, bound);
            let outcome = wait(&mut child, &out, &err, limits.timeout(), &mut clock);
            let failure = match outcome {
                Wait::Done(code, stdout, stderr) => {
                    return Ok(Output {
                        code,
                        stdout,
                        stderr,
                        elapsed: clock.elapsed(),
                    });
                }
                Wait::Over(stream) => ExecError::OutputTooLarge {
                    command: command_text.clone(),
                    stream,
                    limit: bound,
                    elapsed: clock.elapsed(),
                },
                Wait::Late => ExecError::Timeout {
                    command: command_text.clone(),
                    timeout: limits.timeout(),
                    elapsed: clock.elapsed(),
                },
                Wait::Failed(error) => ExecError::Io {
                    command: command_text.clone(),
                    message: error.to_string(),
                },
            };
            Err(abandon(&mut child, failure, command_text))
        }
    }
    
    /// Kill the process that `failure` gave up on. When it cannot be killed the
    /// error says so, since the process may still be running.
    fn abandon(child: &mut Child, failure: ExecError, command: String) -> ExecError {
        match terminate(child) {
            Ok(()) => failure,
            Err(error) => ExecError::Io {
                command,
                message: format!(
                    "{failure}; and the process could not be killed, so it may still be running: {error}"
                ),
            },
        }
    }
    
    /// A thread that reads one stream up to a bound.
    struct Capture {
        overflow: Arc<AtomicBool>,
        result: Receiver<io::Result<Vec<u8>>>,
    }
    
    impl Capture {
        /// Read `pipe` to its end on a thread of its own, keeping at most `bound`
        /// bytes. Reading stops at the first byte past the bound and the overflow
        /// flag is set before the result is sent.
        fn start(pipe: impl Read + Send + 'static, bound: usize) -> Self {
            let overflow = Arc::new(AtomicBool::new(false));
            let flag = Arc::clone(&overflow);
            let (send, result) = mpsc::channel();
            let limit = u64::try_from(bound).map_or(u64::MAX, |bound| bound.saturating_add(1));
            thread::spawn(move || {
                let mut kept = Vec::new();
                let read = pipe.take(limit).read_to_end(&mut kept);
                let answer = read.map(|length| {
                    if length > bound {
                        flag.store(true, Ordering::SeqCst);
                    }
                    kept
                });
                // The receiver is gone when the run was abandoned.
                let _ = send.send(answer);
            });
            Self { overflow, result }
        }
    
        fn overflowed(&self) -> bool {
            self.overflow.load(Ordering::SeqCst)
        }
    }
    
    /// How the wait for a process ended.
    enum Wait {
        /// The process exited and both streams reached their end.
        Done(Option<i32>, Vec<u8>, Vec<u8>),
        /// A stream passed its bound.
        Over(Stream),
        /// The wall-clock limit passed.
        Late,
        /// Waiting or reading failed.
        Failed(io::Error),
    }
    
    /// What one look at a run decides, from what the look found.
    #[derive(Debug, Clone, Copy, PartialEq, Eq)]
    enum Look {
        /// A stream passed its bound.
        Over(Stream),
        /// The process exited and both streams reached their end.
        Done,
        /// The wall-clock limit passed.
        Late,
        /// Nothing is decided; look again after a pause.
        Again,
    }
    
    /// The stream that passed its bound, the standard output first.
    fn overflow(overflow_out: bool, overflow_err: bool) -> Option<Stream> {
        if overflow_out {
            Some(Stream::Stdout)
        } else if overflow_err {
            Some(Stream::Stderr)
        } else {
            None
        }
    }
    
    /// Decide a look. A passed bound beats a finished run, and a finished run
    /// beats the limit; the limit is passed once `elapsed` reaches `timeout`.
    fn verdict(
        overflow_out: bool,
        overflow_err: bool,
        done: bool,
        elapsed: Duration,
        timeout: Duration,
    ) -> Look {
        if let Some(stream) = overflow(overflow_out, overflow_err) {
            Look::Over(stream)
        } else if done {
            Look::Done
        } else if elapsed >= timeout {
            Look::Late
        } else {
            Look::Again
        }
    }
    
    /// The pause after this one: twice as long, up to [`POLL_CEILING`].
    fn next_pause(pause: Duration) -> Duration {
        (pause * 2).min(POLL_CEILING)
    }
    
    /// Poll the process and the two captures until all three are finished, a bound
    /// is passed, the limit passes, or something fails.
    fn wait<C: Clock>(
        child: &mut Child,
        out: &Capture,
        err: &Capture,
        timeout: Duration,
        clock: &mut C,
    ) -> Wait {
        let mut status: Option<ExitStatus> = None;
        let mut stdout: Option<Vec<u8>> = None;
        let mut stderr: Option<Vec<u8>> = None;
        let mut pause = Duration::from_millis(1);
        loop {
            if let Some(stream) = overflow(out.overflowed(), err.overflowed()) {
                return Wait::Over(stream);
            }
            if status.is_none() {
                match child.try_wait() {
                    Ok(found) => status = found,
                    Err(error) => return Wait::Failed(error),
                }
            }
            for (capture, slot) in [(out, &mut stdout), (err, &mut stderr)] {
                if slot.is_none() {
                    match capture.result.try_recv() {
                        Ok(Ok(bytes)) => *slot = Some(bytes),
                        Ok(Err(error)) => return Wait::Failed(error),
                        Err(TryRecvError::Empty) => {}
                        Err(TryRecvError::Disconnected) => {
                            return Wait::Failed(io::Error::other("an output reader stopped"));
                        }
                    }
                }
            }
            // A stream that overflowed has just been reported as a result too.
            let done = status.is_some() && stdout.is_some() && stderr.is_some();
            let elapsed = clock.elapsed();
            match verdict(out.overflowed(), err.overflowed(), done, elapsed, timeout) {
                Look::Over(stream) => return Wait::Over(stream),
                Look::Done => {
                    return Wait::Done(
                        status.and_then(|status| status.code()),
                        stdout.unwrap_or_default(),
                        stderr.unwrap_or_default(),
                    );
                }
                Look::Late => return Wait::Late,
                Look::Again => {}
            }
            clock.pause(pause.min(timeout.saturating_sub(elapsed)));
            pause = next_pause(pause);
        }
    }
    
    /// Kill the process, and on Unix the group it leads, and collect it.
    ///
    /// A failure to signal is reported: the process may then still be running.
    fn terminate(child: &mut Child) -> io::Result<()> {
        #[cfg(unix)]
        let group = kill_group(child);
        #[cfg(not(unix))]
        let group = Ok(());
        let own = terminate_process(child);
        group.and(own)
    }
    
    /// Kill and reap only the direct child, without signalling its descendants.
    fn terminate_process(child: &mut Child) -> io::Result<()> {
        // `Child::kill` succeeds on a process that has already exited. Still wait
        // after a signal failure, and report a wait failure when signalling worked.
        let killed = child.kill();
        let waited = child.wait().map(|_| ());
        killed.and(waited)
    }
    
    /// Send SIGKILL to the process group the child leads. The child is started
    /// with `process_group(0)` (see `run`), so its id is the group id. The group
    /// holds what `cargo` started, such as a test binary that keeps the output
    /// pipes open. A group with no member left is not an error.
    #[cfg(unix)]
    fn kill_group(child: &mut Child) -> io::Result<()> {
        use rustix::process::{Pid, Signal, kill_process_group};
    
        let raw = i32::try_from(child.id()).map_err(io::Error::other)?;
        let pid = Pid::from_raw(raw).ok_or_else(|| io::Error::other("the process id is not valid"))?;
        let signalled = kill_process_group(pid, Signal::KILL);
        // macOS answers EPERM, not ESRCH, when the group holds only processes that
        // are exiting or have exited but are not reaped yet. The leader having
        // ended is what tells that case from a group that cannot be signalled, and
        // it is looked at only then. Elsewhere EPERM means a live member that may
        // not be signalled, and is always reported.
        let leader_ended = cfg!(target_os = "macos")
            && signalled == Err(rustix::io::Errno::PERM)
            && leader_ended(child);
        group_signal_outcome(signalled, leader_ended)
    }
    
    /// Whether the leader has ended, collecting it when it has.
    ///
    /// A leader that is still exiting is not yet seen by `try_wait` on macOS,
    /// although the group signal already passed it over. The leader is the child
    /// itself, so it is sent SIGKILL directly: when that is delivered, or finds the
    /// leader already on its way out, the leader ends and the wait that follows
    /// returns once it has (`terminate` waits for it in any case). A leader that
    /// refuses its own signal has not ended and is not waited for.
    #[cfg(unix)]
    fn leader_ended(child: &mut Child) -> bool {
        if matches!(child.try_wait(), Ok(Some(_))) {
            return true;
        }
        leader_signal_ends_it(&child.kill()) && child.wait().is_ok()
    }
    
    /// Whether a SIGKILL sent to the leader itself means it ends: it was delivered,
    /// or the leader has no live process left (`ESRCH`). Any other failure, such as
    /// `EPERM`, means the leader may go on running.
    #[cfg(unix)]
    fn leader_signal_ends_it(signalled: &io::Result<()>) -> bool {
        match signalled {
            Ok(()) => true,
            Err(error) => rustix::io::Errno::from_io_error(error) == Some(rustix::io::Errno::SRCH),
        }
    }
    
    /// What a failed or successful group signal means. A group with no live member
    /// (`ESRCH`) is done; so is one that answers `EPERM` while its leader has
    /// ended, because a group signal fails only when it reached no member, and the
    /// leader could be signalled, so it was already exiting (macOS only: `kill_group` passes `leader_ended` as
    /// false elsewhere). Any other
    /// failure means the group may still be running and is reported.
    #[cfg(unix)]
    fn group_signal_outcome(
        signalled: Result<(), rustix::io::Errno>,
        leader_ended: bool,
    ) -> io::Result<()> {
        use rustix::io::Errno;
    
        match signalled {
            Ok(()) | Err(Errno::SRCH) => Ok(()),
            Err(Errno::PERM) if leader_ended => Ok(()),
            Err(errno) => Err(errno.into()),
        }
    }

    //! Tests of the system runner.
    //!
    //! The runs of real processes go through [`SystemRunner::run_with`] with a
    //! clock the test controls, so no test here asserts on, or waits for, the
    //! clock of the machine; only the list pass in [`cargo`] runs under it, as a
    //! watchdog whose reading nothing asserts. Under [`Frozen`] every look reads
    //! the same time, so only the end of the process or an output bound can end
    //! the run, and the time a run reports is that reading exactly. Under
    //! [`Stepped`] each pause the runner asks for moves the reading by that much,
    //! so the limit passes after a bounded number of looks whatever the speed of
    //! the host. Both count every reading and every pause and fail the test past
    //! a ceiling, so a runner that never decides, or that stops pausing, fails
    //! instead of spinning.
    //!
    //! A process that has to outlive its limit first ignores SIGHUP, SIGINT and
    //! SIGTERM, writes its process id to a FIFO (the pids) and blocks reading
    //! another that the test holds open (the hold); a leader that exits and leaves
    //! it behind first waits on a third (go) until its pidfd is taken. On Linux the
    //! stepped clock reads the pids without blocking and does not move until it
    //! has a pidfd of each process, so a script that ends early ends the run
    //! instead of stopping the test. Once the run is over the test releases the
    //! hold, sends SIGCONT to each process and blocks on each pidfd until the
    //! process has ended. A process killed with SIGKILL never ran the rest of its
    //! script, so the marker file that the script writes after the hold is
    //! absent; one that survived the kill, or was only stopped, writes the marker
    //! and ends, so no step can block for ever.
    //!
    //! [`cargo`] runs a bound test and a bound example that never end through the
    //! executor and the real `cargo` in the same way.

        let mut output = io::BufReader::new(leader.stdout.take().expect("leader output"));

            let mut ready = false;
            for _ in 0..16 {

                assert_ne!(output.read_line(&mut line).expect("read readiness"), 0);
                if line.trim_end() == READY {

        drop(release);

        if leader.try_wait().expect("inspect owned leader").is_none() {

        }

    // ------------------------------------------------- the clocks of the tests
    
    /// A borrowed clock is the clock of a run, so the test can read what the
    /// clock saw once the run is over.
    #[cfg(unix)]
    impl<C: Clock + ?Sized> Clock for &mut C {
        fn elapsed(&self) -> Duration {
            (**self).elapsed()
        }
    
        fn pause(&mut self, pause: Duration) {
            (**self).pause(pause);
        }
    
        fn spawned(&mut self, process: u32) {
            (**self).spawned(process);
        }
    }
    
    /// The most looks, and the most pauses, a run under [`Frozen`] may make, and a
    /// run under [`Stepped`] before its gate opens. Each pause there only yields
    /// the thread, so a short process ends within a small part of this; a runner
    /// that never decides reaches it and fails the test instead of spinning.
    #[cfg(unix)]
    const FROZEN_CEILING: usize = 5_000_000;
    
    /// The most looks, and the most pauses, a run under [`Stepped`] may make once
    /// its clock moves: twice the number of pauses of [`POLL_CEILING`] in `limit`.
    /// The limit passes within that many pauses and five more (at most five of
    /// the doubling pauses, then 20 ms each, then what remains), and the runner
    /// reads the clock once at each look and once for the result; for the limit
    /// of 2007 ms that is 105 pauses against a ceiling of 200.
    #[cfg(unix)]
    fn stepped_ceiling(limit: Duration) -> usize {
        let pauses = limit.as_millis() / POLL_CEILING.as_millis();
        usize::try_from(pauses).expect("a limit of the tests") * 2
    }
    
    /// The count of one kind of call to a test clock. Past its ceiling it fails
    /// the test, so a runner that never decides fails whether it pauses or not.
    #[cfg(unix)]
    struct Count {
        what: &'static str,
        done: std::cell::Cell<usize>,
        ceiling: usize,
    }
    
    #[cfg(unix)]
    impl Count {
        fn new(what: &'static str, ceiling: usize) -> Self {
            Self {
                what,
                done: std::cell::Cell::new(0),
                ceiling,
            }
        }
    
        fn tick(&self) {
            let done = self.done.get() + 1;
            self.done.set(done);
            assert!(
                done <= self.ceiling,
                "the run made {done} {}, past the ceiling of {}",
                self.what,
                self.ceiling
            );
        }
    }
    
    /// A clock whose reading never moves: each reading and each pause is counted,
    /// and a pause yields the thread.
    #[cfg(unix)]
    struct Frozen {
        elapsed: Duration,
        looks: Count,
        pauses: Count,
    }
    
    #[cfg(unix)]
    impl Frozen {
        /// A clock that reads `elapsed` at every look.
        fn at(elapsed: Duration) -> Self {
            Self {
                elapsed,
                looks: Count::new("looks under a clock that never moves", FROZEN_CEILING),
                pauses: Count::new("pauses under a clock that never moves", FROZEN_CEILING),
            }
        }
    }
    
    #[cfg(unix)]
    impl Clock for Frozen {
        fn elapsed(&self) -> Duration {
            self.looks.tick();
            self.elapsed
        }
    
        fn pause(&mut self, _pause: Duration) {
            self.pauses.tick();
            std::thread::yield_now();
        }
    }
    
    /// A clock that moves by each pause the runner asks for, and nothing else;
    /// each reading and each pause is counted, against the ceiling of its limit
    /// (see [`stepped_ceiling`]). On Linux it has a gate (an [`Opening`]): until
    /// the gate opens, a pause does not move the clock, so the processes of a
    /// [`Subject`] are known before the limit can pass.
    #[cfg(unix)]
    struct Stepped {
        now: Duration,
        looks: Count,
        pauses: Count,
        /// The ceiling once the gate has opened.
        #[cfg(target_os = "linux")]
        ceiling: usize,
        #[cfg(target_os = "linux")]
        gate: Option<Box<dyn Opening>>,
        #[cfg(target_os = "linux")]
        seen: Option<Seen>,
    }
    
    #[cfg(unix)]
    impl Stepped {
        fn moving(ceiling: usize) -> (Count, Count) {
            (
                Count::new("looks under a stepped clock", ceiling),
                Count::new("pauses under a stepped clock", ceiling),
            )
        }
    
        /// A clock that moves from its first pause, for a run of `limit`.
        #[cfg(not(target_os = "linux"))]
        fn new(limit: Duration) -> Self {
            let (looks, pauses) = Self::moving(stepped_ceiling(limit));
            Self {
                now: Duration::ZERO,
                looks,
                pauses,
            }
        }
    
        /// A clock for a run of `limit` that moves once `gate` has opened.
        #[cfg(target_os = "linux")]
        fn gated(gate: impl Opening + 'static, limit: Duration) -> Self {
            Self {
                now: Duration::ZERO,
                looks: Count::new("looks before the gate opened", FROZEN_CEILING),
                pauses: Count::new("pauses before the gate opened", FROZEN_CEILING),
                ceiling: stepped_ceiling(limit),
                gate: Some(Box::new(gate)),
                seen: None,
            }
        }
    }
    
    #[cfg(unix)]
    impl Clock for Stepped {
        fn elapsed(&self) -> Duration {
            self.looks.tick();
            self.now
        }
    
        fn pause(&mut self, pause: Duration) {
            #[cfg(target_os = "linux")]
            if let Some(gate) = &mut self.gate {
                if let Some(seen) = gate.turn() {
                    self.seen = Some(seen);
                    self.gate = None;
                    (self.looks, self.pauses) = Self::moving(self.ceiling);
                } else {
                    self.pauses.tick();
                    std::thread::yield_now();
                    return;
                }
            }
            self.pauses.tick();
            self.now += pause;
        }
    
        fn spawned(&mut self, process: u32) {
            #[cfg(target_os = "linux")]
            if let Some(gate) = &mut self.gate {
                gate.spawned(process);
            }
            #[cfg(not(target_os = "linux"))]
            let _ = process;
        }
    }

    /// A command of `/bin/sh -c <script>`, as the caller's own command runs:
    /// no program is looked up on `PATH` and no file is written to be executed.
    #[cfg(unix)]
    fn sh(script: &str) -> Invocation {
        let words = ["/bin/sh", "-c", script].map(str::to_string).to_vec();
        Invocation::trusted(&TrustedArgv::new(words).expect("a command"))
    }
    
    /// A runner in the directory of temporary files.
    #[cfg(unix)]
    fn runner() -> SystemRunner {
        SystemRunner::new(Programs::new(std::env::temp_dir()))
    }
    
    /// A native exclusively created temporary directory, removed by its owner.
    #[cfg(unix)]
    struct Scratch(purrdf_testkit::TempDir);
    
    #[cfg(unix)]
    impl Scratch {
        fn new() -> Self {
            Self(purrdf_testkit::TempDir::new_in(std::env::temp_dir()).expect("create process fixture"))
        }
    }
    
    /// What a process that outlives its limit leaves behind it.
    #[cfg(unix)]
    #[derive(Debug, Clone, Copy, PartialEq, Eq)]
    enum Shape {
        /// The process itself blocks.
        Alone,
        /// The process starts one that blocks, holding the output pipes, and waits
        /// for it.
        Waits,
        /// The process starts one that blocks, holding the output pipes, and exits.
        Leaves,
    }
    
    /// The limit of the runs that outlive it: two seconds and seven milliseconds.
    /// The pauses before the gate opens do not move a stepped clock, so the
    /// moving pauses may start anywhere in the runner's schedule and add up to
    /// 31, 30, 28, 24, 16 or 0 ms and then 20 ms each; none of those sums is this
    /// limit, so a pause that is not cut to what remains passes it.
    #[cfg(unix)]
    const LIMIT: Duration = Duration::from_millis(2007);
    
    /// A run of a process of `shape` that outlives its limit, and on Linux what
    /// became of its processes.
    #[cfg(unix)]
    struct Outlived {
        command: String,
        result: Result<crate::exec::Output, ExecError>,
        #[cfg(target_os = "linux")]
        seen: Seen,
        #[cfg(target_os = "linux")]
        settled: Settled,
    }
    
    /// Run a process of `shape` under a stepped clock with the limit [`LIMIT`].
    #[cfg(target_os = "linux")]
    fn outlive(shape: Shape) -> Outlived {
        let subject = Subject::new();
        let invocation = sh(&subject.script(shape));
        let mut clock = Stepped::gated(subject.gate(shape), LIMIT);
        let limits = Limits::trusted().with_timeout(LIMIT);
        let result = runner().run_with(&invocation, &limits, || &mut clock);
        let Some(seen) = clock.seen.take() else {
            panic!("the run ended before the gate opened on the processes of its script: {result:?}");
        };
        let settled = subject.settle(&seen);
        Outlived {
            command: invocation.to_string(),
            result,
            seen,
            settled,
        }
    }
    
    /// Run a process of `shape` under a stepped clock with the limit [`LIMIT`].
    /// Without the FIFOs of Linux the process never ends by running a loop of the
    /// shell, and only the error is asserted.
    #[cfg(all(unix, not(target_os = "linux")))]
    fn outlive(shape: Shape) -> Outlived {
        let script = match shape {
            Shape::Alone => "while :; do :; done",
            Shape::Waits => "{ while :; do :; done; } & wait",
            Shape::Leaves => "{ while :; do :; done; } & exit 0",
        };
        let invocation = sh(script);
        let mut clock = Stepped::new(LIMIT);
        let limits = Limits::trusted().with_timeout(LIMIT);
        let result = runner().run_with(&invocation, &limits, || &mut clock);
        Outlived {
            command: invocation.to_string(),
            result,
        }
    }
    
    /// The error of a run killed at its limit, with the limit as the time it ran.
    #[cfg(unix)]
    fn killed_at_the_limit(run: &Outlived) {
        assert_eq!(
            run.result,
            Err(ExecError::Timeout {
                command: run.command.clone(),
                timeout: LIMIT,
                elapsed: LIMIT,
            })
        );
    }
    
    // --------------------------------------- the processes of a run, on Linux
    
    /// The FIFOs and the marker of one process that outlives its limit.
    #[cfg(target_os = "linux")]
    struct Subject {
        scratch: Scratch,
        /// The pid FIFO, open for reading and writing since before the spawn, so
        /// the script's write never blocks, and without blocking, so the gate
        /// never waits on it.
        pids: std::fs::File,
        /// The hold FIFO, open for reading and writing, so the release is a plain
        /// write that lands whether or not the script has opened it yet.
        hold: std::fs::File,
        /// The FIFO a leader that leaves its descendant behind reads before it
        /// exits, open the same way: the leader stays unreaped until its pidfd is
        /// taken.
        go: std::fs::File,
    }
    
    #[cfg(target_os = "linux")]
    impl Subject {
        fn new() -> Self {
            use std::os::unix::fs::OpenOptionsExt as _;
    
            let scratch = Scratch::new();
            let fifo = |name: &str, flags: rustix::fs::OFlags| {
                let path = scratch.0.path().join(name);
                rustix::fs::mknodat(
                    rustix::fs::CWD,
                    &path,
                    rustix::fs::FileType::Fifo,
                    rustix::fs::Mode::from_raw_mode(0o600),
                    0,
                )
                .unwrap_or_else(|error| panic!("{}: {error}", path.display()));
                let flags = i32::try_from(flags.bits()).expect("open flags");
                std::fs::OpenOptions::new()
                    .read(true)
                    .write(true)
                    .custom_flags(flags)
                    .open(&path)
                    .unwrap_or_else(|error| panic!("{}: {error}", path.display()))
            };
            let pids = fifo("pids", rustix::fs::OFlags::NONBLOCK);
            let hold = fifo("hold", rustix::fs::OFlags::empty());
            let go = fifo("go", rustix::fs::OFlags::empty());
            Self {
                scratch,
                pids,
                hold,
                go,
            }
        }
    
        /// The absolute path of `name` in the scratch directory, quoted for the
        /// shell.
        fn quoted(&self, name: &str) -> String {
            let path = self.scratch.0.path().join(name);
            let text = path.to_str().expect("a UTF-8 temporary directory");
            assert!(!text.contains('\''), "{text}");
            format!("'{text}'")
        }
    
        fn marker(&self) -> std::path::PathBuf {
            self.scratch.0.path().join("marker")
        }
    
        /// The script: it first ignores SIGHUP, SIGINT and SIGTERM, before any
        /// fork, so that only a signal that cannot be ignored ends it or what it
        /// starts; the process ids go to the pid FIFO, the process that blocks
        /// reads the hold FIFO, and after the hold it writes the marker. A leader
        /// that exits reads the go FIFO first. Every word is a builtin of the
        /// shell, so it runs with an empty `PATH`.
        fn script(&self, shape: Shape) -> String {
            let (pids, hold, go, marker) = (
                self.quoted("pids"),
                self.quoted("hold"),
                self.quoted("go"),
                self.quoted("marker"),
            );
            let blocks = format!("read x < {hold}; echo survived > {marker}");
            let body = match shape {
                Shape::Alone => format!("echo $$ > {pids}; {blocks}; exit 0"),
                Shape::Waits => format!("{{ {blocks}; }} & echo \"$$ $!\" > {pids}; wait"),
                Shape::Leaves => {
                    format!("{{ {blocks}; }} & echo \"$$ $!\" > {pids}; read y < {go}; exit 0")
                }
            };
            format!("trap '' HUP INT TERM; {body}")
        }
    
        /// The gate of a stepped clock for a process of `shape`.
        fn gate(&self, shape: Shape) -> Gate {
            let clone = |file: &std::fs::File| file.try_clone().expect("a FIFO is shared");
            Gate {
                pids: clone(&self.pids),
                line: Vec::new(),
                go: (shape == Shape::Leaves).then(|| clone(&self.go)),
                found: None,
            }
        }
    
        /// What became of the processes once the run is over: whether the leader
        /// is reaped as the run returns; then, with the hold released, each
        /// process sent SIGCONT (so one that a stop left behind runs on) and each
        /// process ended, whether the marker was written. A process the kill
        /// reached never read the release, so the marker is there only if one
        /// survived the kill.
        fn settle(&self, seen: &Seen) -> Settled {
            use rustix::process::{Signal, pidfd_send_signal};
            use std::io::Write as _;
            use std::os::fd::AsFd as _;
    
            let reaped = waitid(
                WaitId::PidFd(seen.leader.fd.as_fd()),
                WaitIdOptions::EXITED | WaitIdOptions::NOHANG,
            );
            (&self.hold).write_all(b"\n").expect("the hold is released");
            for process in seen.processes() {
                // A process that has ended is not an error here.
                let _ = pidfd_send_signal(&process.fd, Signal::CONT);
            }
            for process in seen.processes() {
                process.wait_for_end();
            }
            Settled {
                leader_reaped: matches!(reaped, Err(Errno::CHILD)),
                marker: self.marker().exists(),
            }
        }
    }
    
    /// What a [`Stepped`] clock waits for before it moves.
    #[cfg(target_os = "linux")]
    trait Opening {
        /// The runner has started the process of the run with this id.
        fn spawned(&mut self, _process: u32) {}
    
        /// One turn, at a pause of the runner: the processes once the limit may
        /// pass, or `None` to look again at the next pause.
        fn turn(&mut self) -> Option<Seen>;
    }
    
    /// The gate of a [`Subject`] script: the process ids the script writes, a
    /// pidfd of each, and for a leader that exits, its end. Each turn looks
    /// without blocking.
    #[cfg(target_os = "linux")]
    struct Gate {
        /// The pid FIFO, read without blocking.
        pids: std::fs::File,
        /// What has been read of the line of process ids.
        line: Vec<u8>,
        /// The go FIFO, for a leader that exits.
        go: Option<std::fs::File>,
        /// The processes, once their pidfds are taken.
        found: Option<Seen>,
    }
    
    #[cfg(target_os = "linux")]
    impl Opening for Gate {
        fn turn(&mut self) -> Option<Seen> {
            use std::io::{Read as _, Write as _};
    
            if self.found.is_none() {
                let mut chunk = [0_u8; 64];
                match (&self.pids).read(&mut chunk) {
                    // The test holds a writer, so a read never finds the end.
                    Ok(0) => panic!("the pid FIFO ended"),
                    Ok(read) => self.line.extend_from_slice(&chunk[..read]),
                    Err(error)
                        if matches!(
                            error.kind(),
                            io::ErrorKind::WouldBlock | io::ErrorKind::Interrupted
                        ) => {}
                    Err(error) => panic!("the pid FIFO is read: {error}"),
                }
                if !self.line.contains(&b'\n') {
                    return None;
                }
                let line = String::from_utf8(std::mem::take(&mut self.line)).expect("process ids");
                let mut pids = line.split_whitespace().map(Process::open);
                let leader = pids.next().expect("the leader's process id");
                let descendant = pids.next();
                if let Some(go) = &self.go {
                    (&*go).write_all(b"\n").expect("the leader is let go");
                } else {
                    assert!(
                        !leader.ended(),
                        "the process ended before its limit could pass"
                    );
                }
                self.found = Some(Seen { leader, descendant });
            }
            let leader_exits = self.go.is_some();
            let seen = self.found.as_ref()?;
            if leader_exits && !seen.leader.ended() {
                return None;
            }
            self.found.take()
        }
    }
    
    /// The processes of a run as the gate found them. Dropping it sends SIGKILL to
    /// each of them through its pidfd, and on a failing test to the group as
    /// well, so a run that fails leaves nothing behind; after a clean end the
    /// group's id may already belong to another group.
    #[cfg(target_os = "linux")]
    struct Seen {
        leader: Process,
        descendant: Option<Process>,
    }
    
    #[cfg(target_os = "linux")]
    impl Seen {
        fn processes(&self) -> impl Iterator<Item = &Process> {
            std::iter::once(&self.leader).chain(&self.descendant)
        }
    }
    
    #[cfg(target_os = "linux")]
    impl Drop for Seen {
        fn drop(&mut self) {
            use rustix::process::{Signal, kill_process_group, pidfd_send_signal};
    
            for process in self.processes() {
                let _ = pidfd_send_signal(&process.fd, Signal::KILL);
            }
            if std::thread::panicking() {
                let _ = kill_process_group(self.leader.pid, Signal::KILL);
            }
        }
    }
    
    /// One process, held by a pidfd, and its group when the pidfd was taken.
    #[cfg(target_os = "linux")]
    struct Process {
        pid: Pid,
        fd: std::os::fd::OwnedFd,
        pgrp: i32,
    }
    
    #[cfg(target_os = "linux")]
    impl Process {
        fn open(word: &str) -> Self {
            use rustix::process::{PidfdFlags, pidfd_open};
    
            let raw = purrdf_xsd::numeric::parse_integer(word)
                .ok()
                .and_then(|number| number.as_i128())
                .and_then(|number| i32::try_from(number).ok())
                .expect("a process id in the signed 32-bit range");
            let pid = Pid::from_raw(raw).expect("a process id");
            // The process is blocked reading a FIFO the test holds and is not
            // reaped, so the id is still this process.
            let fd = pidfd_open(pid, PidfdFlags::empty())
                .unwrap_or_else(|error| panic!("a pidfd of process {raw}, still running: {error}"));
            let pgrp = group_of(pid).expect("the stat of a live process");
            Self { pid, fd, pgrp }
        }
    
        /// Whether the process has ended, without blocking.
        fn ended(&self) -> bool {
            use rustix::event::{PollFd, PollFlags, Timespec, poll};
    
            let no_wait = Timespec {
                tv_sec: 0,
                tv_nsec: 0,
            };
            let mut fds = [PollFd::new(&self.fd, PollFlags::IN)];
            loop {
                match poll(&mut fds, Some(&no_wait)) {
                    Ok(_) => return fds[0].revents().contains(PollFlags::IN),
                    Err(Errno::INTR) => {}
                    Err(error) => panic!("poll of the pidfd of {}: {error}", self.pid.as_raw_pid()),
                }
            }
        }
    
        /// Block until the process has ended: an event of the kernel, not a time.
        /// It is asked for only after the hold is released and SIGCONT sent, when
        /// a process that survived the kill has nothing left to wait on.
        fn wait_for_end(&self) {
            use rustix::event::{PollFd, PollFlags, poll};
    
            let mut fds = [PollFd::new(&self.fd, PollFlags::IN)];
            loop {
                match poll(&mut fds, None) {
                    Ok(_) if fds[0].revents().contains(PollFlags::IN) => return,
                    Ok(_) | Err(Errno::INTR) => {}
                    Err(error) => panic!("poll of the pidfd of {}: {error}", self.pid.as_raw_pid()),
                }
            }
        }
    }
    
    /// The process group of `pid`, the fifth field of `/proc/<pid>/stat` (the
    /// fields after the command name, which ends at the last `)`, begin with the
    /// third).
    #[cfg(target_os = "linux")]
    fn group_of(pid: Pid) -> Option<i32> {
        let text = std::fs::read_to_string(format!("/proc/{}/stat", pid.as_raw_pid())).ok()?;
        let (_, rest) = text.rsplit_once(')')?;
        purrdf_xsd::numeric::parse_integer(rest.split_whitespace().nth(2)?)
            .ok()?
            .as_i128()
            .and_then(|number| i32::try_from(number).ok())
    }
    
    /// What became of the processes of a run once it was over.
    #[cfg(target_os = "linux")]
    #[derive(Debug)]
    struct Settled {
        leader_reaped: bool,
        marker: bool,

    /// The kill reached the leader and what it started: both are in the group
    /// the run started, the leader is reaped as the run returns, and the marker
    /// the script writes after the hold was never written. The script ignores
    /// SIGHUP, SIGINT and SIGTERM and every process gets SIGCONT after the
    /// release, so only SIGKILL keeps the marker away.
    #[cfg(target_os = "linux")]
    fn nothing_outlived_the_kill(seen: &Seen, settled: &Settled) {
        let leader = seen.leader.pid.as_raw_pid();
        assert_eq!(seen.leader.pgrp, leader, "the leader leads a group");
        if let Some(descendant) = &seen.descendant {
            assert_eq!(descendant.pgrp, leader, "in the leader's group");
        }
        assert!(
            settled.leader_reaped,
            "the leader was not reaped: {settled:?}"
        );
        assert!(
            !settled.marker,
            "a process ran on past the kill and wrote the marker: {settled:?}"
        );
    }

        const ROOT: &str = "RAF_EXEC_ESCAPED_GROUP_ROOT";
        const LEADER: &str = "RAF_EXEC_ESCAPED_GROUP_LEADER";
        const METHOD: &str = "RAF_EXEC_ESCAPED_GROUP_METHOD";
        const NAME: &str =
            "exec::system::tests::a_descendant_that_leaves_its_group_survives_the_group_kill";

            let root = std::path::PathBuf::from(root);

                .expect("the selected escape operation")
                .as_str()
            {
                "setsid" => {
                    rustix::process::setsid().expect("a nonleader starts a session");
                }
                "setpgid" => {
                    rustix::process::setpgid(None, None).expect("a nonleader starts a group");
                }
                other => panic!("unrecognized fixture escape operation: {other:?}"),
            }
            assert_eq!(
                rustix::process::getpgrp().as_raw_pid(),
                i32::try_from(std::process::id()).unwrap()
            );
            let leader = std::env::var(LEADER).expect("the original leader's pid");
            let mut pids = std::fs::OpenOptions::new()
                .write(true)
                .open(root.join("pids"))
                .expect("the parent holds the receipt FIFO");
            writeln!(pids, "{leader} {}", std::process::id()).unwrap();
            drop(pids);
            let hold =
                std::fs::File::open(root.join("hold")).expect("the parent holds the release FIFO");
            let mut released = String::new();
            io::BufReader::new(hold).read_line(&mut released).unwrap();
            assert_eq!(
                released, "\n",
                "the parent explicitly releases the escaped process"
            );
            std::fs::write(root.join("marker"), "survived\n").unwrap();
            return;
        }
    
        for method in ["setsid", "setpgid"] {
            let subject = Subject::new();
            let child = TrustedArgv::new(vec![
                std::env::current_exe()
                    .unwrap()
                    .to_str()
                    .unwrap()
                    .to_string(),
                "--exact".to_string(),
                NAME.to_string(),
                "--nocapture".to_string(),
            ])
            .unwrap()
            .text();
            // The shell starts a nonleader in its group. That child acknowledges its
            // new group before the gate releases the original leader to exit.
            let invocation = sh(&format!(
                "{ROOT}={} {LEADER}=$$ {METHOD}={method} {child} & read release < {}; exit 0",
                subject.quoted(""),
                subject.quoted("go")
            ));
            let mut clock = Stepped::gated(subject.gate(Shape::Leaves), LIMIT);
            let result = runner().run_with(&invocation, &Limits::trusted().with_timeout(LIMIT), || {
                &mut clock
            });
            let seen = clock
                .seen
                .take()
                .expect("both processes acknowledged before the controlled limit");
            assert_eq!(
                result,
                Err(ExecError::Timeout {
                    command: invocation.to_string(),
                    timeout: LIMIT,

                "the group kill cannot reach the new group"
            );
            assert!(
                !subject.marker().exists(),
                "the escaped child is still held"
            );
            let settled = subject.settle(&seen);
            assert!(settled.leader_reaped);
            assert!(
                settled.marker,
                "the escaped process continued after the group kill"
            );
            assert!(escaped.ended(), "the released escaped process has ended");
        }
    }

    //! A bound test and a bound example that never end, run by the executor
    //! through the real `cargo` against a crate generated into a scratch
    //! directory, on Linux.
    //!
    //! The executor runs every process of a binding through one runner, and the
    //! runner here picks the clock of each invocation ([`Clocked`]): the clock of
    //! the machine for the list pass of a test, which ends by itself, and a
    //! [`Stepped`] clock for the one other run, which moves only by the pauses the
    //! runner asks for. A limit of five seconds then passes after a bounded number
    //! of looks whatever the speed of the host, and nothing waits for it.
    //!
    //! The test and the example of the crate call one function of its library: it
    //! ignores SIGHUP, SIGINT and SIGTERM, writes the id of its process to the pid
    //! FIFO of a [`Subject`], blocks reading the hold FIFO and then writes the
    //! marker. The gate of the stepped clock ([`Watch`]) takes a pidfd of the
    //! process the runner starts (`cargo`) as it starts, and blocks until the id
    //! is written or that process has ended, so a `cargo` that ends first ends the
    //! run instead of stopping the test. With the id read it takes a pidfd of the
    //! process that wrote it and lets the clock move. Once the run is over the
    //! hold is released, each process is sent SIGCONT and waited for through its
    //! pidfd: a process that the kill reached never wrote the marker, and only
    //! SIGKILL keeps it away.

    use std::fmt::Write as _;
    use std::path::PathBuf;
    use std::process::{Command, Stdio};
    use std::time::Duration;
    
    use raf_source::{DiskTree, MemoryTree, Workspace};
    use rustix::io::Errno;
    
    use super::{Opening, Process, Seen, Settled, Stepped, Subject, nothing_outlived_the_kill};

    /// The limit of the runs that never end: five seconds of the stepped clock.
    const LIMIT: Duration = Duration::from_secs(5);
    
    /// The packages of the generated crates: one for the test and one for the
    /// example. A target directory named by the environment (`CARGO_TARGET_DIR`)
    /// overrides the one each crate names, and is then shared by every crate; two
    /// names, and the example in only one of them, keep the two tests of this
    /// process from rewriting each other's executables there.
    const TEST_PACKAGE: &str = "fx-forever-test";

    const EXAMPLE_PACKAGE: &str = "fx-forever-example";

    fn cargo() -> PathBuf {

    // ------------------------------------------------------ the runner and gate
    
    /// The system runner with a clock chosen for each invocation. A list pass
    /// runs under the clock of the machine and the limit of the gate
    /// ([`BUILD_TIMEOUT`]): it ends by itself, and what it takes is not what is
    /// tested. Any other run takes the stepped clock, once, under the limits the
    /// executor asked for. The limits asked for each invocation are kept.
    struct Clocked {
        runner: SystemRunner,
        stepped: RefCell<Option<Stepped>>,
        ran: RefCell<Option<Stepped>>,
        asked: RefCell<Vec<(Vec<String>, Limits)>>,
    }
    
    impl Clocked {
        fn new(runner: SystemRunner, stepped: Stepped) -> Self {
            Self {
                runner,
                stepped: RefCell::new(Some(stepped)),
                ran: RefCell::new(None),
                asked: RefCell::new(Vec::new()),
            }
        }
    }
    
    impl Runner for Clocked {
        fn run(&self, invocation: &Invocation, limits: &Limits) -> Result<Output, ExecError> {
            let argv = invocation.argv();
            let list = argv.iter().any(|word| word == "--list");
            self.asked.borrow_mut().push((argv, *limits));
            if list {
                let gate = limits.with_timeout(BUILD_TIMEOUT);
                return self.runner.run_with(invocation, &gate, WallClock::start);
            }
            let mut clock = self
                .stepped
                .borrow_mut()
                .take()
                .expect("one run under the stepped clock");
            let result = self.runner.run_with(invocation, limits, || &mut clock);
            *self.ran.borrow_mut() = Some(clock);
            result
        }
    }
    
    /// The gate of a run through `cargo`: a pidfd of the process the runner
    /// started, taken as it starts, then the process id the test or the example
    /// writes to the pid FIFO. A turn blocks until the id is written or the
    /// started process has ended, events of the kernel and not a time.
    struct Watch {
        /// The pid FIFO, open for reading and writing since before the build and
        /// read without blocking once the poll has found it readable.
        pids: std::fs::File,
        /// What has been read of the line.
        line: Vec<u8>,
        /// The process the runner started, until the gate opens.
        leader: Option<Process>,
        /// Whether the process the runner started ended before the id came.
        leader_ended: bool,
    }
    
    impl Watch {
        fn new(subject: &Subject) -> Self {
            Self {
                pids: subject.pids.try_clone().expect("a FIFO is shared"),
                line: Vec::new(),
                leader: None,
                leader_ended: false,
            }
        }
    }
    
    impl Opening for Watch {
        fn spawned(&mut self, process: u32) {
            assert!(
                self.leader.is_none(),
                "one process is started under the stepped clock"
            );
            // The runner has not collected it yet, so the id is still this process.
            self.leader = Some(Process::open(&process.to_string()));
        }
    
        fn turn(&mut self) -> Option<Seen> {
            use rustix::event::{PollFd, PollFlags, poll};
            use std::io::Read as _;
    
            if self.leader_ended {
                return None;
            }
            let leader = self
                .leader
                .as_ref()
                .expect("the runner names the process it started");
            let (written, ended) = {
                let mut fds = [
                    PollFd::new(&self.pids, PollFlags::IN),
                    PollFd::new(&leader.fd, PollFlags::IN),
                ];
                match poll(&mut fds, None) {
                    Ok(_) => {}
                    Err(Errno::INTR) => return None,
                    Err(error) => panic!("poll of the pid FIFO and of `cargo`: {error}"),
                }
                (
                    fds[0].revents().contains(PollFlags::IN),
                    fds[1].revents().contains(PollFlags::IN),
                )
            };
            if written {
                let mut chunk = [0_u8; 64];
                match (&self.pids).read(&mut chunk) {
                    // The test holds a writer, so a read never finds the end.
                    Ok(0) => panic!("the pid FIFO ended"),
                    Ok(read) => self.line.extend_from_slice(&chunk[..read]),
                    Err(error)
                        if matches!(
                            error.kind(),
                            std::io::ErrorKind::WouldBlock | std::io::ErrorKind::Interrupted
                        ) => {}
                    Err(error) => panic!("the pid FIFO is read: {error}"),
                }
            }
            if !self.line.contains(&b'\n') {
                self.leader_ended = ended;
                return None;
            }
            let line = String::from_utf8(std::mem::take(&mut self.line)).expect("a process id");
            let word = line.trim();
            let leader = self.leader.take()?;
            // `cargo run` on Unix replaces itself with the example, which then
            // writes the id of the process the runner started.
            let descendant = (word != leader.pid.as_raw_pid().to_string()).then(|| Process::open(word));
            assert!(!leader.ended(), "`cargo` ended before the limit could pass");
            Some(Seen { leader, descendant })
        }
    }
    
    impl Drop for Watch {
        /// A run that fails before the gate opens leaves nothing behind.
        fn drop(&mut self) {
            use rustix::process::{Signal, kill_process_group, pidfd_send_signal};
    
            if let Some(leader) = &self.leader {
                let _ = pidfd_send_signal(&leader.fd, Signal::KILL);
                if std::thread::panicking() {
                    let _ = kill_process_group(leader.pid, Signal::KILL);
                }
            }
        }
    }

        /// Build what a binding needs before its run, so that the run starts the
        /// built test binary or example. It takes real time and nothing is
        /// asserted about it.
        fn warm(&self, args: &[&str]) {
            let output = Command::new(cargo())
                .args(args)
                .arg("--frozen")
                .current_dir(&self.dir)
                .stdin(Stdio::null())
                .output()
                .expect("cargo starts");
            assert!(
                output.status.success(),
                "{}",
                String::from_utf8_lossy(&output.stderr)
            );
        }
    
        /// A runner for the real `cargo`.
        fn runner(&self) -> SystemRunner {
            SystemRunner::new(Programs::with_cargo_var(&self.dir, Some(cargo().into())))
        }
    
        /// Run the binding `entry` of `list` through `runner`.
        fn judge<R: Runner>(
            &self,
            runner: R,
            list: &str,
            entry: &str,
            limits: Option<(Limits, Limits)>,
        ) -> Result<Evidence, ExecError> {
            let doc = capability(list, entry);
            let docs = MemoryTree::from([("docs/capabilities/fixture.md", doc.as_str())]);
            let plan = Workplan::load(&docs, &self.source);
            let binding = plan
                .graph()
                .bindings()
                .first()
                .expect("the capability lists a binding");
            assert!(
                binding.resolution().is_ok(),
                "{entry} must resolve: {:?}",
                binding.resolution()
            );
            let mut executor = Executor::new(&self.source, runner).expect("the index lists its tests");
            if let Some((build, example)) = limits {
                executor = executor.with_limits(build, example);
            }
            executor.run(binding)
        }
    }
    
    /// The library of the crate. The paths of the FIFOs and of the marker are
    /// written into it as Rust string literals.
    fn library(subject: &Subject) -> String {
        let literal = |name: &str| {
            let path = subject.scratch.0.path().join(name);
            format!("{:?}", path.to_str().expect("a UTF-8 temporary directory"))
        };
        let (pids, hold, marker) = (literal("pids"), literal("hold"), literal("marker"));
        format!(
            r#"//! A crate whose test and example never end by themselves.
    
    use std::io::{{Read as _, Write as _}};
    
    extern "C" {{
        fn signal(signum: i32, handler: usize) -> usize;
    }}
    
    /// Ignore SIGHUP, SIGINT and SIGTERM, write the id of this process to the
    /// pid FIFO, block reading the hold FIFO, then write the marker.
    pub fn outlive() {{
        for signum in [1, 2, 15] {{
            // SIG_IGN is 1.
            unsafe {{
                signal(signum, 1);
            }}
        }}
        // The test holds the FIFO open for reading, so the open does not block.
        let mut pids = std::fs::OpenOptions::new()
            .write(true)
            .open({pids})
            .expect("the pid FIFO opens");
        writeln!(pids, "{{}}", std::process::id()).expect("the process id is written");
        // The test holds the FIFO open for writing, so only the read blocks.
        let mut hold = std::fs::File::open({hold}).expect("the hold FIFO opens");
        let _ = hold.read(&mut [0_u8; 1]);
        std::fs::write({marker}, "survived\n").expect("the marker is written");
    }}
    
    #[cfg(test)]
    mod tests {{
        #[test]
        fn passes() {{}}
    
        #[test]
        fn forever() {{
            super::outlive();
        }}
    }}
    "#
        )
    }
    
    /// The capability document that lists `entry` in `list`.
    fn capability(list: &str, entry: &str) -> String {
        let mut sections = String::new();
        for name in [
            "Purpose",
            "Contract",
            "Usage",
            "Examples",
            "Constraints",
            "Explanation",
            "Verification",
        ] {
            write!(sections, "## {name}\n\nText for {name}.\n\n").expect("a string");
        }
        let entry = format!("\"{entry}\"");
        let (tests, examples) = match list {
            "tests" => (entry.as_str(), ""),
            _ => ("", entry.as_str()),
        };
        format!(
            "---\ntype: Capability\nid: cap:fixture\ntitle: Fixture\ndescription: Fixture.\n\
             state: implementing\nbindings:\n  code: []\n  tests: [{tests}]\n  \
             examples: [{examples}]\ndepends: []\n---\n\n# Fixture\n\n{sections}"
        )
    }

    /// A run of a binding that never ends, and what became of its processes.
    struct NeverEnding {
        crate_: Crate,
        evidence: Evidence,
        asked: Vec<(Vec<String>, Limits)>,
        seen: Seen,
        settled: Settled,
        /// Kept to the end: its scratch directory holds the crate.
        _subject: Subject,
    }
    
    /// Generate the crate of `package`, with an example when `list` is
    /// `examples`, build it with `warm`, and run the binding `entry` of `list`
    /// with `limits` through a [`Clocked`] runner.
    fn never_ending(
        package: &str,
        list: &str,
        entry: &str,
        warm: &[&str],
        limits: (Limits, Limits),
    ) -> NeverEnding {
        let subject = Subject::new();
        let crate_ = Crate::generate(&subject, package, list == "examples");
        crate_.warm(warm);
        let runner = Clocked::new(crate_.runner(), Stepped::gated(Watch::new(&subject), LIMIT));
        let evidence = crate_
            .judge(&runner, list, entry, Some(limits))
            .expect("a kill is a verdict");
        let mut clock = runner
            .ran
            .take()
            .expect("the binding ran under the stepped clock");
        let Some(seen) = clock.seen.take() else {
            panic!("the run ended before the gate opened on the process that never ends: {evidence}");
        };
        let settled = subject.settle(&seen);
        NeverEnding {
            crate_,
            evidence,
            asked: runner.asked.take(),
            seen,
            settled,
            _subject: subject,
        }
    }
    
    fn words(parts: &[&str]) -> Vec<String> {
        parts.iter().map(|part| (*part).to_string()).collect()
    }
    
    /// The reason of a run killed at [`LIMIT`], word for word.
    fn killed_at_the_limit(evidence: &Evidence) {
        let killed = evidence.runs().last().expect("a run");
        assert_eq!(killed.ended(), Ended::TimedOut, "{evidence}");
        assert_eq!(
            killed.elapsed(),
            LIMIT,
            "the reading of the stepped clock at the kill"
        );
        assert_eq!(
            evidence.outcome(),
            &BoundOutcome::Failed {
                reason: format!(
                    "`{}` ran 5.0 s of its 5 s limit and was killed; the run has no verdict",
                    killed.invocation()
                ),
            }
        );
    }

    /// The name of a variant. The match has no wildcard arm, so a variant added to
    /// [`Arg`] stops this file compiling until it is walked below.
    fn variant(arg: &Arg) -> &'static str {
        match arg {
            Arg::Literal(_) => "literal",
            Arg::Package(_) => "package",
            Arg::Target(Target::Lib) => "target-lib",
            Arg::Target(Target::Bin(_)) => "target-bin",
            Arg::Target(Target::Test(_)) => "target-test",
            Arg::Example(_) => "example",
            Arg::Filter(_) => "filter",
            Arg::DocFilter(_) => "doc-filter",
            Arg::RefCommit(_) => "ref-commit",
            Arg::Commit(_) => "commit",
            Arg::RevPath { .. } => "rev-path",
            Arg::Path(_) => "path",
            Arg::LineLog { .. } => "line-log",
            Arg::Pickaxe(_) => "pickaxe",
            Arg::Trusted(_) => "trusted",
        }
    }

    #[test]
    fn a_rejected_value_never_becomes_an_argument() {
        let mut accepted = 0;
        let mut rejected = 0;
        for (name, pattern, make) in builders() {
            for (text, matching) in corpus() {
                let expected = matching.contains(&pattern);
                let built = make(&text);
                assert_eq!(
                    built.is_ok(),
                    expected,
                    "`{name}` built from {text:?} must be {} by {pattern:?}",
                    if expected { "accepted" } else { "rejected" },
                );
                match built {
                    Ok(arg) => {
                        accepted += 1;
                        assert_eq!(variant(&arg), name);
                        let rendered = arg.to_string();
                        assert!(
                            rendered.contains(&text),
                            "`{rendered}` must carry its value {text:?} unchanged"
                        );
                        if pattern == Pattern::TreePath {
                            // A tree path may look like an option or hold any
                            // printable character: it is a path after `--`, or
                            // the tail of one `-L<start>,<end>:` argument.
                            assert!(
                                rendered == text || rendered == format!("-L1,2:{text}"),
                                "`{rendered}` places the tree path {text:?} where an option is read"
                            );
                            assert!(
                                !text.chars().any(char::is_control),
                                "{text:?} holds a control character"
                            );
                        } else {
                            assert!(!text.starts_with('-'), "{text:?} was accepted");
                            assert!(
                                text.chars().all(|c| c.is_ascii_graphic())
                                    && !text.contains([';', '|', '$', '`', '*', '(', ')', '"']),
                                "{text:?} holds a character no pattern allows"
                            );
                        }
                    }
                    Err(error) => {
                        rejected += 1;
                        assert_eq!(error.pattern(), pattern);
                        assert_ne!(error.to_string(), "");
                    }
                }
            }
        }
        // Both directions are exercised, so neither the accepting nor the
        // rejecting half can vanish unseen.
        assert!(accepted > 30, "only {accepted} values were accepted");
        assert!(rejected > 150, "only {rejected} values were rejected");
    }

    /// The command a person or an agent typed after `--` on the command line of
    /// `evidence record`: the program and its arguments, run exactly as given.
    ///
    /// It is trusted build input, as the workspace's own build commands are: it
    /// runs with no shell, its standard input null, its output bounded and a
    /// wall-clock limit, in the workspace root, with the caller's environment
    /// unchanged (no `LC_ALL=C`, and no `--frozen` added to a `cargo` in it). It is
    /// the one process of the workplan whose argv is not a row of an argv table,
    /// and it can be built from one place only: the command line of the running
    /// process ([`TrustedArgv::from_command_line`] takes the [`std::env::ArgsOs`]
    /// that only [`std::env::args_os`] makes). The text of a loaded document or
    /// record, such as the `command` an evidence record stores for its readers,
    /// cannot become one:
    ///
    /// ```compile_fail,E0308
    /// use raf_workplan::exec::TrustedArgv;
    ///
    /// // The `command` key of a loaded evidence record is text, and text is not
    /// // the command line of the process.
    /// let command: Option<String> = Some("cargo bench parse".to_string());
    /// let _ = TrustedArgv::from_command_line(command.unwrap_or_default());
    /// ```
    ///
    /// ```compile_fail,E0451
    /// use raf_workplan::exec::TrustedArgv;
    ///
    /// // Nor can its words be filled in directly.
    /// let _ = TrustedArgv { words: Vec::new() };
    /// ```
    ///
    /// ```compile_fail,E0624
    /// use raf_workplan::exec::TrustedArgv;
    ///
    /// // And the crate's own constructor, for its tests, is not public.
    /// let _ = TrustedArgv::new(vec!["sh".to_string()]);
    /// ```
    #[derive(Debug, Clone, PartialEq, Eq, Hash)]
    pub struct TrustedArgv {
        words: Vec<TrustedWord>,
    }
    
    /// The word that ends the options of `evidence record` and starts the command.
    const SEPARATOR: &str = "--";
    
    impl TrustedArgv {
        /// The words after the first `--` of the command line of this process.
        ///
        /// # Errors
        ///
        /// [`TrustedArgvError::NoSeparator`] when no word is `--`,
        /// [`TrustedArgvError::Empty`] when nothing follows it,
        /// [`TrustedArgvError::EmptyProgram`] when the program is the empty word,
        /// and [`TrustedArgvError::NotUtf8`] when a word of the command is not
        /// UTF-8 (the record stores the command as text for its readers, so a word
        /// it cannot hold is refused rather than altered).
        pub fn from_command_line(args: std::env::ArgsOs) -> Result<Self, TrustedArgvError> {
            Self::after_separator(args)
        }
    
        /// The words after the first `--` of `args`.
        fn after_separator(args: impl IntoIterator<Item = OsString>) -> Result<Self, TrustedArgvError> {
            let mut args = args.into_iter();
            if !args.any(|arg| arg == SEPARATOR) {
                return Err(TrustedArgvError::NoSeparator);
            }
            let words = args
                .enumerate()
                .map(|(index, arg)| {
                    arg.into_string()
                        .map_err(|_| TrustedArgvError::NotUtf8 { index })
                })
                .collect::<Result<Vec<String>, _>>()?;
            Self::new(words)
        }
    
        /// The command of these words, as the crate's tests build one.
        pub(crate) fn new(words: Vec<String>) -> Result<Self, TrustedArgvError> {
            match words.first() {
                None => return Err(TrustedArgvError::Empty),
                Some(program) if program.is_empty() => return Err(TrustedArgvError::EmptyProgram),
                Some(_) => {}
            }
            Ok(Self {
                words: words.into_iter().map(TrustedWord).collect(),
            })
        }
    
        /// The words: the program, then its arguments.
        #[must_use]
        pub fn words(&self) -> &[TrustedWord] {
            &self.words
        }
    
        /// The program: the first word.
        #[must_use]
        pub fn program(&self) -> &str {
            self.words.first().map_or("", TrustedWord::as_str)
        }
    
        /// The command as text for a reader: the words joined by one space, each
        /// that holds anything but `A-Z a-z 0-9` and `_ @ % + = : , . / -` (or is
        /// empty) in single quotes, a `'` in it written `'\''`, as a POSIX shell
        /// would read the text back. It is never run: the workplan starts no shell.
        #[must_use]
        pub fn text(&self) -> String {
            text_of(self.words.iter().map(TrustedWord::as_str))
        }
    }
    
    /// The words as [`TrustedArgv::text`] writes them, each quoted as a POSIX
    /// shell reads it back and joined by one space. The message of an error about
    /// the caller's command quotes it with this, so it reads as the record stores
    /// it.
    pub(crate) fn text_of<'a>(words: impl IntoIterator<Item = &'a str>) -> String {
        words.into_iter().map(quote).collect::<Vec<_>>().join(" ")
    }
    
    /// `word` as a POSIX shell reads it back.
    fn quote(word: &str) -> String {
        let plain = !word.is_empty()
            && word
                .chars()
                .all(|c| c.is_ascii_alphanumeric() || "_@%+=:,./-".contains(c));
        if plain {
            word.to_string()
        } else {
            format!("'{}'", word.replace('\'', "'\\''"))
        }
    }
    
    /// Why the command line holds no command to run.
    #[derive(Debug, Clone, PartialEq, Eq)]
    #[non_exhaustive]
    pub enum TrustedArgvError {
        /// No word of the command line is `--`.
        NoSeparator,
        /// Nothing follows the `--`.
        Empty,
        /// The program is the empty word.
        EmptyProgram,
        /// A word of the command is not UTF-8.
        NotUtf8 {
            /// The place of the word in the command, from 0 for the program.
            index: usize,
        },
    }
    
    impl fmt::Display for TrustedArgvError {
        fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
            match self {
                Self::NoSeparator => f.write_str(
                    "no command to run: put the command after `--`, as in `-- cargo bench parse`",
                ),
                Self::Empty => f.write_str("nothing follows `--`: name the command to run after it"),
                Self::EmptyProgram => f.write_str("the program after `--` is the empty word"),
                Self::NotUtf8 { index } => write!(
                    f,
                    "word {index} of the command is not UTF-8, and the record stores the command as text; nothing was run"
                ),
            }
        }
    }

    /// Complete the change `change`: refuse with every reason it cannot be
        /// completed, or rewrite the generated blocks the removal stales and
        /// remove `work/changes/<slug>.md`.
        ///
        /// `change` is the slug, with or without the `change:` prefix. The six
        /// conditions of the completion gate are judged as the module documents:
        ///
        /// 1. the obligation list of [`Lifecycle::reconcile`](Lifecycle::reconcile)
        ///    is empty (every kind, `contract_effect: none` leaving out the
        ///    untouched contract alone; the list also holds
        ///    `contract-effect-understated`);
        /// 2. every item has an `outcome`, and a `moved` outcome names a change
        ///    record that exists; a `dropped` or `moved` item
        ///    also has a `reason` (`outcome-without-reason`);
        /// 3. no PROPOSED block of the change remains in any capability document,
        ///    whether the document loads or not;
        /// 4. `docs sync` would write nothing;
        /// 5. each test of `bindings.tests` and each example of
        ///    `bindings.examples` of a listed capability passes, by the proof its
        ///    kind requires ([`Executor`]). A capability with empty lists satisfies
        ///    this only while it is `designed` or `considered`; in any other state
        ///    the empty list is a `binding-missing` obligation of condition 1;
        /// 6. each authored section in `affects` has a current assessment with the
        ///    verdict `accepted` ([`assessment`](crate::assessment)): a section
        ///    with no assessment, one whose inputs moved, and one that is current
        ///    and `rejected` each refuse, whether the assessor was a human or an
        ///    agent. A section that holds only generated blocks would need none,
        ///    but the profile refuses one, so every section of a loaded capability
        ///    is authored.
        ///    This is judged with the other static conditions, before any bound
        ///    test or example runs.
        ///
        /// A failing example is both an `example-failing` obligation (condition 1)
        /// and a failure of condition 5, and a section without a current accepted
        /// assessment is one or more `assessment-*` obligations (condition 1), one
        /// `assessment-stale` for each input that moved, each also a failure of
        /// condition 6; the headline of the refusal counts the section once.
        ///
        /// The repository without the record must also pass `check`, apart from
        /// the generated blocks the removal stales (the `Changed by` rows of the
        /// relationship tables), which `complete` rewrites. This is where the
        /// state of a listed capability is judged: a capability that is not
        /// `delivered` and is listed by no other open record is `no-open-record`,
        /// so the author sets `state: delivered` before running `complete`.
        /// `complete` never edits prose, and it does not judge the repository
        /// before the removal by the state rules: a `delivered` capability beside
        /// its still-open record is exactly what it resolves.
        ///
        /// That check fails on every finding in the repository, not only on the
        /// ones about the listed capabilities: a broken link in an unrelated
        /// capability blocks completion, and so does a link in prose to the record
        /// itself, which is broken once the record is removed.
        ///
        /// An `outcome` that is not `implemented`, `dropped` or
        /// `moved:change:<slug>` is a value the profile refuses and makes the
        /// record not load, so `complete` returns the error of
        /// [`Workplan::change`] and not a condition-2 refusal, as
        /// [`Lifecycle::reconcile`](Lifecycle::reconcile) does. Condition 2
        /// refuses an item with no outcome and a `moved` outcome whose target is
        /// missing.
        ///
        /// ```rust,no_run
        /// use raf_workplan::exec::Programs;
        /// use raf_workplan::lifecycle::Lifecycle;
        ///
        /// let lifecycle = Lifecycle::open(Programs::new("."))?;
        /// let completion = lifecycle.complete("incremental-parsing")?;
        /// print!("{completion}");
        /// std::process::exit(i32::from(!completion.is_completed()));
        /// # Ok::<(), raf_workplan::lifecycle::LifecycleError>(())
        /// ```
        ///
        /// # Writes
        ///
        /// A refusal writes nothing. A completion writes the generated blocks
        /// that the removal stales (through [`Workplan::write`], which stages
        /// every file and then renames) and then removes the record, as the very
        /// last step. If the rewrite fails the record is not removed. If the
        /// removal fails after the rewrite, the error says which files were
        /// rewritten. No other file is written and no state is changed.
        ///
        /// A document changed between the second evaluation and the write is not
        /// detected: [`Workplan::write`] replaces a file without comparing it with
        /// the text the rewrite was computed from.
        ///
        /// # Errors
        ///
        /// [`LifecycleError::Change`] and [`LifecycleError::Reconcile`] when the
        /// record or a listed capability does not load; [`LifecycleError::Git`]
        /// and [`LifecycleError::Document`] when a section cannot be compared;
        /// [`LifecycleError::Exec`] when a process of a bound run cannot be
        /// started or read to its end; [`LifecycleError::RecordRefused`] when
        /// `work` or `work/changes` is a symbolic link or not a directory, or the
        /// record is not a regular file there, judged with each evaluation, the
        /// first before anything runs and the second after the bound runs and
        /// before the writes; [`LifecycleError::Gate`] when the
        /// repository cannot be opened again; [`LifecycleError::Rewrite`] and
        /// [`LifecycleError::Remove`] when a write fails. Each is the tool being
        /// unable to give an answer, which is not a refusal.
        pub fn complete(&self, change: &str) -> Result<Completion, LifecycleError> {
            let slug = bare_slug(change);
    
            // The source index at the merge base is loaded once for both
            // judgements: the files of one commit do not change, and the second
            // judgement asks for the merge base again and loads it again only when
            // it, or the packages it is read from, differ.
            let mut kept = None;
            let before = self.evaluate(&self.opened, slug, false, &mut kept)?;
            if !before.failures.is_empty() {
                return Ok(Completion::Refused {
                    failures: before.failures,
                });
            }
    
            let failing = self.run_bound(&self.opened, &before.runs)?;
            if !failing.is_empty() {
                return Ok(Completion::Refused { failures: failing });
            }
    
            // The documents and the source can change while `cargo` runs, and
            // `Opened` holds the source as it was.
            let opened = self.gate.open()?;
            let after = self.evaluate(&opened, slug, true, &mut kept)?;
            let different = run_keys(&after.runs) != run_keys(&before.runs);
            if different || !after.failures.is_empty() {
                let what = if different {
                    "the bound tests and examples to run differ from the ones that ran"
                } else {
                    "a condition that held before the bound tests and examples ran fails now"
                };
                let mut failures = vec![Failure::Changed {
                    what: what.to_string(),
                }];
                failures.extend(after.failures);
                return Ok(Completion::Refused { failures });
            }
    
            Self::write(&opened, &after.record, &after.rewrite)
        }

    /// Run the bound tests and the bound examples of the change `change`, or of
    /// every capability when `change` is `None`, through `runner`, and report each
    /// binding.
    ///
    /// A test is judged by the proof that exactly one test ran and passed, an
    /// example target by its exit status within the limit, a documentation test by
    /// the `ok` line of each entry the list pass selected (see
    /// [`Executor`]). A binding that does not resolve, is not a `#[test]`, or whose
    /// value fails its pattern (`-Zunstable-options`, `--include-ignored`,
    /// `--exact`, a path with `;` or a space) is reported as unresolved and
    /// never reaches `cargo`. `code` bindings are not run.
    ///
    /// This does not ask `git` anything and does not go through the Git version
    /// gate of [`Lifecycle::open`](super::Lifecycle::open). It writes nothing in
    /// the repository; `cargo` builds into its own target directory.
    ///
    /// For a change, every capability it lists must be loaded. For every
    /// capability, no capability document may be unloaded: what it binds is
    /// unknown, and a run that skipped it would pass.
    ///
    /// # Errors
    ///
    /// [`LifecycleError::Change`] when the slug names no loaded record;
    /// [`LifecycleError::Reconcile`] when a listed capability did not load;
    /// [`LifecycleError::Unloaded`] for a run over every capability while one did
    /// not load; [`LifecycleError::Exec`] when a process cannot be started or
    /// read to its end, which is no verdict.
    pub fn verify<R: Runner + ?Sized>(
        opened: &Opened,
        runner: &R,
        change: Option<&str>,
    ) -> Result<VerifyReport, LifecycleError> {
        let plan = opened.workplan();
        let mut bindings = match change {
            Some(slug) => bindings_of_change(&plan, slug)?,
            None => bindings_of_every_capability(&plan)?,
        };
        bindings.retain(|binding| matches!(binding.kind(), BindingKind::Test | BindingKind::Example));
        bindings.sort_by(|a, b| {
            (a.capability(), a.kind(), a.path()).cmp(&(b.capability(), b.kind(), b.path()))
        });
        let executor = Executor::new(opened.workspace(), runner)?;
        let evidence = bindings
            .iter()
            .map(|binding| executor.run(binding))
            .collect::<Result<Vec<_>, _>>()?;
        Ok(VerifyReport {
            change: change.map(|slug| crate::lookup::bare_slug(slug).to_string()),
            evidence,
        })
    }
