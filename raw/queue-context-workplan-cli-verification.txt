    /// `assess`: the context, a recorded verdict, or the list. Each prints
    /// outcomes only, so each exits 0 when it has an answer: a rejected verdict
    /// was recorded, and a stale or rejected assessment in the list is shown with
    /// its currency and left to `change reconcile`, which lists what it owes. A
    /// section given to `--list` must be one `--context` answers for, so an
    /// unknown capability is exit 2 in all three modes.
    pub(crate) fn assess(root: &Path, args: &AssessArgs) -> Result<Report, Failure> {
        let opened = Gate::workspace(root).open()?;
        let plan = opened.workplan();
        if args.list {
            // A section to list is one the library can assess: a capability or a
            // section that no loaded document has is an error, as for `--context`.
            if let Some(subject) = &args.subject {
                plan.assess_context(subject)?;
            }
            let states = plan.list_assessments();
            let shown: Vec<&AssessmentState> = states
                .iter()
                .filter(|state| args.subject.as_ref().is_none_or(|s| *s == state.subject))
                .collect();
            let mut report = Report::of(shown.iter().map(|state| state_record(state)));
            if shown.is_empty() {
                report.note(match &args.subject {
                    Some(subject) => format!("no assessment of {subject} loaded"),
                    None => "no assessment record loaded".to_string(),
                });
            }
            return Ok(report);
        }
        let subject = args
            .subject
            .as_deref()
            .ok_or_else(|| Failure::new("give the section to assess, `cap:<slug>#<section>`"))?;
        let Some(verdict) = args.verdict else {
            let context = plan.assess_context(subject)?;
            return Ok(Report::of([context_record(&context)]));
        };
        let by = args
            .by
            .ok_or_else(|| Failure::new("--verdict needs --by human or --by agent: who assessed"))?;
        let mut options = RecordOptions::new();
        if let Some(copyright) = &args.copyright {
            options = options.copyright(copyright.as_str());
        }
        if let Some(licence) = &args.licence {
            options = options.licence(licence.as_str());
        }
        let request =
            AssessmentRequest::new(subject, Verdict::verdict(verdict), by.actor()).options(options);
        let assessed = plan.record_assessment(root, &request)?;
        Ok(Report::of([assessed_record(&assessed)]))
    }

    /// Run the command of `args` and return what it produced.
    ///
    /// # Errors
    ///
    /// A [`Failure`] when the tool could not run: the root cannot be found, the
    /// repository cannot be opened, or the command's own input is refused. The
    /// command of `evidence record` is checked before the root is looked for, so
    /// a refusal of it starts no process.
    pub(crate) fn run(args: &Args) -> Result<Report, Failure> {
        let root = || workspace_root(args.root.as_deref());
        match &args.command {
            Command::Check => check(&root()?, args.strict),
            Command::Status => status(&root()?),
            Command::Docs(Docs::Sync(sync_args)) => sync(&root()?, sync_args),
            Command::Docs(Docs::Preview(preview_args)) => preview(&root()?, &preview_args.slug),
            Command::Change(Change::New(new_args)) => change::new(&root()?, new_args),
            Command::Change(Change::Impact(slug)) => change::impact(&root()?, &slug.slug),
            Command::Change(Change::Reconcile(slug)) => {
                change::reconcile(&root()?, &slug.slug, args.strict)
            }
            Command::Change(Change::Complete(slug)) => {
                change::complete(&root()?, &slug.slug, args.strict)
            }
            Command::Verify(verify_args) => change::verify(&root()?, verify_args.slug.as_deref()),
            Command::History(history_args) => history::history(&root()?, history_args),
            Command::Evidence(Evidence::Record(record_args)) => {
                let source = evidence::source(record_args)?;
                evidence::record(&root()?, record_args, source)
            }
            Command::Evidence(Evidence::Claims(id)) => evidence::claims(&root()?, &id.id),
            Command::Evidence(Evidence::For(cap)) => evidence::support(&root()?, &cap.capability),
            Command::Assess(assess_args) => assess::assess(&root()?, assess_args),
            Command::Coverage(Coverage::Code(coverage_args)) => {
                coverage::coverage(&root()?, coverage_args, GapKind::PublicItem)
            }
            Command::Coverage(Coverage::Tests(coverage_args)) => {
                coverage::coverage(&root()?, coverage_args, GapKind::Test)
            }
            Command::Coverage(Coverage::Claims(claims_args)) => coverage::claims(&root()?, claims_args),
            Command::Coverage(Coverage::Lines(lines_args)) => {
                coverage::lines(&root()?, lines_args, args.strict)
            }
            Command::Debt(debt_args) => debt::debt(&root()?, debt_args, args.strict),
        }
    }

    /// A coverage report of one direction, of the whole workspace or of the
    /// package `--crate` names. The report is a finding when it names a gap, or
    /// in the code direction a surface that cannot be listed, so the run exits 1;
    /// otherwise it is an outcome and exits 0. It starts no process beyond the

    pub(crate) fn coverage(root: &Path, args: &CoverageArgs, kind: GapKind) -> Result<Report, Failure> {
        let opened = Gate::workspace(root).open()?;
        let plan = opened.workplan();
        let whole = plan.coverage_report();
        let report = match &args.package {
            None => whole,
            Some(name) => selected(&whole, name)?,
        };
        let errors = kind.is_code() && !report.errors().is_empty();
        let class = if report.gaps(kind).is_empty() && !errors {
            Class::Outcome
        } else {
            Class::Finding
        };
        Ok(Report::rendered(
            class,
            render::table(&report, kind),
            render::json(&report, kind),
        ))
    }

    /// `evidence record`: run the source of the verdict and write the record. A
    /// record written with the verdict `pass` is an outcome; one written with
    /// `fail` is a finding, so the run exits 1. A run that writes no record is a
    /// failure (exit 2).
    pub(crate) fn record(
        root: &Path,
        args: &RecordArgs,
        source: VerdictSource,
    ) -> Result<Report, Failure> {
        let opened = Gate::workspace(root).open()?;
        let plan = opened.workplan();
        let mut options = RecordOptions::new();
        if let Some(copyright) = &args.copyright {
            options = options.copyright(copyright.as_str());
        }
        if let Some(licence) = &args.licence {
            options = options.licence(licence.as_str());
        }
        let mut request = EvidenceRequest::new(
            args.id.as_str(),
            args.covers.clone(),
            args.by.actor(),
            source,
        )
        .limits(args.limit.clone())
        .options(options);
        if let Some(revision) = &args.revision {
            request = request.revision(revision.as_str());
        }
        if args.update {
            request = request.mode(WriteMode::Replace);
        }
        let runner = SystemRunner::new(Programs::new(root));
        let recorded = plan.record_evidence(root, &runner, &request)?;
        Ok(Report::of([recorded_record(&recorded)]))
    }

    /// The source of the verdict of `evidence record`: the bound test or example
    /// `--from-binding` names, or the command after `--`. Nothing is read and
    /// nothing is started here, so a refusal comes before root discovery.
    ///
    /// # Errors
    ///
    /// A [`Failure`] when the program name of the process is `--`, when
    /// `--from-binding` is given beside a `--`, when the command line holds no
    /// command after `--`, its program is the empty word or a word of it is not
    /// UTF-8, or when the words after its first `--` are not the command the
    /// parser read.
    pub(crate) fn source(args: &RecordArgs) -> Result<VerdictSource, Failure> {
        line(std::env::args_os(), args.from_binding.is_some())?;
        if let Some(path) = &args.from_binding {
            return Ok(VerdictSource::Binding(path.clone()));
        }
        let trusted = TrustedArgv::from_command_line(std::env::args_os())?;
        let words: Vec<&OsStr> = trusted
            .words()
            .iter()
            .map(|word| OsStr::new(word.as_str()))
            .collect();
        agree(&words, &args.command)?;
        Ok(VerdictSource::Command(trusted))
    }

    //! Text from a document, a record or a commit, made safe to print: each
    //! character that changes how a terminal shows the text around it is written
    //! as its `\u{..}` escape. The history and coverage tables escape their cells
    //! with it, and the command line escapes the text output of the commands that
    //! print document text, so no text can move the cursor of a terminal, set its
    //! title, clear it, ring it, or reorder how a line reads.
    
    use std::fmt::Write as _;
    
    /// Whether `c` changes how a terminal shows the text around it: a control
    /// character, or a bidirectional formatting character (the marks U+061C,
    /// U+200E and U+200F, the embeddings and overrides U+202A to U+202E, and the
    /// isolates U+2066 to U+2069).
    fn hidden(c: char) -> bool {
        c.is_control()
            || matches!(
                c,
                '\u{61c}' | '\u{200e}' | '\u{200f}' | '\u{202a}'..='\u{202e}' | '\u{2066}'..='\u{2069}'
            )
    }
    
    /// `text` with each character that `keep` does not keep and that changes how
    /// a terminal shows the text around it as its `\u{..}` escape.
    fn escaped(text: &str, keep: impl Fn(char) -> bool) -> String {
        let mut out = String::with_capacity(text.len());
        for c in text.chars() {
            if hidden(c) && !keep(c) {
                let _ = write!(out, "\\u{{{:x}}}", u32::from(c));
            } else {
                out.push(c);
            }
        }
        out
    }
    
    /// `text` on one line: every control character, line feeds and tabs
    /// included, and every bidirectional formatting character as its escape.
    pub(crate) fn shown(text: &str) -> String {
        escaped(text, |_| false)
    }
    
    /// `text` as lines: each control character other than the line feed and the
    /// tab, a carriage return and an escape included, and each bidirectional
    /// formatting character as its `\u{..}` escape, so the text keeps its lines
    /// and indentation and nothing else of it reaches the terminal as a control.
    /// Other text, non-ASCII letters and symbols included, is kept as it is.
    ///
    /// ```rust
    /// use raf_workplan::escape::shown_lines;
    ///
    /// let text = "It holds.\u{1b}]0;title\u{7}\n\tSo \u{202e}does this.\n";
    /// assert_eq!(
    ///     shown_lines(text),
    ///     "It holds.\\u{1b}]0;title\\u{7}\n\tSo \\u{202e}does this.\n"
    /// );
    /// ```
    #[must_use]
    pub fn shown_lines(text: &str) -> String {
        escaped(text, |c| c == '\n' || c == '\t')
    }
    
    #[cfg(test)]
    mod tests {
        use super::*;
    
        #[test]
        fn each_control_and_bidirectional_formatting_character_is_escaped() {
            assert_eq!(shown("x\u{1b}[31m\u{1f}"), "x\\u{1b}[31m\\u{1f}");
            assert_eq!(shown("one\ntwo\tthree"), "one\\u{a}two\\u{9}three");
            assert_eq!(
                shown("a\u{202e}b\u{2066}c\u{200f}\u{61c}"),
                "a\\u{202e}b\\u{2066}c\\u{200f}\\u{61c}"
            );
            for c in (0u32..0x20).chain(0x7f..0xa0).filter_map(char::from_u32) {
                assert!(
                    !shown(&c.to_string()).chars().any(hidden),
                    "{:x}",
                    u32::from(c)
                );
            }
            // The neighbour: other non-ASCII text is kept.
            assert_eq!(shown("été \u{2014} 名前"), "été \u{2014} 名前");
        }
    
        #[test]
        fn lines_keep_their_line_feeds_and_tabs_and_lose_every_other_control() {
            assert_eq!(
                shown_lines("A\u{1b}[2J\u{1b}]0;PWNED\u{7}\n\tB\r\n\u{202e}C\n"),
                "A\\u{1b}[2J\\u{1b}]0;PWNED\\u{7}\n\tB\\u{d}\n\\u{202e}C\n"
            );
            assert_eq!(shown_lines("plain\n  text\n"), "plain\n  text\n");
            assert_eq!(shown_lines("été \u{2014} 名前"), "été \u{2014} 名前");
        }
    }

    //! Process-free workspace discovery through `raf-source`.
    
    use raf_source::{DiskTree, LoadError, RelPath, TreeError};
    use std::error::Error;
    use std::fmt;
    use std::path::{Path, PathBuf};
    
    /// Why a workspace root cannot be discovered.
    #[derive(Debug)]
    #[non_exhaustive]
    pub enum RootError {
        /// The starting directory cannot be canonicalized.
        Directory(std::io::Error),
        /// The filesystem tree cannot be opened.
        Tree(TreeError),
        /// A manifest or membership finding prevents discovery.
        Source(LoadError),
        /// A filesystem path cannot be represented as a portable tree path.
        Path(String),
    }
    impl fmt::Display for RootError {
        fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
            match self {
                Self::Directory(error) => write!(f, "cannot open discovery directory: {error}"),
                Self::Tree(error) => write!(f, "{error}"),
                Self::Source(error) => write!(f, "{error}"),
                Self::Path(path) => write!(
                    f,
                    "cannot represent discovery path `{}` as a tree path",
                    path.escape_debug()
                ),
            }
        }
    }
    impl Error for RootError {
        fn source(&self) -> Option<&(dyn Error + 'static)> {
            match self {
                Self::Directory(error) => Some(error),
                Self::Tree(error) => Some(error),
                Self::Source(error) => Some(error),
                Self::Path(_) => None,
            }
        }
    }
    /// Find the workspace containing `directory`, without starting any process.
    ///
    /// # Errors
    ///
    /// Returns [`RootError`] for an inaccessible directory, a malformed manifest,
    /// an unlisted package, or conflicting membership. See
    /// [`raf_source::discover_root`] for the tree-based discovery contract.
    pub fn discover(directory: &Path) -> Result<PathBuf, RootError> {
        let directory = directory.canonicalize().map_err(RootError::Directory)?;
        let boundary = directory
            .ancestors()
            .last()
            .ok_or_else(|| RootError::Path(directory.display().to_string()))?;
        let relative = directory
            .strip_prefix(boundary)
            .map_err(|_| RootError::Path(directory.display().to_string()))?;
        let text = relative
            .to_str()
            .ok_or_else(|| RootError::Path(relative.display().to_string()))?;
        let start = RelPath::new(&text.replace(std::path::MAIN_SEPARATOR, "/"))
            .map_err(|_| RootError::Path(text.to_owned()))?;
        let tree = DiskTree::new(boundary).map_err(RootError::Tree)?;
        let root = raf_source::discover_root(&tree, &start).map_err(RootError::Source)?;
        Ok(boundary.join(root.as_str()))
    }
