    Usage
    
    ```sh
    cargo workplan check
    cargo workplan status
    cargo workplan change new incremental-parsing --capability cap:streaming-parser
    cargo workplan docs preview incremental-parsing
    cargo workplan change complete incremental-parsing
    cargo-workplan --format json change reconcile incremental-parsing
    ```
    
    `make docs-sync` in this repository runs `cargo-workplan docs sync` with the SPDX text
    of a new index.
    
    The record commands are used the same way:
    
    - `cargo workplan history cap:workplan-reconcile --since 2026-01-01` prints the
      commits of the capability and of its bound code.
    - `cargo workplan evidence record --id peak-rss --for cap:streaming-parser --by agent --limit peak=64 -- cargo bench parse`
      runs the benchmark and writes `work/evidence/peak-rss.md`.
    - `cargo workplan assess cap:streaming-parser#usage --context` prints what an
      assessor reads, and `cargo workplan assess cap:streaming-parser#usage --verdict accepted --by human`
      records the verdict.
    - `cargo workplan --format json coverage code --crate raf-workplan` prints the
      public items of one package that no capability binds.
    - `cargo workplan --strict check` judges the repository under strict mode before
      its coverage record says `strictness: strict`, and `cargo workplan check` judges
      it so once the record says it.
    - `cargo workplan coverage claims --capability cap:streaming-parser` prints each
      claim of the capability and the tests its markers name.
    
    > [!PROPOSED change:workplan-strict]
    > The other commands of strict mode are used the same way:
    >
    > - `cargo workplan coverage lines --crate raf-okf`, or
    >   `cargo workplan coverage --reverse --crate raf-okf`, prints the orphan lines of
    >   one package by crate, module and item.
    > - `cargo workplan debt --write --copyright '2026 Example Inc.' --licence CC-BY-4.0`
    >   writes `work/debt/default.md` with its two SPDX lines.

    /// `assess`: the context, a recorded verdict, or the list. Each prints
    /// outcomes only, so each exits 0 when it has an answer: a rejected verdict
    /// was recorded, and a stale or rejected assessment in the list is shown with
    /// its currency and left to `change reconcile`, which lists what it owes. A
    /// section given to `--list` must be one `--context` answers for, so an
    /// unknown capability is exit 2 in all three modes.
    pub(crate) fn assess(root: &Path, args: &AssessArgs) -> Result<Report, Failure> {
        let opened = Gate::workspace(root).open()?;
        let plan = opened.workplan();
        if args.list {
            // A section to list is one the library can assess: a capability or a
            // section that no loaded document has is an error, as for `--context`.
            if let Some(subject) = &args.subject {
                plan.assess_context(subject)?;
            }
            let states = plan.list_assessments();
            let shown: Vec<&AssessmentState> = states
                .iter()
                .filter(|state| args.subject.as_ref().is_none_or(|s| *s == state.subject))
                .collect();
            let mut report = Report::of(shown.iter().map(|state| state_record(state)));
            if shown.is_empty() {
                report.note(match &args.subject {
                    Some(subject) => format!("no assessment of {subject} loaded"),
                    None => "no assessment record loaded".to_string(),
                });
            }
            return Ok(report);
        }
        let subject = args
            .subject
            .as_deref()
            .ok_or_else(|| Failure::new("give the section to assess, `cap:<slug>#<section>`"))?;
        let Some(verdict) = args.verdict else {
            let context = plan.assess_context(subject)?;
            return Ok(Report::of([context_record(&context)]));
        };
        let by = args
            .by
            .ok_or_else(|| Failure::new("--verdict needs --by human or --by agent: who assessed"))?;
        let mut options = RecordOptions::new();
        if let Some(copyright) = &args.copyright {
            options = options.copyright(copyright.as_str());
        }
        if let Some(licence) = &args.licence {
            options = options.licence(licence.as_str());
        }
        let request =
            AssessmentRequest::new(subject, Verdict::verdict(verdict), by.actor()).options(options);
        let assessed = plan.record_assessment(root, &request)?;
        Ok(Report::of([assessed_record(&assessed)]))
    }

    /// Run the command of `args` and return what it produced.
    ///
    /// # Errors
    ///
    /// A [`Failure`] when the tool could not run: the root cannot be found, the
    /// repository cannot be opened, or the command's own input is refused. The
    /// command of `evidence record` is checked before the root is looked for, so
    /// a refusal of it starts no process.
    pub(crate) fn run(args: &Args) -> Result<Report, Failure> {
        let root = || workspace_root(args.root.as_deref());
        match &args.command {
            Command::Check => check(&root()?, args.strict),
            Command::Status => status(&root()?),
            Command::Docs(Docs::Sync(sync_args)) => sync(&root()?, sync_args),
            Command::Docs(Docs::Preview(preview_args)) => preview(&root()?, &preview_args.slug),
            Command::Change(Change::New(new_args)) => change::new(&root()?, new_args),
            Command::Change(Change::Impact(slug)) => change::impact(&root()?, &slug.slug),
            Command::Change(Change::Reconcile(slug)) => {
                change::reconcile(&root()?, &slug.slug, args.strict)
            }
            Command::Change(Change::Complete(slug)) => {
                change::complete(&root()?, &slug.slug, args.strict)
            }
            Command::Verify(verify_args) => change::verify(&root()?, verify_args.slug.as_deref()),
            Command::History(history_args) => history::history(&root()?, history_args),
            Command::Evidence(Evidence::Record(record_args)) => {
                let source = evidence::source(record_args)?;
                evidence::record(&root()?, record_args, source)
            }
            Command::Evidence(Evidence::Claims(id)) => evidence::claims(&root()?, &id.id),
            Command::Evidence(Evidence::For(cap)) => evidence::support(&root()?, &cap.capability),
            Command::Assess(assess_args) => assess::assess(&root()?, assess_args),
            Command::Coverage(Coverage::Code(coverage_args)) => {
                coverage::coverage(&root()?, coverage_args, GapKind::PublicItem)
            }
            Command::Coverage(Coverage::Tests(coverage_args)) => {
                coverage::coverage(&root()?, coverage_args, GapKind::Test)
            }
            Command::Coverage(Coverage::Claims(claims_args)) => coverage::claims(&root()?, claims_args),
        }
    }

    /// A coverage report of one direction, of the whole workspace or of the
    /// package `--crate` names. The report is a finding when it names a gap, or
    /// in the code direction a surface that cannot be listed, so the run exits 1;
    /// otherwise it is an outcome and exits 0. It starts no process beyond the
    /// `git` that lists the tracked files, and writes nothing.
    pub(crate) fn coverage(root: &Path, args: &CoverageArgs, kind: GapKind) -> Result<Report, Failure> {
        let opened = Gate::workspace(root).open()?;
        let plan = opened.workplan();
        let whole = plan.coverage_report();
        let report = match &args.package {
            None => whole,
            Some(name) => selected(&whole, name)?,
        };
        let errors = kind.is_code() && !report.errors().is_empty();
        let class = if report.gaps(kind).is_empty() && !errors {
            Class::Outcome
        } else {
            Class::Finding
        };
        Ok(Report::rendered(
            class,
            render::table(&report, kind),
            render::json(&report, kind),
        ))
    }

    /// `evidence record`: run the source of the verdict and write the record. A
    /// record written with the verdict `pass` is an outcome; one written with
    /// `fail` is a finding, so the run exits 1. A run that writes no record is a
    /// failure (exit 2).
    pub(crate) fn record(
        root: &Path,
        args: &RecordArgs,
        source: VerdictSource,
    ) -> Result<Report, Failure> {
        let opened = Gate::workspace(root).open()?;
        let plan = opened.workplan();
        let mut options = RecordOptions::new();
        if let Some(copyright) = &args.copyright {
            options = options.copyright(copyright.as_str());
        }
        if let Some(licence) = &args.licence {
            options = options.licence(licence.as_str());
        }
        let mut request = EvidenceRequest::new(
            args.id.as_str(),
            args.covers.clone(),
            args.by.actor(),
            source,
        )
        .limits(args.limit.clone())
        .options(options);
        if let Some(revision) = &args.revision {
            request = request.revision(revision.as_str());
        }
        if args.update {
            request = request.mode(WriteMode::Replace);
        }
        let runner = SystemRunner::new(Programs::new(root));
        let recorded = plan.record_evidence(root, &runner, &request)?;
        Ok(Report::of([recorded_record(&recorded)]))
    }

    /// The source of the verdict of `evidence record`: the bound test or example
    /// `--from-binding` names, or the command after `--`. Nothing is read and
    /// nothing is started here, so a refusal comes before root discovery.
    ///
    /// # Errors
    ///
    /// A [`Failure`] when the program name of the process is `--`, when
    /// `--from-binding` is given beside a `--`, when the command line holds no
    /// command after `--`, its program is the empty word or a word of it is not
    /// UTF-8, or when the words after its first `--` are not the command the
    /// parser read.
    pub(crate) fn source(args: &RecordArgs) -> Result<VerdictSource, Failure> {
        line(std::env::args_os(), args.from_binding.is_some())?;
        if let Some(path) = &args.from_binding {
            return Ok(VerdictSource::Binding(path.clone()));
        }
        let trusted = TrustedArgv::from_command_line(std::env::args_os())?;
        let words: Vec<&OsStr> = trusted
            .words()
            .iter()
            .map(|word| OsStr::new(word.as_str()))
            .collect();
        agree(&words, &args.command)?;
        Ok(VerdictSource::Command(trusted))
    }

    //! Text from a document, a record or a commit, made safe to print: each
    //! character that changes how a terminal shows the text around it is written
    //! as its `\u{..}` escape. The history and coverage tables escape their cells
    //! with it, and the command line escapes the text output of the commands that
    //! print document text, so no text can move the cursor of a terminal, set its
    //! title, clear it, ring it, or reorder how a line reads.
    
    use std::fmt::Write as _;
    
    /// Whether `c` changes how a terminal shows the text around it: a control
    /// character, or a bidirectional formatting character (the marks U+061C,
    /// U+200E and U+200F, the embeddings and overrides U+202A to U+202E, and the
    /// isolates U+2066 to U+2069).
    fn hidden(c: char) -> bool {
        c.is_control()
            || matches!(
                c,
                '\u{61c}' | '\u{200e}' | '\u{200f}' | '\u{202a}'..='\u{202e}' | '\u{2066}'..='\u{2069}'
            )
    }
    
    /// `text` with each character that `keep` does not keep and that changes how
    /// a terminal shows the text around it as its `\u{..}` escape.
    fn escaped(text: &str, keep: impl Fn(char) -> bool) -> String {
        let mut out = String::with_capacity(text.len());
        for c in text.chars() {
            if hidden(c) && !keep(c) {
                let _ = write!(out, "\\u{{{:x}}}", u32::from(c));
            } else {
                out.push(c);
            }
        }
        out
    }
    
    /// `text` on one line: every control character, line feeds and tabs
    /// included, and every bidirectional formatting character as its escape.
    pub(crate) fn shown(text: &str) -> String {
        escaped(text, |_| false)
    }
    
    /// `text` as lines: each control character other than the line feed and the
    /// tab, a carriage return and an escape included, and each bidirectional
    /// formatting character as its `\u{..}` escape, so the text keeps its lines
    /// and indentation and nothing else of it reaches the terminal as a control.
    /// Other text, non-ASCII letters and symbols included, is kept as it is.
    ///
    /// ```rust
    /// use raf_workplan::escape::shown_lines;
    ///
    /// let text = "It holds.\u{1b}]0;title\u{7}\n\tSo \u{202e}does this.\n";
    /// assert_eq!(
    ///     shown_lines(text),
    ///     "It holds.\\u{1b}]0;title\\u{7}\n\tSo \\u{202e}does this.\n"
    /// );
    /// ```
    #[must_use]
    pub fn shown_lines(text: &str) -> String {
        escaped(text, |c| c == '\n' || c == '\t')
    }
    
    #[cfg(test)]
    mod tests {
        use super::*;
    
        #[test]
        fn each_control_and_bidirectional_formatting_character_is_escaped() {
            assert_eq!(shown("x\u{1b}[31m\u{1f}"), "x\\u{1b}[31m\\u{1f}");
            assert_eq!(shown("one\ntwo\tthree"), "one\\u{a}two\\u{9}three");
            assert_eq!(
                shown("a\u{202e}b\u{2066}c\u{200f}\u{61c}"),
                "a\\u{202e}b\\u{2066}c\\u{200f}\\u{61c}"
            );
            for c in (0u32..0x20).chain(0x7f..0xa0).filter_map(char::from_u32) {
                assert!(
                    !shown(&c.to_string()).chars().any(hidden),
                    "{:x}",
                    u32::from(c)
                );
            }
            // The neighbour: other non-ASCII text is kept.
            assert_eq!(shown("été \u{2014} 名前"), "été \u{2014} 名前");
        }
    
        #[test]
        fn lines_keep_their_line_feeds_and_tabs_and_lose_every_other_control() {
            assert_eq!(
                shown_lines("A\u{1b}[2J\u{1b}]0;PWNED\u{7}\n\tB\r\n\u{202e}C\n"),
                "A\\u{1b}[2J\\u{1b}]0;PWNED\\u{7}\n\tB\\u{d}\n\\u{202e}C\n"
            );
            assert_eq!(shown_lines("plain\n  text\n"), "plain\n  text\n");
            assert_eq!(shown_lines("été \u{2014} 名前"), "été \u{2014} 名前");
        }
    }

    // SPDX-License-Identifier: MIT OR Apache-2.0 OR MulanPSL-2.0
    
    //! Root discovery: the workspace root of the Cargo workspace the runner's
    //! directory is in.
    //!
    //! [`discover`] starts exactly one process, through the runner:
    //!
    //! ```text
    //! cargo metadata --format-version 1 --no-deps --frozen
    //! ```
    //!
    //! `--no-deps` skips dependency resolution, which finding the root does not
    //! need, and `--frozen` forbids network access and any change to `Cargo.lock`.
    //! The output is bounded at 64 MiB ([`METADATA_OUTPUT_BOUND`](crate::exec::METADATA_OUTPUT_BOUND)) and the
    //! `workspace_root` member of its top-level object is the only thing read.
    //! `raf-workplan` takes no JSON dependency, so a small scanner validates the
    //! text as JSON and finds that one member; a text that is not JSON, or whose
    //! top-level object does not have the member exactly once as a string, is an
    //! error and not a guess (see [`RootError`]).
    //!
    //! No value from a document is placed on the command line. Cargo reads the
    //! workspace's `.cargo/config.toml`, and the rustup proxy reads its
    //! `rust-toolchain.toml`, so the workspace is trusted build input, as it is for
    //! `cargo build`; cargo does not run the `build.rustc` or `build.rustc-wrapper`
    //! program of that configuration for this command.
    
    mod json;
    
    use std::error::Error;
    use std::fmt;
    use std::path::{Path, PathBuf};
    
    use crate::exec::{ExecError, Invocation, Limits, Runner};
    
    use json::JsonError;
    
    /// Why the workspace root cannot be found.
    #[derive(Debug, Clone, PartialEq, Eq)]
    #[non_exhaustive]
    pub enum RootError {
        /// `cargo` did not run to a result: it could not be started, ran too long,
        /// or wrote more than the bound.
        Exec(ExecError),
        /// `cargo metadata` ran and failed, as it does outside a Cargo workspace.
        Failed {
            /// The exit code, or `None` when a signal ended the process.
            code: Option<i32>,
            /// What it wrote to standard error.
            stderr: String,
        },
        /// The output is not UTF-8, so it is not JSON.
        NotUtf8,
        /// The output is not the JSON object the field is read from.
        Json(String),
        /// The field is not an absolute path.
        NotAbsolute {
            /// The value.
            value: String,
        },
    }
    
    impl fmt::Display for RootError {
        fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
            match self {
                Self::Exec(error) => write!(f, "{error}"),
                Self::Failed { code, stderr } => {
                    write!(
                        f,
                        "`cargo metadata --format-version 1 --no-deps --frozen` failed"
                    )?;
                    match code {
                        Some(code) => write!(f, " (exit {code})")?,
                        None => write!(f, " (ended by a signal)")?,
                    }
                    if !stderr.is_empty() {
                        write!(f, ": {stderr}")?;
                    }
                    write!(
                        f,
                        "; run it inside a Cargo workspace, or name the workspace root directly"
                    )
                }
                Self::NotUtf8 => write!(
                    f,
                    "the output of `cargo metadata` is not UTF-8, so it is not JSON; the workspace root is not read"
                ),
                Self::Json(reason) => write!(
                    f,
                    "the output of `cargo metadata` does not give the workspace root: {reason}"
                ),
                Self::NotAbsolute { value } => write!(
                    f,
                    "`workspace_root` of `cargo metadata` is `{}`, which is not an absolute path",
                    value.escape_debug()
                ),
            }
        }
    }
    
    impl Error for RootError {
        fn source(&self) -> Option<&(dyn Error + 'static)> {
            match self {
                Self::Exec(error) => Some(error),
                _ => None,
            }
        }
    }
    
    impl From<ExecError> for RootError {
        fn from(error: ExecError) -> Self {
            Self::Exec(error)
        }
    }
    
    impl From<JsonError> for RootError {
        fn from(error: JsonError) -> Self {
            Self::Json(error.to_string())
        }
    }
    
    /// The workspace root of the Cargo workspace the runner's directory is in, as
    /// the `workspace_root` field of `cargo metadata` gives it.
    ///
    /// # Errors
    ///
    /// A [`RootError`] when `cargo` cannot run or fails, writes more than 64 MiB, or
    /// its output does not give the root as the string member `workspace_root` of
    /// the top-level JSON object, an absolute path.
    pub fn discover(runner: &dyn Runner) -> Result<PathBuf, RootError> {
        let output = runner.run(&Invocation::metadata(), &Limits::metadata())?;
        if !output.success() {
            return Err(RootError::Failed {
                code: output.code,
                stderr: output.stderr_text(),
            });
        }
        let text = std::str::from_utf8(&output.stdout).map_err(|_| RootError::NotUtf8)?;
        from_metadata(text)
    }
    
    /// The root from the text of a `cargo metadata` output.
    fn from_metadata(text: &str) -> Result<PathBuf, RootError> {
        let value = json::workspace_root(text)?;
        if value.contains('\0') || !Path::new(&value).is_absolute() {
            return Err(RootError::NotAbsolute { value });
        }
        Ok(PathBuf::from(value))
    }

    //! The one field of `cargo metadata` the workplan reads.
    //!
    //! The scanner validates the whole text as JSON (RFC 8259: no comments, no
    //! trailing commas, no raw control character in a string, numbers of the
    //! grammar, escapes of the set) while it looks for the member `workspace_root`
    //! of the top-level object, and keeps nothing else. A `workspace_root` key
    //! anywhere deeper, as a package may carry in its own `metadata` table, is not
    //! the field. A text that is not JSON, nested deeper than [`MAX_DEPTH`], not an
    //! object at the top, without the member, with the member twice, or with a
    //! member that is not a string is an error: the scanner never guesses.
    
    use std::fmt;
    
    /// The deepest nesting accepted. `cargo metadata` nests a few levels; a text
    /// deeper than this is not its output.
    const MAX_DEPTH: usize = 128;
    
    /// The name of the member.
    const MEMBER: &str = "workspace_root";
    
    /// Why the text is not the output of `cargo metadata`.
    #[derive(Debug, Clone, PartialEq, Eq)]
    pub(super) enum JsonError {
        /// The text is not valid JSON, at this byte offset.
        Malformed {
            /// The offset of the byte the scanner stopped at.
            offset: usize,
            /// What was expected there.
            reason: &'static str,
        },
        /// The top-level value is not an object.
        NotAnObject,
        /// The top-level object has no member `workspace_root`.
        Missing,
        /// The top-level object has the member twice.
        Duplicate,
        /// The member is not a string.
        NotAString,
        /// The string holds an unpaired surrogate escape, so it is not text.
        NotText,
    }
    
    impl fmt::Display for JsonError {
        fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
            match self {
                Self::Malformed { offset, reason } => {
                    write!(f, "not JSON: {reason} at byte {offset}")
                }
                Self::NotAnObject => write!(f, "the top-level value is not an object"),
                Self::Missing => write!(f, "the top-level object has no `{MEMBER}` member"),
                Self::Duplicate => write!(f, "the top-level object has `{MEMBER}` twice"),
                Self::NotAString => write!(f, "`{MEMBER}` is not a string"),
                Self::NotText => write!(f, "`{MEMBER}` holds an unpaired surrogate escape"),
            }
        }
    }
    
    /// The value of the top-level member `workspace_root`.
    pub(super) fn workspace_root(text: &str) -> Result<String, JsonError> {
        let mut scan = Scan {
            bytes: text.as_bytes(),
            at: 0,
        };
        scan.space();
        if scan.peek() != Some(b'{') {
            // Text that is JSON, but not an object, is the wrong shape; text that
            // is not JSON is reported as that.
            scan.value(0)?;
            scan.space();
            scan.end()?;
            return Err(JsonError::NotAnObject);
        }
        let root = scan.top_object()?;
        scan.space();
        scan.end()?;
        root.ok_or(JsonError::Missing)
    }
    
    /// A cursor over the bytes of a UTF-8 text.
    struct Scan<'t> {
        bytes: &'t [u8],
        at: usize,
    }
    
    impl Scan<'_> {
        fn peek(&self) -> Option<u8> {
            self.bytes.get(self.at).copied()
        }
    
        fn fail(&self, reason: &'static str) -> JsonError {
            JsonError::Malformed {
                offset: self.at,
                reason,
            }
        }
    
        fn space(&mut self) {
            while matches!(self.peek(), Some(b' ' | b'\t' | b'\n' | b'\r')) {
                self.at += 1;
            }
        }
    
        fn eat(&mut self, byte: u8, reason: &'static str) -> Result<(), JsonError> {
            if self.peek() == Some(byte) {
                self.at += 1;
                Ok(())
            } else {
                Err(self.fail(reason))
            }
        }
    
        fn end(&self) -> Result<(), JsonError> {
            if self.at == self.bytes.len() {
                Ok(())
            } else {
                Err(self.fail("the end of the text"))
            }
        }
    
        /// The members of the top-level object. The value of `workspace_root` is
        /// returned; every other value is validated and dropped.
        fn top_object(&mut self) -> Result<Option<String>, JsonError> {
            self.eat(b'{', "`{`")?;
            let mut found: Option<String> = None;
            self.space();
            if self.peek() == Some(b'}') {
                self.at += 1;
                return Ok(found);
            }
            loop {
                self.space();
                let key = self.string(true)?.unwrap_or_default();
                self.space();
                self.eat(b':', "`:` after a key")?;
                self.space();
                if key == MEMBER {
                    if found.is_some() {
                        return Err(JsonError::Duplicate);
                    }
                    if self.peek() != Some(b'"') {
                        // Validate the value first: a text that is not JSON is
                        // reported as that, whatever the member holds.
                        self.value(1)?;
                        return Err(JsonError::NotAString);
                    }
                    found = Some(self.string(true)?.ok_or(JsonError::NotText)?);
                } else {
                    self.value(1)?;
                }
                self.space();
                match self.peek() {
                    Some(b',') => self.at += 1,
                    Some(b'}') => {
                        self.at += 1;
                        return Ok(found);
                    }
                    _ => return Err(self.fail("`,` or `}`")),
                }
            }
        }
    
        /// Validate one value at this nesting depth.
        fn value(&mut self, depth: usize) -> Result<(), JsonError> {
            if depth > MAX_DEPTH {
                return Err(self.fail("nesting no deeper than 128"));
            }
            match self.peek() {
                Some(b'{') => self.object(depth),
                Some(b'[') => self.array(depth),
                Some(b'"') => self.string(false).map(drop),
                Some(b'-' | b'0'..=b'9') => self.number(),
                Some(b't') => self.word(b"true"),
                Some(b'f') => self.word(b"false"),
                Some(b'n') => self.word(b"null"),
                _ => Err(self.fail("a value")),
            }
        }
    
        fn object(&mut self, depth: usize) -> Result<(), JsonError> {
            self.eat(b'{', "`{`")?;
            self.space();
            if self.peek() == Some(b'}') {
                self.at += 1;
                return Ok(());
            }
            loop {
                self.space();
                self.string(false)?;
                self.space();
                self.eat(b':', "`:` after a key")?;
                self.space();
                self.value(depth + 1)?;
                self.space();
                match self.peek() {
                    Some(b',') => self.at += 1,
                    Some(b'}') => {
                        self.at += 1;
                        return Ok(());
                    }
                    _ => return Err(self.fail("`,` or `}`")),
                }
            }
        }
    
        fn array(&mut self, depth: usize) -> Result<(), JsonError> {
            self.eat(b'[', "`[`")?;
            self.space();
            if self.peek() == Some(b']') {
                self.at += 1;
                return Ok(());
            }
            loop {
                self.space();
                self.value(depth + 1)?;
                self.space();
                match self.peek() {
                    Some(b',') => self.at += 1,
                    Some(b']') => {
                        self.at += 1;
                        return Ok(());
                    }
                    _ => return Err(self.fail("`,` or `]`")),
                }
            }
        }
    
        fn word(&mut self, word: &'static [u8]) -> Result<(), JsonError> {
            if self.bytes[self.at..].starts_with(word) {
                self.at += word.len();
                Ok(())
            } else {
                Err(self.fail("`true`, `false` or `null`"))
            }
        }
    
        /// `-? (0 | [1-9][0-9]*) (. [0-9]+)? ([eE] [+-]? [0-9]+)?`.
        fn number(&mut self) -> Result<(), JsonError> {
            if self.peek() == Some(b'-') {
                self.at += 1;
            }
            match self.peek() {
                Some(b'0') => self.at += 1,
                Some(b'1'..=b'9') => self.digits(),
                _ => return Err(self.fail("a digit")),
            }
            if self.peek() == Some(b'.') {
                self.at += 1;
                self.some_digits()?;
            }
            if matches!(self.peek(), Some(b'e' | b'E')) {
                self.at += 1;
                if matches!(self.peek(), Some(b'+' | b'-')) {
                    self.at += 1;
                }
                self.some_digits()?;
            }
            Ok(())
        }
    
        fn digits(&mut self) {
            while self.peek().is_some_and(|byte| byte.is_ascii_digit()) {
                self.at += 1;
            }
        }
    
        fn some_digits(&mut self) -> Result<(), JsonError> {
            if self.peek().is_some_and(|byte| byte.is_ascii_digit()) {
                self.digits();
                Ok(())
            } else {
                Err(self.fail("a digit"))
            }
        }
    
        /// One string. With `keep` the decoded text is returned, `None` when an
        /// escape pairs no surrogate; without it the string is only validated and
        /// `None` is returned.
        fn string(&mut self, keep: bool) -> Result<Option<String>, JsonError> {
            self.eat(b'"', "`\"`")?;
            let mut out: Vec<u8> = Vec::new();
            let mut text = true;
            loop {
                let Some(byte) = self.peek() else {
                    return Err(self.fail("the end of the string"));
                };
                match byte {
                    b'"' => {
                        self.at += 1;
                        break;
                    }
                    0..=0x1f => return Err(self.fail("an escaped control character")),
                    b'\\' => {
                        self.at += 1;
                        let Some(escape) = self.peek() else {
                            return Err(self.fail("an escape"));
                        };
                        self.at += 1;
                        let plain = match escape {
                            b'"' | b'\\' | b'/' => escape,
                            b'b' => 0x08,
                            b'f' => 0x0c,
                            b'n' => b'\n',
                            b'r' => b'\r',
                            b't' => b'\t',
                            b'u' => {
                                let unit = self.unit()?;
                                match self.scalar(unit)? {
                                    Some(c) => {
                                        if keep {
                                            let mut buffer = [0; 4];
                                            out.extend_from_slice(
                                                c.encode_utf8(&mut buffer).as_bytes(),
                                            );
                                        }
                                    }
                                    None => text = false,
                                }
                                continue;
                            }
                            _ => {
                                self.at -= 1;
                                return Err(self.fail("one of \" \\ / b f n r t u after a backslash"));
                            }
                        };
                        if keep {
                            out.push(plain);
                        }
                    }
                    _ => {
                        // A byte of the text itself; the text is UTF-8, so a
                        // multi-byte character passes through byte by byte.
                        if keep {
                            out.push(byte);
                        }
                        self.at += 1;
                    }
                }
            }
            if !keep {
                return Ok(None);
            }
            if !text {
                return Ok(None);
            }
            // The raw bytes came from a `str` and each escape was encoded whole.
            String::from_utf8(out)
                .map(Some)
                .map_err(|_| self.fail("UTF-8 text"))
        }
    
        /// The four hexadecimal digits after `\u`.
        fn unit(&mut self) -> Result<u16, JsonError> {
            let mut value: u16 = 0;
            for _ in 0..4 {
                let digit = self
                    .peek()
                    .and_then(|byte| char::from(byte).to_digit(16))
                    .ok_or_else(|| self.fail("four hexadecimal digits after `\\u`"))?;
                // A digit is below 16, so it fits.
                value = value * 16 + u16::try_from(digit).unwrap_or(0);
                self.at += 1;
            }
            Ok(value)
        }
    
        /// The character a code unit stands for, pairing a high surrogate with the
        /// `\u` escape that must follow it. `None` for an unpaired surrogate,
        /// which is valid JSON and not text.
        fn scalar(&mut self, unit: u16) -> Result<Option<char>, JsonError> {
            match unit {
                0xd800..=0xdbff => {
                    if self.bytes[self.at..].starts_with(b"\\u") {
                        let save = self.at;
                        self.at += 2;
                        let low = self.unit()?;
                        if (0xdc00..=0xdfff).contains(&low) {
                            let high = u32::from(unit) - 0xd800;
                            let low = u32::from(low) - 0xdc00;
                            return Ok(char::from_u32(0x1_0000 + (high << 10) + low));
                        }
                        // Not a pair: the high surrogate is alone, and the escape
                        // that followed is read again as its own.
                        self.at = save;
                    }
                    Ok(None)
                }
                0xdc00..=0xdfff => Ok(None),
                other => Ok(char::from_u32(u32::from(other))),
            }
        }
    }

    /// The workspace root of the Cargo workspace the runner's directory is in, as
    /// the `workspace_root` field of `cargo metadata` gives it.
    ///
    /// # Errors
    ///
    /// A [`RootError`] when `cargo` cannot run or fails, writes more than 64 MiB, or
    /// its output does not give the root as the string member `workspace_root` of
    /// the top-level JSON object, an absolute path.
    pub fn discover(runner: &dyn Runner) -> Result<PathBuf, RootError> {
        let output = runner.run(&Invocation::metadata(), &Limits::metadata())?;
        if !output.success() {
            return Err(RootError::Failed {
                code: output.code,
                stderr: output.stderr_text(),
            });
        }
        let text = std::str::from_utf8(&output.stdout).map_err(|_| RootError::NotUtf8)?;
        from_metadata(text)
    }
